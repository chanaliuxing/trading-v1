```python
def _pass_cost_gate(self, sig: Dict[str,Any], feat: Dict[str,Any],
                    df1: pd.DataFrame, df5: pd.DataFrame, eps: float) -> Dict[str, Any]:
    px = float(feat['close'])
    commission = float(self.cfg.get("commission_per_share", 0.0035))
    slip_cost = px * eps  # 单边 half-spread

    # ① 预期毛利（维持你原有三类估算）
    if sig['type'].startswith("A1"):
        a1_profit_pct = self.cfg.get("A1", {}).get("exit", {}).get("profit_take_pct", 0.7)
        vwap = float(_vwap_session(df1).iloc[-1])
        gross = max(0.0, vwap - px) * a1_profit_pct
        target_px, profit_type = vwap, "VWAP"

    elif sig['type'].startswith("A2"):
        a2_profit_pct = self.cfg.get("A2", {}).get("exit", {}).get("anchor_profit_pct", 0.6)
        sma30 = float(_ema(df5['close'], span=30).iloc[-1])
        gross = max(0.0, sma30 - px) * a2_profit_pct
        target_px, profit_type = sma30, "ANCHOR"

    elif sig['type'].startswith("B1"):
        orb_hi = float(self.state.get("_orb_state", {}).get("hi", px))
        b1_profit_pct = float(self.cfg.get("B1", {}).get("exit", {}).get("profit_take_pct", 0.4))
        gross = max(0.0, orb_hi - px) * max(0.4, b1_profit_pct)
        target_px, profit_type = orb_hi, "ORB_HI"

    else:
        gross = 0.0
        target_px, profit_type = px, "UNKNOWN"

    # ② 动态 edge_floor：P60(近笔 PnL) 与 ATR_5m 的 2% 二者取大
    atr5 = 0.0

    except Exception:
        atr5 = 0.0

    recent = list(self.guard.state.get("recent_net_pnl_list", []))
    if len(recent) >= 5:
        pnl_p60 = float(np.percentile(np.array(recent, dtype=float), 60))
    else:
        pnl_p60 = float(self.cfg.get("edge_floor", 0.18))  # 配置兜底

    edge_from_atr = 0.02 * atr5
    edge_floor = float(max(pnl_p60, edge_from_atr))

    # ③ 最终判定

    passed = bool(net_edge >= edge_floor)

        "target_px": round(target_px, 2),

        "reason": f"net_edge={net_edge:.4f} {'>=' if passed else '<'} floor={edge_floor:.4f}"
```

```python
        gross = max(0.0, orb_hi * b1_mult - px)
        target_px, profit_type = orb_hi * b1_mult, "ORB_HI"
```

```python
    if len(df5) < 60:

    sma30 = _ema(df5['close'], span=30)
    close = float(df5['close'].iloc[-1])
    anchor = float(sma30.iloc[-1])
    g = float(p['A2_grid_g'])

    # 反饥饿：放宽 g，但不低于 floor
    starv_lvl = int(self.guard.state.get("starvation_level", 0))
    if starv_lvl >= 1:
        relax = self.cfg.get("anti_starvation", {}).get("relax_g", [0.9, 0.8])
        idx = min(starv_lvl - 1, len(relax) - 1)
        g = max(g * float(relax[idx]), float(p.get('A2_grid_g_floor', g)))

    # 单边过滤（避免大幅单边追价）
    oneway_thr = float(p.get('A2_oneway_thr', 2.2))  # 按 %
    day_open = float(df5['open'].iloc[0])
    if abs(close - day_open) / max(1e-6, day_open) >= (oneway_thr / 100.0):
        return None

    max_layers = int(p.get("A2_max_layers", 4))
    cur_layers = int(self.guard.state.get("a2_layers", 0))
    if cur_layers >= max_layers:
        return None

    next_layer = cur_layers + 1
    want_px = anchor - g * next_layer

    if (close <= want_px) and (not self._paused()):
        # 命中新层：更新分层状态
        self.guard.state["a2_layers"] = next_layer
        self.guard._persist()
        return {
            "type": "A2_LAYER",
            "reason": f"close<=anchor-g*{next_layer}",
            "layer": next_layer,
            "anti_starvation_probe": (starv_lvl > 0)
        }
    return None
```

```python
async def _execute(self, sig: Dict[str, Any], features: Dict[str,Any], mult: float, eps: float):
    px = float(features['close'])

    # 1) 基准单位
    base_pct = float(self.cfg.get("positioning", {}).get("base_unit_pct", 0.012))
    base_usd = self.params.capital_limit_usd * base_pct
    base_qty = max(1, int(base_usd / max(px, 1e-6)))

    target_qty = base_qty
    sizing_reason = "Base"

    # 2) A2 金字塔分层
    if sig.get("type") == "A2_LAYER":
        layer = int(sig.get("layer", 1))
        ratios = list(self.cfg.get("A2", {}).get("sizing_ratios", [1.0, 0.8, 0.6, 0.4]))
        r = float(ratios[layer-1]) if 1 <= layer <= len(ratios) else 1.0
        target_qty = max(1, int(base_qty * r))
        sizing_reason = f"A2 L{layer} (ratio={r})"

    # 3) 反饥饿探测单（缩量）
    if bool(sig.get("anti_starvation_probe", False)):
        pr = float(self.cfg.get("anti_starvation", {}).get("probe_ratio", 0.5))
        target_qty = max(1, int(target_qty * pr))
        sizing_reason += f" | Probe({pr})"

    # 4) 应用风控乘数
    final_qty = int(max(1, target_qty) * float(mult))

    self.logger.info(
        "Position Sizing (Advanced)",
        current_px=round(px, 2),
        capital_limit=self.params.capital_limit_usd,
        base_qty=base_qty,
        sizing_reason=sizing_reason,
        target_qty_before_mult=target_qty,
        risk_multiplier=mult,
        final_qty=final_qty,
        notional_usd=round(final_qty * px, 2)
    )

    if final_qty <= 0:
        self.logger.warning("Final qty is 0 after multipliers", target_qty=target_qty, mult=mult)
        return

    limit_px = self._marketable_limit(features, side="BUY", eps=eps)

    if self.portfolio_guard:
        final_qty = self.portfolio_guard.request_allocation(self.symbol, final_qty, limit_px)
    if final_qty <= 0:
        return

    decision = DecisionEnvelope(
        decision_id=self.generate_decision_id(),
        strategy_id=self.strategy_id,
        symbol=self.symbol,
        side="BUY",
        target_qty=final_qty,
        limit_px=limit_px,
        rationale=f"{sig['type']} {sig.get('reason','')} | {sizing_reason}"
    )

    self.logger.info("Triggering Order", decision_id=decision.decision_id, qty=decision.target_qty, limit_px=decision.limit_px, rationale=decision.rationale)
```

```python
def _ema(s: pd.Series, span: int) -> pd.Series:
    return s.ewm(span=span, adjust=False).mean()

def _atr_wilder(df: pd.DataFrame, n: int = 14) -> pd.Series:
    h, l, c = df['high'], df['low'], df['close']
    tr = pd.concat([h-l, (h-c.shift()).abs(), (l-c.shift()).abs()], axis=1).max(axis=1)
    return tr.ewm(alpha=1/n, adjust=False).mean()

def _vwap_session(df: pd.DataFrame) -> pd.Series:
    q = df['volume'].clip(lower=0.0)
    pq = df['close'] * q
    return (pq.cumsum() / q.cumsum()).replace([np.inf, -np.inf], np.nan).ffill()

    # 去极端跳变
    df = df[~(df['close'].pct_change().abs() > 0.25)]
    if isinstance(df.index, pd.DatetimeIndex):
        return df.between_time("09:30","16:00")
    return df

        try:
            df1p = self.data_dir / f"{self.symbol.lower()}_1m.parquet"

        # A1: VWAP 偏离阈值（60D 统计，取 |dev| 的 P60 并裁剪）
        vwap1 = _vwap_session(self.df1.last('60D'))
        dev = (self.df1.last('60D')['close'] - vwap1) / vwap1
        thr_sigma = float(np.clip(self._percentile(dev.abs(), 60, 0.007), 0.004, 0.020))

        # 止损系数 k_atr（以 14ATR 的均值做基线+分位比率）
        atr5 = _atr_wilder(self.df5.last('60D'), 14)
        atr_mean = float(np.nanmean(atr5))
        ret_abs = self.df5.last('60D')['close'].diff().abs().dropna()
        ratio = self._percentile(ret_abs, 90, 1.2) / max(atr_mean, 1e-6)
        k_atr = float(np.clip(1.2 + 0.4 * ((ratio - 1.2)/(3.0 - 1.2)), 1.4, 3.0))

        self.params = {
            "symbol": self.symbol,
            "k_atr": round(k_atr, 2),
            "A1": {"thr_sigma": round(thr_sigma, 4), "rvol_lo": 0.85, "rvol_hi": 1.8, "time_exit_min": 40},
            "A2": {"grid_g": round(a2_grid_g, 2), "grid_g_floor": 0.8, "oneway_thr": 2.2, "max_layers": 4},
            "B1": {"orb_len_min": 30, "rvol_min": 0.85, "pullback_window_M": 6}
        }

        # 简化版触发率调节（近7D A1 触发日均 < 目标时放宽）
```

#### **2. 实现策略端的“平滑吸收器” (`_mini_recalibrate`)**

现在，我们给策略装上“吸收”新参数的器官。`_mini_recalibrate` 不再是空函数，它将拥有平滑更新的核心逻辑。

**找到 `TSLASimplifiedV7_2` 类中的 `_mini_recalibrate` 方法，并用以下代码完整替换它：**

```python
# 在 strategies/tsla_v7_2.py 文件中

# ... 其他代码 ...

class TSLASimplifiedV7_2(StrategyAlgorithm):
    # ... 其他方法 ...

    # 【↓↓↓ 使用下面的函数完整替换掉旧的 _mini_recalibrate ↓↓↓】
    async def _mini_recalibrate(self):
        """
        Performs smooth updates of dynamic parameters during the trading session.
        This function is called periodically by on_features.
        """
        suggestion_file = f"strategies/configs/{self.symbol.lower()}_v7_2.suggestion.json"
        
        try:
            # 1. 尝试读取“建议文件”
            with open(suggestion_file, 'r') as f:
                suggestions = json.load(f)
            
            if not suggestions:
                return

            self.logger.info("New parameter suggestions found, applying smooth update...", suggestions=suggestions)
            
            # 2. 定义平滑因子 (alpha)，可以从配置中读取
            alpha = float(self.cfg.get("recalibration", {}).get("smoothing_alpha", 0.2))
            
            updated_params = []

            # 3. 遍历建议，平滑更新每一个参数
            for key, suggested_value in suggestions.items():
                # 假设参数是一层嵌套，例如 "A1_thr_sigma"
                if key in self.state["dyn"]:
                    current_value = self.state["dyn"][key]
                elif key in self.state["dyn"].get("A1", {}):
                    current_value = self.state["dyn"]["A1"][key]
                else:
                    continue # 如果参数不存在，则跳过

                # 核心平滑公式: new = alpha * suggestion + (1 - alpha) * current
                new_value = (alpha * float(suggested_value)) + ((1 - alpha) * float(current_value))
                
                # 更新状态
                if key in self.state["dyn"]:
                    self.state["dyn"][key] = new_value
                elif key in self.state["dyn"].get("A1", {}):
                    self.state["dyn"]["A1"][key] = new_value

                updated_params.append(f"{key}: {current_value:.4f} -> {new_value:.4f}")

            if updated_params:
                self.logger.info("Parameters smoothly updated.", changes=updated_params)

        except FileNotFoundError:
            # 文件不存在是正常情况（例如盘中大脑还没来得及第一次运行）
            pass 
        except Exception as e:
            self.logger.error("Error during mini_recalibrate", error=str(e))
    # 【↑↑↑ 替换到此结束 ↑↑↑】
```
