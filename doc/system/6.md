# ① PG 咨询锁 & 存储过程（审批→预留→下单→释放）

## 1.1 锁ID计算函数（策略×标的 64bit）

```sql
-- 使用内置 hashtextextended，按策略×symbol 生成 64bit 锁ID
CREATE OR REPLACE FUNCTION core_hash_lock_id(p_strategy text, p_symbol text)
RETURNS bigint LANGUAGE sql IMMUTABLE AS $$
  SELECT (hashtextextended(p_strategy, 0) # hashtextextended(p_symbol, 0));
$$;
```

## 1.2 审批 + 资金预留（幂等、原子、串行）

```sql
-- 输入：策略、标的、名义金额、缓冲bps、幂等键
-- 输出：状态码、reservation_id、预留金额、拒绝原因
CREATE OR REPLACE FUNCTION core_sp_risk_approve_and_reserve(
    p_strategy_id   text,
    p_symbol        text,
    p_notional_usd  numeric,          -- 预计名义额
    p_buffer_bps    int,              -- 缓冲bps(含滑点/佣金)
    p_idem_key      text
) RETURNS TABLE (
    status          text,             -- APPROVED / DENIED_LIMIT / RETRY_LATER / IDEMPOTENT
    reservation_id  uuid,
    reserved_usd    numeric,
    reason          text
) LANGUAGE plpgsql AS $$
DECLARE
    v_lock_id     bigint := core_hash_lock_id(p_strategy_id, p_symbol);
    v_cap         numeric;
    v_reserved    numeric;
    v_buf_ratio   numeric;
    v_need        numeric;
    v_res_id      uuid;
    v_exists      core_order_reservation;
BEGIN
    -- 幂等：已有记录直接返回
    SELECT * INTO v_exists FROM core_order_reservation WHERE idempotency_key = p_idem_key;
    IF FOUND THEN
        status := 'IDEMPOTENT';
        reservation_id := v_exists.reservation_id;
        reserved_usd := v_exists.reserved_usd;
        reason := v_exists.status;
        RETURN;
    END IF;

    -- 事务+咨询锁
    PERFORM pg_advisory_xact_lock(v_lock_id);  -- 阻塞式；若要非阻塞可换 pg_try_advisory_xact_lock

    -- 读取资金池
    SELECT s.capital_limit_usd, l.reserved_usd, l.buffer_ratio
      INTO v_cap, v_reserved, v_buf_ratio
      FROM core_strategy s
      JOIN core_strategy_ledger l USING(strategy_id)
     WHERE s.strategy_id = p_strategy_id
     FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'strategy % not found', p_strategy_id;
    END IF;

    -- 计算预留金额：notional*(1 + max(配置缓冲, 账本缓冲))
    v_need := p_notional_usd * (1 + GREATEST(p_buffer_bps/10000.0, COALESCE(v_buf_ratio,0)));

    -- 超额检查（这里只做资金池，板块/黑名单/时窗等可在服务层先判掉）
    IF v_reserved + v_need > v_cap THEN
        status := 'DENIED_LIMIT';
        reservation_id := NULL; reserved_usd := 0; reason := 'CAP_EXCEEDED';
        RETURN;
    END IF;

    -- 写入 reservation（幂等键唯一）
    v_res_id := gen_random_uuid();
    INSERT INTO core_order_reservation(reservation_id, strategy_id, symbol, notional_usd,
        buffer_bps, reserved_usd, status, idempotency_key)
    VALUES (v_res_id, p_strategy_id, p_symbol, p_notional_usd,
        p_buffer_bps, v_need, 'RESERVED', p_idem_key);

    -- 更新 ledger.reserved
    UPDATE core_strategy_ledger
       SET reserved_usd = reserved_usd + v_need,
           updated_at = now(), version = version + 1
     WHERE strategy_id = p_strategy_id;

    status := 'APPROVED';
    reservation_id := v_res_id;
    reserved_usd := v_need;
    reason := NULL;
    RETURN;
END;
$$;
```

## 1.3 订单终态释放（按幂等键释放，防脏账）

```sql
-- 将订单置终态 + 释放资金预留（一次性释放；如需部分释放可扩展）
CREATE OR REPLACE FUNCTION core_sp_release_on_terminal(
    p_idem_key      text,             -- 与 reservation/order 同幂等键
    p_order_id      uuid,             -- 执行器侧下单后回填
    p_terminal      text,             -- FILLED / CANCELLED / REJECTED
    p_reason        text
) RETURNS TABLE (
    ok              boolean,
    released_usd    numeric,
    msg             text
) LANGUAGE plpgsql AS $$
DECLARE
    v_res core_order_reservation;
BEGIN
    -- 找预留
    SELECT * INTO v_res FROM core_order_reservation
     WHERE idempotency_key = p_idem_key
     FOR UPDATE;
    IF NOT FOUND THEN
        ok := FALSE; released_usd := 0; msg := 'RESERVATION_NOT_FOUND'; RETURN;
    END IF;

    -- 幂等：已释放则返回OK
    IF v_res.status IN ('RELEASED','CANCELLED') THEN
        ok := TRUE; released_usd := 0; msg := 'ALREADY_RELEASED'; RETURN;
    END IF;

    -- 更新订单状态（若存在）
    UPDATE core_order
       SET state = p_terminal, last_update_at = now()
     WHERE idempotency_key = p_idem_key OR order_id = p_order_id;

    -- 标记释放
    UPDATE core_order_reservation
       SET status='RELEASED', released_at=now(), reason=p_reason, order_id=p_order_id
     WHERE reservation_id = v_res.reservation_id;

    -- 回收 ledger 预留
    UPDATE core_strategy_ledger
       SET reserved_usd = GREATEST(reserved_usd - v_res.reserved_usd, 0),
           updated_at = now(), version = version + 1
     WHERE strategy_id = v_res.strategy_id;

    ok := TRUE; released_usd := v_res.reserved_usd; msg := 'RELEASED';
    RETURN;
END;
$$;
```

## 1.4 过期预留清理（容灾）

```sql
-- 超过 TTL 仍未产生订单的预留，自动取消并回收（定时任务跑）
CREATE OR REPLACE FUNCTION core_sp_sweep_stale_reservations(p_ttl_minutes int DEFAULT 15)
RETURNS TABLE (reservation_id uuid, reclaimed numeric) LANGUAGE plpgsql AS $$
DECLARE r record;
BEGIN
  FOR r IN
    SELECT * FROM core_order_reservation
     WHERE status='RESERVED' AND created_at < now() - (p_ttl_minutes || ' minutes')::interval
  LOOP
    UPDATE core_order_reservation
       SET status='CANCELLED', released_at=now(), reason='TTL_EXPIRED'
     WHERE reservation_id = r.reservation_id;

    UPDATE core_strategy_ledger
       SET reserved_usd = GREATEST(reserved_usd - r.reserved_usd, 0),
           updated_at = now(), version = version + 1
     WHERE strategy_id = r.strategy_id;

    reservation_id := r.reservation_id; reclaimed := r.reserved_usd; RETURN NEXT;
  END LOOP;
END; $$;
```

**建议**：对 `core_sp_*` 加 `SECURITY DEFINER` 并建立**最小权限**角色，仅风险/执行服务可调用。

# ③ 风险/执行侧调用流程（伪代码）

**Risk Service：审批与预留**

```python
def approve_and_reserve(strategy_id, symbol, notional_usd, buffer_bps, idem_key):
    row = sql.fetchrow("""
        SELECT * FROM core_sp_risk_approve_and_reserve($1,$2,$3,$4,$5);
    """, [strategy_id, symbol, notional_usd, buffer_bps, idem_key])
    if row.status == 'APPROVED':
        return {"ok": True, "reservation_id": row.reservation_id, "reserved_usd": row.reserved_usd}
    elif row.status == 'IDEMPOTENT':
        return {"ok": True, "reservation_id": row.reservation_id, "reserved_usd": row.reserved_usd, "idempotent": True}
    elif row.status == 'DENIED_LIMIT':
        raise Denied("CAP_EXCEEDED")
    elif row.status == 'RETRY_LATER':
        raise Retry("LOCK_BUSY")
```

# ④ 关键测试用例（上线前必须通过）

1. **并发审批同标的**：两个策略/实例并发调用 `core_sp_risk_approve_and_reserve`（同策略×symbol） → 只有一个 `APPROVED`，另一个阻塞或走 `RETRY_LATER`。
2. **幂等**：相同 `idempotency_key` 连续调用审批 → 第二次返回 `IDEMPOTENT`；下单与释放同理。
3. **资金上限**：构造接近上限的 `notional`，使一次 `DENIED_LIMIT`；账本不增长。
4. **终态释放**：下单后 `FILLED` 或 `CANCELLED` → `reserved_usd` 恢复原值（零漂移）。
5. **过期清理**：制造 `RESERVED` 但未下单的记录，运行 `core_sp_sweep_stale_reservations(1)` → 变为 `CANCELLED` 且回收。
6. **重启对账配合**：持有实挂单重启 → Reconcile 领养并补挂 OCO；随后终态释放资金，与账本一致。
