# RiskService（最小闭环版）：意图 → 校验 → 预留 → 决策

特点：

* 消费 `order.intent.v1`（来自策略引擎）
* 做**基础风控**（交易时段、黑白名单、单笔名义上限等）
* 到 **PG `reservations`** 表做**幂等预留**（与 Portfolio 的表结构一致）
* 通过则发 `risk.decision.v2`（携带 `idempotency_key/decision_id`）
* 拒绝则发 `risk.decision.v2`（REJECTED）+ `risk.violation.v1`
* Prom 指标：批准/拒绝/原因分布、延迟

> 说明：为简洁起见，这个最小版**直接写 PG** 做 reservation（与我们给的 PortfolioService 共享同一张表），不额外引入 RPC。后续要解耦时，再把“预留”改为调用 Portfolio 的 HTTP/gRPC。

---

## 主题契约（简述）

### 输入：`order.intent.v1`

```json
{
  "schema_v":1,
  "metadata":{
    "ts_us": 1730540000000000,
    "idempotency_key": "engine:SID:SYM:UUID",
    "decision_id": "same-as-idemp-or-uuid",
    "producer": "engine"
  },
  "data":{
    "strategy_id":"ORB",
    "symbol":"AAPL",
    "side":"BUY",
    "order_type":"LMT",
    "limit_px":190.25,
    "qty":100,
    "tif":"DAY",
    "exec_overrides": {"stop_policy": {...}},
    "allow_prepost": false   // 可选，是否允许盘前/盘后
  }
}
```

### 输出：`risk.decision.v2`

```json
{
  "schema_v": 2,
  "metadata": {
    "ts_us": 1730540001000000,
    "idempotency_key": "engine:SID:SYM:UUID",
    "decision_id": "same-as-input",
    "producer": "risk"
  },
  "decision": "APPROVED",  // or "REJECTED"
  "reason": "ok",          // 若拒绝，填具体原因
  "data": { /* 原 data 回传给执行内核用 */ }
}
```

### 侧路：`risk.violation.v1`（只有拒绝时发）

```json
{
  "schema_v":1,
  "metadata":{"ts_us":1730540001000000,"producer":"risk"},
  "data":{"strategy_id":"ORB","symbol":"AAPL","reason":"notional_exceed","ctx":{...}}
}
```

## 代码（直接粘贴即可运行；不生成文件）

```python
import os, asyncio, json, time, uuid, contextlib, math, pytz, datetime as dt
from dataclasses import dataclass
from typing import Optional, Dict, Any
from aiokafka import AIOKafkaConsumer, AIOKafkaProducer
import asyncpg
from prometheus_client import Counter, Histogram, Gauge, start_http_server

# ---------- 指标 ----------
start_http_server(int(os.getenv("RISK_METRICS_PORT","9107")))
RM = dict(
  rsk_in = Counter("risk_intent_in_total","order intents in"),
  rsk_approved = Counter("risk_approved_total","approved"),
  rsk_rejected = Counter("risk_rejected_total","rejected",["reason"]),
  rsk_latency = Histogram("risk_eval_latency_ms","eval latency ms",
                          buckets=(5,10,20,50,100,200,400,800,1600,3200)),
  rsk_reserve_ms = Histogram("risk_reserve_latency_ms","reserve latency ms"),
  rsk_cap_avail = Gauge("risk_strategy_capital_available","strategy capital available",["strategy"]),
)

# ---------- 配置 ----------
@dataclass
class RiskCfg:
    kafka_bootstrap: str = os.getenv("KAFKA_BOOTSTRAP","localhost:9092")
    group_id: str = os.getenv("RISK_GROUP","risk-v1")
    topic_intent: str = os.getenv("TOPIC_INTENT","order.intent.v1")
    topic_decision: str = os.getenv("TOPIC_DECISION","risk.decision.v2")
    topic_violation: str = os.getenv("TOPIC_VIOLATION","risk.violation.v1")
    pg_dsn: str = os.getenv("PG_DSN","postgres://user:pass@localhost:5432/trade")

    # 基础风控参数（可做成 per-strategy 覆盖）
    trading_tz: str = os.getenv("TRADING_TZ","America/New_York")
    allow_rth_only: bool = os.getenv("RTH_ONLY","true").lower()=="true"
    session_start: str = os.getenv("RTH_START","09:30")
    session_end: str   = os.getenv("RTH_END","16:00")

    max_notional_usd: float = float(os.getenv("MAX_ORDER_NOTIONAL","500000"))  # 单笔名义上限
    slip_bps: int = int(os.getenv("RISK_SLIP_BPS","10"))                       # 预留成本缓冲
    lot_mult: float = float(os.getenv("LOT_MULT","1.0"))                       # 期权/期货可调

    blacklist_syms: str = os.getenv("BLACKLIST","")  # 逗号分隔

# ---------- 工具 ----------
def now_us(): return int(time.time()*1e6)

def in_rth(tzname:str, start_hm:str, end_hm:str, allow_prepost:bool) -> bool:
    if not allow_prepost:
        tz = pytz.timezone(tzname)
        now = dt.datetime.now(tz)
        hm = now.strftime("%H:%M")
        return start_hm <= hm <= end_hm
    return True

# ---------- Risk Service ----------
class RiskService:
    def __init__(self, cfg:RiskCfg):
        self.cfg=cfg
        self.producer=None
        self.consumer=None
        self.pg: Optional[asyncpg.Pool]=None
        self.black = {s.strip().upper() for s in (cfg.blacklist_syms or "").split(",") if s.strip()}
        # 内存镜像策略上限与已预留（与 Portfolio 共用同一套数据表）
        self.cap_limit: Dict[str,float] = {}
        self.reserved: Dict[str,float] = {}

    async def start(self):
        self.pg = await asyncpg.create_pool(self.cfg.pg_dsn, min_size=1, max_size=4)
        async with self.pg.acquire() as con:
            rows = await con.fetch("SELECT strategy_id, capital_limit FROM strategies")
            self.cap_limit = {r["strategy_id"]: float(r["capital_limit"]) for r in rows}
            # 初始已预留（重启恢复）
            rows = await con.fetch("SELECT strategy_id, sum(amount) AS amt FROM reservations WHERE state='HELD' GROUP BY strategy_id")
            self.reserved = {r["strategy_id"]: float(r["amt"]) for r in rows}

        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.cfg.kafka_bootstrap, acks="all", linger_ms=5, compression_type="lz4",
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda k: k.encode("utf-8") if k else None
        )
        await self.producer.start()

        self.consumer = AIOKafkaConsumer(
            self.cfg.topic_intent, bootstrap_servers=self.cfg.kafka_bootstrap, group_id=self.cfg.group_id,
            value_deserializer=lambda b: json.loads(b.decode("utf-8")), auto_offset_reset="latest", enable_auto_commit=True
        )
        await self.consumer.start()
        asyncio.create_task(self._loop())

    async def stop(self):
        if self.consumer: with contextlib.suppress(Exception): await self.consumer.stop()
        if self.producer: with contextlib.suppress(Exception): await self.producer.stop()
        if self.pg: with contextlib.suppress(Exception): await self.pg.close()

    # 可用资金（策略独立池）：上限 - 已预留 + （可加 realized pnl；这里最小闭环先不加）
    def cap_avail(self, sid:str) -> float:
        limit = self.cap_limit.get(sid, 0.0)
        held = self.reserved.get(sid, 0.0)
        avail = max(0.0, limit - held)
        RM["rsk_cap_avail"].labels(strategy=sid).set(avail)
        return avail

    async def _loop(self):
        async for msg in self.consumer:
            RM["rsk_in"].inc()
            t0=time.time()
            try:
                meta = msg.value.get("metadata",{})
                d = msg.value["data"]
                # 提取
                sid = d["strategy_id"]; sym = d["symbol"].upper()
                side = d["side"].upper(); typ = d.get("order_type","LMT").upper()
                px = float(d.get("limit_px") or d.get("price") or 0.0)
                qty = float(d["qty"]); tif = d.get("tif","DAY")
                allow_prepost = bool(d.get("allow_prepost", False))
                idem_key = meta.get("idempotency_key") or f"risk:{sid}:{sym}:{uuid.uuid4()}"
                decision_id = meta.get("decision_id") or idem_key

                # 0) 基本校验
                if sym in self.black:
                    await self._reject(meta, d, decision_id, "symbol_blacklisted", {"symbol":sym}); continue
                if not in_rth(self.cfg.trading_tz, self.cfg.session_start, self.cfg.session_end, allow_prepost and not self.cfg.allow_rth_only):
                    await self._reject(meta, d, decision_id, "not_in_rth", {}); continue
                if side not in ("BUY","SELL") or qty <= 0:
                    await self._reject(meta, d, decision_id, "invalid_side_or_qty", {}); continue
                # 价格检查
                if typ=="LMT" and px <= 0:
                    await self._reject(meta, d, decision_id, "limit_price_missing", {}); continue

                # 1) 单笔名义金额限制
                notional = (px if typ!="MKT" and px>0 else px) * qty * self.cfg.lot_mult
                if notional > self.cfg.max_notional_usd:
                    await self._reject(meta, d, decision_id, "notional_exceed", {"notional":notional}); continue

                # 2) 资金预留（幂等）
                est_cost = notional * (1 + self.cfg.slip_bps/1e4)
                if est_cost > self.cap_avail(sid):
                    await self._reject(meta, d, decision_id, "insufficient_capital", {"need":est_cost, "avail":self.cap_avail(sid)})
                    continue

                t1=time.time()
                ok = await self._reserve_pg(sid, idem_key, est_cost)
                RM["rsk_reserve_ms"].observe((time.time()-t1)*1000.0)
                if not ok:
                    await self._reject(meta, d, decision_id, "reserve_failed", {}); continue

                # 更新内存已预留
                self.reserved[sid] = self.reserved.get(sid,0.0) + est_cost

                # 3) 放行
                await self._approve(meta, d, decision_id)
            except Exception as e:
                await self._reject(msg.value.get("metadata",{}), msg.value.get("data",{}), 
                                   (msg.value.get("metadata",{}) or {}).get("decision_id") or "NA",
                                   "risk_exception", {"err":repr(e)})
            finally:
                RM["rsk_latency"].observe((time.time()-t0)*1000.0)

    async def _reserve_pg(self, strategy_id:str, order_ref:str, amt:float) -> bool:
        async with self.pg.acquire() as con:
            async with con.transaction():
                # 幂等：存在则视为成功
                ex = await con.fetchval("SELECT 1 FROM reservations WHERE strategy_id=$1 AND order_ref=$2", strategy_id, order_ref)
                if ex: 
                    return True
                # 可用性再检（事务内防竞态）
                held = await con.fetchval("SELECT COALESCE(sum(amount),0) FROM reservations WHERE strategy_id=$1 AND state='HELD' FOR UPDATE", strategy_id)
                limit = await con.fetchval("SELECT capital_limit FROM strategies WHERE strategy_id=$1", strategy_id) or 0
                if amt > float(limit - float(held or 0)):
                    return False
                await con.execute("INSERT INTO reservations(res_id,strategy_id,order_ref,amount,state) VALUES($1,$2,$3,$4,'HELD')",
                                  uuid.uuid4(), strategy_id, order_ref, amt)
                return True

    async def _approve(self, meta:Dict[str,Any], data:Dict[str,Any], decision_id:str):
        RM["rsk_approved"].inc()
        out = {
          "schema_v":2,
          "metadata":{
            "ts_us": now_us(),
            "idempotency_key": meta.get("idempotency_key"),
            "decision_id": decision_id,
            "producer":"risk"
          },
          "decision":"APPROVED",
          "reason":"ok",
          "data": data
        }
        await self.producer.send_and_wait(self.cfg.topic_decision, key=f"{data['strategy_id']}:{data['symbol']}", value=out)

    async def _reject(self, meta:Dict[str,Any], data:Dict[str,Any], decision_id:str, reason:str, ctx:Dict[str,Any]):
        RM["rsk_rejected"].labels(reason=reason).inc()
        # 决策
        out = {
          "schema_v":2,
          "metadata":{
            "ts_us": now_us(),
            "idempotency_key": meta.get("idempotency_key"),
            "decision_id": decision_id,
            "producer":"risk"
          },
          "decision":"REJECTED",
          "reason":reason,
          "data": data
        }
        await self.producer.send_and_wait(self.cfg.topic_decision, key=f"{data.get('strategy_id','NA')}:{data.get('symbol','NA')}", value=out)
        # 侧路违规
        vio = {
          "schema_v":1,
          "metadata":{"ts_us": now_us(), "producer":"risk"},
          "data":{"strategy_id": data.get("strategy_id","NA"), "symbol": data.get("symbol","NA"),
                  "reason": reason, "ctx": ctx}
        }
        await self.producer.send_and_wait(self.cfg.topic_violation, key=f"{data.get('strategy_id','NA')}:{data.get('symbol','NA')}", value=vio)
```

**运行方式**

* 配好 PG（与前面 Portfolio 的 DDL 相同）、Kafka；
* 环境变量中指定 `PG_DSN/KAFKA_BOOTSTRAP`；
* 起本服务后，向 `order.intent.v1` 发一条意图消息；
* 如果通过，会看到 `risk.decision.v2` 里 `decision="APPROVED"`；同时在 `reservations` 表写入 `HELD`；执行内核接到批准即可下单；订单走到终态，Portfolio 服务会自动 release/commit。

## 烟测顺序（1–2 小时全链路）

1. **ClickHouse**：已有连续 5s bar（聚合器+Writer+回补已在跑）。
2. **Portfolio**：启动，确认 `strategies` 表中每策略 `capital_limit` 已填。
3. **RiskService**：启动（本回帖代码）。
4. **执行内核**：启动（上次给的安全内核+CRS）；IB 可先用 Mock。
5. **策略引擎**：起两个策略（5s / 10s），把 `order.intent.v1` 打出来。
6. **观测**：

   * Risk：`risk_approved_total` 增长；PG `reservations` 有 `HELD`；
   * 执行：收到 `risk.decision.v2(APPROVED)` → 下单 → 回流状态/成交；
   * Portfolio：消费状态，`HELD` 被 `RELEASED/COMMITTED`，快照输出可用资金变化；
   * 追踪止损：从 CH 拉数据更新 OCO（可看 `crs_replace_ok_total`）。
