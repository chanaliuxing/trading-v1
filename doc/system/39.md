* 共享限流：**Redis Token Bucket**（place/modify/cancel 三路 + 全局/每策略/每标的粒度）

做完这三块，整套系统就能稳定“开机→对账→限流保护→进盘”。

# 1) Redis 共享令牌桶（跨进程限流）

> 目标：把 place / modify / cancel 的速率**集中限流**（全局/每策略/每标的），避免 IB pacing error 与改单风暴。
> 特性：单条 Lua 脚本**原子**扣桶；支持**容量**与**补给速率**；支持**多维 key**（global / strategy / symbol）。

```python
# pip install redis
import asyncio, time, os
from typing import Optional, Tuple, Dict
import redis.asyncio as redis

class RedisTokenBucket:
    """
    共享令牌桶：
      - capacity: 桶容量（最大突发）
      - refill_per_sec: 每秒补给令牌数
      - cost: 本次消耗令牌数（默认1）
    采用 Lua 脚本保证原子“补给+扣减”。
    """
    LUA = """
    local key = KEYS[1]
    local cap = tonumber(ARGV[1])
    local rate = tonumber(ARGV[2])
    local now = tonumber(ARGV[3])
    local cost = tonumber(ARGV[4])

    local data = redis.call('HMGET', key, 'tokens', 'ts')
    local tokens = tonumber(data[1]) or cap
    local ts = tonumber(data[2]) or now

    if now > ts then
      local delta = (now - ts) / 1000.0
      tokens = math.min(cap, tokens + delta * rate)
      ts = now
    end

    local ok = 0
    if tokens >= cost then
      tokens = tokens - cost
      ok = 1
    end

    redis.call('HMSET', key, 'tokens', tokens, 'ts', ts)
    -- 设置合理过期，避免冷桶永久占用
    redis.call('PEXPIRE', key, math.max(60000, math.floor(1000 * cap / math.max(rate, 0.001))))
    return {ok, tokens}
    """

    def __init__(self, r: redis.Redis, namespace: str, capacity: int, refill_per_sec: float):
        self.r = r
        self.ns = namespace
        self.cap = capacity
        self.rate = refill_per_sec
        self.sha = None

    async def _ensure_script(self):
        if not self.sha:
            self.sha = await self.r.script_load(RedisTokenBucket.LUA)

    def _key(self, *parts) -> str:
        return ":".join([self.ns, *[str(p) for p in parts]])

    async def try_acquire(self, key_suffix: Tuple, cost: int = 1) -> bool:
        """非阻塞获取；返回 True 表示成功扣桶。"""
        await self._ensure_script()
        k = self._key(*key_suffix)
        now = int(time.time() * 1000)
        try:
            ok, _tokens = await self.r.evalsha(self.sha, 1, k, self.cap, self.rate, now, cost)
            return ok == 1
        except redis.ResponseError:
            # 脚本失效重载
            self.sha = None
            return await self.try_acquire(key_suffix, cost)

    async def acquire(self, key_suffix: Tuple, cost: int = 1, timeout_s: float = 1.0) -> bool:
        """阻塞获取，最多等待 timeout_s；返回 True 表示成功。"""
        deadline = time.time() + timeout_s
        while time.time() < deadline:
            if await self.try_acquire(key_suffix, cost):
                return True
            await asyncio.sleep(0.01)
        return False
```

# 2) RTH / 交易日历组件（统一口径）

> 目标：统一“是否交易时段”的判断，避免策略/风控/聚合器口径不一致。
> 特性：**无外部依赖也能跑**；支持**节假日与半日**通过配置注入；支持盘前/盘后开关；提供“下一开/收盘时间”。

```python
import datetime as dt
import pytz
from typing import Optional, Set, Tuple, Dict

class MarketCalendar:
    """
    轻量美股日历（NYSE 口径）：
      - 默认 RTH 09:30–16:00 America/New_York
      - 支持外部注入 holiday_set（YYYY-MM-DD 字符串）
      - 支持 half_days（YYYY-MM-DD -> close_hm，例如 '13:00'）
      - allow_prepost 控制是否视盘前/盘后为可交易
    """
    def __init__(self,
                 tz: str = "America/New_York",
                 rth_open_hm: str = "09:30",
                 rth_close_hm: str = "16:00",
                 holiday_set: Optional[Set[str]] = None,
                 half_days: Optional[Dict[str, str]] = None):
        self.tz = pytz.timezone(tz)
        self.rth_open_hm = rth_open_hm
        self.rth_close_hm = half_days or {}  # 注意：这里暂存 half_days，便于 next_open_close 里替换
        self._holidays = set(holiday_set or set())
        self._half_days = dict(half_days or {})  # date -> close_hm

    def _local_date(self, t: Optional[dt.datetime] = None) -> dt.date:
        now = t or dt.datetime.now(tz=self.tz)
        return now.date()

    def _hm_to_time(self, hm: str) -> dt.time:
        hh, mm = hm.split(":")
        return dt.time(int(hh), int(mm), tzinfo=self.tz)

    def is_trading_day(self, d: Optional[dt.date] = None) -> bool:
        d = d or self._local_date()
        if d.weekday() >= 5:  # Sat/Sun
            return False
        if d.strftime("%Y-%m-%d") in self._holidays:
            return False
        return True

    def rth_window(self, d: Optional[dt.date] = None) -> Tuple[dt.datetime, dt.datetime]:
        d = d or self._local_date()
        open_hm = self.rth_open_hm
        close_hm = self._half_days.get(d.strftime("%Y-%m-%d"), "16:00")
        opens = self.tz.localize(dt.datetime.combine(d, self._hm_to_time(open_hm)))
        closes = self.tz.localize(dt.datetime.combine(d, self._hm_to_time(close_hm)))
        return opens, closes

    def is_rth(self, now: Optional[dt.datetime] = None, allow_prepost: bool = False) -> bool:
        now = now.astimezone(self.tz) if now else dt.datetime.now(self.tz)
        if not self.is_trading_day(now.date()):
            return False if not allow_prepost else True  # 休市日即使盘前/后也不建议交易；若 insist，可开关
        o, c = self.rth_window(now.date())
        if o <= now <= c:
            return True
        return allow_prepost  # 非 RTH，若允许盘前/盘后则视为可交易

    def next_open_close(self, now: Optional[dt.datetime] = None) -> Tuple[dt.datetime, dt.datetime]:
        now = now.astimezone(self.tz) if now else dt.datetime.now(self.tz)
        d = now.date()
        for i in range(8):  # 向前找最多一周
            dd = d if i == 0 else (d + dt.timedelta(days=i))
            if not self.is_trading_day(dd):
                continue
            o, c = self.rth_window(dd)
            if now <= c:
                # 如果还没开盘，返回今天的开/收；如果开着，返回今天的开/收
                return o, c
        # 理论不达此处
        return self.rth_window(d)
```

# 3) 冷启动对账（Reconciliation 2.0）

> 目标：重启后 5 秒内，使**内存状态**与**IB/TWS**完全一致，避免“幽灵单 / 幽灵仓位 / 保护缺失”。
> 步骤：
>
> 1. **采集**：IB open orders + IB portfolio；
> 2. **领养**：按 `orderRef` 直连；缺失 `orderRef` 走**模糊匹配**（symbol/side/价位/时间窗）；
> 3. **重建**：重建 `order_map`、`plan → stop/tp` 绑定；
> 4. **清理**：DB 有、IB 无 → 标记 `STALE_CLOSED`；
> 5. **补护**：有持仓无保护 → 立即挂保护（OCO），并写审计日志。

```python
import time, math, asyncio
from typing import Dict, List, Optional, Tuple

class Reconciler:
    def __init__(self, adapter, kernel, pg_pool, price_tol_bps:int=15, time_window_s:int=6*3600):
        """
        adapter: IBInsyncAdapter（提供 open_orders() / ib.portfolio()）
        kernel:  执行安全内核（提供 adopt_* 与 ensure_oco）
        pg_pool: asyncpg.Pool （读写 orders/fills 标志）
        """
        self.adapter = adapter
        self.kernel = kernel
        self.pg = pg_pool
        self.price_tol_bps = price_tol_bps
        self.time_window_s = time_window_s

    async def run_once(self):
        # 1) 拉取 IB 端现状
        ib_orders = await self.adapter.open_orders()      # List[PlacedOrder]
        ib_pos = await asyncio.to_thread(self.adapter.ib.portfolio)  # PortfolioItem 列表

        # 2) 查 DB 端未终态订单（用于清理与绑定）
        async with self.pg.acquire() as con:
            rows = await con.fetch("""
              SELECT order_id, symbol, side, limit_px, stop_px, tif, oca_group, idempotency_key, state, created_at
              FROM orders WHERE state NOT IN ('FILLED','CANCELLED','INACTIVE')
            """)
            db_open = {r["order_id"]: dict(r) for r in rows}

        # 3) 先按 orderRef 直连领养
        ref_index: Dict[str, List] = {}
        for o in ib_orders:
            if o.order_ref:
                ref_index.setdefault(o.order_ref, []).append(o)

        adopted: set = set()
        for plan in self.kernel.live_plans():  # 你内核里应当提供当前活跃计划列表
            ref_stp = f"{plan.idempotency_key}:STP"
            ref_tp  = f"{plan.idempotency_key}:TP"
            ref_ent = f"{plan.idempotency_key}:ENT"
            for ref in (ref_stp, ref_tp, ref_ent):
                for o in ref_index.get(ref, []):
                    await self.kernel.adopt_external_order(plan, o)  # 把 o 注入 plan.order_map / 绑定 stop/tp/ent
                    adopted.add(o.order_id)

        # 4) 模糊匹配（缺失 orderRef 的情况）
        def _close(px1, px2, bps):
            if not px1 or not px2: return False
            return abs(px1 - px2) / max(px1, px2) * 1e4 <= bps

        for o in ib_orders:
            if o.order_id in adopted:
                continue
            # 以 symbol+side 匹配最近的 plan
            cand = self.kernel.find_recent_plan(symbol=o.symbol, side=o.side, within_s=self.time_window_s)
            if not cand:
                # 找不到计划，视为“孤儿单”：强制撤销或挂入 Orphan 队列
                await self.kernel.cancel_or_flag_orphan(o)
                continue
            # 价格容忍
            if (o.order_type in ("LMT","STP_LMT") and not _close(o.limit_px, cand.last_intent_px, self.price_tol_bps)) \
               and (o.order_type in ("STP","STP_LMT") and not _close(o.stop_px, cand.last_stop_px, self.price_tol_bps)):
                await self.kernel.cancel_or_flag_orphan(o)
                continue
            # 领养成功
            await self.kernel.adopt_external_order(cand, o)
            adopted.add(o.order_id)

        # 5) 清理“幽灵单”（DB 有，IB 无）
        for oid, rec in list(db_open.items()):
            if oid in adopted or any(oo.order_id == oid for oo in ib_orders):
                continue
            # 在 IB 不存在，但 DB 仍标记未终态 → 标为 STALE_CLOSED
            async with self.pg.acquire() as con:
                await con.execute("UPDATE orders SET state='STALE_CLOSED', updated_at=now() WHERE order_id=$1", oid)

        # 6) 为有持仓无保护者补护（OCO）
        for it in ib_pos:
            sym = it.contract.symbol
            qty = float(it.position)
            if abs(qty) < 1e-9: 
                continue
            plan = self.kernel.find_or_create_plan_for_position(sym, qty, avg_px=float(it.avgCost))
            await self.kernel.ensure_oco(plan)  # 我们内核的 `_ensure_oco`，会立即挂保护

        # 7) 指标/审计
        self.kernel.metrics.reconcile_runs.inc()
        self.kernel.log.info("reconcile done: adopted=%d, ib_open=%d, db_open=%d, pos=%d",
                             len(adopted), len(ib_orders), len(db_open), len(list(ib_pos)))
```
