# B) 代码接线（插入 RiskService 审批前链路）

> 下面给出三段：① 轻量 Guard（如果你已粘过，可跳过）、② 将 MarketState → MarketSnap 的映射、③ 在 `approve()` 开头调用并统一返回。

## B.1 轻量 Guard（放在 `risk_service.py` 顶部或公共模块）

```python
# ---- 轻量 Reg-Lite ----
from dataclasses import dataclass
from typing import Optional, Dict, Any, List

@dataclass
class IntentLite:
    decision_id: str
    symbol: str
    side: str              # BUY/SELL/SELL_SHORT
    qty: float
    order_type: str        # MKT/LMT/STP/STP_LMT
    limit_px: Optional[float]
    stop_px: Optional[float]
    tif: str
    allow_prepost: bool = False

@dataclass
class MarketSnap:
    last_px: Optional[float]
    bid_px: Optional[float]
    ask_px: Optional[float]
    mid_px: Optional[float]
    luld_low: Optional[float]
    luld_high: Optional[float]
    ssr_active: bool
    halted: bool
    tick_size: float
    lot_size: float

@dataclass
class RuleHit:
    rule_id: str
    reason: str
    meta: Dict[str, Any]

class RegLiteGuard:
    def __init__(self, price_collar_pct: float = 5.0):
        self.collar_pct = price_collar_pct

    def _round_tick(self, px: float, tick: float) -> float:
        if tick <= 0 or px is None: return px
        return round(round(px / tick) * tick, 10)

    def check(self, it: IntentLite, m: MarketSnap) -> List[RuleHit]:
        hits: List[RuleHit] = []
        # qty
        if it.qty <= 0:
            hits.append(RuleHit("bad_qty", "non-positive qty", {"qty": it.qty}))
            return hits
        # 停牌
        if m.halted:
            hits.append(RuleHit("halted", "symbol halted/news pending", {}))
        # lot
        if m.lot_size and (it.qty % m.lot_size) > 1e-9:
            hits.append(RuleHit("lot_size", "qty not multiple of lot", {"lot": m.lot_size, "qty": it.qty}))
        # 需要价格的订单
        pxs = []
        if it.order_type in ("LMT","STP_LMT") and it.limit_px: pxs.append(("limit_px", it.limit_px))
        if it.order_type in ("STP","STP_LMT") and it.stop_px:  pxs.append(("stop_px", it.stop_px))
        for name, px in pxs:
            adj = self._round_tick(px, m.tick_size)
            if px is not None and adj is not None and abs(adj - px) > 1e-12:
                hits.append(RuleHit("tick_size", f"{name} violates tick", {"tick": m.tick_size, "px": px, "adj": adj}))
        # LULD
        def _in_luld(px: Optional[float]) -> bool:
            if px is None: return True
            if m.luld_low is None or m.luld_high is None: return True
            return m.luld_low - 1e-10 <= px <= m.luld_high + 1e-10
        for name, px in pxs:
            if not _in_luld(px):
                hits.append(RuleHit("luld_band", f"{name} outside LULD", {"low": m.luld_low, "high": m.luld_high, "px": px}))
        # SSR（卖空抬价）
        if it.side.upper() in ("SELL_SHORT","SHORT","SLD_SHORT") and m.ssr_active:
            bid = m.bid_px or (m.mid_px or m.last_px or 0.0)
            if it.order_type == "MKT":
                hits.append(RuleHit("ssr", "SSR active: short MKT forbidden", {"bid": bid}))
            elif it.order_type in ("LMT","STP_LMT") and it.limit_px is not None:
                need = bid + (m.tick_size or 0.01)
                if it.limit_px <= need - 1e-12:
                    hits.append(RuleHit("ssr", "limit must be above bid", {"bid": bid, "limit": it.limit_px, "need_min": need}))
        # 价格围栏（限价）
        ref = m.mid_px or m.last_px
        if ref and it.order_type in ("LMT","STP_LMT") and it.limit_px:
            dev = abs(it.limit_px - ref) / ref * 100.0
            if dev > self.collar_pct:
                hits.append(RuleHit("price_collar", "limit too far from ref", {"ref": ref, "limit": it.limit_px, "dev_pct": dev}))
        # 市价单 NBBO
        if it.order_type == "MKT" and not (m.bid_px and m.ask_px):
            hits.append(RuleHit("no_nbbo", "MKT without NBBO", {}))
        return hits
```

## B.2 在 RiskService 初始化里注入（构造函数处）

> 你已有 `MarketStateProvider`。在 `RiskService.__init__` 里加：

```python
class RiskService:
    def __init__(self, cfg, market_state_provider, metrics, logger, ...):
        self.cfg = cfg
        self.ms  = market_state_provider
        self.log = logger
        self.m   = metrics

        rl_cfg = cfg["risk"]["reglite"]
        self.reglite_enabled = bool(rl_cfg.get("enabled", True))
        self.reglite_enforce = rl_cfg.get("enforce", {"luld":True,"ssr":True,"halt":True})
        self.reglite_guard   = RegLiteGuard(price_collar_pct=float(rl_cfg.get("price_collar_pct", 3.0)))
        self.reglite_require_nbbo = bool(rl_cfg.get("require_nbbo", True))
```

## B.3 在“批准前”调用（`approve()` 开头加入 12 行）

> 位置：**所有资金/风控规则之前**。把策略的 Envelope/ExecPlan 映射成轻量 Intent，取行情快照后调用 `check()`。

```python
    async def approve(self, env, plan, ttl_ms: int):
        """
        env: DecisionEnvelope
        plan: ExecPlan（翻译层输出）
        """
        # --- Reg-Lite 前置闸 ---
        if self.reglite_enabled:
            m = await self.ms.get(env.symbol)
            snap = MarketSnap(
                last_px=m.last_px, bid_px=m.bid_px, ask_px=m.ask_px,
                mid_px=m.mid_px, luld_low=m.luld_low, luld_high=m.luld_high,
                ssr_active=bool(getattr(m, "ssr_active", False)),
                halted=bool(getattr(m, "halted", False)),
                tick_size=float(m.tick_size or 0.01),
                lot_size=float(m.lot_size or 1.0)
            )
            it = IntentLite(
                decision_id=env.decision_id, symbol=env.symbol, side=plan.side,
                qty=plan.qty, order_type=plan.order_type, limit_px=plan.limit_px,
                stop_px=plan.stop_px, tif=plan.tif
            )
            hits = self.reglite_guard.check(it, snap)

            hard_rules = {"tick_size","lot_size","price_collar","no_nbbo"}
            # 对权威状态执行硬拒；无权威→只 warn（由 MarketStateProvider/Bridge 提供）
            if self.reglite_enforce.get("luld", True) and snap.luld_low is not None and snap.luld_high is not None:
                hard_rules.add("luld_band")
            if self.reglite_enforce.get("ssr", True) and snap.bid_px is not None:
                hard_rules.add("ssr")
            if self.reglite_enforce.get("halt", True):
                hard_rules.add("halted")

            blocks = [h for h in hits if h.rule_id in hard_rules]
            warns  = [h for h in hits if h.rule_id not in hard_rules]

            for h in warns:
                self.m.reglite_warn.labels(rule=h.rule_id).inc()
                self.log.warning(f"[RegLite:WARN] {env.decision_id} {env.symbol} {h.rule_id} {h.reason} {h.meta}")

            if blocks:
                for h in blocks:
                    self.m.reglite_reject.labels(rule=h.rule_id).inc()
                reason = {"hits":[{"rule":h.rule_id,"reason":h.reason,**h.meta} for h in blocks]}
                # 统一拒绝出口（沿用你原 approve/reject 流程）
                return False, {"kind":"reglite_block", **reason}
        # --- 后续：资金池/名义/回撤/行业集中度等原有规则 ---
        ...
```

> 说明：
>
> * **无权威状态**（例如 LULD 尚未接通）→ `luld_band/ssr/halted` 会被计为 WARN，但不会阻塞。
> * `tick/lot/price_collar/no_nbbo` **始终硬拒**（不会依赖外部状态源）。
> * 你的后续风控（资金/集中度等）保持原逻辑。

```python
from prometheus_client import Counter

reglite_reject = Counter(

    "reglite_warn_total", "Reg-Lite soft warnings", ["rule"]
)
```

**建议加两条告警（Grafana/PromQL）：**

* `increase(reglite_reject_total[5m]) > 0` 且 `by(rule)` 展示 → 快速定位触发类型
* `reglite_warn_total / (reglite_warn_total + reglite_reject_total)` 突增 → 可能状态源不稳定

# D) 自测剧本（本地/影子环境 10 分钟跑完）

> 直接在控制台或你的集成测试里，构造不同场景的 `MarketSnap` + `IntentLite`，看返回。

1. **无 NBBO 的市价单** → 命中 `no_nbbo`（硬拒）
2. **限价越 tick** → 命中 `tick_size`（硬拒）
3. **越 LULD**（确保有 luld_low/high）→ 命中 `luld_band`（硬拒）
4. **SSR 卖空**（`ssr_active=True` 且 `limit_px <= bid+tick`）→ 命中 `ssr`（硬拒）
5. **价格围栏过远**（`limit_px` 偏离 mid > collar%）→ 命中 `price_collar`（硬拒）
6. **停牌**（`halted=True`）→ 命中 `halted`（硬拒）
7. **只 WARN**：断开权威 LULD/SSR 时再测 3/4，应该仅计入 `reglite_warn_total` 不阻塞

> 验收：拒单有返回 `{kind:"reglite_block", hits:[...]}`；指标相应累加；日志含 `decision_id/symbol/rule`。
