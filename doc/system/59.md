# 1) SDK 总览（面向策略开发者）

### 1.1 生命周期接口（StrategyAlgorithm）

```python
from dataclasses import dataclass
from typing import Dict, Any, List, Optional

@dataclass
class StrategyParams:
    cadence_s: int = 5              # 策略独立节奏（秒）
    ttl_ms: int = 2000              # 本策略默认意图TTL（可被上层硬上限截断）
    qscore_threshold: int = 70      # 特征质量门槛
    max_notional_usd: float = 100000  # 单策略资金池硬上限由 Risk/Reserve 决定，这里是策略自限
    per_symbol_max_usd: float = 50000
    risk_per_trade_usd: float = 2000 # 风险预算（ATR sizing可引用）
    oco: Dict[str, Any] = None      # {'tp_px':None,'sl_px':None,'trail_atr_mult':1.5,'time_stop_s':600}
    extras: Dict[str, Any] = None   # 策略自定义参数

class StrategyAlgorithm:
    def __init__(self, ctx, params: StrategyParams):
        self.ctx = ctx                  # StrategyContext（见下）
        self.params = params
        self.state: Dict[str,Any] = {}  # 可持久化小状态（热重启恢复）
        self.log = ctx.log
        self.m = ctx.metrics

    async def Initialize(self):
        """启动时只调用一次。订阅数据/加载历史/设置节奏等。"""
        pass

    async def OnFeatures(self, symbol: str, f: Dict[str, Any]):
        """每收到一条 features（如5s）回调一次——策略核心逻辑写这里。"""
        pass

    async def OnOrderUpdate(self, upd: Dict[str, Any]):
        """订单状态更新（Submitted/PartiallyFilled/Filled/Cancelled/Rejected）。"""
        pass

    async def OnPositionUpdate(self, pos: Dict[str, Any]):
        """持仓变动通知。"""
        pass

    async def OnShutdown(self):
        """进程退出/卸载策略时清理资源。"""
        pass
```

### 1.2 上下文 API（StrategyContext）

```python
class StrategyContext:
    """
    由引擎注入，每个策略实例独立。屏蔽底层执行/风控/数据来源。
    """
    def __init__(self, engine, strategy_id, logger, metrics):
        self.engine = engine
        self.strategy_id = strategy_id
        self.log = logger
        self.metrics = metrics

    # 数据订阅 / 历史回读
    async def SubscribeFeatures(self, symbols: List[str], bar_sec: int = 5):
        """订阅 md.features.{bar_sec}s.v1，内部做RingBuffer与回补填充。"""
        return await self.engine.subscribe_features(self.strategy_id, symbols, bar_sec)

    async def History(self, symbol: str, bar_sec: int, start_ts, end_ts, fields: List[str]):
        """从 ClickHouse 回读features_* 历史。"""
        return await self.engine.history_features(symbol, bar_sec, start_ts, end_ts, fields)

    # 策略节奏
    async def SetCadence(self, seconds: int):
        """调度器为该策略-符号设置独立节奏（覆盖全局默认）。"""
        return await self.engine.set_cadence(self.strategy_id, seconds)

    # 组合与状态
    async def Portfolio(self) -> Dict[str, Any]:
        """查询组合：现金、总权益、每标的头寸（pos_qty/avg_px/UPL/RPL）。"""
        return await self.engine.portfolio_view(self.strategy_id)

    async def Position(self, symbol: str) -> Dict[str, Any]:
        return await self.engine.position_view(self.strategy_id, symbol)

    # 发布意图（交给：Validator→Arbiter→Risk(Reserve)→Exec）
    async def SubmitIntent(self, env: Dict[str, Any]) -> None:
        """env = DecisionEnvelope（见下）。"""
        return await self.engine.submit_intent(self.strategy_id, env)

    # 工具
    def NowMs(self) -> int: ...
    def Param(self, key: str, default=None): ...
```

### 1.3 DecisionEnvelope（高层“意图”）

> 由策略构造，交给引擎管线（**先仲裁，再风控预留，再执行**）。
> **必须字段**：`decision_id, strategy_id, symbol, ts_bar_end_us, target_qty, order_style, tif, freshness.ttl_ms, extras.ref_px/edge_bp`

```python
from dataclasses import dataclass, field

@dataclass
class Freshness:
    ttl_ms: int

@dataclass
class OrderStyle:
    kind: str                  # 'MKT'|'LMT'|'PEG'...（Translator/Reg-Lite会做修正）
    limit_px: float = None
    stop_px: float = None

@dataclass
class DecisionEnvelope:
    decision_id: str
    strategy_id: str
    symbol: str
    ts_bar_end_us: int
    target_qty: float          # 目标净仓（+多/-空），引擎负责净变更与反手编舞
    tif: str                   # 'DAY'|'IOC'...
    order_style: OrderStyle
    freshness: Freshness
    extras: dict = field(default_factory=dict)   # {'ref_px':..., 'edge_bp':..., 'oco_cfg':{...}}
```

# 2) 策略模板：5s 均值回归（含 ATR sizing / 品质门控 / TTL）

```python
import math, uuid
class MeanRevert5s(StrategyAlgorithm):
    async def Initialize(self):
        symbols = self.params.extras.get("symbols", [])
        await self.ctx.SubscribeFeatures(symbols, bar_sec=5)
        await self.ctx.SetCadence(self.params.cadence_s)
        self.log.info(f"[{self.ctx.strategy_id}] init with {symbols}, cadence={self.params.cadence_s}s")

    async def OnFeatures(self, symbol: str, f: dict):
        # 质量门控
        q = f.get("qscore", 0)
        if q < self.params.qscore_threshold: 
            return

        close = f.get("close"); z20 = f.get("zscore_20"); atr = f.get("atr_14")
        mid  = f.get("mid_last") or close
        if not all([close is not None, z20 is not None, atr is not None, mid is not None]):
            return  # 数据不足，跳过

        # 简单规则：zscore<-1 开多；zscore>+1 开空；中性区间不动
        band = float(self.params.extras.get("z_entry", 1.0))
        tgt = 0
        if z20 <= -band:
            tgt = +1
        elif z20 >= +band:
            tgt = -1
        else:
            # 回归区间内回归到0：轻仓回补/减仓（可选，示例直接保持）
            tgt = 0

        # 头寸 sizing：按照 ATR 风险预算
        # qty = risk_per_trade_usd / (atr * k_atr)；可乘以杠杆/调仓系数
        k_atr = float(self.params.extras.get("atr_mult", 1.5))
        dollar_per_share = max(atr * k_atr, 0.01)
        risk_usd = min(self.params.risk_per_trade_usd, self.params.per_symbol_max_usd)
        qty_abs = max(0, int(risk_usd / dollar_per_share))
        target_qty = tgt * qty_abs

        # 读取当前仓位，避免重复提交同一方向
        pos = await self.ctx.Position(symbol)
        cur_qty = int(pos.get("pos_qty", 0) or 0)
        if target_qty == cur_qty:
            return

        # OCO 配置：灾难止损 + ATR 跟踪
        oco_cfg = {
            "tp_px": None, 
            "sl_px": None, 
            "trail_atr_mult": float(self.params.oco.get("trail_atr_mult", 1.5)) if self.params.oco else 1.5,
            "time_stop_s": int(self.params.oco.get("time_stop_s", 900)) if self.params.oco else 900
        }

        # 构造 Envelope
        decision_id = str(uuid.uuid4())
        env = DecisionEnvelope(
            decision_id=decision_id,
            strategy_id=self.ctx.strategy_id,
            symbol=symbol,
            ts_bar_end_us=f["ts_end"],             # 来自 features 的 bar 结束时间
            target_qty=float(target_qty),
            tif="DAY",
            order_style=OrderStyle(
                kind="LMT",
                limit_px=float(mid)                # 以 mid 作为参考，Reg-Lite/矩阵会修正
            ),
            freshness=Freshness(ttl_ms=self.params.ttl_ms),
            extras={
                "ref_px": float(mid),
                "edge_bp": float(abs(z20)*10),     # 伪装一个边际得分给仲裁器
                "oco_cfg": oco_cfg
            }
        )

        await self.ctx.SubmitIntent(env.__dict__)  # SDK 内部会做序列化/验证
```

> 说明
>
> * **跳过逻辑**：质量、不足、仓位未变 → 都不下单；
> * **自适应秒级节奏**：`Initialize` 设置 cadence=5/10/…；
> * **仓位表达**：用 **目标净仓**，由引擎负责净变更/反手编舞（第 4 步）；
> * **TTL** 全链路生效（第 6 步）；
> * **价格修正/错误重试**：交给 **Reg-Lite + Error Matrix**（第 1–2 步）。

_REGISTRY = {}

def register(name):
    def deco(cls):
        _REGISTRY[name] = cls
        return cls
    return deco

**加载器**

```python
import importlib, hashlib, json

class StrategyLoader:
    def __init__(self, logger):
        self.log = logger

    def _param_hash(self, params: dict) -> str:
        s = json.dumps(params, sort_keys=True, ensure_ascii=False)
        return hashlib.sha1(s.encode('utf-8')).hexdigest()[:10]

    def load(self, name: str, params: dict, ctx) -> StrategyAlgorithm:
        cls = _REGISTRY[name]
        phash = self._param_hash(params)
        ctx.log.info(f"[loader] spawn {name} param_hash={phash}")
        return cls(ctx, StrategyParams(**params))
```

> **热加载**：变更参数→新实例在无缝过渡窗口内接管（暂停旧实例的 SubmitIntent，等其未决回调清空后卸载）。

# 5) 回测/影子/实盘“三位一体”Harness

### 5.1 统一驱动（同一策略不改代码）

```python
class StrategyHarness:
    def __init__(self, mode: str, engine, ch, logger, metrics):
        self.mode = mode              # 'backtest'|'shadow'|'live'
        self.engine = engine
        self.ch = ch
        self.log = logger
        self.m = metrics

    async def run_backtest(self, strat: StrategyAlgorithm, symbol: str, start, end):
        # 从 CH 扫 features_5s（或CSV）→ 逐行馈送给 OnFeatures
        rows = await self.ch.query("""
            SELECT * FROM features_5s 
            WHERE symbol=%(s)s AND ts_end BETWEEN %(st)s AND %(ed)s 
            ORDER BY ts_end
        """, {"s":symbol, "st":start, "ed":end})
        for f in rows:
            await strat.OnFeatures(symbol, f)
            # 将 SubmitIntent 送入“影子撮合器”（或 backtest fill 模型）
        return self._evaluate()

    def _evaluate(self):
        # 计算收益、回撤、胜率、持仓/换手等
        return {"pnl":..., "sharpe":..., "maxdd":..., "hit":...}
```

### 5.2 影子撮合（Paper/Shadow）

* 价差/滑点模型：`next bar VWAP ± x * spread`
* LMT 达成条件：若 LMT≤low≤high 包含则成交（更严格可用逐笔回放）。
* OCO：按我们第 4 步逻辑模拟子单触发。

# 7) 关键不变量（防踩线）

* **策略不直接触达券商/数据库**；只能用 `SubmitIntent()`。
* **不得绕过**仲裁/预留；资金与每标的上限由风控链路执行。
* **必须**为每个意图带上 `ts_bar_end_us` 与 `freshness.ttl_ms`；
* 避免使用**未来数据**（只读 features 当前行 + 历史 lag）；
* 不做长阻塞（每次决策≤20ms），重任务放异步任务/CH批处理；
* **日志轻量**：info 级别≤1行/决策，debug 才打印细节（避免 I/O 限速）。

---

# 8) 与前 1–8 步的“即插即用”对齐

* **Freshness/TTL（第 6 步）**：SDK 在 SubmitIntent 前补齐/校验 TTL；超时各层自动作废。
* **仲裁（第 5 步）**：Envelope 带 `edge_bp/ref_px`，仲裁器按优先级/净边际择优与裁剪。
* **资金预留（第 3 步）**：RiskService 依据 `target_qty*ref_px` 预留；终态释放。
* **Error Matrix（第 2 步）**：策略无需关心 pacing/围栏/LULD/SSR 修复；
* **OCO（第 4 步）**：由 `extras.oco_cfg` 指定，首次成交即挂，部分成交按比例调，反手原子切换；
* **数据层（第 7–8 步）**：策略只订阅 `md.features.{5s|10s}.v1`，质量不达标自动跳过。

# 9) 迁移指引（把老策略搬进新模板）

1. **把信号条件**改写成 `OnFeatures()` 基于 `features_*` 字段（如 ATR/RSI/ZScore 已有）。
2. **把下单逻辑**改为设置 `target_qty`（净仓），用 `extras.oco_cfg` 描述保护，而不是自己下 OCO。
3. **把参数**落到 `StrategyParams`/`extras`，保证 `param_hash` 可复现。
4. **把风控/资金**依赖删掉（统一由链路执行）。
5. **回测**用 `StrategyHarness.run_backtest()`，校对与老系统（如 1m 模型）在样本期的一致性。
