好，我来把**IB 适配层（ib_insync 版）**一次性补齐，做到“能用就跑”的程度：

* 提供 `IBInsyncAdapter(IBAdapter)`，实现我们内核要求的 6 个方法：`place_limit/stop/stop_limit/market/modify/cancel/open_orders`
* 自带**可靠连接/重连**、**合约构建**、**OCO 字段映射**（`orderRef / ocaGroup / ocaType`）
* 附送一个**事件桥（Polling 版）**，将 IB 的订单状态/成交**轮询**发布到 Kafka：`exec.order.status.v2` 与 `exec.fill.v2`（签名稳定，避免 ib_insync 事件 API 差异带来的坑）

> 说明：ib_insync 的事件回调在不同版本上参数签名略有差异。为保证稳定，我给出“**轮询桥**”——用 `openTrades()` 和 `reqExecutionsAsync()` 定期拉取，**100% 可跑**。后续你可切换成事件回调型以降低延迟。

# 一、IB 适配器：IBInsyncAdapter（直接粘贴）

```python
# 依赖：pip install ib-insync aiokafka
import asyncio, os, time, contextlib
from dataclasses import dataclass
from typing import Optional, List, Dict

from ib_insync import IB, Stock, Order, util, Contract  # type: ignore

# ---- 复用你内核里定义的 PlacedOrder/IBAdapter ----
@dataclass
class PlacedOrder:
    order_id: int
    order_ref: str
    symbol: str
    side: str         # BUY / SELL
    order_type: str   # LMT / STP / STP_LMT / MKT
    qty: float
    limit_px: Optional[float]=None
    stop_px: Optional[float]=None
    tif: str="DAY"
    oco_group: Optional[str]=None
    status: str="Submitted"   # PreSubmitted / Submitted / Filled / Cancelled / Inactive ...
    ts_us: int = int(time.time()*1e6)

class IBAdapter:
    async def place_limit(self, symbol:str, side:str, qty:float, px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def place_stop(self, symbol:str, side:str, qty:float, stop_px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def place_stop_limit(self, symbol:str, side:str, qty:float, stop_px:float, limit_px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def place_market(self, symbol:str, side:str, qty:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def modify(self, order_id:int, **fields) -> bool: ...
    async def cancel(self, order_id:int) -> bool: ...
    async def open_orders(self) -> List[PlacedOrder]: ...

# ---------------------------

def _now_us() -> int: return int(time.time()*1e6)

class IBInsyncAdapter(IBAdapter):
    """
    - 线程模型：ib_insync 本身就是 asyncio 友好；这里所有方法都是 async。
    - OCO：设置 Order.ocaGroup = <group>，Order.ocaType = 1（OCA_CANCEL_WITH_BLOCK，典型 OCO）。
    - orderRef：贯穿我们系统的 idempotency_key，便于冷启动领养。
    """
    def __init__(self,
                 host: str = None,
                 port: int = None,
                 client_id: int = None,
                 account: Optional[str] = None,
                 default_exchange: str = "SMART",
                 default_currency: str = "USD",
                 primary_exchange_hint: Optional[str] = None,
                 outside_rth: bool = False):

        self.host = host or os.getenv("IB_HOST","127.0.0.1")
        self.port = port or int(os.getenv("IB_PORT","7497"))
        self.client_id = client_id or int(os.getenv("IB_CLIENT_ID","101"))
        self.account = account or os.getenv("IB_ACCOUNT","")
        self.exch = default_exchange
        self.ccy = default_currency
        self.primary = primary_exchange_hint  # e.g. "NASDAQ" / "NYSE"（可为空）
        self.outside_rth = outside_rth or (os.getenv("IB_OUTSIDE_RTH","false").lower()=="true")

        self.ib = IB()
        self._lock = asyncio.Lock()     # 保护 trades 映射
        self._trades: Dict[int, 'Trade'] = {}  # orderId -> Trade

    async def start(self):
        await self._connect()
        # 定期保活/重连
        asyncio.create_task(self._reconnector())

    async def stop(self):
        with contextlib.suppress(Exception):
            self.ib.disconnect()

    async def _connect(self):
        if self.ib.isConnected():
            return
        await self.ib.connectAsync(self.host, self.port, clientId=self.client_id, timeout=5.0)
        # 账户切换（如需要）
        if self.account:
            try:
                await self.ib.reqAccountUpdatesAsync(True, self.account)
            except Exception:
                pass

    async def _reconnector(self):
        while True:
            try:
                if not self.ib.isConnected():
                    await self._connect()
            except Exception:
                pass
            await asyncio.sleep(2.0)

    # --------- 合约构造 ----------
    async def _contract(self, symbol:str) -> Contract:
        # 对美股：SMART 路由 + currency USD；必要时可补充 primaryExchange
        c = Stock(symbol, self.exch, self.ccy)
        if self.primary:
            c.primaryExchange = self.primary
        # 确认合约（resolve conId）
        await self._connect()
        cc = await self.ib.qualifyContractsAsync(c)
        return cc[0] if cc else c

    # --------- 下单公共路径 ----------
    async def _place(self, c:Contract, o:Order, order_ref:str, oco_group:Optional[str]) -> PlacedOrder:
        await self._connect()
        o.account = self.account or o.account
        o.outsideRth = bool(self.outside_rth)
        o.orderRef = order_ref[:45]  # IB 对 orderRef 有长度限制；保守截断
        if oco_group:
            o.ocaGroup = oco_group[:30]
            o.ocaType = 1  # Cancel remaining（典型 OCO）

        trade = self.ib.placeOrder(c, o)
        # 等到 IB 接受（收到 PreSubmitted/Submitted）
        try:
            await trade.waitUntilAccepted(timeout=3)
        except Exception:
            # 可能已经被接受，只是没及时回；忽略
            pass

        async with self._lock:
            self._trades[trade.order.orderId] = trade

        po = PlacedOrder(
            order_id = trade.order.orderId,
            order_ref = o.orderRef or order_ref,
            symbol = c.symbol,
            side = o.action.upper(),
            order_type = o.orderType.replace(" ", "_"),
            qty = float(o.totalQuantity),
            limit_px = float(getattr(o, "lmtPrice", 0) or 0) if hasattr(o,"lmtPrice") else None,
            stop_px  = float(getattr(o, "auxPrice", 0) or 0) if hasattr(o,"auxPrice") else None,
            tif = o.tif,
            oco_group = o.ocaGroup or None,
            status = trade.orderStatus.status or "Submitted",
            ts_us = _now_us()
        )
        return po

    # --------- 6 个方法实现 ----------
    async def place_limit(self, symbol:str, side:str, qty:float, px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder:
        c = await self._contract(symbol)
        o = Order()
        o.orderType = "LMT"; o.lmtPrice = float(px)
        o.action = side.upper(); o.totalQuantity = float(qty); o.tif = tif
        return await self._place(c, o, order_ref, oco_group)

    async def place_stop(self, symbol:str, side:str, qty:float, stop_px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder:
        c = await self._contract(symbol)
        o = Order()
        o.orderType = "STP"; o.auxPrice = float(stop_px)
        o.action = side.upper(); o.totalQuantity = float(qty); o.tif = tif
        return await self._place(c, o, order_ref, oco_group)

    async def place_stop_limit(self, symbol:str, side:str, qty:float, stop_px:float, limit_px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder:
        c = await self._contract(symbol)
        o = Order()
        o.orderType = "STP LMT"; o.auxPrice = float(stop_px); o.lmtPrice = float(limit_px)
        o.action = side.upper(); o.totalQuantity = float(qty); o.tif = tif
        return await self._place(c, o, order_ref, oco_group)

    async def place_market(self, symbol:str, side:str, qty:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder:
        c = await self._contract(symbol)
        o = Order()
        o.orderType = "MKT"
        o.action = side.upper(); o.totalQuantity = float(qty); o.tif = tif
        return await self._place(c, o, order_ref, oco_group)

    async def modify(self, order_id:int, **fields) -> bool:
        """修改现有订单：更新 Order 字段，然后重新 place（IB 会当作修改）"""
        await self._connect()
        async with self._lock:
            t = self._trades.get(order_id)
        if not t:
            # 尝试从 openTrades 拉
            for tr in self.ib.openTrades():
                if tr.order.orderId == order_id:
                    t = tr; break
        if not t: 
            return False

        o = t.order
        # 支持的字段：qty/limit_px/stop_px/tif
        if "qty" in fields:
            o.totalQuantity = float(fields["qty"])
        if "limit_px" in fields:
            o.lmtPrice = float(fields["limit_px"])
            o.orderType = "LMT" if o.orderType in ("LMT","STP LMT") and not getattr(o, "auxPrice", 0) else o.orderType
        if "stop_px" in fields:
            o.auxPrice = float(fields["stop_px"])
            if o.orderType == "LMT":
                o.orderType = "STP LMT"
        if "tif" in fields:
            o.tif = str(fields["tif"])

        # 关键：重新 place 即为修改
        self.ib.placeOrder(t.contract, o)
        try:
            await t.waitUntilAccepted(timeout=2)
        except Exception:
            pass
        return True

    async def cancel(self, order_id:int) -> bool:
        await self._connect()
        async with self._lock:
            t = self._trades.get(order_id)
        if not t:
            for tr in self.ib.openTrades():
                if tr.order.orderId == order_id:
                    t = tr; break
        if not t:
            # 可能已被取消/成交
            return True
        self.ib.cancelOrder(t.order)
        try:
            await t.waitUntilCancelled(timeout=3)
        except Exception:
            pass
        return True

    async def open_orders(self) -> List[PlacedOrder]:
        await self._connect()
        out: List[PlacedOrder] = []
        trades = list(self.ib.openTrades())
        for t in trades:
            o = t.order; c = t.contract; s = t.orderStatus
            out.append(PlacedOrder(
                order_id=o.orderId, order_ref=o.orderRef or "",
                symbol=c.symbol, side=o.action.upper(),
                order_type=o.orderType.replace(" ","_"),
                qty=float(o.totalQuantity),
                limit_px=float(getattr(o,"lmtPrice",0) or 0) or None,
                stop_px=float(getattr(o,"auxPrice",0) or 0) or None,
                tif=o.tif or "DAY",
                oco_group=o.ocaGroup or None,
                status=s.status or "Submitted",
                ts_us=_now_us()
            ))
        return out
```

**要点说明**

* **OCO**：我们把 `ocaGroup` 设为 `OCO:<decision_id>`，`ocaType=1`（一腿成交，另一腿自动取消）。
* **修改**：直接对 `Trade.order` 改字段再 `placeOrder(contract, order)`，IB 会当作“改单”。
* **orderRef**：截断到 ~45 字符（IB 有限制），但仍保留足够的幂等信息（我们的 `idempotency_key` 通常短）。
* **重连**：后台协程每 2s 检查一次，断线即 `connectAsync`。

# 二、IB 事件桥（Polling 版）：把状态/成交推到 Kafka

> 这个组件用于给**执行安全内核**和**组合账本**提供 `exec.order.status.v2` 与 `exec.fill.v2` 事件。
> 它不依赖 ib_insync 的事件回调，而是**稳妥轮询**（openTrades + reqExecutionsAsync），避免签名兼容问题。

```python
import json, time, asyncio, os, contextlib
from typing import Dict, Tuple, Set
from aiokafka import AIOKafkaProducer
from ib_insync import IB, util, ExecutionFilter  # type: ignore

def _now_us() -> int: return int(time.time()*1e6)

class IBEventBridge:
    def __init__(self, ib: IB,
                 kafka_bootstrap:str = None,
                 topic_status:str = None,
                 topic_fill:str = None,
                 poll_open_s:float = 0.5,
                 poll_exec_s:float = 1.0):
        self.ib = ib
        self.bootstrap = kafka_bootstrap or os.getenv("KAFKA_BOOTSTRAP","localhost:9092")
        self.t_status = topic_status or os.getenv("TOPIC_STATUS","exec.order.status.v2")
        self.t_fill   = topic_fill   or os.getenv("TOPIC_FILL","exec.fill.v2")
        self.poll_open_s = poll_open_s
        self.poll_exec_s = poll_exec_s
        self.producer: AIOKafkaProducer = None
        # 去重缓存
        self._seen_exec_ids: Set[str] = set()
        self._last_exec_time = None  # IB 过滤器时间戳（“YYYYMMDD HH:MM:SS”）

    async def start(self):
        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.bootstrap, acks="all", linger_ms=5, compression_type="lz4",
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda k: k.encode("utf-8") if k else None
        )
        await self.producer.start()
        asyncio.create_task(self._loop_open_orders())
        asyncio.create_task(self._loop_executions())

    async def stop(self):
        if self.producer: with contextlib.suppress(Exception): await self.producer.stop()

    # 轮询 openTrades -> 发布状态
    async def _loop_open_orders(self):
        while True:
            try:
                if self.ib.isConnected():
                    for t in list(self.ib.openTrades()):
                        o = t.order; s = t.orderStatus; c = t.contract
                        msg = {
                          "schema_v":2,
                          "metadata":{"ts_us":_now_us(),"producer":"ib-bridge","idempotency_key":o.orderRef},
                          "data":{
                             "order_id": o.orderId,
                             "status": (s.status or "Submitted"),
                             "symbol": c.symbol,
                             "side": (o.action or "").upper(),
                             "qty": float(o.totalQuantity),
                             "tif": o.tif or "DAY",
                             "limit_px": float(getattr(o,"lmtPrice",0) or 0) or None,
                             "stop_px": float(getattr(o,"auxPrice",0) or 0) or None,
                             "oca_group": o.ocaGroup or None,
                             "strategy_id": self._sid_from_ref(o.orderRef)
                          }
                        }
                        await self.producer.send_and_wait(self.t_status, key=f"{c.symbol}:{o.orderId}", value=msg)
            except Exception:
                pass
            await asyncio.sleep(self.poll_open_s)

    # 轮询 Executions -> 发布成交（去重）
    async def _loop_executions(self):
        while True:
            try:
                if self.ib.isConnected():
                    flt = ExecutionFilter()
                    if self._last_exec_time:
                        flt.time = self._last_exec_time  # 'YYYYMMDD HH:MM:SS'
                    # reqExecutionsAsync 返回 ExecutionDetails 列表
                    details = await self.ib.reqExecutionsAsync(flt)
                    if details:
                        # 下一轮从最后一条的时间继续
                        self._last_exec_time = details[-1].exec.time
                    for d in details:
                        key = f"{d.exec.permId}:{d.exec.time}:{d.exec.cumQty}"
                        if key in self._seen_exec_ids:
                            continue
                        self._seen_exec_ids.add(key)
                        c = d.contract; e = d.exec
                        side = ("BUY" if (e.side or "").upper()=="BOT" else "SELL")
                        msg = {
                          "schema_v":2,
                          "metadata":{"ts_us":_now_us(),"producer":"ib-bridge","idempotency_key":None},
                          "data":{
                             "order_id": int(e.orderId),
                             "symbol": c.symbol,
                             "side": side,
                             "qty": float(e.shares),
                             "price": float(e.price),
                             "ts_us": _now_us(),  # 也可解析 e.time
                             "strategy_id": self._sid_from_ref(self._orderref_of(e.orderId))
                          }
                        }
                        await self.producer.send_and_wait(self.t_fill, key=f"{c.symbol}:{e.orderId}", value=msg)
            except Exception:
                pass
            await asyncio.sleep(self.poll_exec_s)

    # 从 orderRef 推断策略ID（我们约定 idempotency_key 形如 'ord:req:<strategy_id>:<symbol>:<uuid>'）
    def _sid_from_ref(self, order_ref: Optional[str]) -> Optional[str]:
        if not order_ref: return None
        try:
            parts = order_ref.split(":")
            # ord:req:SID:SYM:UUID
            if len(parts) >= 5 and parts[0]=="ord" and parts[1]=="req":
                return parts[2]
        except Exception:
            pass
        return None

    # 通过 orderId 找 orderRef（从 openTrades 映射一遍）
    def _orderref_of(self, order_id:int) -> Optional[str]:
        try:
            for t in self.ib.trades():
                if t.order.orderId == order_id:
                    return t.order.orderRef
        except Exception:
            return None
        return None
```

**使用方式**

* 在你的进程里**共享同一个 `IB()` 实例**：

  ```python
  adapter = IBInsyncAdapter(...); await adapter.start()
  bridge  = IBEventBridge(adapter.ib); await bridge.start()
  # 把 adapter 注入到执行安全内核 SafetyKernel(adapter)
  ```
* 现在，“安全内核”通过 `adapter` 下单/改单/撤单，**事件桥**把状态/成交推到 Kafka，**组合账本**和**安全内核**都能消费。

# 三、字段映射与风控/OCO 细节确认（避免踩坑）

1. **OCO 行为**

   * 我们设置 `order.ocaGroup = "OCO:<decision_id>"`，`ocaType = 1`（一腿成交取消另一腿）。
   * **注意**：在 IB 端，如果两腿一起触发，有可能出现**同时成交**的 race。我们已经在“执行安全内核 + CRS”里做了**巡检强撤**，可防残留。

2. **止损类型**

   * `STP`（市价止损）与 `STP LMT`（限价止损）我们都支持；由 `USE_STOP_LIMIT` 控制。
   * IB 的字段：`auxPrice=stop_px`，`lmtPrice=限价`。

3. **orderRef 长度**

   * IB 对 `orderRef` 有长度限制（官方未公开固定上限，实操中过长会报错），我们**保守截断**到 ~45 字符。建议 `idempotency_key` 控制长度（`ord:req:<sid>:<sym>:<uuid8>` 足够）。

4. **outsideRth（盘前/盘后）**

   * 映射到 `Order.outsideRth`；默认关闭。你的策略/风控在意图里有 `allow_prepost` 字段，我们在 Risk 里已经校验；**执行层建议跟随 Risk**整体开关即可。

5. **修改订单**

   * 在 IB/ib_insync 中，**重新 place 同一个订单对象**（同 `orderId`）即可视为“修改”；我们用 `Trade.order` 改字段+place 再提交。
   * 若状态是 `Filled/Cancelled`，修改会失败；CRS 会做重试并在失败时触发兜底流程。

6. **open_orders 的口径**

   * `ib.openTrades()` 汇总“尚未终态”的订单（PreSubmitted/Submitted/InActive 等）。
   * 我们把它转成 `PlacedOrder` 提供给执行内核**领养/巡检**。
