# 1) 目标与约束（交易员视角 → 工程落地）

* **单向收紧**（Ratchet）：多头只上不下，空头只下不上；绝不“放宽”已生效的止损。
* **稳健更新**：间隔/冷却、最小变更幅度（hysteresis）、最小距离（防“贴脸止损”）、入场后宽限 N 根。
* **多策略独立**：每个策略/品种可选不同策略、不同参数、不同更新节拍（5s/10s/1m…）。
* **无数据不动作**：数据不够或异常 → 用硬止损兜底；不盲改。
* **订单速率友好**：价格变化->“聚合更新”，不超过 `modify_rpm`；失败升级由 CRS 兜底。
* **开盘噪声治理**：可配置开盘前 X 分钟只用硬止损或更温和参数。
* **RTH/延长时段**：可开关；默认仅 RTH 生效。

# 3) 三种策略实现

## 3.1 ATR 追踪（波动自适应）

```python
class ATRStop(StopPolicy):
    name = "ATR"
    # params: {"tf_s":60, "n":14, "k":3.0, "grace_bars":2, "floor_bps":30, "cap_bps":400, "rth_only":True}
    async def compute(self, *, symbol, side, entry_px, avg_px, md:MarketDataFacade) -> StopDecision:
        tf = int(self.p.get("tf_s",60)); n = int(self.p.get("n",14)); k=float(self.p.get("k",3.0))
        grace = int(self.p.get("grace_bars",2)); floor_bps=int(self.p.get("floor_bps",30))
        cap_bps = int(self.p.get("cap_bps",400))
        bars = await md.bars(symbol, tf, n+1)
        if not bars or len(bars) < n+1:
            return StopDecision(ok=False, reason="insufficient_bars")
        # 计算 ATR（经典 Wilder）
        tr = []
        for i in range(1, len(bars)):
            h,l,pc = bars[i].h, bars[i].l, bars[i-1].c
            tr.append(max(h-l, abs(h-pc), abs(l-pc)))
        atr = tr[0]
        alpha = 1.0/n
        for v in tr[1:]:
            atr = (1-alpha)*atr + alpha*v
        last = bars[-1].c
        # 应用 grace：入场后前 N 根用 floor/cap 区间
        base = avg_px
        floor = base*(1 - floor_bps/1e4) if side=="LONG" else base*(1 + floor_bps/1e4)
        band  = k*atr
        cand = (last - band) if side=="LONG" else (last + band)
        # 限制在 floor/cap 之间
        cap = base*(1 - cap_bps/1e4) if side=="LONG" else base*(1 + cap_bps/1e4)
        # 对多头：止损不能低于 cap（最大亏损上限），也不能高于 floor（早期不太紧）
        lo, hi = (cap, floor) if side=="LONG" else (floor, cap)  # 注意空头方向相反
        stop = min(max(cand, lo), hi) if side=="LONG" else max(min(cand, hi), lo)
        # 简易 TP：对称 bps（也可独立参数）
        tp_bps = int(self.p.get("tp_bps",40))
        tp = avg_px*(1+tp_bps/1e4) if side=="LONG" else avg_px*(1-tp_bps/1e4)
        return StopDecision(ok=True, stop_px=round(stop,4), tp_px=round(tp,4),
                            reason=f"atr={atr:.4f},band={band:.4f}", valid_ms=tf*1000,
                            tighten_only=True, min_change_bps=int(self.p.get("min_change_bps",5)))
```

## 3.2 通道（Donchian）追踪

```python
class ChannelStop(StopPolicy):
    name="CHANNEL"
    # params: {"tf_s":60, "n":20, "offset_bps":5, "tp_bps":60}
    async def compute(self, *, symbol, side, entry_px, avg_px, md:MarketDataFacade) -> StopDecision:
        tf = int(self.p.get("tf_s",60)); n=int(self.p.get("n",20)); off=int(self.p.get("offset_bps",5))
        bars = await md.bars(symbol, tf, n)
        if not bars or len(bars) < n:
            return StopDecision(ok=False, reason="insufficient_bars")
        highs = [b.h for b in bars]; lows=[b.l for b in bars]
        ch_high = max(highs); ch_low = min(lows)
        if side=="LONG":
            base = ch_low
            stop = base*(1 - off/1e4)
            tp = ch_high
        else:
            base = ch_high
            stop = base*(1 + off/1e4)
            tp = ch_low
        return StopDecision(ok=True, stop_px=round(stop,4), tp_px=round(tp,4),
                            reason=f"donchian[{n}] off={off}bps", valid_ms=tf*1000,
                            tighten_only=True, min_change_bps=int(self.p.get("min_change_bps",5)))
```

## 3.3 斜率（线回归）趋势止损

```python
class SlopeStop(StopPolicy):
    name="SLOPE"
    # params: {"tf_s":60, "n":20, "r2_min":0.2, "k_mult":2.0, "tp_bps":50}
    async def compute(self, *, symbol, side, entry_px, avg_px, md:MarketDataFacade) -> StopDecision:
        tf = int(self.p.get("tf_s",60)); n=int(self.p.get("n",20))
        r2_min=float(self.p.get("r2_min",0.2)); k_mult=float(self.p.get("k_mult",2.0))
        bars = await md.bars(symbol, tf, n)
        if not bars or len(bars) < n:
            return StopDecision(ok=False, reason="insufficient_bars")
        xs = list(range(n)); ys=[b.c for b in bars]
        # 线回归（无依赖实现）
        xbar = sum(xs)/n; ybar = sum(ys)/n
        num = sum((x-xbar)*(y-ybar) for x,y in zip(xs,ys))
        den = sum((x-xbar)**2 for x in xs) or 1e-9
        slope = num/den
        # R^2
        ss_tot = sum((y-ybar)**2 for y in ys) or 1e-9
        ss_res = sum((y - (ybar + slope*(x-xbar)))**2 for x,y in zip(xs,ys))
        r2 = 1-ss_res/ss_tot
        if r2 < r2_min:
            return StopDecision(ok=False, reason=f"r2<{r2_min}")
        # 以回归线 - k倍残差标准差作为 trailing 参考
        resid = [y - (ybar + slope*(x-xbar)) for x,y in zip(xs,ys)]
        import math
        sd = math.sqrt(sum(r*r for r in resid)/n)
        ref = (ybar + slope*(xs[-1]-xbar))     # 最新点的回归值
        band = k_mult*sd
        stop = (ref - band) if side=="LONG" else (ref + band)
        tp_bps = int(self.p.get("tp_bps",50))
        tp = avg_px*(1+tp_bps/1e4) if side=="LONG" else avg_px*(1-tp_bps/1e4)
        return StopDecision(ok=True, stop_px=round(stop,4), tp_px=round(tp,4),
                            reason=f"slope={slope:.5f},r2={r2:.3f},sd={sd:.4f}",
                            valid_ms=tf*1000, tighten_only=True,
                            min_change_bps=int(self.p.get("min_change_bps",5)))
```

# 4) 组合器与单向收紧（Ratchet）

```python
class CompositeStop(StopPolicy):
    name="COMPOSITE"
    # params: {"policies":[("ATR", {...}), ("CHANNEL", {...})], "mode":"MAX_PROTECT"} 
    # mode: MAX_PROTECT(多头取更高的止损、空头取更低的止损) / MIN_MOVE(更新幅度更大的那个) 等
    def __init__(self, params:Dict, registry:Dict[str,StopPolicy]):
        super().__init__(params); self._reg = registry

    async def compute(self, *, symbol, side, entry_px, avg_px, md:MarketDataFacade) -> StopDecision:
        mode = self.p.get("mode","MAX_PROTECT")
        decisions: List[StopDecision] = []
        for name, p in self.p.get("policies", []):
            d = await self._reg[name](p).compute(symbol=symbol, side=side, entry_px=entry_px, avg_px=avg_px, md=md)
            if d.ok: decisions.append(d)
        if not decisions:
            return StopDecision(ok=False, reason="no_subpolicy_ok")
        # 聚合
        if side=="LONG":
            if mode=="MAX_PROTECT":
                stop = max(d.stop_px for d in decisions)
            else:
                stop = sorted(decisions, key=lambda x: abs((x.stop_px-entry_px)/entry_px))[-1].stop_px
        else:
            if mode=="MAX_PROTECT":
                stop = min(d.stop_px for d in decisions)
            else:
                stop = sorted(decisions, key=lambda x: abs((x.stop_px-entry_px)/entry_px))[-1].stop_px
        tp = max((d.tp_px for d in decisions if d.tp_px), default=None) if side=="LONG" else \
             min((d.tp_px for d in decisions if d.tp_px), default=None)
        return StopDecision(ok=True, stop_px=round(stop,4), tp_px=(round(tp,4) if tp else None),
                            reason=f"composite[{mode}]", valid_ms=min(d.valid_ms for d in decisions),
                            tighten_only=True, min_change_bps=min(d.min_change_bps for d in decisions))
```

# 5) 策略解析器（按策略/计划加载正确 StopPolicy）

```python
class StopPolicyResolver:
    """
    从 plan.exec_overrides 或 全局策略配置 解析出 StopPolicy。
    resolver 支持：每策略/每symbol/每计划 覆盖参数与更新节拍。
    """
    def __init__(self, default_conf:Dict, registry:Dict[str,type]=None):
        self.default = default_conf or {}
        self.registry = registry or {
            "ATR": ATRStop, "CHANNEL": ChannelStop, "SLOPE": SlopeStop,
            "COMPOSITE": lambda p: CompositeStop(p, {"ATR":ATRStop, "CHANNEL":ChannelStop, "SLOPE":SlopeStop})
        }

    def resolve(self, plan) -> StopPolicy:
        # 优先：plan.exec_overrides.stop_policy > 策略级默认 > 系统默认
        sp = (plan.exec_overrides or {}).get("stop_policy") or self.default.get(plan.strategy_id) or self.default.get("DEFAULT")
        if not sp:  # 兜底：ATR
            return ATRStop({"tf_s":60,"n":14,"k":3.0})
        name = sp["name"].upper()
        params = sp.get("params",{})
        cls = self.registry[name]
        return cls(params)
```

把 `_ensure_oco()` 的“计算 stop_px / tp_px”的部分替换为：

```python
# 计算方向
long = plan.pos.qty > 0
side = "LONG" if long else "SHORT"
entry_px = plan.exec_overrides.get("entry_px") or plan.pos.avg_px or 0.0
avg = plan.pos.avg_px

# 计算候选价格（策略可 per-plan 配置）
policy = self.stop_resolver.resolve(plan)
t0 = time.time()
dec = await policy.compute(symbol=plan.symbol, side=side, entry_px=entry_px, avg_px=avg, md=self.md)

# 回退到硬止损/固定TP（与之前一致）
if not dec.ok or dec.stop_px is None:
    hard_pct = float(plan.exec_overrides.get("hard_stop_pct", self.cfg.hard_stop_pct))
    tp_bps   = int(plan.exec_overrides.get("take_profit_bps", self.cfg.take_profits_bps if hasattr(self.cfg,"take_profits_bps") else self.cfg.take_profit_bps))
    stop_px = avg*hard_pct if long else avg*(2-hard_pct)
    tp_px   = avg*(1+tp_bps/1e4) if long else avg*(1-tp_bps/1e4)
else:
    stop_px = dec.stop_px
    tp_px = dec.tp_px or (avg*(1+int(self.cfg.take_profit_bps)/1e4) if long else avg*(1-int(self.cfg.take_profit_bps)/1e4))

# 单向收紧（Ratchet）
if plan.stop_order_id and plan.protection_attached:
    # 需要当前已挂止损价（建议你在适配器或状态流里缓存当前价格；这里以 exec_overrides 暂存演示）
    current_stp = plan.exec_overrides.get("current_stop_px")
    if current_stp:
        if long:   stop_px = max(stop_px, current_stp)
        else:      stop_px = min(stop_px, current_stp)

# Hysteresis：若变化 < min_change_bps 则跳过更新
last_stp = plan.exec_overrides.get("last_proposed_stop_px")
if last_stp:
    delta_bps = abs((stop_px - last_stp)/max(1e-6, last_stp))*1e4
    if delta_bps < dec.min_change_bps:
        # 仅调整数量，不改价
        await self.crs.replace(plan.stop_order_id, qty=abs(plan.pos.qty))
        return

# 应用/调整 OCO
oco_side = side_flip("BUY" if long else "SELL")
need_qty = abs(plan.pos.qty)

if not plan.protection_attached or (plan.stop_order_id is None and plan.tp_order_id is None):
    plan.oco_group = plan.oco_group or f"OCO:{plan.decision_id}"
    if self.cfg.use_stop_limit:
        slip = int(plan.exec_overrides.get("stop_limit_slip_bps", getattr(self.cfg,"stop_limit_slip_bps",20)))
        lim_px = stop_px*(1 - slip/1e4) if long else stop_px*(1 + slip/1e4)
        so = await self.adapter.place_stop_limit(plan.symbol, oco_side, need_qty, stop_px, lim_px, plan.tif, order_ref=f"{plan.idempotency_key}:STP", oco_group=plan.oco_group)
    else:
        so = await self.adapter.place_stop(plan.symbol, oco_side, need_qty, stop_px, plan.tif, order_ref=f"{plan.idempotency_key}:STP", oco_group=plan.oco_group)
    to = await self.adapter.place_limit(plan.symbol, oco_side, need_qty, tp_px, plan.tif, order_ref=f"{plan.idempotency_key}:TP", oco_group=plan.oco_group)
    self.orders[so.order_id]=plan.decision_id; self.orders[to.order_id]=plan.decision_id
    plan.stop_order_id = so.order_id; plan.tp_order_id = to.order_id
    plan.protection_attached=True
else:
    # 调价（CRS 替换，强一致）
    await self.crs.replace(plan.stop_order_id, qty=need_qty, stop_px=stop_px)  # 字段名按你的适配器实现
    await self.crs.replace(plan.tp_order_id, qty=need_qty, limit_px=tp_px)

# 记录提案止损价与耗时
plan.exec_overrides["last_proposed_stop_px"] = stop_px
# 你也可以上报 Prom：stop_policy_compute_latency_ms / stop_policy_updates_total{policy}
```

> 注意：`replace(...)` 的字段名需要与你的 `IBAdapter.modify()` 对齐（有的 API 是 `lmtPrice/auxPrice`）。这里只是占位。

# 8) MarketDataFacade 对接建议

* **实时**用你 5s Bar（ClickHouse `bar_5s_v2`），策略参数若指定 1m，可在 Facade 里“取 12 根 5s 聚合为 1m”，或直接下发 CH `GROUP BY toStartOfMinute(ts_start)`。
* **缓存与节流**：同一 symbol 的同一 `tf_s`，5s 内不要重复查 CH；放入 LRU/memory cache。
* **异常回退**：CH 查询失败 → 返回空；策略收到 `insufficient_bars` 自动走硬止损兜底。

# 9) 风险与边界条件处理

* **Gap 风险**：若最新价即刻越过 `stop_px`（滑点/跳空）→ `_ensure_oco` 内直接 `emergency_flatten(plan)`（我们已提供）。
* **开盘前 X 分钟**：在 Resolver 里对 `grace_bars` 或 `k` 做放宽；或强制使用硬止损。
* **修改风暴抑制**：在安全内核已有 `TokenBucket(modify_rpm)` 基础上，再叠加 **hysteresis + valid_ms**，只在窗口到期/幅度足够时改价。
* **策略切换**：`stop_policy.name` 动态变更时，第一笔采用“更保守”的 stop（对多头取 max(旧,新)，空头取 min）。

# 10) 验收清单（10 分钟冒烟）

1. 为策略 `ORB` 配置 `COMPOSITE`（ATR+CHANNEL），`cadence_s=5`。
2. 产生入场 → 首填 → `_ensure_oco` 首次挂 OCO（看指标 `kern_oco_attach_total`）。
3. 模拟价格上行（多头）→ `StopDecision.stop_px` 逐阶上移；`crs_replace_ok_total` 增长；止损价只上不下。
4. 降噪：把价格小抖动控制在 `< min_change_bps`，观察**不触发**改单。
5. 断开 CH（Facade 返回空）→ 策略改用硬止损；无频繁报错。
6. 打开市场跳空场景：价格低于止损 → 触发 `emergency_flatten`，另一腿被强撤（`crs_cancel_ok_total{kind="TP"}` 增长）。
