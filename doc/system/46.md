太好了。接下来我把**“IB 快照缓存（NBBO + 状态补丁）门面”**一次性贴出来，直接当作 `MarketStateProvider` 的 `secondary` 数据源用。它解决三件事：

* 把 IB / ib-insync 的 Ticker 更新**收集成可读快照**：`bid/ask/last/mid/tick/lot/ts_us`
* 允许从**外部状态源**（监管/交易所/内部服务）**补充**：`luld_low/high`、`ssr_active`、`halted`
* 内置 **TTL 与并发安全**，缺字段也能稳健返回（给 Reg-Lite 做“拒/放行+告警”的依据）

# 1) 快照缓存：NBBOSnapshotCache

```python
# -*- coding: utf-8 -*-
import time, asyncio
from dataclasses import dataclass, asdict
from typing import Optional, Dict, Any, Iterable

def _now_us() -> int:
    return int(time.time() * 1e6)

@dataclass
class NBBOSnapshot:
    symbol: str
    ts_us: int = 0
    bid: Optional[float] = None
    bid_size: Optional[float] = None
    ask: Optional[float] = None
    ask_size: Optional[float] = None
    last: Optional[float] = None
    last_size: Optional[float] = None
    mid: Optional[float] = None
    tick: float = 0.01          # 最小价位（缺省）
    lot: float = 1.0            # 最小交易单位（缺省）
    # 状态补丁（可由外部源写入）
    luld_low: Optional[float] = None
    luld_high: Optional[float] = None
    ssr_active: bool = False
    halted: bool = False
    # 来源标注（便于排查）
    sources: Dict[str, int] = None  # {"ib": ts_us, "patch": ts_us, "xchg": ts_us}

    def to_market_state_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # MarketStateProvider 只取这些关键字段，其他用于观测/调试
        return {
            "symbol": self.symbol,
            "ts_us": self.ts_us,
            "bid": self.bid, "ask": self.ask, "last": self.last,
            "mid": self.mid if self.mid is not None and not (self.mid != self.mid) else None,
            "tick": self.tick, "lot": self.lot,
            "luld_low": self.luld_low, "luld_high": self.luld_high,
            "ssr": self.ssr_active, "halted": self.halted
        }

class NBBOSnapshotCache:
    """
    - 接收 ib-insync Ticker 或手动 update()，形成每 symbol 的最新快照
    - 允许 apply_status() 从外部源写入 LULD/SSR/halt/tick/lot
    - 提供 TTL；过期不删，但 read() 会返回 None（让上层走 fallback）
    """
    def __init__(self, ttl_ms: int = 800):
        self.ttl_ms = ttl_ms
        self._lock = asyncio.Lock()
        self._snap: Dict[str, NBBOSnapshot] = {}
        self._subscribed: set[str] = set()
        self._closed = False

        # 可选：在你的指标系统里暴露
        self.metrics = type("M", (), {
            "nbbo_updates_total": 0,
            "status_patches_total": 0,
            "reads_total": 0,
            "reads_stale_total": 0
        })()

    # ------------------ 对外主接口 ------------------

    async def read(self, symbol: str) -> Optional[Dict[str, Any]]:
        """读取缓存；若超过 TTL 视为陈旧，返回 None（让上游 fallback）"""
        async with self._lock:
            self.metrics.reads_total += 1
            s = self._snap.get(symbol)
            if not s:
                self.metrics.reads_stale_total += 1
                return None
            if (_now_us() - s.ts_us) > self.ttl_ms * 1000:
                self.metrics.reads_stale_total += 1
                return None
            # 派生 mid
            if s.mid is None and s.bid is not None and s.ask is not None:
                s.mid = (s.bid + s.ask) / 2.0
            return s.to_market_state_dict()

    async def update(self, symbol: str, **fields) -> None:
        """
        手动更新（你也可以从别的行情门面推入）
        可填字段：bid, bid_size, ask, ask_size, last, last_size, tick, lot
        """
        if self._closed:
            return
        async with self._lock:
            s = self._snap.get(symbol) or NBBOSnapshot(symbol=symbol, sources={})
            changed = False
            for k, v in fields.items():
                if not hasattr(s, k): 
                    continue
                if v is None:
                    continue
                setattr(s, k, float(v) if isinstance(getattr(s, k, None), (float, type(None))) and isinstance(v, (int, float)) else v)
                changed = True
            if changed:
                s.ts_us = _now_us()
                s.sources["manual"] = s.ts_us
                # 更新 mid
                if s.bid is not None and s.ask is not None:
                    s.mid = (s.bid + s.ask) / 2.0
                self._snap[symbol] = s
                self.metrics.nbbo_updates_total += 1

    async def apply_status(self, symbol: str, **status_fields) -> None:
        """
        外部状态补丁（交易所/监管/内部服务）：
          luld_low, luld_high, ssr_active, halted, tick, lot
        """
        if self._closed:
            return
        async with self._lock:
            s = self._snap.get(symbol) or NBBOSnapshot(symbol=symbol, sources={})
            for k in ("luld_low","luld_high","ssr_active","halted","tick","lot"):
                if k in status_fields and status_fields[k] is not None:
                    setattr(s, k, status_fields[k])
            s.ts_us = s.ts_us or _now_us()  # 若尚无 ts，给一个
            s.sources["patch"] = _now_us()
            self._snap[symbol] = s
            self.metrics.status_patches_total += 1

    async def close(self):
        self._closed = True
        # 如有 ib 订阅在这里清理

    # ------------------（可选）直接接 ib-insync ------------------

    async def subscribe_ib(self, ib, symbols: Iterable[str], generic_ticks: str = "", snapshot: bool = False):
        """
        直接用 ib-insync 订阅，并将 ticker.updateEvent 挂到 _on_ib_update
        - generic_ticks：按需填写（例如 LULD/SSR 相关 tick；留空也可）
        - snapshot=True 仅拉一次快照
        """
        if self._closed:
            return
        todo = [s for s in symbols if s not in self._subscribed]
        if not todo:
            return

        # 合约构造按你现有 adapter 封装；这里示意用 Stock(symbol, 'SMART', 'USD')
        from ib_insync import Stock
        contracts = [Stock(sym, 'SMART', 'USD') for sym in todo]
        tickers = await ib.reqMktDataMulti(contracts, genericTickList=generic_ticks, snapshot=snapshot)
        for sym, tk in zip(todo, tickers):
            # 强绑定 symbol 方便回调识别
            setattr(tk, "_q_symbol", sym)
            tk.updateEvent += self._on_ib_update
            self._subscribed.add(sym)

    def _on_ib_update(self, ticker):
        """
        ib-insync 的 Ticker.updateEvent 回调；尽量不阻塞事件线程
        可获得字段（常见）：ticker.bid/ask/last/bidSize/askSize/lastSize/close
        其他状态（halt/SSR/LULD）未必有 → 通过 apply_status 外补
        """
        try:
            sym = getattr(ticker, "_q_symbol", None) or (getattr(ticker.contract, "symbol", None) or "")
            if not sym:
                return
            # 采样最新快照
            fields = {}
            if getattr(ticker, "bid", None) not in (None, float("nan")):
                fields["bid"] = float(ticker.bid)
            if getattr(ticker, "ask", None) not in (None, float("nan")):
                fields["ask"] = float(ticker.ask)
            if getattr(ticker, "last", None) not in (None, float("nan")):
                fields["last"] = float(ticker.last)
            if getattr(ticker, "bidSize", None):
                fields["bid_size"] = float(ticker.bidSize)
            if getattr(ticker, "askSize", None):
                fields["ask_size"] = float(ticker.askSize)
            if getattr(ticker, "lastSize", None):
                fields["last_size"] = float(ticker.lastSize)

            # 若 ib 提供最小价位/合约乘数/lot，可在你的 adapter 额外塞入（演示从 contract multipliers）
            if getattr(ticker, "contract", None):
                # 如果你有更准确 tick/lot 来源，可以在别处 apply_status 覆盖
                pass

            # 将更新丢到后台任务，避免阻塞 ib 线程
            asyncio.get_event_loop().create_task(self.update(sym, **fields))
        except Exception:
            # 不抛异常以免影响 ib 事件循环
            pass
```

```python
# 你的 MarketStateProvider.secondary 来源换成 IB 缓存门面
class MarketStateIB(MarketStateSource):
    def __init__(self, nbbo_cache: NBBOSnapshotCache):
        self.cache = nbbo_cache
    async def get(self, symbol: str) -> Optional["MarketState"]:
        s = await self.cache.read(symbol)
        if not s:
            return None
        # 组装为 MarketState（与之前给你的 Provider 契约一致）
        from dataclasses import dataclass
        @dataclass
        class MarketState:
            symbol: str; ts_us: int
            bid_px: Optional[float]=None; ask_px: Optional[float]=None
            mid_px: Optional[float]=None; last_px: Optional[float]=None
            luld_low: Optional[float]=None; luld_high: Optional[float]=None
            ssr_active: bool=False; halted: bool=False
            tick_size: float=0.01; lot_size: float=1.0; source: str="ib"
        return MarketState(
            symbol=s["symbol"], ts_us=s["ts_us"],
            bid_px=s["bid"], ask_px=s["ask"], last_px=s["last"], mid_px=s["mid"],
            luld_low=s["luld_low"], luld_high=s["luld_high"],
            ssr_active=bool(s["ssr"]), halted=bool(s["halted"]),
            tick_size=s["tick"], lot_size=s["lot"], source="ib"
        )
```

async def _self_test():
    cache = NBBOSnapshotCache(ttl_ms=800)
    await cache.update("AAPL", bid=200.0, ask=200.1)                # 1
    await cache.update("AAPL", last=200.05, bid_size=100, ask_size=200)  # 2

# 4) 集成注意点（避免踩坑）

* **TTL 要短**（~0.8–1s）：MarketStateProvider 会在 TTL 过期时自动走 `fallback`；这样 Reg-Lite 能“安全降级”，不会因快照陈旧误判。
* **tick/lot 的权威来源**：

  * 若 IB 合约元数据不可靠（个别场景会这样），你可以在**合约加载**时将正确的 tick/lot 写入 `apply_status()`，保证 Reg-Lite 的 tick/lot 校验口径**唯一**。
* **LULD/SSR/Halt**：

  * IB 的 Ticker 并**不总是**稳定提供这些监管状态，所以我们用 `apply_status()` 预留了外部补丁口；你可以订阅交易所/数据商的状态源（或内部规则服务），然后按 symbol 写入缓存。
  * 拿不到时，Reg-Lite 对 LULD/SSR/Halt 只**WARN**；但对 tick/lot、价格围栏、无 NBBO 的 **MKT** 仍**硬拒**（这个逻辑我们已在 Reg-Lite 实现）。
* **并发安全**：`update()`/`apply_status()`/`read()` 都加了锁；`_on_ib_update` 用 `create_task()` 异步落地，**不阻塞 IB 事件线程**。
* **监控**：把 `nbbo_updates_total / status_patches_total / reads_stale_total` 挂到你的指标系统，Grafana 上加个“快照过期率”图，超过阈值（比如 5%）报警。
