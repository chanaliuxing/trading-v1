# ② MarketStateProvider（LULD/SSR/停牌 数据融合）

**目的**：给 Reg-Lite 提供**权威/一致**的市场状态：
`luld_low/luld_high`、`ssr_active`、`halted`、`nbbo(bid/ask)`、`tick_size/lot_size`。
来源可多元：**主**（交易所/数据商 API）、**备**（IB/TWS 快照）、**回退**（基于历史波动的近似）。

下面是**接口与聚合器骨架**（带本地缓存与 TTL；拿不到就降级为“WARN 放行”）：

```python
import time, asyncio
from dataclasses import dataclass
from typing import Optional, Dict, Any

@dataclass
class MarketState:
    symbol: str
    ts_us: int
    bid_px: Optional[float]=None
    ask_px: Optional[float]=None
    mid_px: Optional[float]=None
    last_px: Optional[float]=None
    luld_low: Optional[float]=None
    luld_high: Optional[float]=None
    ssr_active: bool=False
    halted: bool=False
    tick_size: float=0.01
    lot_size: float=1.0
    source: str="unknown"

class MarketStateSource:
    async def get(self, symbol:str) -> Optional[MarketState]:
        raise NotImplementedError

class MarketStateIB(MarketStateSource):
    """从 IB/TWS 快照读取（实现时：调用你已有的行情门面/快照缓存）"""
    def __init__(self, snap_cache):
        self.snap = snap_cache
    async def get(self, symbol:str) -> Optional[MarketState]:
        s = await self.snap.read(symbol)  # 返回 {bid,ask,last,mid,tick,lot,halt,ssr,luld_low,high} 可能部分缺失
        if not s: return None
        return MarketState(
            symbol=symbol, ts_us=s["ts_us"], bid_px=s.get("bid"), ask_px=s.get("ask"),
            mid_px=s.get("mid"), last_px=s.get("last"), tick_size=s.get("tick",0.01), lot_size=s.get("lot",1.0),
            halted=bool(s.get("halted",False)), ssr_active=bool(s.get("ssr",False)),
            luld_low=s.get("luld_low"), luld_high=s.get("luld_high"), source="ib"
        )

class MarketStateXchg(MarketStateSource):
    """交易所/数据商权威源（留接口；你可接纳斯达克/OPRA 等）"""
    async def get(self, symbol:str) -> Optional[MarketState]:
        return None  # 待接第三方

class MarketStateFallback(MarketStateSource):
    """基于近期 5s bar 推估 LULD 近似带（保守）：±(α·ATR% + β%)"""
    def __init__(self, bar_facade, alpha=3.0, beta_pct=2.0):
        self.bar = bar_facade; self.a=alpha; self.b=beta_pct
    async def get(self, symbol:str) -> Optional[MarketState]:
        bar = await self.bar.last_bar(symbol)  # {o,h,l,c,ts_us,atr_pct}
        if not bar: return None
        ref = bar["c"]; atr_pct = bar.get("atr_pct", 1.0)
        band = (self.a * atr_pct + self.b) / 100.0
        low = ref * (1 - band); high = ref * (1 + band)
        return MarketState(symbol=symbol, ts_us=bar["ts_us"], last_px=ref, mid_px=ref,
                           luld_low=low, luld_high=high, tick_size=0.01, lot_size=1.0, source="fallback")

class MarketStateProvider:
    def __init__(self, primary:MarketStateSource, secondary:MarketStateSource, fallback:MarketStateSource,
                 ttl_ms:int=800, prefer:str="primary"):
        self.primary=primary; self.secondary=secondary; self.fallback=fallback
        self.ttl = ttl_ms
        self.cache: Dict[str, MarketState] = {}
        self.prefer = prefer

    def _fresh(self, ms:MarketState) -> bool:
        return (time.time()*1e6 - ms.ts_us) <= (self.ttl*1000)

    async def get(self, symbol:str) -> MarketState:
        # 1) 先看缓存
        ms = self.cache.get(symbol)
        if ms and self._fresh(ms): return ms

        # 2) 取源：主→次→回退，按 prefer
        order = [self.primary, self.secondary, self.fallback]
        if self.prefer=="secondary": order=[self.secondary,self.primary,self.fallback]

        out=None
        for src in order:
            try:
                s = await src.get(symbol)
            except Exception:
                s = None
            if s and (s.bid_px or s.ask_px or s.last_px):
                out=s; break
        if not out:
            out = MarketState(symbol, int(time.time()*1e6))  # 极端：返回“空状态”，由 Reg-Lite 决定 WARN/放行

        # 3) 补全派生字段
        if out.mid_px is None and out.bid_px and out.ask_px:
            out.mid_px = (out.bid_px + out.ask_px)/2.0

        self.cache[symbol]=out
        return out
```

**与 Reg-Lite 对接（风控侧一处改动）**：

```python
mstate = await market_state_provider.get(intent.symbol)
snap = MarketSnap(
  last_px=mstate.last_px, bid_px=mstate.bid_px, ask_px=mstate.ask_px,
  mid_px=mstate.mid_px, luld_low=mstate.luld_low, luld_high=mstate.luld_high,
  ssr_active=mstate.ssr_active, halted=mstate.halted,
  tick_size=mstate.tick_size, lot_size=mstate.lot_size
)
hits = reglite_guard.check(intent, snap)
```

**落地建议**

* 先接 **IB 快照** 做 `secondary`（我们已有 IBEventBridge/行情门面）；
* `primary` 先空实现（或你有第三方数据源就接上）；
* `fallback` 用 5s bar 的 `ATR%` 粗略估带，**仅作保守围栏，不作强制 LULD**；
* 缺权威数据时，Reg-Lite 对 LULD/SSR/Halt 命中只**WARN**，但 tick/lot、价格围栏、无 NBBO 的 **MKT** 仍**硬拒**。
