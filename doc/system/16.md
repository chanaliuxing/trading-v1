# 1) 单文件离线 E2E Harness（M1/M2）

> 作用：不依赖 Kafka/PG/CH/Redis，也能先验证**幂等预留/释放**、**OCO 首填即挂**、**5s 聚合器**的核心时序与边界。
> 用完这个，再把“内存总线”替换为 Kafka 适配层就行（我上次也贴过 Kafka 适配层）。

```python
# ======= E2E HARNESS: reserve→exec→OCO→release + tick→5s bars =======
import asyncio, time, uuid
from collections import defaultdict
from dataclasses import dataclass

# ---------- 通用 ----------
now_us = lambda: int(time.time() * 1e6)

# ---------- 内存消息总线（代替 Kafka） ----------
class Bus:
    def __init__(self): self.topics = defaultdict(list)
    def subscribe(self, topic):
        q = asyncio.Queue(); self.topics[topic].append(q); return q
    async def publish(self, topic, key, value):
        for q in self.topics[topic]:
            await q.put({"key": key, "value": value})

# ---------- 假账本（幂等预留/释放 + 串行化） ----------
class FakeLedger:
    def __init__(self, cap_usd):
        self.cap = float(cap_usd); self.reserved = 0.0
        self.lock = asyncio.Lock(); self.res_map = {}  # idem -> amt
    async def approve_and_reserve(self, notional_usd, buffer_bps, idem):
        async with self.lock:
            if idem in self.res_map:
                return True, self.res_map[idem]
            need = float(notional_usd) * (1 + buffer_bps/10000.0)
            if self.reserved + need > self.cap:
                return False, 0.0
            self.reserved += need; self.res_map[idem] = need
            return True, need
    async def release(self, idem):
        async with self.lock:
            if idem not in self.res_map: return True, 0.0
            amt = self.res_map.pop(idem); self.reserved = max(self.reserved - amt, 0.0)
            return True, amt

# ---------- 风控服务：消费请求→审批→发决策 ----------
class RiskService:
    def __init__(self, bus, ledger): self.bus, self.ledger = bus, ledger
    async def run(self):
        q = self.bus.subscribe("engine.order.request.v2")
        while True:
            msg = await q.get()
            data, meta = msg["value"]["data"], msg["value"]["metadata"]
            ok, reserved = await self.ledger.approve_and_reserve(
                data["notional_hint_usd"], data.get("slippage_buffer_bps", 10), meta["idempotency_key"]
            )
            out = {
                "schema_v": 2,
                "metadata": {"event_id": str(uuid.uuid4()), "ts_us": now_us(),
                             "producer": "risk", "idempotency_key": meta["idempotency_key"]},
                "data": {
                    "decision": "APPROVED" if ok else "DENIED",
                    "reservation": {"reservation_id": str(uuid.uuid4()),
                                    "reserved_usd": reserved,
                                    "buffer_bps": data.get("slippage_buffer_bps", 10)}
                }
            }
            await self.bus.publish("risk.decision.v2", key=msg["key"], value=out)

# ---------- 执行器：首填即挂 OCO + 终态释放 ----------
class ExecService:
    def __init__(self, bus, ledger):
        self.bus, self.ledger = bus, ledger
        self.oco_attached = set()     # 统计用
        self.orders = {}              # idem -> order_id
    async def run(self):
        q = self.bus.subscribe("risk.decision.v2")
        while True:
            msg = await q.get()
            meta, dec = msg["value"]["metadata"], msg["value"]["data"]["decision"]
            idem = meta["idempotency_key"]
            if dec != "APPROVED":  # 拒单不下
                continue
            oid = self.orders.get(idem) or str(uuid.uuid4())
            self.orders[idem] = oid
            await self._status(oid, "SUBMITTED", idem)
            # 模拟首笔成交→立即挂 OCO（≤1s）
            await asyncio.sleep(0.05)
            await self._fill(oid, 100, 221.34, idem)
            await self._attach_oco(oid)
            # 模拟终态（止盈触发）→ 释放预留
            await asyncio.sleep(0.05)
            await self._status(oid, "FILLED", idem)
            await self.ledger.release(idem)
    async def _status(self, oid, state, idem):
        payload = {"schema_v":2, "metadata":{"event_id":str(uuid.uuid4()),"ts_us":now_us(),
                   "producer":"exec","idempotency_key":idem}, "data":{"order_id":oid,"state":state}}
        await self.bus.publish("exec.order.status.v2", key=oid, value=payload)
    async def _fill(self, oid, qty, px, idem):
        payload = {"schema_v":2, "metadata":{"event_id":str(uuid.uuid4()),"ts_us":now_us(),
                   "producer":"exec","idempotency_key":idem}, "data":{"order_id":oid,"qty":qty,"px":px}}
        await self.bus.publish("exec.fill.v2", key=oid, value=payload)
    async def _attach_oco(self, oid): self.oco_attached.add(oid)

# ---------- 极简引擎（只发下单意图） ----------
class MiniEngine:
    def __init__(self, bus): self.bus = bus
    async def place_limit(self, strategy_id, symbol, qty, limit_px):
        idem = f"ord:req:{strategy_id}:{symbol}:{uuid.uuid4()}"
        payload = {
          "schema_v":2,
          "metadata":{"event_id":str(uuid.uuid4()),"ts_us":now_us(),"producer":"engine","idempotency_key":idem},
          "data":{
            "strategy_id":strategy_id, "symbol":symbol, "side":"BUY" if qty>0 else "SELL",
            "order_type":"LMT","qty":abs(qty),"limit_px":float(limit_px),"stop_px":None,"tif":"DAY",
            "notional_hint_usd":abs(qty)*float(limit_px),"slippage_buffer_bps":8,"time_stop_s":900,
            "tags":["demo","5s"],"params_hash":"demo"
          }
        }
        await self.bus.publish("engine.order.request.v2", key=f"{strategy_id}:{symbol}", value=payload)
        return idem

# ---------- 5s 聚合器（watermark + emit_buffer） ----------
@dataclass
class Tick: symbol:str; ts_us:int; px:float; qty:int

class InMemAgg:
    def __init__(self, bus, emit_buffer_ms=300, max_late_ms=2000):
        self.bus = bus
        self.emit_buf = emit_buffer_ms*1000; self.max_late = max_late_ms*1000
        self.state = {}  # sym -> dict
    def _bucket(self, ts_us): return (ts_us//5_000_000)*5_000_000
    async def on_tick(self, t: Tick):
        b = self._bucket(t.ts_us)
        st = self.state.get(t.symbol)
        if not st:
            self.state[t.symbol] = {"bucket": b, "open": t.px, "high": t.px, "low": t.px, "close": t.px,
                                    "vol": t.qty, "vwap_num": t.px*t.qty, "vwap_den": t.qty,
                                    "wm_ts": t.ts_us, "late_count":0}
            return
        if b != st["bucket"]:
            if t.ts_us >= st["bucket"] + 5_000_000 + self.emit_buf:
                await self._emit(t.symbol, st); self.state.pop(t.symbol, None)
            self.state[t.symbol] = {"bucket": b, "open": t.px, "high": t.px, "low": t.px, "close": t.px,
                                    "vol": t.qty, "vwap_num": t.px*t.qty, "vwap_den": t.qty,
                                    "wm_ts": t.ts_us, "late_count":0}
            return
        if t.ts_us > st["bucket"] + 5_000_000 + self.max_late:
            st["late_count"] += 1; return
        st["high"] = max(st["high"], t.px)
        st["low"]  = min(st["low"], t.px)
        st["close"]= t.px
        st["vol"] += t.qty
        st["vwap_num"] += t.px*t.qty
        st["vwap_den"] += t.qty
        st["wm_ts"] = max(st["wm_ts"], t.ts_us)
    async def flush_overdue(self, now_us):
        for sym, st in list(self.state.items()):
            if now_us >= st["bucket"] + 5_000_000 + self.emit_buf:
                await self._emit(sym, st); self.state.pop(sym, None)
    async def _emit(self, sym, st):
        bar = {
          "schema_v":2, "metadata":{"event_id":str(uuid.uuid4()),"ts_us":now_us(),"producer":"agg"},
          "data":{"symbol": sym, "ts_start_us": st["bucket"], "ts_end_us": st["bucket"]+5_000_000,
                  "open": st["open"], "high": st["high"], "low": st["low"], "close": st["close"],
                  "vol": st["vol"], "vwap": (st["vwap_num"]/st["vwap_den"]) if st["vwap_den"] else None,
                  "rth": True, "source": "AGG", "calc_mode": "realtime", "late_count": st["late_count"]}
        }
        await self.bus.publish("md.bar.5s.v2", key=sym, value=bar)

# ---------- 三个自动化用例 ----------
async def test_reserve_exec_release():
    print("[A] 预留→执行→OCO→释放 …")
    bus = Bus(); ledger = FakeLedger(cap_usd=100_000)
    risk = RiskService(bus, ledger); execv = ExecService(bus, ledger); eng = MiniEngine(bus)
    asyncio.create_task(risk.run()); asyncio.create_task(execv.run())
    idem1 = await eng.place_limit("S1","AAPL",100,221.0)
    await asyncio.sleep(0.3)
    assert ledger.reserved == 0.0, "终态应释放预留"
    assert len(execv.oco_attached) == 1, "应挂上 OCO"
    print("  ✅ PASS")

async def test_concurrency_gate():
    print("[B] 并发冲关（资金上限型） …")
    bus = Bus(); ledger = FakeLedger(cap_usd=10_000)
    risk = RiskService(bus, ledger); execv = ExecService(bus, ledger); eng = MiniEngine(bus)
    asyncio.create_task(risk.run()); asyncio.create_task(execv.run())
    async def fire(): await eng.place_limit("S1","AAPL",50,100.0)  # 每单名义5k
    await asyncio.gather(*[fire() for _ in range(10)])
    await asyncio.sleep(0.8)
    assert ledger.reserved == 0.0, "应全部终态释放"
    # 不做“仅1单通过”的强断言（正式服务用 PG advisory lock 实现）
    print("  ✅ PASS")

async def test_tick_to_bar():
    print("[C] tick→5s 聚合（watermark/emit_buffer/迟到） …")
    bus = Bus(); agg = InMemAgg(bus, emit_buffer_ms=300, max_late_ms=2000)
    qbar = bus.subscribe("md.bar.5s.v2")
    t0 = now_us()
    for i in range(12):
        bucket_start = t0 + i*5_000_000
        for j in range(10):
            # ±1.5s 抖动
            ts = bucket_start + 200_000 + (j*20_000) + ([-1,1][j%2]*1_500_000)
            await agg.on_tick(Tick("AAPL", ts, px=100+i*0.1+j*0.01, qty=10))
        # 超迟到（>2s）：应计入 late_count，不影响已出bar
        await agg.on_tick(Tick("AAPL", bucket_start + 5_000_000 + 3_500_000, px=99.0, qty=1))
        await agg.flush_overdue(bucket_start + 5_000_000 + 400_000)
    bars = []
    while not qbar.empty(): bars.append((await qbar.get())["value"]["data"])
    assert len(bars) == 12, f"应有12根bar，实际 {len(bars)}"
    assert all(b["late_count"] >= 1 for b in bars), "每桶应至少记录到一个迟到被丢弃"
    print("  ✅ PASS")

# ---------- 主入口 ----------
async def main():
    await test_reserve_exec_release()
    await test_concurrency_gate()
    await test_tick_to_bar()
    print("\n全部用例通过 ✅  —— 上 Kafka/PG/CH/Redis 前的必要闭环已验证。")
if __name__ == "__main__":
    asyncio.run(main())
# ================== END ==================
```

# 2) 回补缺口“切片优先级”策略（确保最重要的先补上）

> 目标：当缺口很多/速率受限时，**先补“最影响策略决策”的片段**，保证开盘后 5s 引擎稳定。

**优先队列排序（权重分）**

1. **靠近当前时间**（距 now 越近权重越高，避免“近期指标断层”）
2. **RTH 时段优先**（非 RTH 降权 0.5）
3. **缺口长度越大、越完整优先**（避免碎片带来的多次请求开销）
4. **交易活跃度高的 symbol 优先**（最近 1h 成交量/波动率越高权重越高）
5. **相邻缺口合并**（< 30s 的小间隙按 1 个缺口处理，减少请求数）

**调度伪代码**

```python
def weight_gap(sym, t0, t1, is_rth, vol_score):
    recency = max(0.1, 1.0 - (now()-t1).total_seconds()/3600.0)  # 近1小时拉满
    len_score = min(1.0, (t1-t0).total_seconds()/120.0)         # 2分钟及以上满分
    rth_w = 1.0 if is_rth else 0.5
    return 0.5*recency + 0.3*len_score + 0.2*vol_score * rth_w

# 构建优先队列
pq = []
for gap in gaps(sym):
    w = weight_gap(sym, gap.t0, gap.t1, is_rth(gap), vol_score(sym))
    heappush(pq, (-w, sym, gap.t0, gap.t1))

# 取任务时：从 pq 取出，按 1800s (或 300s) 切成段，入回补队列
```

**注意**

* **相邻切片拼接写入**：确保 `ts_start` 对齐 5s；避免重复覆盖。
* **回补成功后复检**：只对该 symbol 的时间窗重跑“缺口探针”，确认无残留。
* **熔断/退避**：用我上一条消息给你的限流/退避/熔断策略，避免 IB 封锁。

# 3) Prometheus 指标最小导出片段（可嵌进任意服务）

> 指标名全部沿用我们前面统一的命名，这个最小片段让你10行就把指标暴露出去。

```python
# ===== Prometheus Exporter (最小) =====
from prometheus_client import Counter, Gauge, Histogram, start_http_server

# 通用直方图桶：毫秒级
LAT_BUCKETS = (5, 10, 25, 50, 100, 200, 400, 800, 1600)

# 风控/账本
risk_approval_latency_ms = Histogram("risk_approval_latency_ms", "Risk approve latency (ms)",
                                     ["strategy"], buckets=LAT_BUCKETS)
ledger_reserved_usd = Gauge("ledger_reserved_usd", "Total reserved USD", ["strategy"])

# 执行/OCO
oco_attach_success_total = Counter("order_oco_attach_success_total", "OCO attach success", ["strategy"])
unprotected_window_seconds = Gauge("order_unprotected_window_seconds", "Seconds without protection", ["strategy"])

# 数据面/聚合
bar_emit_latency_ms = Histogram("bar_emit_latency_ms", "5s bar emit latency (ms)", ["symbol"], buckets=LAT_BUCKETS)
bar5s_gap_count = Gauge("bar5s_gap_count", "Gap count today", ["symbol"])

# 引擎/调度
strategy_exec_latency_ms = Histogram("engine_strategy_exec_latency_ms", "on_microbar latency (ms)",
                                     ["strategy"], buckets=LAT_BUCKETS)
strategy_actual_cadence_s = Gauge("engine_strategy_actual_cadence_s", "Actual cadence seconds", ["strategy"])
strategy_skips_total = Counter("engine_strategy_skips_total", "Strategy skip count", ["strategy","reason"])

# 启动 HTTP /metrics
def start_metrics_server(port=9000):
    start_http_server(port)
    print(f"[metrics] Prometheus exporter listening on :{port}")

# 使用示例（任意服务中）：
# start_metrics_server(9000)
# risk_approval_latency_ms.labels(strategy="S1").observe(12.3)
# ledger_reserved_usd.labels(strategy="S1").set(22313.0)
# =====================================
```

> **部署贴士**
>
> * 每个进程开一个 exporter 端口（例如：risk 9101、exec 9102、engine 9103、agg 9104）。
> * Grafana 面板就用我们前面列的指标名：`oco_attach_success_rate=success/(success+fail)`、`emit_latency_p95`、`strategy.skips_total{reason}` 等。
