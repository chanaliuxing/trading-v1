那我来定：**先把全链路 Prometheus 指标导出做起来**。可观测性是你上线最需要的“安全气囊”，而且改动最小、收益最大：做完立刻能发现 5s 聚合延迟、ClickHouse 写入卡顿、策略节奏漂移、OCO 保护窗口等问题。下面直接贴**可拷贝**的指标模块 + 在**聚合器**和**CH Writer**里的接线点（保持我们之前所有接口，直接加几行就能跑），不生成文件。

# 一、通用指标模块（所有服务共用）

```python
# ===== metrics_exporter.py =====
# 依赖: prometheus_client>=0.20
from prometheus_client import Counter, Gauge, Histogram, start_http_server
from collections import defaultdict
import threading, os

# 避免多次注册
_METRICS = {}
_LOCK = threading.Lock()

def _mk(name, typ, desc, labels=()):
    key = (typ, name, tuple(labels))
    with _LOCK:
        if key in _METRICS: return _METRICS[key]
        if typ == "counter":
            m = Counter(name, desc, labels) if labels else Counter(name, desc)
        elif typ == "gauge":
            m = Gauge(name, desc, labels) if labels else Gauge(name, desc)
        elif typ == "hist":
            # 统一毫秒直方图桶
            m = Histogram(name, desc, labels, buckets=(5,10,20,50,100,200,400,800,1600,3200))
        else:
            raise ValueError("unknown metric type")
        _METRICS[key] = m
        return m

def boot_metrics(service_name:str, default_port:int=9100):
    """
    在进程内启动 /metrics HTTP 端口。
    端口优先读 env: METRICS_PORT；没有就用 default_port。
    """
    port = int(os.getenv("METRICS_PORT", default_port))
    # 只启动一次
    once_key = f"__started__:{port}"
    with _LOCK:
        if once_key in _METRICS: return port
        start_http_server(port)
        _METRICS[once_key] = True
    # 通用元信息
    _mk("service_info", "gauge", "static label carrier", ("service",)).labels(service=service_name).set(1)
    return port

# 统一的度量句柄（各服务用得到的）
def registry():
    return dict(
        # 聚合器
        agg_tick_ingested=_mk("agg_tick_ingested_total","counter","ticks ingested",{ "symbol" }),
        agg_bar_emit=_mk("agg_bar_emit_total","counter","5s bars emitted",{ "symbol" }),
        agg_bar_emit_latency=_mk("agg_bar_emit_latency_ms","hist","bar emit latency ms",{ "symbol" }),
        agg_repair_emit=_mk("agg_repair_emit_total","counter","repair bars emitted",{ "symbol" }),
        agg_late_ticks=_mk("agg_late_ticks_total","counter","late ticks",{ "symbol" }),
        agg_gap_events=_mk("agg_gap_events_total","counter","gap events",{ "symbol","reason" }),
        agg_buckets_active=_mk("agg_buckets_active","gauge","active 5s buckets"),
        agg_kafka_fail=_mk("agg_kafka_send_fail_total","counter","kafka send failures"),
        agg_redis_fail=_mk("agg_redis_persist_fail_total","counter","redis persist failures"),

        # ClickHouse Writer
        chw_rows_ok=_mk("chw_rows_ok_total","counter","rows inserted ok"),
        chw_rows_err=_mk("chw_rows_err_total","counter","rows failed"),
        chw_flush=_mk("chw_flush_total","counter","flush batches"),
        chw_retry=_mk("chw_retry_total","counter","flush retries"),
        chw_flush_latency=_mk("chw_flush_latency_ms","hist","flush latency ms"),
        chw_batch_size=_mk("chw_batch_size","hist","batch size",{ "topic" }),

        # 策略引擎（若你要一并接线）
        eng_exec_latency=_mk("engine_strategy_exec_latency_ms","hist","on_* latency ms",{ "strategy" }),
        eng_actual_cadence=_mk("engine_strategy_actual_cadence_s","gauge","actual cadence seconds",{ "strategy" }),
        eng_skips=_mk("engine_strategy_skips_total","counter","readycheck skips",{ "strategy","reason" }),
        eng_orders=_mk("engine_emit_requests_total","counter","order intents",{ "strategy" }),
        eng_dedup=_mk("engine_dedup_suppressed_total","counter","dedup suppressed",{ "strategy" }),
    )
```

在 `_ingest_ticks()` 开头统计 tick；生成/更新桶时更新活动桶数；迟到统计；发出 bar / repair / gap 时计数与延迟：

```python
    async def _ingest_ticks(self, tickers):
        now_ms = int(time.time()*1000)
        for t in tickers:
            sym = t.contract.symbol
            px = float(t.last) if t.last else None
            size = float(t.lastSize) if t.lastSize else None
            if px is None or not size or size <= 0:
                continue
            _MET["agg_tick_ingested"].labels(symbol=sym).inc()

            ts_ms = now_ms
            win_start = floor_5s(ts_ms)
            win_end = win_start + 5000
            key = (sym, win_start)
            new_bucket = False
            b = self.bucket.get(key)
            if not b:
                new_bucket = True
                b = BarBucket(sym, win_start, win_end)
                b.bar_seq = self.last_bar_seq.get(sym, 0) + 1
                self.last_bar_seq[sym] = b.bar_seq
                self.bucket[key] = b
                self.emit_schedule[key] = win_end + self.cfg.emit_buffer_ms
                _MET["agg_buckets_active"].inc()
            b.on_trade(px, size)

            if ts_ms > b.ts_end_ms + self.cfg.emit_buffer_ms:
                b.late_count += 1
                _MET["agg_late_ticks"].labels(symbol=sym).inc()

            try:
                await self._persist_bucket(b)
            except Exception:
                _MET["agg_redis_fail"].inc()
                # 不中断聚合
```

在 `_flusher()` 发 bar / gap：

```python
    async def _flusher(self):
        while True:
            await asyncio.sleep(0.05)
            now_ms = int(time.time()*1000)
            to_emit = [(k,b) for k,b in list(self.bucket.items()) if self.emit_schedule.get(k, 1e18) <= now_ms]
            for key, b in to_emit:
                rth_flag = is_rth(b.ts_end_ms/1000.0, self.cfg.calendar_tz) if self.cfg.rth_only else True
                msg = b.to_msg(rth_flag, source="AGG", calc_mode="realtime")
                try:
                    await self._kafka_send(self.cfg.topic_bar, f"{b.symbol}:{b.ts_start_ms}", msg)
                    _MET["agg_bar_emit"].labels(symbol=b.symbol).inc()
                    emit_latency = max(0, now_ms - b.ts_end_ms)  # ms from window end to emit
                    _MET["agg_bar_emit_latency"].labels(symbol=b.symbol).observe(emit_latency)
                except Exception:
                    _MET["agg_kafka_fail"].inc()
                self.watermark_ms[b.symbol] = max(self.watermark_ms.get(b.symbol, 0), b.ts_end_ms)
                self.bucket.pop(key, None); self.emit_schedule.pop(key, None)
                await self._delete_bucket(b)
                _MET["agg_buckets_active"].dec()

            for sym in self.cfg.symbols:
                wm = self.watermark_ms.get(sym)
                if not wm: continue
                if now_ms - wm > 20_000:
                    gap = {
                        "schema_v":1,"metadata":{"event_id":str(uuid.uuid4()),"ts_us":now_us(),"producer":"agg5s"},
                        "data":{"symbol":sym,"last_bar_ts":int(wm/1000),"now_ts":int(now_ms/1000),"reason":"no_bar_20s"}
                    }
                    try:
                        await self._kafka_send(self.cfg.topic_gap, sym, gap)
                        _MET["agg_gap_events"].labels(symbol=sym, reason="no_bar_20s").inc()
                    except Exception:
                        _MET["agg_kafka_fail"].inc()
```

在 `_loop()` 处记录 batch 大小（按 topic）；在 `_flush()` 记录耗时、行数、重试：

```python
    async def _loop(self):
        try:
            while True:
                msg = await self.consumer.getone()
                row = self._to_row(msg.value)
                if row is not None:
                    self._batch.append(row)
                    self._batch_offsets.append((msg.topic, msg.partition, msg.offset))
                    _MET["chw_batch_size"].labels(topic=msg.topic).observe(1)
                await self._maybe_flush()
        except asyncio.CancelledError:
            pass
```

```python
    async def _flush(self):
        if not self._batch:
            self._last_flush = time.time()
            return
        rows = self._batch; offsets = self._batch_offsets
        self._batch = []; self._batch_offsets = []
        sql = self._mk_insert_sql(rows)

        t0 = time.time()
        backoff = self.cfg.base_backoff_s
        for attempt in range(self.cfg.max_retry):
            try:
                resp = await self.client.post(f"/?database={self.cfg.ch_db}", content=sql)
                resp.raise_for_status()
                await self._commit_offsets(offsets)
                _MET["chw_rows_ok"].inc(len(rows))
                _MET["chw_flush"].inc()
                _MET["chw_flush_latency"].observe((time.time()-t0)*1000.0)
                self._last_flush = time.time()
                return
            except Exception:
                _MET["chw_retry"].inc()
                await asyncio.sleep(backoff)
                backoff *= 2
        _MET["chw_rows_err"].inc(len(rows))
        _MET["chw_flush_latency"].observe((time.time()-t0)*1000.0)
        self._last_flush = time.time()
```

# 五、快速验证 & 面板要点

**本地验证**

1. 分别启动聚合器、CH Writer、策略引擎后，打开：

   * `http://localhost:9101/metrics`（agg）
   * `http://localhost:9102/metrics`（ch-writer）
   * `http://localhost:9103/metrics`（engine）
2. 观察是否出现：

   * `agg_bar_emit_total{symbol="AAPL"}`
   * `agg_bar_emit_latency_ms_bucket{...}`
   * `chw_rows_ok_total`
   * `engine_strategy_exec_latency_ms_bucket{strategy="ORBBreakout"}`

**Grafana 建议 6 张核心图**

* `rate(agg_bar_emit_total[5m]) by (symbol)`（5s Bar 产出速率）
* `histogram_quantile(0.95, sum(rate(agg_bar_emit_latency_ms_bucket[5m])) by (le))`（聚合 p95）
* `histogram_quantile(0.95, sum(rate(chw_flush_latency_ms_bucket[5m])) by (le))`（CH 写入 p95）
* `sum(rate(chw_rows_ok_total[5m]))` vs `sum(rate(chw_rows_err_total[5m]))`（写入健康度）
* `histogram_quantile(0.95, sum(rate(engine_strategy_exec_latency_ms_bucket[10m])) by (le, strategy))`（策略 p95）
* `avg_over_time(engine_strategy_actual_cadence_s{strategy="ORBBreakout"}[10m])`（节奏漂移）

# 六、为什么这一票优先

* 这是**最小侵入**的改动：只加几行代码，但能立刻看见链路瓶颈。
* 指标名称与我们之前的 SLO/PromQL 完全对齐，后面做告警直接复用。
* 有了它，你可以**自信地**继续推进 Time&Sales 精确时间戳、执行内核节流/OCO巡检等重活，因为任何回归都会在面板上“报警”。
