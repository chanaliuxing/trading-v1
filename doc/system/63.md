# 2) ClickHouse 表结构

## 2.1 Fills（实时成交落地）

```sql
CREATE TABLE fills_live (
  ts        DateTime64(3),                       -- 成交时间（交易所/网关时间）
  strategy_id LowCardinality(String),
  symbol    LowCardinality(String),
  side      Enum8('BUY'=1,'SELL'=-1),
  qty       Int64,
  price     Float64,
  fee       Float64 DEFAULT 0,
  order_id  String,
  decision_id String,
  param_hash FixedString(10),
  liquidity LowCardinality(String) DEFAULT 'UNK', -- Taker/Maker/UNK
  ver       UInt64,
  ingest_ts DateTime64(3) DEFAULT now64()
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY toDate(ts)
ORDER BY (strategy_id, symbol, ts, order_id)
SETTINGS index_granularity = 4096;
```

## 2.2 逐笔 PnL 账本（Python 作业写入）

```sql
-- 每一笔成交之后的账本状态（加权平均法）
CREATE TABLE pnl_fills (
  ts        DateTime64(3),
  strategy_id LowCardinality(String),
  symbol    LowCardinality(String),
  side      Int8,                                 -- +1 买 / -1 卖
  qty       Int64,
  price     Float64,
  fee       Float64,
  pos_qty_after  Int64,                           -- 该笔后净仓
  avg_cost_after Float64,                         -- 该笔后仓位均价（含方向符号）
  realized_pnl   Float64,                         -- 该笔带来的已实现盈亏（含 fee）
  cash_after     Float64,                         -- 该笔后现金
  order_id  String,
  decision_id String,
  param_hash FixedString(10),
  ver       UInt64,
  ingest_ts DateTime64(3) DEFAULT now64()
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY toDate(ts)
ORDER BY (strategy_id, symbol, ts, order_id)
SETTINGS index_granularity = 4096;
```

## 2.3 5s 持仓快照（Python 作业写入）

```sql
CREATE TABLE pnl_pos_5s (
  ts_end    DateTime,                              -- 对齐 bars_5s.ts_end
  strategy_id LowCardinality(String),
  symbol    LowCardinality(String),
  pos_qty   Int64,
  avg_cost  Float64,
  mark_px   Float64,                               -- 取 bars_5s.mid_last 或 close
  mtm       Float64,                               -- 市值 = pos_qty * mark_px
  upl       Float64,                               -- 未实现盈亏 = pos_qty * (mark_px - avg_cost)
  ver       UInt64,
  ingest_ts DateTime64(3) DEFAULT now64()
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY toDate(ts_end)
ORDER BY (strategy_id, symbol, ts_end)
SETTINGS index_granularity = 4096;
```

## 2.4 日聚合（SummingMergeTree）

```sql
CREATE TABLE pnl_day (
  d         Date,
  strategy_id LowCardinality(String),
  symbol    LowCardinality(String),
  sector    LowCardinality(String),
  traded_shares UInt64,             -- 当日成交量（我们自己的）
  traded_notional Float64,          -- 成交金额（绝对值）
  realized_pnl   Float64,
  fees           Float64,
  upl_close      Float64,           -- 收盘时未实现
  nav_close      Float64,           -- 收盘 NAV（start_cash + RPL + UPL）
  cap_used_close Float64,           -- 收盘资金使用（绝对市值）
  mkt_volume     Float64,           -- 当日市场总成交量（bars_5s 求和）
  participation  Float64            -- 参与率 = traded_shares / mkt_volume
)
ENGINE = SummingMergeTree()
PARTITION BY d
ORDER BY (strategy_id, symbol, d)
SETTINGS index_granularity = 4096;
```

# 3) 逐笔账本作业（Python 骨架）

> 要点：**方向有符号**的加权均价；卖出/回补时**计入 realized**；fee 计入现金 & RPL。
> 该作业可订阅 `events.fills.v1` 或轮询 PG `fills`，再写入 `pnl_fills` 与 `pnl_pos_5s`。

```python
# -*- coding: utf-8 -*-
from collections import defaultdict
from dataclasses import dataclass

@dataclass
class Lot:
    pos: int = 0               # 净仓（带符号：多>0，空<0）
    avg: float = 0.0           # 仓位均价（同符号，空头为负均价以便统一计算）
    cash: float = 0.0          # 现金（策略资金账）

def step_fill(state: Lot, side: int, qty: int, px: float, fee: float):
    """
    side: +1 买入，-1 卖出；avg 与 pos 同号，空头 avg<0。
    返回：pos_after, avg_after, realized_pnl_delta, cash_after
    """
    pos0, avg0, cash0 = state.pos, state.avg, state.cash
    # 方向与数量规范化
    qty = abs(qty) * side

    realized = 0.0
    # 情况 A：同向加仓（符号相同或从 0 到非 0）
    if pos0 == 0 or (pos0 > 0 and qty > 0) or (pos0 < 0 and qty < 0):
        # 新均价 = (旧仓金额 + 新成交金额) / 新仓量
        notional_old = abs(pos0) * abs(avg0)
        notional_new = abs(qty)  * abs(px)
        pos1 = pos0 + qty
        avg1 = (notional_old + notional_new) / max(1, abs(pos1))
        avg1 = avg1 * (1 if pos1 > 0 else -1)   # 方向符号
    else:
        # 情况 B：减仓/反手；对冲部分计入 realized
        # 以旧均价结算被对冲数量
        close_qty = min(abs(pos0), abs(qty)) * (1 if pos0>0 else -1)  # 被对冲的方向 = 原仓方向
        # 多仓被卖出：realized += (卖价-多均价)*数量；空仓被买回：realized += (空均价绝对值-买价)*数量
        if pos0 > 0:   # 平多
            realized += (px - abs(avg0)) * abs(close_qty)
        else:          # 平空
            realized += (abs(avg0) - px) * abs(close_qty)

        pos1 = pos0 + qty
        if pos1 == 0:
            avg1 = 0.0
        elif (pos0 > 0 and pos1 > 0) or (pos0 < 0 and pos1 < 0):
            # 部分减仓后仍同向，均价不变
            avg1 = avg0
        else:
            # 反手：剩余部分来自本次剩余成交
            residual_qty = abs(pos1)
            residual_px  = px
            avg1 = residual_px * (1 if pos1>0 else -1)

    cash1 = cash0 - side*abs(qty)*px - fee   # 买入减现金，卖出加现金（side=-1 → 减号变加）
    realized -= fee                           # 手续费计入已实现
    # 更新状态
    state.pos, state.avg, state.cash = pos1, avg1, cash1
    return pos1, avg1, realized, cash1
```

**将逐笔结果写入 CH：**

* 每笔写 `pnl_fills` 一行（含 `pos_qty_after/avg_cost_after/realized_pnl/cash_after`）。
* 每 1s 对齐最近一根 `bars_5s.ts_end`，用最新 `pos/avg` + `mark_px` 生成 `pnl_pos_5s` 快照（`upl = pos*(mark-avg)`）。

# 4) T+0（盘中）核心 SQL

## 4.1 当前持仓 & NAV & 资金使用率

```sql
-- 选最新一条 5s 快照作为当前
WITH last_pos AS (
  SELECT strategy_id, symbol,
         anyLast(pos_qty)  AS pos_qty,
         anyLast(avg_cost) AS avg_cost,
         anyLast(mark_px)  AS mark_px,
         anyLast(upl)      AS upl,
         anyLast(ts_end)   AS ts_end
  FROM pnl_pos_5s
  WHERE ts_end > now() - INTERVAL 20 MINUTE
  GROUP BY strategy_id, symbol
),
rpl_today AS (
  SELECT strategy_id, symbol, sum(realized_pnl) AS rpl
  FROM pnl_fills
  WHERE toDate(ts) = toDate(now())
  GROUP BY strategy_id, symbol
),
cap_cfg AS (  -- 你的策略资金池上限（可从 PG/配置表同步到 CH）
  SELECT 'meanrev_v2:A' AS strategy_id, 150000.0 AS cap_usd
)
SELECT
  p.strategy_id,
  groupArray((symbol, pos_qty, mark_px, upl)) AS positions,
  sum(p.pos_qty * p.mark_px) AS gross_notional,          -- 市值（带方向）
  sum(abs(p.pos_qty * p.mark_px)) AS cap_used,           -- 资金使用（绝对）
  sum(coalesce(r.rpl,0)) AS realized_today,
  sum(p.upl) AS upl_now,
  (sum(coalesce(r.rpl,0)) + sum(p.upl)) AS pnl_total,
  (sum(abs(p.pos_qty * p.mark_px)) / any(cap.cap_usd)) AS cap_usage_ratio
FROM last_pos p
LEFT JOIN rpl_today r USING (strategy_id, symbol)
LEFT JOIN cap_cfg cap USING (strategy_id)
GROUP BY p.strategy_id;
```

## 4.2 参与率（Participation Rate）

```sql
-- 我们自家成交股数 vs 市场总量（当日）
WITH my_traded AS (
  SELECT strategy_id, symbol, sum(abs(qty)) AS shares
  FROM fills_live
  WHERE toDate(ts) = toDate(now())
  GROUP BY strategy_id, symbol
),
mkt_volume AS (
  SELECT symbol, sum(volume) AS shares
  FROM bars_5s
  WHERE toDate(ts_start) = toDate(now())
  GROUP BY symbol
)
SELECT
  m.strategy_id, m.symbol,
  m.shares AS my_shares,
  coalesce(v.shares, 0) AS mkt_shares,
  m.shares / nullIf(v.shares,0) AS participation
FROM my_traded m
LEFT JOIN mkt_volume v USING (symbol)
ORDER BY participation DESC;
```

## 4.3 滑点（相对基准中价或下一根 VWAP）

```sql
-- 基准：成交当根的 mid_last（或下一根 bar 的 vwap，更保守）
WITH ref AS (
  SELECT f.order_id, anyLast(b.mid_last) AS ref_px
  FROM fills_live f
  JOIN bars_5s b
    ON f.symbol=b.symbol
   AND b.ts_start <= f.ts
   AND b.ts_end   >  f.ts
  GROUP BY f.order_id
)
SELECT
  strategy_id, symbol,
  count() AS fills,
  quantile(0.5)(
    10000 * ( if(side=1, price - r.ref_px, r.ref_px - price) / nullIf(r.ref_px,0) )
  ) AS slippage_bps_median,
  quantile(0.9)(
    10000 * ( if(side=1, price - r.ref_px, r.ref_px - price) / nullIf(r.ref_px,0) )
  ) AS slippage_bps_p90
FROM fills_live f
LEFT JOIN ref r USING (order_id)
WHERE toDate(ts)=toDate(now())
GROUP BY strategy_id, symbol
ORDER BY slippage_bps_median DESC;
```

# 5) T+1（日终）核心 SQL

## 5.1 每策略/每标的日报（RPL/UPL/NAV/参与率）

```sql
WITH eod_mark AS (
  SELECT symbol, anyLast(close) AS px_close
  FROM bars_5s
  WHERE toDate(ts_start) = toDate(today())
  GROUP BY symbol
),
eod_pos AS (
  SELECT strategy_id, symbol,
         anyLast(pos_qty) AS pos_qty,
         anyLast(avg_cost) AS avg_cost
  FROM pnl_pos_5s
  WHERE toDate(ts_end)=toDate(today())
  GROUP BY strategy_id, symbol
),
rpl_day AS (
  SELECT strategy_id, symbol, sum(realized_pnl) AS rpl, sum(fee) AS fees
  FROM pnl_fills
  WHERE toDate(ts)=toDate(today())
  GROUP BY strategy_id, symbol
),
mkt_volume AS (
  SELECT symbol, sum(volume) AS mkt_shares
  FROM bars_5s
  WHERE toDate(ts_start)=toDate(today())
  GROUP BY symbol
),
my_traded AS (
  SELECT strategy_id, symbol, sum(abs(qty)) AS my_shares, sum(abs(qty*price)) AS my_notional
  FROM fills_live
  WHERE toDate(ts)=toDate(today())
  GROUP BY strategy_id, symbol
),
dim AS ( SELECT * FROM dim_symbol )
SELECT
  e.strategy_id, e.symbol, d.sector,
  r.rpl, r.fees,
  e.pos_qty,
  e.avg_cost,
  m.px_close,
  (e.pos_qty * (m.px_close - e.avg_cost)) AS upl_close,
  my.my_notional AS traded_notional,
  my.my_shares   AS traded_shares,
  v.mkt_shares,
  my.my_shares / nullIf(v.mkt_shares,0) AS participation
FROM eod_pos e
LEFT JOIN eod_mark m USING (symbol)
LEFT JOIN rpl_day r USING (strategy_id, symbol)
LEFT JOIN my_traded my USING (strategy_id, symbol)
LEFT JOIN mkt_volume v USING (symbol)
LEFT JOIN dim d USING (symbol)
ORDER BY e.strategy_id, d.sector, e.symbol;
```

## 5.2 行业暴露（收盘）

```sql
SELECT
  d.sector,
  sum(abs(e.pos_qty * m.px_close)) AS notional_abs_close,
  sum(e.pos_qty * m.px_close)      AS net_notional_close,
  sum(e.pos_qty * (m.px_close - e.avg_cost)) AS upl_close
FROM pnl_pos_5s e
JOIN (
  SELECT symbol, anyLast(close) AS px_close
  FROM bars_5s
  WHERE toDate(ts_start)=toDate(today())
  GROUP BY symbol
) m USING(symbol)
JOIN dim_symbol d USING(symbol)
WHERE toDate(e.ts_end)=toDate(today())
GROUP BY d.sector
ORDER BY notional_abs_close DESC;
```

# 6) 每日复盘脚本骨架（生成 Markdown 文本）

```python
def daily_review(ch, strategy_id):
    # 1) 读取 T+1 日报聚合
    day = ch.query("/* 上面 5.1 SQL 简化版，汇总到策略级 */ ...")
    # 2) Top5 贡献 / 拖累
    winners = ch.query("""
      SELECT symbol, (rpl + upl_close) AS pnl
      FROM <日报明细视图> WHERE strategy_id=%(sid)s
      ORDER BY pnl DESC LIMIT 5
    """, {"sid":strategy_id})
    losers = ch.query("""
      SELECT symbol, (rpl + upl_close) AS pnl
      FROM <日报明细视图> WHERE strategy_id=%(sid)s
      ORDER BY pnl ASC LIMIT 5
    """, {"sid":strategy_id})
    # 3) 参与率与滑点异常
    slip = ch.query("""
      SELECT symbol, slippage_bps_median AS bps, fills
      FROM <滑点视图>
      WHERE strategy_id=%(sid)s AND d=today() ORDER BY bps DESC LIMIT 5
    """, {"sid":strategy_id})

    md = []
    md.append(f"## {strategy_id} T+1 日报")
    md.append(f"- 当日 RPL: {day['rpl']:.2f}, UPL收盘: {day['upl']:.2f}, NAV收盘: {day['nav']:.2f}")
    md.append(f"- 资金池使用峰值: {day['cap_peak']:.2f}, 参与率中位: {day['participation_p50']:.4f}")
    md.append("### Top5 贡献")
    for r in winners: md.append(f"- {r['symbol']}: {r['pnl']:.2f}")
    md.append("### Top5 拖累")
    for r in losers: md.append(f"- {r['symbol']}: {r['pnl']:.2f}")
    md.append("### 滑点异常")
    for r in slip: md.append(f"- {r['symbol']}: {r['bps']:.1f} bps (fills={r['fills']})")
    return "\n".join(md)
```
