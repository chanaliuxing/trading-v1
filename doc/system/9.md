# 1) 引擎主循环（伪代码）

```python
class Engine:
    def __init__(self, cfg):
        self.cfg = cfg
        self.ch = ClickHouseClient(cfg.ch_dsn)
        self.prod = KafkaProducer(cfg.kafka)
        self.cons = KafkaConsumerGroup(cfg.kafka, topics=[
            "md.bar.5s.v2",
            "risk.decision.v2",
            "exec.order.status.v2",
            "exec.fill.v2",
            "portfolio.snapshot.v2",
        ])
        self.health = HealthProbe()  # pre-flight 里已经检查通过
        self.token_bucket = GlobalLimiter(qps=cfg.scheduler.global_qps)
        self.strategies = {}  # id -> StrategyRuntime

    def start(self):
        # 1) 装载策略与运行时
        for scfg in load_strategy_cfgs(self.cfg):
            srt = StrategyRuntime(scfg, self.ch, self.prod, self.token_bucket, self.health)
            self.strategies[scfg.id] = srt

        # 2) 订阅消费线程/协程
        spawn(self._consume_md_5s)
        spawn(self._consume_risk_decision)
        spawn(self._consume_exec_status)
        spawn(self._consume_exec_fill)
        spawn(self._consume_portfolio)

        # 3) 启动每策略调度循环
        for srt in self.strategies.values():
            spawn(srt.loop)

        # 4) 监控与优雅关闭
        wait_forever()
```

# 2) Per-Strategy Runtime（调度 + ReadyCheck + 调用策略）

```python
class StrategyRuntime:
    def __init__(self, scfg, ch, producer, token_bucket, health):
        self.cfg = scfg
        self.ch = ch
        self.kprod = producer
        self.tb = token_bucket
        self.health = health
        self.last_exec_ts = 0.0
        self.next_cadence = scfg.cadence.default_s
        self.snapshot = {}  # feat_daily_snapshot_v2 缓存
        self.md_cache = RingCache(capacity=2048)  # 按 symbol 收最近 5s bar
        self.portfolio = PortfolioView()
        self.open_orders = {}  # order_id->OrderUpdate
        self.strategy = load_strategy_class(scfg)  # StrategyAlgorithm 子类
        self.params_hash = scfg.params_hash
        self.ready = False

    async def loop(self):
        # 初始化：订阅、加载快照、策略 initialize()
        self._hydrate_snapshot()
        self.strategy.initialize(ctx=self._ctx())

        while not shutting_down():
            # 等待 cadence
            await sleep_until(self.last_exec_ts + self.next_cadence)

            # 系统健康/限流
            if not self.health.ok():
                self._skip("SYSTEM_DEGRADED"); self._backoff(); continue
            if not self.tb.take(1):
                self._skip("RATE_LIMIT"); self._backoff(); continue

            # ReadyCheck：数据/特征/时窗
            ok, reason, bars_5s_batch = await self._ready_check()
            if not ok:
                self._skip(reason); self._backoff(); continue

            # 调用策略
            t0 = now()
            try:
                self.strategy.on_microbar(ctx=self._ctx(), bars_5s=bars_5s_batch)
                metrics.observe("strategy.exec.latency_ms", (now()-t0)*1000, tags={"id": self.cfg.id})
            except Exception as e:
                self._handle_err(e, "on_microbar")
            finally:
                self.last_exec_ts = now()
                self._adapt_cadence()

    def _ctx(self) -> "Context":
        # 轻量对象：只读视图 + 交易 API 代理（发 Kafka）
        return Context(
            engine=self,
            strategy_id=self.cfg.id,
            portfolio=self.portfolio,
            open_orders=list(self.open_orders.values()),
            ch=self.ch,
            kprod=self.kprod,
            params_hash=self.params_hash,
            cadence_bounds=self.cfg.cadence
        )
```

# 3) ReadyCheck（数据就绪与“智能跳过”）

```python
async def _ready_check(self) -> tuple[bool, str, list[Bar5s] | None]:
    # 3.1 最近 5s bar 是否齐备
    bars_batch = []
    for sym in self.cfg.symbols:
        # 本地缓存优先，不足则 CH 查询
        bars = self.md_cache.get(sym, n=self.cfg.deps.bars_5s_required)
        if len(bars) < self.cfg.deps.bars_5s_required:
            bars = self._query_bars(sym, "5s", self.cfg.deps.bars_5s_required)
        if len(bars) < self.cfg.deps.bars_5s_required:
            return False, "DATA_GAP_5S", None
        if not all(b.rth for b in bars):  # RTH 校验（若策略要求）
            return False, "DATA_RTH_MISMATCH", None
        bars_batch.extend(bars[-min(12, len(bars)):])  # 传给策略最近 1 分钟窗口

    # 3.2 1m 依赖（可选）
    if self.cfg.deps.bars_1m_required:
        for sym in self.cfg.symbols:
            if not self._has_recent_bars(sym, "1m", self.cfg.deps.bars_1m_required):
                return False, "DATA_GAP_1M", None

    # 3.3 特征新鲜度（intraday ≤ 15s）
    intr = self.cfg.deps.intraday_features
    if intr and not self._intraday_fresh(self.cfg.symbols, intr, max_staleness_s=15):
        return False, "FEATURE_INTRADAY_STALE", None

    # 3.4 快照（收市后产物）
    snap = self.cfg.deps.snapshot_features
    if snap and not self._has_snapshot(self.cfg.symbols, snap):
        return False, "FEATURE_SNAPSHOT_MISSING", None

    return True, "", bars_batch
```

**CH 查询模板（与前面规范一致）**

* 最近 N 根 5s 是否齐全：

```sql
SELECT count() FROM qsys.bar_5s_v2
WHERE symbol = {sym} AND ts_start >= now() - INTERVAL {N}*5 second AND rth=1;
```

* 最近 N 根 1m：

```sql
SELECT count() FROM qsys.bar_1m_v2
WHERE symbol = {sym} AND ts_start >= now() - INTERVAL {N} minute AND rth=1;
```

* Intraday 特征新鲜度：

```sql
SELECT max(ts) FROM qsys.feat_intraday_v2
WHERE symbol IN {syms} AND name IN {names};
```

* Daily Snapshot：

```sql
SELECT 1 FROM qsys.feat_daily_snapshot_v2 WHERE asof_date=today() AND symbol IN {syms};
```

```python
class Context:
    def __init__(self, engine, strategy_id, portfolio, open_orders, ch, kprod, params_hash, cadence_bounds):

        self._pf = portfolio; self._oo = open_orders
        self._ch = ch; self._k = kprod
        self._params_hash = params_hash
        self._cadence = cadence_bounds

    # ---- 只读 ----
    def now(self): return now()
    @property
    def portfolio(self): return self._pf
    @property
    def open_orders(self): return self._oo

    # ---- 数据访问（封装 CH）----
    def history(self, symbol, *, resolution, limit): return ch_read_bars(self._ch, symbol, resolution, limit)
    def features_intraday(self, symbol, names, *, max_staleness_s=15): return ch_read_intraday(self._ch, symbol, names, max_staleness_s)
    def features_snapshot(self, symbol): return ch_read_snapshot(self._ch, symbol)

    # ---- 下单（发布 engine.order.request.v2）----
    def limit_order(self, symbol, qty, limit_px, *, tif="DAY", notional_hint_usd=None, tags=[]):

            "event_id": str(uuid4()),
```

# 6) 自适应节奏与退避（与前文一致）

```python
def _adapt_cadence(self):
    if not self.cfg.cadence.adaptive:
        self.next_cadence = self.cfg.cadence.default_s; return
    vol = recent_micro_atr(self.cfg.symbols, self.ch)  # CH 查询/本地估算
    load = system_load()
    cad = self.cfg.cadence.default_s
    if vol < VOL_LOW: cad = min(self.cfg.cadence.max_s, cad * 2)
    if load > LOAD_HIGH: cad = max(self.cfg.cadence.min_s, cad + 5)
    self.next_cadence = clamp(cad, self.cfg.cadence.min_s, self.cfg.cadence.max_s)

def _skip(self, reason):
    metrics.inc("strategy.skips_total", tags={"id": self.cfg.id, "reason": reason})

def _backoff(self):
    self.next_cadence = max(self.cfg.cadence.min_s, self.cfg.cadence.default_s)
```

# 9) 验收用例（引擎维度）

1. **5s 独立节奏**：两个策略分别 5s/20s，日志中 `actual_cadence_s` 符合预期且互不影响。
2. **智能跳过**：拔掉部分 symbol 的 5s 流，`DATA_GAP_5S` 增加，策略不触发；恢复后自动继续。
3. **事件幂等**：同一回调内重复触发下单，`engine.order.request.v2` 只发布一次（引擎去重）。
4. **高负载退避**：模拟 CH 慢查询/CPU 高，`actual_cadence_s` 自适应上调。
5. **回调预算**：`on_microbar` 人为 sleep 超预算，观测被记账并拉长 cadence（守门生效）。
