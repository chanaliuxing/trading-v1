# ① 最小 “RMS → Cache” 消费者（把 `reg.market_state.v1` 写入快照缓存）

> 作用：订阅标准事件 **`reg.market_state.v1`**，把 `luld/ssr/halt/tick/lot` 写进我们之前做好的 `NBBOSnapshotCache.apply_status()`，让 Reg-Lite 风控即时可用。
> 说明：示例用 **aiokafka** + JSON/Avro 皆可；如果你已经用了 confluent-kafka，也只需改 2–3 行消费接口。

```python
# -*- coding: utf-8 -*-
import asyncio, json, time
from typing import Optional, Callable

def _now_us(): return int(time.time()*1e6)

class RMS2CacheConsumer:
    """
    订阅 reg.market_state.v1 → 写 nbbo_cache.apply_status()
    - 兼容两种负载：JSON dict 或 Avro 解码后的 dict
    - key: symbol (bytes/str)，value: {schema_v, metadata, data}
    """
    def __init__(self, kafka_bootstrap:str, group_id:str, topic:str, nbbo_cache, avro_decoder: Optional[Callable]=None):
        self.bootstrap = kafka_bootstrap
        self.group_id = group_id
        self.topic = topic
        self.cache = nbbo_cache
        self.avro_decoder = avro_decoder
        self._task = None
        self._stopped = False

    async def start(self):
        from aiokafka import AIOKafkaConsumer
        self.consumer = AIOKafkaConsumer(
            self.topic,
            bootstrap_servers=self.bootstrap,
            group_id=self.group_id,
            enable_auto_commit=True,
            auto_offset_reset="latest",
            value_deserializer=lambda v: v,  # 先拿 raw bytes
            key_deserializer=lambda k: k
        )
        await self.consumer.start()
        self._task = asyncio.create_task(self._loop())

    async def stop(self):
        self._stopped = True
        if self._task: self._task.cancel()
        await self.consumer.stop()

    def _decode(self, raw: bytes):
        if self.avro_decoder:
            return self.avro_decoder(raw)  # 交给外部 Avro 解码器
        try:
            return json.loads(raw.decode("utf-8"))
        except Exception:
            return None

    async def _loop(self):
        try:
            async for msg in self.consumer:
                if self._stopped: break
                sym = msg.key.decode("utf-8") if isinstance(msg.key, (bytes, bytearray)) else str(msg.key)
                v = self._decode(msg.value)
                if not v or "data" not in v: 
                    continue
                d = v["data"]
                try:
                    await self.cache.apply_status(
                        sym,
                        luld_low=d.get("luld_low"),
                        luld_high=d.get("luld_high"),
                        ssr_active=bool(d.get("ssr_active", False)),
                        halted=bool(d.get("halted", False)),
                        # 可选：tick/lot 也可能在 RMS 里下发
                        tick=d.get("tick_size"),
                        lot=d.get("lot_size")
                    )
                except Exception:
                    # 记录告警但不影响消费循环
                    pass
        except asyncio.CancelledError:
            pass
```

```python
    kafka_bootstrap="localhost:9092",
    group_id="state-bridge",
    topic="reg.market_state.v1",

)
# await rms_consumer.start()  # 异步启动
```

> 若你使用 **Confluent Avro**，`avro_decoder` 形如：
>
> ```python
> from confluent_kafka.schema_registry import SchemaRegistryClient
> from confluent_kafka.serialization import MessageField, SerializationContext
> from confluent_kafka.schema_registry.avro import AvroDeserializer
>
> sr = SchemaRegistryClient({"url": "http://schema-registry:8081"})
> avro = AvroDeserializer(sr, schema_str=REG_MARKET_STATE_AVRO)
> def avro_decode(raw: bytes):
>     return avro(raw, SerializationContext("reg.market_state.v1", MessageField.VALUE))
> ```
>
> 然后把 `avro_decoder=avro_decode` 传进去即可。

# ② 合约主数据加载器（`contract.master.v1` 生产者 + 落 Cache）

> 作用：周期或启动时拉取**权威合约元数据**（tickSize/lotSize/multiplier/secType），产出标准事件 **`contract.master.v1`**，并**直接写入快照缓存**（tick/lot）作为 Reg-Lite 的唯一口径。
> 数据来源：IB `contractDetails`（ib-insync）、或你已有的主数据表/第三方接口。

```python
# -*- coding: utf-8 -*-
import asyncio, time
from dataclasses import dataclass
from typing import List, Optional, Dict, Any

def _now_us(): return int(time.time()*1e6)

@dataclass
class ContractInfo:
    symbol: str
    con_id: Optional[int] = None
    exchange: str = "SMART"
    currency: str = "USD"
    tick_size: Optional[float] = None
    lot_size: Optional[float] = 1.0
    multiplier: Optional[float] = None
    sec_type: str = "STK"

class ContractMasterLoader:
    """
    - 从 IB 或自有主数据源拉取合约主数据
    - 产出 contract.master.v1 事件（可选）
    - 同时把 tick/lot 写入 nbbo_cache.apply_status()，确保风控 tick/lot 口径唯一
    """
    def __init__(self, nbbo_cache, kafka_producer=None, topic:str="contract.master.v1"):
        self.cache = nbbo_cache
        self.producer = kafka_producer
        self.topic = topic

    async def fetch_from_ib(self, ib, symbols: List[str]) -> List[ContractInfo]:
        """
        使用 ib-insync 的 reqContractDetails 拉取合约元数据。
        注意：minTick 通常在 ContractDetails.minTick；lotSize 需要从规则或交易所信息推断，IB 对股票一般为1。
        """
        from ib_insync import Stock
        out: List[ContractInfo] = []
        for sym in symbols:
            cds = await ib.reqContractDetails(Stock(sym, "SMART", "USD"))
            if not cds: 
                continue
            cd = cds[0]
            tick = float(getattr(cd, "minTick", 0.01) or 0.01)
            mult = float(getattr(cd.contract, "multiplier", 1) or 1)
            # lotSize：IB 对美股常为 1；若你的交易所规则另有最小下单量，可在外部规则覆盖
            lot = 1.0
            out.append(ContractInfo(
                symbol=sym,
                con_id=int(cd.contract.conId),
                exchange=str(cd.contract.primaryExchange or "SMART"),
                currency=str(cd.contract.currency or "USD"),
                tick_size=tick,
                lot_size=lot,
                multiplier=mult,
                sec_type=str(cd.contract.secType or "STK")
            ))
        return out

    async def apply_to_cache(self, info: ContractInfo):
        await self.cache.apply_status(
            info.symbol,
            tick=info.tick_size,
            lot=info.lot_size
        )

    async def emit_contract_event(self, info: ContractInfo):
        if not self.producer: 
            return
        evt = {
            "schema_v": 1,
            "metadata": {"ts_us": _now_us(), "producer": "contract-master"},
            "data": {
                "symbol": info.symbol,
                "con_id": info.con_id,
                "exchange": info.exchange,
                "currency": info.currency,
                "tick_size": info.tick_size,
                "lot_size": info.lot_size,
                "multiplier": info.multiplier,
                "sec_type": info.sec_type
            }
        }
        key = info.symbol.encode("utf-8")
        await self.producer.send(self.topic, key=key, value=evt)

    async def run_once(self, ib=None, symbols: Optional[List[str]]=None, source_cb=None):
        """
        - ib 模式：传入 ib + symbols
        - 自定义源：传入 source_cb -> List[ContractInfo]
        """
        infos: List[ContractInfo] = []
        if source_cb:
            infos = await source_cb()
        elif ib and symbols:
            infos = await self.fetch_from_ib(ib, symbols)
        else:
            return

        for info in infos:
            # 1) 写入缓存（tick/lot）
            await self.apply_to_cache(info)
            # 2) 广播事件（可选）
            await self.emit_contract_event(info)

    async def run_periodic(self, interval_min: float, ib=None, symbols: Optional[List[str]]=None, source_cb=None):
        while True:
            try:
                await self.run_once(ib=ib, symbols=symbols, source_cb=source_cb)
            except Exception:
                # 记录警告即可，不中断循环
                pass
            await asyncio.sleep(interval_min * 60)
```

**注意事项（主数据口径唯一）：**

* **tick/lot** 以 `contract.master` 为唯一权威口径；行情/券商若带来不同口径，不要在 Reg-Lite 直接使用，以免前后不一致。
* 如你的交易标的包含期权/期货，lot 和 multiplier 需要按合约规则填充（可以扩展 `ContractInfo`）。

## 自检 / 验收（3 分钟）

1. 手动向 Kafka 写入一条 `reg.market_state.v1`（或模拟 JSON 值），确认 `nbbo_cache.read(symbol)` 能读到 `luld/ssr/halt`。
2. 跑一次 `ContractMasterLoader.run_once()`，确认：

   * `nbbo_cache.read(symbol)` 返回的 `tick/lot` 被更新；
   * （可选）Kafka 收到一条 `contract.master.v1`。
3. Reg-Lite：随手发一笔 **SELL_SHORT MKT** + `ssr_active=True` 场景，风控**硬拒**（`rule=ssr`）；
   发一笔 **LMT** 且超出 `luld_high`，风控**硬拒**（`rule=luld_band`）；
   发一笔 **LMT**，价格不贴 tick，风控**硬拒**（`rule=tick_size`）。

## 接下来我建议做的两件“小而关键”的收尾

* **把 RMS 消费和 ContractMasterLoader 都挂到 SystemManager 启动序列**（先主数据 → 再 RMS → 再策略/风控）。
* **Grafana 再加两条曲线**：

  * `nbbo_cache_reads_stale_total / nbbo_cache_reads_total`（快照过期率）；
  * `state_bridge.patch_expired_total / patch_in_total`（过期状态占比）。
