# 目标

* 定期从 TWS/IBGW 读取**账户摘要**与**账户持仓**，标准化后发布到 Kafka：

  * `broker.account.v1`：现金、净值、初保/维保、可用资金、购买力、货币……
  * `broker.position.v1`：合约层持仓（symbol、数量、均价、市场价/市值、未实现盈亏、乘数）。
* **去抖 + 去重**：仅在**显著变化**或**超时**才发布，避免刷屏。
* 兼容我们账本服务（PortfolioService）之前定义的输入（它已经订阅 `broker.account.v1`）。

> 说明：为避免 ib_insync 版本差异，这里采用**轮询**（polling）方式，使用 `ib.accountSummary() / ib.portfolio()`；这些在 ib_insync 中是**安全可用**的同步接口，跑在 asyncio 里时我们用 `asyncio.to_thread(...)` 包装，不会阻塞事件循环。你后续也可以换成 ib_insync 的 `...Async` 版本或事件回调。

# 代码（直接粘贴；不生成文件）

```python
import os, json, time, asyncio, contextlib
from dataclasses import dataclass
from typing import Optional, Dict, Any, Tuple

from aiokafka import AIOKafkaProducer
from prometheus_client import start_http_server, Counter, Gauge, Histogram

from ib_insync import IB  # 复用你已经在 IBInsyncAdapter 初始化的 IB 实例

def _now_us() -> int: return int(time.time()*1e6)

# ----------------- Prom 指标 -----------------
start_http_server(int(os.getenv("IB_ACC_METRICS_PORT","9108")))
AM = dict(
  acc_pub = Counter("ib_account_publish_total","account snapshots published"),
  pos_pub = Counter("ib_position_publish_total","position snapshots published"),
  acc_poll_ms = Histogram("ib_account_poll_latency_ms","account poll latency ms",
                          buckets=(10,20,50,100,200,400,800,1600)),
  pos_poll_ms = Histogram("ib_position_poll_latency_ms","position poll latency ms",
                          buckets=(10,20,50,100,200,400,800,1600)),
  acc_cash = Gauge("ib_account_cash","account cash",["account"]),
  acc_init = Gauge("ib_account_initial_margin","initial margin",["account"]),
  acc_maint = Gauge("ib_account_maintenance_margin","maintenance margin",["account"]),
)

# ----------------- 配置 -----------------
@dataclass
class AccCfg:
  kafka_bootstrap: str = os.getenv("KAFKA_BOOTSTRAP","localhost:9092")
  topic_account: str = os.getenv("TOPIC_ACCOUNT","broker.account.v1")
  topic_position: str = os.getenv("TOPIC_POSITION","broker.position.v1")
  publish_min_interval_s: float = float(os.getenv("ACC_PUB_MIN_SEC","2.0"))  # 账户快照最短间隔
  publish_force_interval_s: float = float(os.getenv("ACC_PUB_FORCE_SEC","10.0"))  # 强制发布间隔
  poll_account_s: float = float(os.getenv("ACC_POLL_SEC","2.0"))
  poll_position_s: float = float(os.getenv("POS_POLL_SEC","3.0"))
  # 变化阈值（避免小抖动反复发布）
  eps_cash: float = float(os.getenv("ACC_EPS_CASH","1.0"))
  eps_margin: float = float(os.getenv("ACC_EPS_MARGIN","1.0"))
  eps_mv: float = float(os.getenv("POS_EPS_MV","0.5"))
  account_hint: str = os.getenv("IB_ACCOUNT","")  # 如果你在适配器里指定了账户，这里也可指定

class IBAccountBridge:
  """
  轮询 IB 账户摘要与持仓，标准化后发布到 Kafka。
  与 PortfolioService 解耦；任何服务都可订阅这两个主题。
  """
  def __init__(self, ib: IB, cfg: Optional[AccCfg] = None):
    self.ib = ib
    self.cfg = cfg or AccCfg()
    self.prod: Optional[AIOKafkaProducer] = None
    self._last_acc: Dict[str, Any] = {}
    self._last_acc_ts: float = 0.0
    self._last_pos_keyvals: Dict[Tuple, Tuple] = {}  # 去重：(acct,sym,conId) -> key fields tuple

  async def start(self):
    self.prod = AIOKafkaProducer(
      bootstrap_servers=self.cfg.kafka_bootstrap, acks="all", linger_ms=5, compression_type="lz4",
      value_serializer=lambda v: json.dumps(v).encode("utf-8"),
      key_serializer=lambda k: k.encode("utf-8") if k else None)
    await self.prod.start()
    asyncio.create_task(self._loop_account())
    asyncio.create_task(self._loop_position())

  async def stop(self):
    if self.prod:
      with contextlib.suppress(Exception):
        await self.prod.stop()

  # --------- 账户摘要轮询 ----------
  async def _loop_account(self):
    while True:
      t0 = time.time()
      try:
        if self.ib.isConnected():
          # 用 to_thread 包装同步调用，避免阻塞
          accsum = await asyncio.to_thread(self.ib.accountSummary)
          data = self._normalize_account(accsum, self.cfg.account_hint)
          if data:
            if self._should_pub_account(data):
              await self._publish_account(data)
      except Exception:
        pass
      finally:
        AM["acc_poll_ms"].observe((time.time()-t0)*1000.0)
      await asyncio.sleep(self.cfg.poll_account_s)

  def _normalize_account(self, accsum_list, account_hint: str) -> Optional[Dict[str,Any]]:
    # accsum_list: list of AccountValue(tag, value, currency, account)
    # 按 account 过滤（如指定）
    pick = None
    if not accsum_list:
      return None
    if account_hint:
      pick = [a for a in accsum_list if getattr(a, "account", "") == account_hint]
      pick = pick or accsum_list
    else:
      pick = accsum_list
    # 取第一个账户
    acct = getattr(pick[0], "account", None) or "UNKNOWN"
    d: Dict[str, Any] = {"account": acct, "currency": "USD"}  # 默认 USD；若有多币可扩展
    tags = {a.tag: a.value for a in pick if getattr(a, "account", "") == acct}
    # IB 常见标签：NetLiquidation, TotalCashValue, InitMarginReq, MaintMarginReq, AvailableFunds, BuyingPower
    def _num(v): 
      try: return float(str(v).replace(",",""))
      except: return 0.0
    d.update(dict(
      cash = _num(tags.get("TotalCashValue", 0.0)),
      net_liquidation = _num(tags.get("NetLiquidation", 0.0)),
      initial_margin = _num(tags.get("InitMarginReq", tags.get("FullInitMarginReq", 0.0))),
      maintenance_margin = _num(tags.get("MaintMarginReq", tags.get("FullMaintMarginReq", 0.0))),
      available_funds = _num(tags.get("AvailableFunds", 0.0)),
      buying_power = _num(tags.get("BuyingPower", 0.0)),
    ))
    return d

  def _should_pub_account(self, d: Dict[str,Any]) -> bool:
    now = time.time()
    if not self._last_acc:
      return True
    if (now - self._last_acc_ts) >= self.cfg.publish_force_interval_s:
      return True
    # 变化阈值判断
    eps_c = self.cfg.eps_cash; eps_m = self.cfg.eps_margin
    last = self._last_acc
    if abs(d["cash"] - last.get("cash", 0.0)) >= eps_c: return True
    if abs(d["initial_margin"] - last.get("initial_margin", 0.0)) >= eps_m: return True
    if abs(d["maintenance_margin"] - last.get("maintenance_margin", 0.0)) >= eps_m: return True
    return False

  async def _publish_account(self, d: Dict[str,Any]):
    msg = {"schema_v":1, "metadata":{"ts_us":_now_us(), "producer":"ib-bridge"}, "data": d}
    await self.prod.send_and_wait(self.cfg.topic_account, key=d["account"], value=msg)
    AM["acc_pub"].inc()
    AM["acc_cash"].labels(account=d["account"]).set(d["cash"])
    AM["acc_init"].labels(account=d["account"]).set(d["initial_margin"])
    AM["acc_maint"].labels(account=d["account"]).set(d["maintenance_margin"])
    self._last_acc = d; self._last_acc_ts = time.time()

  # --------- 持仓轮询 ----------
  async def _loop_position(self):
    while True:
      t0 = time.time()
      try:
        if self.ib.isConnected():
          # ib.portfolio() -> [PortfolioItem]
          items = await asyncio.to_thread(self.ib.portfolio)
          for it in items:
            c = it.contract
            acct = getattr(it, "account", "") or (self.cfg.account_hint or "UNKNOWN")
            sym = c.symbol
            con_id = getattr(c, "conId", 0)
            sec_type = getattr(c, "secType", "STK")
            cur = getattr(c, "currency", "USD")
            exch = getattr(c, "exchange", "SMART")
            mult = float(getattr(c, "multiplier", 1) or 1)
            qty = float(getattr(it, "position", 0.0))
            if qty == 0.0: 
              continue  # 空仓不发
            avg_cost = float(getattr(it, "avgCost", 0.0))
            mkt_px = float(getattr(it, "marketPrice", 0.0))
            mkt_val = float(getattr(it, "marketValue", 0.0))
            upnl = float(getattr(it, "unrealizedPNL", 0.0))
            key = (acct, sym, con_id)
            keyvals = (round(qty,6), round(avg_cost,6), round(mkt_px,6), round(mkt_val,2), round(upnl,2))
            last = self._last_pos_keyvals.get(key)
            # 去重/微小变化抑制
            if last and all(abs(a-b) < self.cfg.eps_mv for a,b in zip(keyvals, last)):
              continue
            self._last_pos_keyvals[key] = keyvals
            msg = {
              "schema_v":1,
              "metadata":{"ts_us":_now_us(), "producer":"ib-bridge"},
              "data":{
                "account": acct, "symbol": sym, "con_id": con_id, "sec_type": sec_type,
                "currency": cur, "exchange": exch, "multiplier": mult,
                "qty": qty, "avg_cost": avg_cost,
                "market_price": mkt_px, "market_value": mkt_val, "unrealized_pnl": upnl
              }
            }
            await self.prod.send_and_wait(self.cfg.topic_position, key=f"{acct}:{sym}:{con_id}", value=msg)
            AM["pos_pub"].inc()
      except Exception:
        pass
      finally:
        AM["pos_poll_ms"].observe((time.time()-t0)*1000.0)
      await asyncio.sleep(self.cfg.poll_position_s)
```

# 如何接入（两步）

1. **与你的 IBInsyncAdapter 共享一个 IB 实例**：

```python
adapter = IBInsyncAdapter(...); await adapter.start()
bridge  = IBEventBridge(adapter.ib); await bridge.start()        # 我们之前贴过
acct    = IBAccountBridge(adapter.ib); await acct.start()        # 👈 新增这一行
```

2. **PortfolioService 已订阅 `broker.account.v1`**：
   它会自动更新 `cash / initial_margin / maintenance_margin`（我们之前的账本骨架已实现 `_loop_acct`）。
   `broker.position.v1` 目前主要用于仪表盘/对账与冷启动“持仓领养”，不强耦合账本（账本以 fills 为权威来源）。

# 小贴士与可选增强

* **多账户**：如果你在一个 TWS 里管理多子账户，上面轮询会返回多条账户摘要；可以把 `account_hint` 设为具体账户（例如 `DU1234567`）以筛选。
* **汇率转换**：若账户非 USD，可在桥里加一层 FX（例如从 IB 获取 `currency rates` 或外部 FX，统一换算为 USD）。
* **事件回调替代**： ib_insync 也支持 `ib.accountSummaryEvent` / `ib.updatePortfolioEvent` 回调；在数据量非常大时可以切回回调模型降低延迟与 CPU。
* **对账/容灾**：`broker.position.v1` 可与我们**冷启动 Reconciliation 2.0**结合，遇到“幽灵仓位”时以 IB 为准重新拉齐。
* **报警**：当 `initial_margin / maintenance_margin` 异常上升（> 3σ）或 `available_funds` 低于阈值，RiskService 可以自动发“降风控”或“全局 Pause”。
