# 一、数据库表结构（DDL）

> 命名空间建议：`core`。所有金额统一 **USD**，时间统一 **UTC**。

```sql
-- 1) 策略主表：资金池上限/缓冲等
CREATE TABLE core_strategy (
  strategy_id        text PRIMARY KEY,
  name               text NOT NULL,
  capital_limit_usd  numeric(18,2) NOT NULL CHECK (capital_limit_usd >= 0),
  reserve_buffer_bps integer NOT NULL DEFAULT 10 CHECK (reserve_buffer_bps BETWEEN 0 AND 200),
  is_enabled         boolean NOT NULL DEFAULT true,
  created_at         timestamptz NOT NULL DEFAULT now()
);

-- 2) 策略资金账本快照（单行）
CREATE TABLE core_strategy_ledger (
  strategy_id        text PRIMARY KEY REFERENCES core_strategy(strategy_id) ON DELETE CASCADE,
  reserved_usd       numeric(18,2) NOT NULL DEFAULT 0,
  realized_pnl_usd   numeric(18,2) NOT NULL DEFAULT 0,
  updated_at         timestamptz NOT NULL DEFAULT now()
);

-- 3) 订单预留（一次下单一个记录；idempotency_key 唯一）
CREATE TABLE core_order_reservation (
  reservation_id     uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  strategy_id        text NOT NULL REFERENCES core_strategy(strategy_id),
  symbol             text NOT NULL,
  idempotency_key    text NOT NULL,
  reserved_usd       numeric(18,2) NOT NULL CHECK (reserved_usd >= 0),
  buffer_bps         integer NOT NULL CHECK (buffer_bps >= 0),
  state              text NOT NULL CHECK (state IN ('HELD','RELEASED','CANCELLED')),
  created_at         timestamptz NOT NULL DEFAULT now(),
  released_at        timestamptz,
  note               text
);
CREATE UNIQUE INDEX ON core_order_reservation (idempotency_key);

-- 4) 订单表（幂等键唯一；对账/执行落地）
CREATE TABLE core_order (
  order_id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  strategy_id        text NOT NULL REFERENCES core_strategy(strategy_id),
  symbol             text NOT NULL,
  idempotency_key    text NOT NULL,
  broker             text,
  ib_perm_id         bigint,
  side               text NOT NULL CHECK (side IN ('BUY','SELL')),
  order_type         text NOT NULL CHECK (order_type IN ('MKT','LMT','STP_LMT')),
  qty                numeric(18,6) NOT NULL CHECK (qty > 0),
  limit_px           numeric(18,6),
  stop_px            numeric(18,6),
  state              text NOT NULL CHECK (state IN ('NEW','SUBMITTED','PARTIAL','FILLED','CANCELLED','REJECTED')),
  created_at         timestamptz NOT NULL DEFAULT now(),
  updated_at         timestamptz NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX core_order_idem_uk ON core_order (idempotency_key);

-- 5) 成交表（去重唯一索引）
CREATE TABLE core_fill (
  fill_id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id           uuid NOT NULL REFERENCES core_order(order_id) ON DELETE CASCADE,
  ts                 timestamptz NOT NULL,
  qty                numeric(18,6) NOT NULL,
  price              numeric(18,6) NOT NULL,
  liquidity          text,  -- 'ADD'/'REMOVE'/NULL
  created_at         timestamptz NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX core_fill_dedup_uk ON core_fill (order_id, ts, qty, price);

-- 6) 仓位表（权威快照）
CREATE TABLE core_position (
  strategy_id        text NOT NULL REFERENCES core_strategy(strategy_id),
  symbol             text NOT NULL,
  pos_qty            numeric(18,6) NOT NULL,
  avg_px             numeric(18,6),
  realized_pnl_usd   numeric(18,2) NOT NULL DEFAULT 0,
  updated_at         timestamptz NOT NULL DEFAULT now(),
  PRIMARY KEY (strategy_id, symbol)
);

-- 7) 风控事件审计
CREATE TABLE core_risk_event (
  event_id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  strategy_id        text NOT NULL,
  symbol             text,
  idempotency_key    text,
  kind               text NOT NULL,   -- 'APPROVE','DENY','RELEASE','SWEEP','ERROR'
  reason             text,
  snapshot_json      jsonb,
  created_at         timestamptz NOT NULL DEFAULT now()
);

-- 8) 对账日志（冷启动/幽灵/领养）
CREATE TABLE core_reconcile_log (
  rec_id             uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  kind               text NOT NULL,  -- 'ADOPT','ORPHAN','STALE_CLOSED','REPAIR_OCO'
  detail             jsonb,
  created_at         timestamptz NOT NULL DEFAULT now()
);
```

# 二、事务与锁语义（Advisory Lock）

* 并发**冲关**点在“**审批+预留**”这一步：
  用 `pg_try_advisory_xact_lock(bigint)` 做**事务级**互斥，键计算规则建议：
  `key = hashtextextended(strategy_id || '|' || symbol, 0)`

  * 粒度：**策略×symbol**（满足实际隔离）
  * 失败返回 `false`：风控以 `RETRY_LATER` 或直接 DENY 返回
  * 成功：在同一事务内**检查 + 预留 + 记账**原子完成

# 三、存储过程（PL/pgSQL）

## 1) 审批并预留：`core_sp_risk_approve_and_reserve(...)`

```sql
CREATE OR REPLACE FUNCTION core_sp_risk_approve_and_reserve(
  p_strategy_id text,
  p_symbol text,
  p_notional_usd numeric,
  p_buffer_bps integer,
  p_idemp_key text
) RETURNS TABLE (
  approved boolean,
  reserved_usd numeric,
  reason text
) LANGUAGE plpgsql AS $$
DECLARE
  v_key bigint;
  v_limit numeric;
  v_reserved numeric;
  v_need numeric;
  v_buf_bps integer;
BEGIN
  IF p_notional_usd IS NULL OR p_notional_usd <= 0 THEN
    RETURN QUERY SELECT false, 0::numeric, 'INVALID_NOTIONAL';
    RETURN;
  END IF;

  -- 幂等：已有记录直接返回（避免重复预留）
  IF EXISTS (SELECT 1 FROM core_order_reservation WHERE idempotency_key = p_idemp_key) THEN
    RETURN QUERY
      SELECT true, reserved_usd, 'IDEMPOTENT'
      FROM core_order_reservation WHERE idempotency_key = p_idemp_key;
    RETURN;
  END IF;

  -- 事务级咨询锁（策略×symbol）
  v_key := hashtextextended(p_strategy_id || '|' || p_symbol, 0);
  IF NOT pg_try_advisory_xact_lock(v_key) THEN
    RETURN QUERY SELECT false, 0::numeric, 'LOCKED';
    RETURN;
  END IF;

  -- 读取上限与当前占用
  SELECT s.capital_limit_usd, l.reserved_usd, COALESCE(NULLIF(p_buffer_bps, NULL), s.reserve_buffer_bps)
  INTO v_limit, v_reserved, v_buf_bps
  FROM core_strategy s
  JOIN core_strategy_ledger l USING(strategy_id)
  WHERE s.strategy_id = p_strategy_id
  FOR UPDATE;

  IF v_limit IS NULL THEN
    RETURN QUERY SELECT false, 0::numeric, 'STRATEGY_NOT_FOUND';
    RETURN;
  END IF;

  v_need := round(p_notional_usd * (1 + v_buf_bps/10000.0), 2);

  IF (v_reserved + v_need) > v_limit THEN
    INSERT INTO core_risk_event(strategy_id, symbol, idempotency_key, kind, reason, snapshot_json)
    VALUES (p_strategy_id, p_symbol, p_idemp_key, 'DENY', 'INSUFFICIENT_CAPITAL',
            jsonb_build_object('limit', v_limit, 'reserved', v_reserved, 'need', v_need));
    RETURN QUERY SELECT false, 0::numeric, 'INSUFFICIENT_CAPITAL';
    RETURN;
  END IF;

  -- 预留 + 记账（同一事务原子）
  INSERT INTO core_order_reservation(strategy_id, symbol, idempotency_key, reserved_usd, buffer_bps, state, note)
  VALUES (p_strategy_id, p_symbol, p_idemp_key, v_need, v_buf_bps, 'HELD', NULL);

  UPDATE core_strategy_ledger
    SET reserved_usd = reserved_usd + v_need,
        updated_at = now()
  WHERE strategy_id = p_strategy_id;

  INSERT INTO core_risk_event(strategy_id, symbol, idempotency_key, kind, reason, snapshot_json)
  VALUES (p_strategy_id, p_symbol, p_idemp_key, 'APPROVE', NULL,
          jsonb_build_object('reserved_usd', v_need, 'buffer_bps', v_buf_bps));

  RETURN QUERY SELECT true, v_need, 'APPROVED';
END;
$$;
```

## 2) 终态释放：`core_sp_release_on_terminal(...)`

```sql
CREATE OR REPLACE FUNCTION core_sp_release_on_terminal(
  p_idemp_key text,
  p_order_id uuid,
  p_terminal text,   -- 'FILLED'|'CANCELLED'|'REJECTED'
  p_reason text DEFAULT NULL
) RETURNS TABLE (
  released boolean,
  released_usd numeric
) LANGUAGE plpgsql AS $$
DECLARE
  v_res numeric;
  v_strategy text;
BEGIN
  IF p_terminal NOT IN ('FILLED','CANCELLED','REJECTED') THEN
    RETURN QUERY SELECT false, 0::numeric;
    RETURN;
  END IF;

  -- 找 reservation
  SELECT r.reserved_usd, r.strategy_id
    INTO v_res, v_strategy
  FROM core_order_reservation r
  WHERE r.idempotency_key = p_idemp_key
    AND r.state = 'HELD'
  FOR UPDATE;

  IF NOT FOUND THEN
    -- 幂等：已释放或不存在也算成功返回0
    RETURN QUERY SELECT true, 0::numeric;
    RETURN;
  END IF;

  -- 更新 reservation
  UPDATE core_order_reservation
     SET state = 'RELEASED', released_at = now(), note = p_terminal
   WHERE idempotency_key = p_idemp_key;

  -- 回收账本
  UPDATE core_strategy_ledger
     SET reserved_usd = GREATEST(reserved_usd - v_res, 0),
         updated_at = now()
   WHERE strategy_id = v_strategy;

  -- 订单状态（可选，若不存在则插入）
  INSERT INTO core_order(order_id, strategy_id, symbol, idempotency_key, side, order_type, qty, state)
  VALUES (p_order_id, v_strategy, 'UNKNOWN', p_idemp_key, 'BUY', 'LMT', 0, p_terminal)
  ON CONFLICT (order_id) DO UPDATE SET state = EXCLUDED.state, updated_at = now();

  INSERT INTO core_risk_event(strategy_id, idempotency_key, kind, reason, snapshot_json)
  VALUES (v_strategy, p_idemp_key, 'RELEASE', p_reason,
          jsonb_build_object('released_usd', v_res, 'terminal', p_terminal));

  RETURN QUERY SELECT true, v_res;
END;
$$;
```

## 3) 清扫滞留：`core_sp_sweep_stale_reservations(cutoff_minutes)`

```sql
CREATE OR REPLACE FUNCTION core_sp_sweep_stale_reservations(
  p_cutoff_minutes integer DEFAULT 30
) RETURNS TABLE (
  swept integer
) LANGUAGE plpgsql AS $$
DECLARE
  v_cnt integer := 0;
BEGIN
  -- 选出长时间未达成订单终态的 HELD 记录
  FOR UPDATE OF r
  SELECT r.reservation_id, r.strategy_id, r.reserved_usd, r.idempotency_key
  FROM core_order_reservation r
  WHERE r.state='HELD'
    AND r.created_at < (now() - make_interval(mins => p_cutoff_minutes))
  LOOP
    -- 回收账本
    UPDATE core_strategy_ledger
       SET reserved_usd = GREATEST(reserved_usd - r.reserved_usd, 0),
           updated_at = now()
     WHERE strategy_id = r.strategy_id;

    UPDATE core_order_reservation
       SET state='CANCELLED', released_at = now(), note='STALE_SWEEP'
     WHERE reservation_id = r.reservation_id;

    INSERT INTO core_risk_event(strategy_id, idempotency_key, kind, reason, snapshot_json)
    VALUES (r.strategy_id, r.idempotency_key, 'SWEEP', 'STALE_RESERVATION',
            jsonb_build_object('swept_usd', r.reserved_usd));

    v_cnt := v_cnt + 1;
  END LOOP;

  RETURN QUERY SELECT v_cnt;
END;
$$;
```

# 四、执行/风控服务端调用约定（最小）

* **审批**：`SELECT * FROM core_sp_risk_approve_and_reserve($1,$2,$3,$4,$5);`

  * 入参：`strategy_id, symbol, notional_usd, buffer_bps(NULL=默认), idempotency_key`
  * 出参：`approved, reserved_usd, reason`
* **终态释放**：`SELECT * FROM core_sp_release_on_terminal($1,$2,$3,$4);`

  * 入参：`idempotency_key, order_id, terminal, reason`
  * 出参：`released, released_usd`
* **清扫**：定时任务 `SELECT * FROM core_sp_sweep_stale_reservations(30);`

# 五、账实一致性校验（常用 SQL）

```sql
-- 1) 账本必须非负
SELECT strategy_id, reserved_usd
FROM core_strategy_ledger
WHERE reserved_usd < 0;

-- 2) 预留求和 = 账本：允许极小差（并发边缘，理论应为0）
SELECT l.strategy_id, l.reserved_usd AS ledger,
       COALESCE(SUM(CASE WHEN r.state='HELD' THEN r.reserved_usd ELSE 0 END),0) AS held_sum
FROM core_strategy_ledger l
LEFT JOIN core_order_reservation r USING(strategy_id)
GROUP BY 1
HAVING ABS(l.reserved_usd - COALESCE(SUM(CASE WHEN r.state='HELD' THEN r.reserved_usd ELSE 0 END),0)) > 0.01;

-- 3) 重复幂等键（不应存在）
SELECT idempotency_key, COUNT(*) FROM core_order GROUP BY 1 HAVING COUNT(*) > 1;

-- 4) 成交去重是否生效（不应返回行）
WITH d AS (
  SELECT order_id, ts, qty, price, COUNT(*) c
  FROM core_fill GROUP BY 1,2,3,4
) SELECT * FROM d WHERE c > 1;
```

# 七、落地建议 & 踩坑提示

* **钱都走存储过程**：应用层不直接改 `ledger/reservation`，只调 `SP`；日志全部写 `core_risk_event`。
* **锁失败即重试/拒单**：`LOCKED` 返回给引擎 → 退避或改期，不要忙等。
* **幂等从入口打穿**：`idempotency_key` 同时用于 `reservation`、`order`、`release`。
* **高可用**：PG 建议开启 `synchronous_commit=on`（至少 WAL 落盘级），执行/风控服务侧做重试。
* **审计可追踪**：任何异常路径（拒单、清扫、错误）都有 `core_risk_event` 记录，便于事后回放。
