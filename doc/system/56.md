## B) 统一时钟与时间戳工具

```python
# timebox.py（或公共工具模块中）
import time
from dataclasses import dataclass

def now_us() -> int:
    # 单调+真实时间混合：以 wall clock 为主，必要时用于日志/审计
    return int(time.time() * 1e6)

@dataclass
class StageMark:
    bar_end_us: int
    env_born_us: int = 0
    approved_us: int = 0
    submitted_us: int = 0

    def bar_to_env_ms(self):      return max(0, (self.env_born_us - self.bar_end_us)//1000)
    def env_to_submit_ms(self):   return max(0, (self.submitted_us - self.env_born_us)//1000)
```

> 说明：为每个 `decision_id` 维护一个 `StageMark`，在关键节点打点。

## D) 策略调度器：节奏/抖动/新鲜度打点/数据不足跳过

```python
# scheduler.py 片段（或放到策略引擎）
import asyncio, random
from timebox import now_us, StageMark

class StrategyScheduler:
    def __init__(self, cfg, bar_store, strategy, metrics, logger):
        self.cfg = cfg; self.bars = bar_store; self.strategy = strategy
        self.m = metrics; self.log = logger
        self.cadence_s = int(cfg["scheduler"]["cadence"]["default_s"])
        self.jitter_ms = int(cfg["scheduler"]["cadence"]["jitter_ms"])
        self.bp_cfg = cfg["scheduler"]["backpressure"]
        self._stop = False

    async def run_symbol(self, symbol: str):
        while not self._stop:
            # 抖动
            await asyncio.sleep(self.cadence_s + random.randint(0, self.jitter_ms)/1000.0)
            # 取最新 1 根 5s bar
            bar = await self.bars.last(symbol)
            if not bar: 
                continue
            pipeline_backlog_lag_ms.labels(symbol=symbol).set(max(0, (now_us()-bar["ts_end_us"])//1000))
            # 数据是否充足（比如策略要求≥N根历史）
            if not await self.strategy.has_enough_data(symbol):
                # 数据不足 → 跳过
                continue

            # 生产 Envelope（策略只产出 env；见第1步定义）
            env = await self.strategy.produce_envelope(symbol)
            # 打点：bar_end → env
            marks = StageMark(bar_end_us=env.ts_bar_end_us)
            marks.env_born_us = now_us()
            fresh_bar_to_env.observe(marks.bar_to_env_ms())

            # 附上 marks（后续 approve/submit 继续补点）
            env.extras = env.extras or {}
            env.extras["__marks__"] = marks  # 直接附在 env 里传下去

            # TTL 默认/上限
            if env.freshness.ttl_ms <= 0:
                env.freshness.ttl_ms = int(self.cfg["slo"]["ttl"]["default_ms"])
            env.freshness.ttl_ms = min(env.freshness.ttl_ms, int(self.cfg["slo"]["ttl"]["hard_ceiling_ms"]))

            # 新鲜度“软门”：若 bar→env 已经接近 TTL 的 70%，可直接 skip，避免浪费
            if marks.bar_to_env_ms() > int(0.7 * env.freshness.ttl_ms):
                intent_expired_total.labels(stage="envelope").inc()
                self.log.info(f"[Fresh] skip early {env.decision_id} t={marks.bar_to_env_ms()}ms ttl={env.freshness.ttl_ms}")
                continue

            # 推送到下一步（validator→arbiter→translator→risk→exec）
            await self.strategy.submit_envelope(env)

    def widen_cadence(self):
        if not self.bp_cfg.get("enabled", True): return
        widen = int(self.bp_cfg["widen_step_s"])
        self.cadence_s = min(self.cadence_s + widen, int(self.bp_cfg["widen_max_s"]))
```

## E) TTL 守门：在每一层都检查并“即时作废”

> 三个插点：**(1) 风控批准入口**、**(2) 执行提交前**、**(3) 消息队列消费处（如用 Kafka）**。

### E.1 风控批准入口（`RiskService.approve()` 之初）

```python
from timebox import now_us, StageMark

class RiskService:
    ...
    async def approve(self, env, plan, ttl_ms: int):
        marks: StageMark = env.extras.get("__marks__")
        if marks:
            marks.approved_us = now_us()

        # --- 硬 TTL ---
        now = now_us()
        age_ms = (now - env.ts_bar_end_us)//1000
        ttl = min(ttl_ms, env.freshness.ttl_ms)
        if age_ms > ttl:
            intent_expired_total.labels(stage="approve").inc()
            return False, {"kind":"expired", "age_ms": int(age_ms), "ttl_ms": int(ttl)}
        ...
        # 其余：Reg-Lite → 资金/集中度 → Reservation
        ...
```

### E.2 执行提交前（在 `executor.submit(env, plan)` 入口处）

```python
class Executor:
    def __init__(self, cfg, exec_wrapper, logger, metrics):
        self.cfg = cfg; self.exec = exec_wrapper; self.log = logger; self.m = metrics

    async def submit(self, env, plan, reservation_id=None):
        # 最终 TTL 闸：执行前再检查一次
        now = now_us()
        age_ms = (now - env.ts_bar_end_us)//1000
        ttl = env.freshness.ttl_ms
        if age_ms > ttl:
            intent_expired_total.labels(stage="submit").inc()
            raise RuntimeError(f"intent_expired age={age_ms}ms ttl={ttl}ms")

        # 发送（走 Error Matrix 包装器）
        res = await self.exec.place(env.symbol, {
            "order_ref": plan.notes.get("order_ref") or env.decision_id,
            "side": plan.side, "order_type": plan.order_type,
            "limit_px": plan.limit_px, "stop_px": plan.stop_px,
            "qty": plan.qty, "tif": plan.tif,
        })
        # 打点：env→submit
        marks = env.extras.get("__marks__")
        if marks:
            marks.submitted_us = now_us()
            fresh_env_to_submit.observe(marks.env_to_submit_ms())
        return res
```

### E.3 队列消费者处（如你用 Kafka/EventBus）

> **强烈建议**：在**消费 loop**最前面做 TTL 过滤，避免把“过期意图”耗到后面的昂贵链路。

async def consume_envelopes(msg):

## F) Freshness SLO 观察与“回退节奏”自动化

> 用一个 5 分钟窗口观察 `p95`；**连续 N 个窗口**越线后触发**放宽节奏**（Backpressure）。

```python
class FreshnessSLOGuard:
    def __init__(self, cfg, scheduler: StrategyScheduler, logger):
        self.cfg = cfg; self.sched = scheduler; self.log = logger
        self.window_s = int(cfg["slo"]["freshness"]["window_s"])
        self.warn_factor = float(cfg["slo"]["freshness"]["warn_factor"])
        self.target_bar_env = int(cfg["slo"]["freshness"]["bar_to_env_ms_p95"])
        self.target_env_submit = int(cfg["slo"]["freshness"]["env_to_submit_ms_p95"])
        self.over_windows = 0
        self.trigger_n = int(cfg["scheduler"]["backpressure"]["trigger_if_p95_over_windows"])

    async def loop(self, pull_quantiles):
        """
        pull_quantiles: 一个回调，返回当前直方图的 p95 估计（由你封装 prometheus_client/本地统计）
        """
        while True:
            await asyncio.sleep(self.window_s)
            p95_a = pull_quantiles("fresh_bar_to_env_ms", 0.95)
            p95_b = pull_quantiles("fresh_env_to_submit_ms", 0.95)
            over = False
            if p95_a and p95_a > self.target_bar_env * self.warn_factor:
                fresh_slo_window_over.labels(kind="bar_to_env").inc(); over = True
            if p95_b and p95_b > self.target_env_submit * self.warn_factor:
                fresh_slo_window_over.labels(kind="env_to_submit").inc(); over = True

            if over:
                self.over_windows += 1
                if self.over_windows >= self.trigger_n:
                    self.log.warning(f"[SLO] p95 over target for {self.over_windows} windows -> widen cadence")
                    self.sched.widen_cadence()
                    self.over_windows = 0
            else:
                self.over_windows = 0
```

> 说明：生产中你可以直接从时序库拉 p95（PromQL）或在进程内做简易分位估计器。这里留了 `pull_quantiles` 钩子。

## G) 与第 1–5 步的衔接

* **第 1 步 Validator** 已做了 `dlb_ms`（判定预算）检查；本步把**硬 TTL**落实到**风控/执行/队列**三处，形成**多闸**。
* **第 2 步 Error Matrix** 不会改变时间语义，提交前 TTL 再闸可避免“修复-重试”把旧意图送出。
* **第 3 步 Reservation**：风控拒于“expired”不会占用资金；若在提交前过期，抛错并由上层捕获 → `ledger.release(reservation_id, cause="expired")`。
* **第 4 步 OCO 编舞**：不受影响；子单的改/撤同样受 Error Matrix 控制。
* **第 5 步 仲裁**：仲裁通过后**务必**在所有出口调用 `arbiter.finalize(env, status=...)`，过期亦同，防“名额泄漏”。

## H) 5 分钟自测清单（本地/影子）

1. **正常路径**：造一根 5s bar，立刻产 env → 观测 `fresh_bar_to_env_ms`；下单前打点 → `fresh_env_to_submit_ms`。
2. **队列过期**：把 env 在队列中 sleep > TTL → 消费后 `intent_expired_total{stage="queue"}`+1。
3. **风控过期**：让 validator 故意 sleep 接近 TTL，再进入 approve → 返回 `{"kind":"expired"}`；`intent_expired_total{stage="approve"}`+1。
4. **执行过期**：让 RiskService 通过，但 `executor.submit` 前 sleep 超 TTL → 抛 `intent_expired`；确认预留释放。
5. **SLO 回退**：人为堵塞让 p95 超线 2 个窗口 → 查看日志 `widen cadence` 与调度器 cadence_s 变大。
