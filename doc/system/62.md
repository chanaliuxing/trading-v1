# ① 影子（Shadow）/纸交（IBKR Paper）对齐回归脚本

> 目标：同一策略、同一数据窗口，**影子撮合轨迹**应与 **IBKR Paper 实际成交**在可接受阈值内一致；不一致要能定位到每笔订单/每根 bar。

```python
# -*- coding: utf-8 -*-
"""
回归对齐：Shadow vs IBKR Paper
需要引擎已接好：
  - StrategyHarness.run_backtest()（第9步）
  - PaperFillFeed.subscribe(order_ref → fill)
  - Features 查询：ClickHouse features_5s
阈值（可调）：
  - 价格偏差：<= 5 bps（按中位数）
  - 成交时间偏差：<= 3s（按p95）
  - 持仓轨迹一致性：>= 95%（逐秒或逐5s比对）
"""

import statistics, math, datetime as dt
from collections import defaultdict

BPS = lambda px, ref: 0.0 if ref<=0 else 1e4*(px-ref)/ref

class AlignReport:
    def __init__(self): 
        self.rows = []  # [(ts, sym, side, sh_px, pa_px, bps, sh_ts, pa_ts, dt_s, ref)]
    def add(self, ts, sym, side, sh_px, pa_px, ref, sh_ts, pa_ts):
        bps = BPS(pa_px, ref) - BPS(sh_px, ref)
        self.rows.append((ts,sym,side,sh_px,pa_px,bps,sh_ts,pa_ts,(pa_ts-sh_ts),ref))
    def stats(self):
        bps_list = [abs(r[5]) for r in self.rows]
        dts = [abs(r[8]) for r in self.rows]
        return {
            "count": len(self.rows),
            "bps_median": statistics.median(bps_list) if bps_list else None,
            "dt_p95": percentile(dts, 95) if dts else None
        }

def percentile(a, p):
    if not a: return None
    a = sorted(a); k=(len(a)-1)*p/100; f=math.floor(k); c=math.ceil(k)
    return a[int(k)] if f==c else a[f] + (a[c]-a[f])*(k-f)

async def collect_paper_fills(api, start, end, strategy_id):
    """
    从 IBKR Paper（或你们的fills表）抓取时间区间内的真实成交
    返回：{order_ref: {"px":..., "ts":epoch_s, "side": "BUY/SELL", "symbol":...}}
    """
    return await api.fetch_fills(start, end, strategy_id=strategy_id)

async def build_shadow_trail(harness, strat, symbol, start, end):
    """
    用影子撮合（同规则：LMT 在 [low, high] 即成交，滑点模型可加常数bp）构建影子成交与持仓轨迹
    返回：
      fills: {order_ref: {"px":..., "ts":epoch_s, "side":..., "symbol":...}}
      pos_ts: [(ts, pos_qty)]  # 逐5s或更细
    """
    fills, pos_ts = await harness.run_backtest(strat, symbol, start, end)
    return fills, pos_ts

async def compare_one_symbol(harness, strat, symbol, start, end, paper_api, px_ref="mid_last"):
    sh_fills, sh_pos = await build_shadow_trail(harness, strat, symbol, start, end)
    pa_fills = await collect_paper_fills(paper_api, start, end, strat.ctx.strategy_id)

    # 用 order_ref 对齐；若 Paper 无 order_ref，则用 (symbol, side, ts±Δ, qty) 模糊比对
    rep = AlignReport()
    misses = []
    for ref, sf in sh_fills.items():
        pf = pa_fills.get(ref)
        if not pf:
            # TODO: 模糊匹配（可选）
            misses.append(("missing_paper", ref, sf))
            continue
        ref_px = sf.get(px_ref) or pf.get(px_ref) or sf["px"]
        rep.add(ts=sf["ts"], sym=symbol, side=sf["side"], sh_px=sf["px"], pa_px=pf["px"],
                ref=ref_px, sh_ts=sf["ts"], pa_ts=pf["ts"])

    # 持仓轨迹一致性（按 5s 采样对比）
    def pos_series(pos): return [(int(t), int(q)) for t,q in pos]  # 简化
   一致 = lambda a,b: sum(1 for (t1,q1),(t2,q2) in zip(a,b) if q1==q2)/max(1,len(a)) * 100
    pos一致 = 一致(pos_series(sh_pos), pos_series(await paper_api.position_trail(symbol, start, end)))

    s = rep.stats()
    return {
        "symbol": symbol,
        "fills": s["count"],
        "bps_median": s["bps_median"],
        "fill_dt_p95_s": s["dt_p95"],
        "pos一致率_pct": pos一致,
        "misses": misses[:10]  # 展示前10条
    }

async def regression_shadow_vs_paper(harness, strat, symbols, start, end, paper_api):
    out = []
    for sym in symbols:
        out.append(await compare_one_symbol(harness, strat, sym, start, end, paper_api))
    # 汇总门槛
    agg = {
        "fills_total": sum(x["fills"] for x in out),
        "bps_median": statistics.median([x["bps_median"] for x in out if x["bps_median"] is not None]) if out else None,
        "fill_dt_p95_s": max(x["fill_dt_p95_s"] or 0 for x in out),
        "pos一致率_pct_min": min(x["pos一致率_pct"] for x in out) if out else None
    }
    return {"by_symbol": out, "aggregate": agg}
```

**验收门槛（建议）**

* `aggregate.bps_median ≤ 5` bps
* `aggregate.fill_dt_p95_s ≤ 3` 秒
* `aggregate.pos一致率_pct_min ≥ 95%`
* `misses` 原因可解释（如：纸交拒单→影子成交）

# ② KillSwitch 演练脚本（暂停→撤单→平仓→释放预留）

> 目标：一键**可回到安全态**，且**不遗留**订单/预留/仲裁名额。

```python
# -*- coding: utf-8 -*-
async def killswitch_drill(strategy_id):
    # 1) 触发 Kill（StateBus）
    await statebus.pause_strategy(strategy_id, reason="drill")
    # 2) 撤全部在挂
    await executor.cancel_all_open(strategy_id)
    # 3) 平全部持仓
    await executor.flatten_all(strategy_id)
    # 4) 释放全部预留
    await ledger.release_all(strategy_id, cause="killswitch")
    # 5) 仲裁 finalize（防名额泄漏）
    await arbiter.finalize_all(strategy_id, status="killswitch")

    # --- 验收断言 ---
    open_orders = await executor.list_open_orders(strategy_id)
    positions   = await portfolio.list_positions(strategy_id)
    snap        = await ledger.snapshot()
    leaks = [r for r in snap["items"].values() if r["strategy_id"]==strategy_id]
    assert not open_orders, "open orders not empty"
    assert all(abs(p["qty"])==0 for p in positions), "positions not flat"
    assert not leaks, "reservations leaked"
    return True
```

**演练步骤**

1. 让策略产生至少 1 笔成交、OCO 在挂；
2. 调用 `killswitch_drill("meanrev_v2:A")`；
3. 检查断言均通过；Grafana 上 `arb_finalize_total{status="killswitch"}` 与 `ledger_release_total{cause="killswitch"}` 增加。
