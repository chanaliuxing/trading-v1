# 二、状态源桥服务（进程内 or 微服务）骨架

> 作用：消费**多来源**状态，做**优先级融合 + TTL**，然后：
>
> * **进程内模式**：直接 `nbbo_cache.apply_status(symbol, ...)`
> * **微服务模式**：生产标准化事件 `reg.market_state.v1`（由下游消费者更新缓存）

```python
# -*- coding: utf-8 -*-
import time, asyncio
from dataclasses import dataclass
from typing import Optional, Dict, Any, List, Iterable

def now_us() -> int: return int(time.time()*1e6)

# ---------- 1) 统一状态模型 ----------
@dataclass
class StatePatch:
    symbol: str
    ts_us: int
    source: str                 # 'xchg','nasdaq','nyse','broker','internal'
    luld_low: Optional[float] = None
    luld_high: Optional[float] = None
    ssr_active: Optional[bool] = None
    halted: Optional[bool] = None
    reason_code: Optional[str] = None
    tick_size: Optional[float] = None
    lot_size: Optional[float] = None
    ttl_ms: int = 2000          # 默认 2s

# ---------- 2) 融合规则 ----------
class PriorityRule:
    """
    简单的优先级：exchange > broker > internal
    同源比时间，过期（ts+ttl<now）丢弃
    每个字段独立选优（避免一个来源覆盖所有字段）
    """
    ORDER = ["xchg","nasdaq","nyse","broker","internal"]

    @classmethod
    def better(cls, a: StatePatch, b: StatePatch) -> bool:
        # True 表示 a 优于 b
        ar = cls.ORDER.index(a.source) if a.source in cls.ORDER else len(cls.ORDER)
        br = cls.ORDER.index(b.source) if b.source in cls.ORDER else len(cls.ORDER)
        if ar != br: return ar < br
        return a.ts_us > b.ts_us

# ---------- 3) 状态融合器（内存态） ----------
class StateFusion:
    def __init__(self):
        # symbol -> field -> best_patch
        self.best: Dict[str, Dict[str, StatePatch]] = {}
        self.metrics = type("M", (), {
            "patch_in_total": 0,
            "patch_accepted_total": 0,
            "patch_expired_total": 0,
            "apply_calls_total": 0
        })()

    def apply(self, p: StatePatch):
        self.metrics.patch_in_total += 1
        now = now_us()
        if p.ts_us + p.ttl_ms*1000 < now:
            self.metrics.patch_expired_total += 1
            return False
        slot = self.best.setdefault(p.symbol, {})
        # 逐字段选优
        for field in ("luld_low","luld_high","ssr_active","halted","reason_code","tick_size","lot_size"):
            val = getattr(p, field)
            if val is None: 
                continue
            old = slot.get(field)
            if (old is None) or PriorityRule.better(p, old):
                slot[field] = p
        self.metrics.patch_accepted_total += 1
        return True

    def materialize(self, symbol: str) -> Dict[str, Any]:
        slot = self.best.get(symbol, {})
        def pick(field):
            pp = slot.get(field)
            return getattr(pp, field) if pp else None
        return {
            "symbol": symbol,
            "luld_low": pick("luld_low"),
            "luld_high": pick("luld_high"),
            "ssr_active": bool(pick("ssr_active")) if pick("ssr_active") is not None else False,
            "halted": bool(pick("halted")) if pick("halted") is not None else False,
            "reason_code": pick("reason_code"),
            "tick_size": pick("tick_size"),
            "lot_size": pick("lot_size"),
        }

# ---------- 4) 适配器：Kafka/REST（按需启用） ----------
class SourceKafka:
    """
    从若干原始主题消费（不同供应商字段不一致），通过 map_fn 统一映射为 StatePatch
    你可以消费我们标准化后的 reg.market_state.v1，或多路原始源。
    """
    def __init__(self, consumer, topics: List[str], map_fn):
        self.consumer = consumer
        self.topics = topics
        self.map_fn = map_fn

    async def run(self, on_patch):
        await self.consumer.start()
        try:
            await self.consumer.subscribe(self.topics)
            async for msg in self.consumer:
                try:
                    p = self.map_fn(msg)
                    if p: await on_patch(p)
                except Exception:
                    # log & continue
                    pass
        finally:
            await self.consumer.stop()

class SourceRESTPoller:
    """
    周期性拉取某 REST/内部接口，转换为 StatePatch 列表
    """
    def __init__(self, interval_s: float, fetch_fn):
        self.interval = interval_s
        self.fetch_fn = fetch_fn
        self._stop = False

    async def run(self, on_patch):
        while not self._stop:
            try:
                patches: List[StatePatch] = await self.fetch_fn()
                for p in patches or []:
                    await on_patch(p)
            except Exception:
                pass
            await asyncio.sleep(self.interval)

    def stop(self):
        self._stop = True

# ---------- 5) 桥接器：写入 NBBOSnapshotCache 或产出标准事件 ----------
class ExchangeStateBridge:
    def __init__(self, fusion: StateFusion, nbbo_cache=None, producer=None, emit_topic=None):
        """
        nbbo_cache: 进程内模式 → 直接 apply_status()
        producer+emit_topic: 微服务模式 → 生产 reg.market_state.v1
        """
        self.fusion = fusion
        self.cache = nbbo_cache
        self.producer = producer
        self.topic = emit_topic

    async def on_patch(self, p: StatePatch):
        ok = self.fusion.apply(p)
        if not ok: 
            return
        state = self.fusion.materialize(p.symbol)
        # 进程内：更新快照缓存（Reg-Lite 直接可见）
        if self.cache is not None:
            await self.cache.apply_status(
                p.symbol,
                luld_low=state["luld_low"],
                luld_high=state["luld_high"],
                ssr_active=state["ssr_active"],
                halted=state["halted"],
                tick=state["tick_size"],
                lot=state["lot_size"]
            )
        # 微服务：产出标准化事件（供所有消费者使用）
        if self.producer is not None and self.topic:
            evt = {
                "schema_v": 1,
                "metadata": {"ts_us": now_us(), "producer":"state-bridge", "source": p.source},
                "data": {
                    "symbol": p.symbol,
                    "luld_low": state["luld_low"],
                    "luld_high": state["luld_high"],
                    "ssr_active": state["ssr_active"],
                    "halted": state["halted"],
                    "reason_code": state["reason_code"],
                    "ttl_ms": 2000
                }
            }
            # 序列化用 Avro/Confluent 库，这里示意直接发送
            await self.producer.send(self.topic, key=p.symbol.encode("utf-8"), value=evt)

# ---------- 6) 最小接线（示例） ----------
async def main_bridge(nbbo_cache, kafka_consumer=None, kafka_producer=None):
    fusion = StateFusion()
    bridge = ExchangeStateBridge(fusion, nbbo_cache=nbbo_cache, producer=kafka_producer, emit_topic="reg.market_state.v1")

    # a) 从标准主题消费（如果上游已产出标准化 RMS 事件）
    def map_from_rms(msg) -> Optional[StatePatch]:
        v = msg.value  # 反序列化后的 dict
        d = v["data"]; m=v["metadata"]
        return StatePatch(
            symbol=d["symbol"], ts_us=v["metadata"]["ts_us"], source=m.get("source","xchg"),
            luld_low=d.get("luld_low"), luld_high=d.get("luld_high"),
            ssr_active=d.get("ssr_active"), halted=d.get("halted"),
            reason_code=d.get("reason_code"), ttl_ms=d.get("ttl_ms") or 2000
        )

    # b) 或者从某第三方“原始状态源”消费，用定制映射函数
    def map_from_vendor_raw(msg) -> Optional[StatePatch]:
        raw = msg.value
        return StatePatch(
            symbol=raw["sym"], ts_us=raw["ts_us"], source="xchg",
            luld_low=raw.get("lb"), luld_high=raw.get("ub"),
            ssr_active=raw.get("ssr", False), halted=raw.get("halt", False),
            reason_code=raw.get("reason")
        )

    # 这里演示用一个源；你可以注册多个 SourceKafka/SourceRESTPoller 并发运行
    tasks = []
    if kafka_consumer:
        src = SourceKafka(kafka_consumer, topics=["reg.market_state.v1"], map_fn=map_from_rms)
        tasks.append(asyncio.create_task(src.run(bridge.on_patch)))

    # 举例：REST 拉取（每 2s）
    async def fetch_from_internal_rest():
        # 返回 List[StatePatch]；此处示意空
        return []
    src_rest = SourceRESTPoller(2.0, fetch_from_internal_rest)
    tasks.append(asyncio.create_task(src_rest.run(bridge.on_patch)))

    await asyncio.gather(*tasks)
```

# 三、进程内 vs 微服务：两种使用姿势

* **进程内（最简单）**：

  * 在风控/策略进程启动时创建 `NBBOSnapshotCache` 与 `ExchangeStateBridge`；
  * 从**任意来源**生产 `StatePatch` 调 `bridge.on_patch()`；
  * Reg-Lite → `MarketStateProvider.secondary` 读 `nbbo_cache.read()`，已能看到合并后的 `luld/ssr/halt/tick/lot`。

* **微服务（更通用）**：

  * Bridge 消费多个源 → 产出标准化主题 `reg.market_state.v1`；
  * 各个进程（风控/执行/策略）都有一个**很轻的**消费者：收到 `reg.market_state.v1` 就调用 `nbbo_cache.apply_status()`；
  * 这样**所有进程**能在 ~毫秒级收敛到同一状态，且**耦合度更低**。

# 四、字段融合与优先级细则（默认策略）

* **优先级**：`exchange/primary feed > (nasdaq|nyse) > broker(IB) > internal(fallback)`
* **逐字段独立决议**：`luld_*`、`ssr_active`、`halted`、`tick/lot`互不拖累（谁新谁权威就用谁）。
* **TTL**：每个 `StatePatch` 自带 `ttl_ms`（默认 2s）；过期自动忽略（避免陈旧误判）。
* **清除逻辑**：某字段若连续 `> TTL×N` 未更新，可把该字段**置空**（由上层 Reg-Lite 对空值做“WARN 放行”而非强拒）。

---

# 五、与现有组件的接线点（一句话对齐）

* **Reg-Lite**：已经对接 `MarketStateProvider` → 读 `NBBOSnapshotCache`；这层桥在更新 cache。
* **风险拒单粒度**：

  * **tick/lot**、**无 NBBO 的 MKT**、**价格围栏**：始终硬拒（已实现）。
  * **LULD/SSR/Halt**：若桥层无权威数据 → 先 `WARN`，否则 `HARD REJECT` 并写规则命中。
* **合约主数据**：建议把 `contract.master.v1` 也消费后写入 `apply_status()`，让 tick/lot 有**唯一权威口径**。
