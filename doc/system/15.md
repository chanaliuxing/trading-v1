# ① Kafka 适配层（与内存 Bus 接口对齐）

> 目标：保留 `subscribe(topic)` 与 `publish(topic, key, value)` 的使用体验；在生产环境下换成 Kafka。

## A. 生产者（幂等等效、低延迟）

```python
from aiokafka import AIOKafkaProducer
import json, asyncio, zlib

class KafkaProducerAdapter:
    def __init__(self, bootstrap, acks="all", compression="lz4", linger_ms=5, batch_size=256*1024, max_in_flight=5):
        self._p = AIOKafkaProducer(
            bootstrap_servers=bootstrap,
            acks=acks,
            compression_type=compression,
            linger_ms=linger_ms,
            batch_size=batch_size,
            enable_idempotence=True,
            max_in_flight_requests_per_connection=max_in_flight # 与幂等一起避免乱序
        )
    async def start(self): await self._p.start()
    async def stop(self): await self._p.stop()

    async def publish(self, topic: str, key: str, value: dict):
        payload = json.dumps(value, separators=(",",":")).encode()
        await self._p.send_and_wait(topic, payload, key=key.encode())
```

**要点**

* `enable_idempotence=True` + `acks="all"`，保证**不重不丢不乱序**（单分区）。
* `linger_ms=5` 降低系统调用频率，延迟≈毫秒级即可满足我们 5s 周期策略。
* **键策略**：继续使用我们统一键（如 `symbol` 或 `{strategy_id}:{symbol}`），确保同键有序。

## B. 消费者（组、反压、优雅关闭）

```python
from aiokafka import AIOKafkaConsumer

class KafkaConsumerAdapter:
    def __init__(self, bootstrap, topics: list[str], group_id: str, max_poll_records=500, session_timeout_ms=10000):
        self.topics = topics
        self._c = AIOKafkaConsumer(
            *topics,
            bootstrap_servers=bootstrap,
            group_id=group_id,
            enable_auto_commit=True,
            auto_offset_reset="latest",
            max_poll_records=max_poll_records,
            session_timeout_ms=session_timeout_ms
        )

    async def start(self): await self._c.start()
    async def stop(self): await self._c.stop()

    async def __aiter__(self):
        try:
            async for msg in self._c:
                yield {
                    "topic": msg.topic,
                    "partition": msg.partition,
                    "offset": msg.offset,
                    "key": msg.key.decode() if msg.key else None,
                    "value": json.loads(msg.value),
                    "ts": msg.timestamp
                }
        finally:
            await self.stop()
```

**使用方式（与 Bus 对齐）**

```python
# Engine/Exec/Risk 任意服务内
prod = KafkaProducerAdapter(bootstrap="kafka:9092"); await prod.start()
cons = KafkaConsumerAdapter(bootstrap="kafka:9092", topics=["risk.decision.v2"], group_id="exec-svc"); await cons.start()

async for m in cons:   # 与内存 Bus 的 subscribe/迭代用法等价
    handle(m["value"])
```

**参数建议**

* 执行/风控类服务 `group_id` 固定，提供**水平扩展**能力；
* 数据类（bar 写入）采用独立消费组，避免回压互相影响；
* **健康检查**：暴露 `kafka.consumer.lag{topic,group}`、`rebalance_total` 指标；lag 超阈报警。

# ② 回补任务：限流 / 退避 / 熔断 策略（含伪代码）

> 目标：**稳定**地填补 `bar_5s_v2` 缺口，**不触发** IB 历史接口限流/封禁；对失败做**分级**处理，**不阻塞**实时链路。

## A. 错误分级（分类决定策略）

* **E_QUOTA**（配额/节流：pacing violation、rate limit、10197 等）
  → 退避：指数 + 抖动；维持**低并发**；记入**节流计数**。
* **E_TRANSIENT**（网络抖动、超时）
  → 重试：短退避 + 抖动；超过阈值后降级为 E_QUOTA。
* **E_PERM**（合约错误、无权限、参数非法）
  → 直接**失败**，标记不可回补，写审计，等待人工处理。
* **E_EMPTY**（查询无数据）
  → 软失败：短期再试 N 次；仍空则记“不可回补（缺历史）”。

## B. 并发/速率控制

* **全局并发上限**：`MAX_GLOBAL=5`；**单 symbol 并发=1**。
* **令牌桶**：`tokens/sec = 0.5~1`（根据 IB 限额）；每任务取 1 个令牌。
* **批处理单位**：每次请求最长 `durationStr='1800 S'`（或 `1 D` for 1m），按缺口分段。

## C. 指标与熔断

* `backfill.success_total / fail_total{class}`、`retry_total{class}`
* `backfill.rate_limited_total`（QUOTA）
* **熔断**：某 symbol 连续失败 ≥ M（比如 5） → `CIRCUIT_OPEN(sym, cooldown=15m)`；过冷却再半开。
* **全局熔断**：全局 QUOTA 率持续高 → 暂停新任务，仅保留已在途任务。

## D. 任务调度器（伪代码）

```python
import asyncio, random, time
from collections import deque, defaultdict
JITTER = lambda base: base * (1 + 0.2 * (random.random()-0.5))

class TokenBucket:
    def __init__(self, rate_per_sec, burst):
        self.rate = rate_per_sec; self.burst = burst; self.tokens = burst; self.ts = time.monotonic()
    async def take(self, n=1):
        while self.tokens < n:
            now = time.monotonic()
            self.tokens = min(self.burst, self.tokens + (now - self.ts)*self.rate)
            self.ts = now
            await asyncio.sleep(0.1)
        self.tokens -= n; return True

class Circuit:
    def __init__(self): self.state = {}  # sym -> (open_until_ts)
    def open(self, sym, cooldown_s): self.state[sym] = time.monotonic() + cooldown_s
    def allow(self, sym): return time.monotonic() >= self.state.get(sym, 0)

class BackfillScheduler:
    def __init__(self, ib, ch, rate=1.0, burst=3, max_global=5):
        self.ib = ib; self.ch = ch
        self.tb = TokenBucket(rate, burst)
        self.sem = asyncio.Semaphore(max_global)
        self.queue = deque()          # (sym, t0, t1, granularity)
        self.cb = Circuit()
        self.inflight = defaultdict(int)

    def enqueue_gaps(self, sym):
        gaps = ch_find_gaps(sym)      # SQL：我们前面给过
        for t0, t1 in gaps:
            # 切片为最大片段（5s/1m 视权限）
            for seg in split_into_chunks(t0, t1, max_secs=1800):
                self.queue.append((sym, seg.start, seg.end, "5s_or_1m"))

    async def run(self):
        while True:
            if not self.queue:
                await asyncio.sleep(5); continue
            sym, t0, t1, gran = self.queue.popleft()
            if not self.cb.allow(sym):  # 熔断中
                continue
            if self.inflight[sym] > 0:  # 单symbol串行
                self.queue.append((sym,t0,t1,gran)); await asyncio.sleep(0); continue

            await self.sem.acquire()
            await self.tb.take(1)
            self.inflight[sym] += 1
            asyncio.create_task(self._do_backfill(sym, t0, t1, gran))

    async def _do_backfill(self, sym, t0, t1, gran):
        try:
            rows = await ib_fetch_hist(self.ib, sym, t0, t1, gran)   # 封装 reqHistoricalData
            if not rows:
                raise EmptyError()
            # 归一化成 5s；calc_mode='backfill'
            await ch_insert_bar5s(rows)
            metrics.inc("backfill.success_total", tags={"sym":sym})
        except QuotaError:
            metrics.inc("backfill.rate_limited_total", tags={"sym":sym})
            self.cb.open(sym, cooldown_s=JITTER(15*60))  # 半小时量级
            # 失败任务分解缩小粒度后重排队
            for seg in split_into_chunks(t0, t1, max_secs=300):
                self.queue.append((sym, seg.start, seg.end, gran))
        except TransientError:
            metrics.inc("backfill.retry_total", tags={"sym":sym,"class":"transient"})
            await asyncio.sleep(JITTER(5))  # 短退避
            self.queue.append((sym, t0, t1, gran))
        except EmptyError:
            metrics.inc("backfill.fail_total", tags={"sym":sym,"class":"empty"})
            # 标记不可回补但不熔断
        except PermError:
            metrics.inc("backfill.fail_total", tags={"sym":sym,"class":"perm"})
            self.cb.open(sym, cooldown_s=JITTER(60*60))
        finally:
            self.inflight[sym] = max(self.inflight[sym]-1, 0)
            self.sem.release()
```

async def ib_fetch_hist(ib, sym, t0, t1, gran):

## E. 监控与阈值（回补）

* `backfill.success_rate = success / (success+fail)` 目标 ≥ **0.99**
* `backfill.rate_limited_total`（每小时）上升 → 提醒**提高退避/降低并发**
* `circuit.open.count`（符号数量）阈值报警
* `gap_probe.gap_count{sym}` 持续>0 → 升级到工单
