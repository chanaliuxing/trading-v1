那我先把**三件最关键的 P0+++ 落地件**一次性贴出来，全部是可以直接拷贝进你现有进程里的**内联骨架**（不生成文件）：

* ① **Reg-Lite 风控规则**：LULD 价格带、SSR（Reg SHO）空头限制、停牌/异常、tick/lot 校验、价格围栏
* ② **IB 错误码 → 动作矩阵**：统一“可重试/不可重试/降级/修正价重试”并带指数退避
* ③ **资金预留泄漏看门狗**：TTL 巡检 + 冷启动自愈 + 异常挂单强一致撤/释放

> 接入位：① 插到 RiskService 的“批准前”链路；② 包裹执行内核的 place/modify/cancel；③ 在 SystemManager 启动后与 CRS 一起常驻。

## ① Reg-Lite 风控规则（LULD / SSR / Halt / tick&lot / 价格围栏）

**用途**：在你已有的“事前风控通过→下单”之间，再加一层**交易所/监管规则闸**，统一拒单理由，避免违规落地到券商。

```python
# ---------- Reg-Lite 规则骨架 ----------
from dataclasses import dataclass
from typing import Optional, Dict, Any, List
import math, time

@dataclass
class Intent:   # 你已有的 order.intent 简化映射
    decision_id: str
    symbol: str
    side: str          # BUY/SELL/SELL_SHORT
    qty: float
    order_type: str    # LMT/MKT/STP/STP_LMT
    limit_px: Optional[float]
    stop_px: Optional[float]
    tif: str
    allow_prepost: bool = False

@dataclass
class MarketSnap:
    # 最少字段，均为“当前刻”的视图；取自行情聚合/券商 top-of-book
    last_px: Optional[float]
    bid_px: Optional[float]
    ask_px: Optional[float]
    mid_px: Optional[float]
    luld_low: Optional[float]   # 当日当前 LULD 下界
    luld_high: Optional[float]  # 当日当前 LULD 上界
    ssr_active: bool            # 是否触发 SSR（Reg SHO）
    halted: bool                # 是否停牌（含 News Pending/Vol Halt）
    tick_size: float            # 最小价位
    lot_size: float             # 最小下单量

@dataclass
class RuleHit:
    rule_id: str
    reason: str
    meta: Dict[str, Any]

class RegLiteGuard:
    def __init__(self, price_collar_pct: float = 5.0):
        # 价格围栏：相对 mid/last 超过 ±X% 拒绝（策略可在 allow_prepost 下放宽）
        self.collar_pct = price_collar_pct

    def _round_to_tick(self, px: float, tick: float) -> float:
        if tick <= 0: return px
        return round(round(px / tick) * tick, 10)

    def check(self, it: Intent, m: MarketSnap) -> List[RuleHit]:
        hits: List[RuleHit] = []
        # 0) 基础健壮
        if it.qty <= 0:
            hits.append(RuleHit("bad_qty", "non-positive qty", {"qty": it.qty}))
            return hits
        if m.halted:
            hits.append(RuleHit("halted", "symbol halted/news pending", {}))
            return hits

        # 1) tick/lot 校验
        if m.lot_size and (it.qty % m.lot_size) > 1e-9:
            hits.append(RuleHit("lot_size", "qty not multiple of lot", {"lot": m.lot_size, "qty": it.qty}))
        # 订单涉及价格者：limit/stop/stop_limit
        pxs = []
        if it.order_type in ("LMT","STP_LMT") and it.limit_px: pxs.append(("limit_px", it.limit_px))
        if it.order_type in ("STP","STP_LMT") and it.stop_px:  pxs.append(("stop_px", it.stop_px))
        for name, px in pxs:
            if m.tick_size:
                adj = self._round_to_tick(px, m.tick_size)
                if abs(adj - px) > 1e-12:
                    hits.append(RuleHit("tick_size", f"{name} violates tick", {"tick": m.tick_size, "px": px, "adj": adj}))

        # 2) LULD 价格带
        def _in_luld(px: float) -> bool:
            if m.luld_low is None or m.luld_high is None: return True  # 无带→放行到下游（或 hits.append warn）
            return (m.luld_low - 1e-10) <= px <= (m.luld_high + 1e-10)

        for name, px in pxs:
            if not _in_luld(px):
                hits.append(RuleHit("luld_band", f"{name} outside LULD", {"low": m.luld_low, "high": m.luld_high, "px": px}))

        # 3) SSR（Reg SHO）空头限制：触发日，卖空不得 ≤ best bid；需抬价
        if it.side.upper() in ("SELL_SHORT","SHORT","SLD_SHORT") and m.ssr_active:
            bid = m.bid_px or (m.mid_px or m.last_px or 0.0)
            # 仅对可成交或触发价有效；市价单拒绝，限价单必须 > bid + 1 tick
            if it.order_type == "MKT":
                hits.append(RuleHit("ssr", "SSR active: short MKT forbidden", {"bid": bid}))
            elif it.order_type in ("LMT","STP_LMT") and it.limit_px is not None:
                need = bid + (m.tick_size or 0.01)
                if it.limit_px <= need - 1e-12:
                    hits.append(RuleHit("ssr", "limit must be above bid", {"bid": bid, "limit": it.limit_px, "need_min": need}))

        # 4) 价格围栏：相对 mid/last 过远
        ref = m.mid_px or m.last_px
        if ref and it.order_type in ("LMT","STP_LMT") and it.limit_px:
            dev = abs(it.limit_px - ref) / ref * 100.0
            if dev > self.collar_pct:
                hits.append(RuleHit("price_collar", "limit too far from ref", {"ref": ref, "limit": it.limit_px, "dev_pct": dev}))
        # 市价单在无报价时拒绝
        if it.order_type == "MKT" and not (m.bid_px and m.ask_px):
            hits.append(RuleHit("no_nbbo", "MKT without NBBO snapshot", {}))

        return hits
```

**接入 RiskService（批准前一刀拦）**：

```python
# 在 approve 之前：
hits = self.reglite_guard.check(intent, market_snap)  # market_snap 来自你行情门面/券商快照
if hits:
    # 统一拒单
    reasons = [{"rule": h.rule_id, "reason": h.reason, **h.meta} for h in hits]
    return self._reject(meta, dto, decision_id, "reglite_block", {"hits": reasons})
```

> 市场状态来源：你可以将 SSR/LULD/Halt 合并到“市场状态服务”，从 IB、纳斯达克数据或外部源拉取；拿不到时默认“放行给下游 + 记录 WARN”，不要静默忽略。

## ② IB 错误码 → 动作矩阵（统一重试/放弃/降级/修正价）

**用途**：把执行侧的各种报错**表驱动**管理，避免“死循环改单/错误处理分散”。动作包含：`RETRY`（指数退避）、`ADJUST_AND_RETRY`（修正价/涨一tick）、`ABORT`（永久拒绝）、`DEGRADE`（降级：减频/减量/改为 STP_LMT 等）。

```python
import re, asyncio, random, math
from dataclasses import dataclass
from typing import Optional, Dict, Any

@dataclass
class ExecErrorAction:
    action: str                 # RETRY / ADJUST_AND_RETRY / ABORT / DEGRADE
    reason: str
    params: Dict[str, Any]
    retry_after_ms: int = 0

# 错误码与消息片段匹配（覆盖常见 IB 错误）
_IB_ERROR_RULES = [
    # 价格无效/越界 → 修正价重试
    (re.compile(r"(10147|Invalid price|Price out of range)", re.I),
     lambda ctx: ExecErrorAction("ADJUST_AND_RETRY","price_out_of_range",
                                 {"adjust":"to_tick_or_luld"}, retry_after_ms=100)),

    # 合约/账户无效 → 放弃
    (re.compile(r"(201|The account.*is invalid|No security definition)", re.I),
     lambda ctx: ExecErrorAction("ABORT","invalid_account_or_contract", {})),

    # Pacing violation → 退避重试 + 降频
    (re.compile(r"(pacing|Exceeded max rate of messages)", re.I),
     lambda ctx: ExecErrorAction("DEGRADE","pacing_violation", {"reduce_rate": True}, retry_after_ms=1200)),

    # 市场关闭/不可交易 → 放弃或延迟
    (re.compile(r"(order is not permitted|outside trading hours|halted)", re.I),
     lambda ctx: ExecErrorAction("ABORT","not_permitted_now", {})),

    # SSR 相关拒绝（券商侧二次校验） → 上浮价重试
    (re.compile(r"(short sale|SSR)", re.I),
     lambda ctx: ExecErrorAction("ADJUST_AND_RETRY","ssr_violation", {"adjust":"lift_above_bid"}, retry_after_ms=150)),
]

def classify_ib_error(err_msg: str, ctx: Dict[str,Any]) -> ExecErrorAction:
    for pat, fn in _IB_ERROR_RULES:
        if pat.search(err_msg or ""):
            return fn(ctx)
    # 默认：小退避重试 N 次
    return ExecErrorAction("RETRY","unknown_ib_error", {}, retry_after_ms=200)

class RetryBudget:
    def __init__(self, max_tries=5, base_ms=100, jitter_ms=50):
        self.max = max_tries; self.base = base_ms; self.jitter = jitter_ms
    def delay_ms(self, attempt:int) -> int:
        expo = self.base * (2 ** max(0, attempt-1))
        jitter = random.randint(0, self.jitter)
        return min(3000, int(expo + jitter))

# 执行内核包裹器（place/modify/cancel 通用）
async def exec_with_policy(op_fn, *, args: tuple, ctx: Dict[str,Any], retry_budget: RetryBudget):
    attempt = 0
    while True:
        try:
            return await op_fn(*args)   # 成功
        except Exception as e:
            attempt += 1
            action = classify_ib_error(str(e), ctx)
            if action.action == "ABORT":
                raise
            if attempt >= retry_budget.max:
                raise
            # 动作：降级/修正价
            if action.action == "DEGRADE" and action.params.get("reduce_rate"):
                # 通知 Redis 限流提高等待或降低桶参数（留给你的限流组件）
                pass
            if action.action == "ADJUST_AND_RETRY":
                # 示例：将价格对齐 tick & LULD；或 SSR 上浮一tick
                adj = await _adjust_price(ctx)
                ctx.update(adj)
            # 延迟（动作自带 or 指数退避）
            delay = action.retry_after_ms or retry_budget.delay_ms(attempt)
            await asyncio.sleep(delay/1000.0)

async def _adjust_price(ctx: Dict[str,Any]) -> Dict[str,Any]:
    # ctx: symbol/side/order_type/limit_px/stop_px/tick/luld_low/high/bid/ask
    tick = ctx.get("tick", 0.01)
    def to_tick(px): return round(round(px/tick)*tick, 10)
    out={}
    if ctx.get("limit_px"):
        px = to_tick(ctx["limit_px"])
        # LULD 夹取
        low = ctx.get("luld_low"); high = ctx.get("luld_high")
        if low is not None:  px = max(px, to_tick(low))
        if high is not None: px = min(px, to_tick(high))
        # SSR 提升
        if ctx.get("ssr_active") and ctx.get("side","").upper().startswith("SELL"):
            bid = ctx.get("bid") or 0.0
            px = max(px, to_tick(bid + tick))
        out["limit_px"] = px
    return out
```


## ③ 资金预留泄漏看门狗（TTL 巡检 + 冷启动自愈）

**用途**：防止“订单卡死/状态丢失”导致预留资金长期不释放；崩溃/重启后自动**以券商状态为准**自愈，确保 Ledger 与真实世界一致。

```python
import asyncio, time, datetime as dt
import asyncpg
from typing import List
from dataclasses import dataclass

@dataclass
class ReservationRec:
    id: int
    strategy_id: str
    order_ref: str
    state: str        # HELD/COMMITTED/RELEASED
    amount: float
    created_at: dt.datetime

class ReservationWatchdog:
    def __init__(self, pg_pool, adapter, ttl_sec:int=180, sweep_sec:int=15, orphan_cancel:bool=True):
        self.pg = pg_pool
        self.adapter = adapter
        self.ttl = ttl_sec
        self.sweep = sweep_sec
        self.orphan_cancel = orphan_cancel
        self._task = None

    async def start(self):
        self._task = asyncio.create_task(self._loop())

    async def _loop(self):
        while True:
            try:
                await self._sweep_once()
            except Exception as e:
                # log warn
                pass
            await asyncio.sleep(self.sweep)

    async def _sweep_once(self):
        now = dt.datetime.utcnow()
        async with self.pg.acquire() as con:
            rows = await con.fetch("""
              SELECT id, strategy_id, order_ref, state, amount, created_at
              FROM reservations
              WHERE state='HELD' AND created_at < (now() - $1::interval)
            """, f"{self.ttl} seconds")
            recs = [ReservationRec(r["id"], r["strategy_id"], r["order_ref"], r["state"], float(r["amount"]), r["created_at"]) for r in rows]

        if not recs: return

        # 券商侧 open orders（做一轮映射）
        ib_open = await self.adapter.open_orders()
        open_refs = {o.order_ref for o in ib_open if o.order_ref}

        async with self.pg.acquire() as con:
            for r in recs:
                if r.order_ref in open_refs:
                    # 真实仍有挂单：标注 “aging_pending”，交给 CRS 强制撤/改单
                    await con.execute("UPDATE reservations SET note='aging_pending', updated_at=now() WHERE id=$1", r.id)
                    if self.orphan_cancel:
                        # 交给你的执行内核做强一致撤（可发指令到控制主题）
                        pass
                else:
                    # 券商侧已无单：直接释放
                    await con.execute("""
                      UPDATE reservations SET state='RELEASED', released_at=now(), note='ttl_release'
                      WHERE id=$1 AND state='HELD'
                    """, r.id)

    # 冷启动自愈：进程启动后立即跑一次
    async def reconcile_on_boot(self):
        ib_open = await self.adapter.open_orders()
        open_refs = {o.order_ref for o in ib_open if o.order_ref}
        async with self.pg.acquire() as con:
            # 对所有 HELD 做核验：券商无对应挂单则释放
            rows = await con.fetch("SELECT id, order_ref FROM reservations WHERE state='HELD'")
            for r in rows:
                if r["order_ref"] not in open_refs:
                    await con.execute("""
                      UPDATE reservations SET state='RELEASED', released_at=now(), note='boot_release'
                      WHERE id=$1 AND state='HELD'
                    """, r["id"])
```

**接入建议**

* **SystemManager** 启动序列：IB 适配器 → CRS → **ReservationWatchdog.reconcile_on_boot()** → Watchdog.start()
* 指标：`reservations_held_total`、`reservations_ttl_release_total`、`reservations_aging_pending_total`；如 TTL 释放异常增多→报警。
