那我来拍板：先把**ClickHouse Writer（消费 5s 实时 + 修复流，批量落库，带重试与限速）**给你贴全。它是整条链最关键的缺口（聚合器→Kafka 已有、引擎→CH 读取也有），把 Writer 落地后，你的 250ms→5s→CH→策略引擎 就闭环了。下面直接给**DDL + 可运行骨架**，不生成文件，全部贴出来。

# 1) ClickHouse 表（推荐最终版）

> 使用 `ReplacingMergeTree(bar_seq)` 数值版本覆盖，保证**迟到修复**自动覆盖旧 Bar。

```sql
CREATE TABLE IF NOT EXISTS qsys.bar_5s_v2
(
  symbol      LowCardinality(String),
  ts_start    DateTime,          -- 秒级UTC（聚合器已传epoch秒）
  ts_end      DateTime,
  open        Float64,
  high        Float64,
  low         Float64,
  close       Float64,
  vol         UInt64,
  vwap        Float64,
  rth         UInt8,
  source      LowCardinality(String),    -- 'AGG' | 'REPAIR' | 'BACKFILL'
  calc_mode   LowCardinality(String),    -- 'realtime' | 'repair' | 'backfill'
  late_count  UInt16,
  bar_seq     UInt32                     -- 越大越新：用于覆盖
)
ENGINE = ReplacingMergeTree(bar_seq)
PARTITION BY toDate(ts_start)
PRIMARY KEY (symbol, ts_start)
ORDER BY (symbol, ts_start)
SETTINGS index_granularity = 8192;
```

# 2) CH Writer：Kafka→批量 INSERT（带重试/限速/度量）

> 依赖：`aiokafka>=0.8`, `httpx>=0.27`, Python 3.10+
> 消费两个 topic：`md.bar.5s.v2`（准实时）与 `md.bar.5s.repair.v1`（迟到修复）。
> 批量策略：`max_rows` 或 `flush_ms` 任一触发即写入；失败**指数退避+重试**；成功后**手动提交 offset**（at-least-once + CH 幂等）。

```python
# ========= CHWriter (Kafka -> ClickHouse) =========
import asyncio, os, json, time, contextlib
from dataclasses import dataclass
from typing import List, Optional
import httpx
from aiokafka import AIOKafkaConsumer

# ---- 配置 ----
@dataclass
class CHWriterConfig:
    kafka_bootstrap: str = "localhost:9092"
    kafka_group: str = "ch-writer-bar5s"
    topics: tuple = ("md.bar.5s.v2", "md.bar.5s.repair.v1")
    ch_dsn: str = "http://localhost:8123"
    ch_db: str = "qsys"
    ch_table: str = "bar_5s_v2"
    ch_user: str = ""
    ch_pass: str = ""
    batch_max_rows: int = 1000
    batch_flush_ms: int = 300
    request_timeout_s: float = 2.0
    max_retry: int = 5
    base_backoff_s: float = 0.25

# ---- Writer 主体 ----
class CHWriter:
    def __init__(self, cfg: CHWriterConfig):
        self.cfg = cfg
        self.consumer: Optional[AIOKafkaConsumer] = None
        self.client: Optional[httpx.AsyncClient] = None
        self._batch: List[dict] = []
        self._batch_offsets = []     # (tp, offset) 列表，用于提交
        self._last_flush = time.time()
        self._metrics = {"rows_ok":0, "rows_err":0, "flush":0, "retry":0}

    async def start(self):
        # HTTP client
        self.client = httpx.AsyncClient(
            base_url=self.cfg.ch_dsn.rstrip("/"),
            timeout=self.cfg.request_timeout_s,
            headers={"Content-Type":"text/plain; charset=utf-8"}
        )
        # Kafka consumer
        self.consumer = AIOKafkaConsumer(
            *self.cfg.topics,
            bootstrap_servers=self.cfg.kafka_bootstrap,
            group_id=self.cfg.kafka_group,
            enable_auto_commit=False,
            auto_offset_reset="latest",
            value_deserializer=lambda b: json.loads(b.decode("utf-8")),
            key_deserializer=lambda b: b.decode("utf-8") if b else "",
        )
        await self.consumer.start()
        asyncio.create_task(self._loop())

    async def stop(self):
        # 最后一批
        with contextlib.suppress(Exception):
            await self._flush()
        if self.consumer:
            with contextlib.suppress(Exception):
                await self.consumer.stop()
        if self.client:
            with contextlib.suppress(Exception):
                await self.client.aclose()

    async def _loop(self):
        try:
            while True:
                msg = await self.consumer.getone()
                row = self._to_row(msg.value)
                if row is not None:
                    self._batch.append(row)
                    self._batch_offsets.append((msg.topic, msg.partition, msg.offset))
                await self._maybe_flush()
        except asyncio.CancelledError:
            pass

    def _to_row(self, m: dict) -> Optional[dict]:
        """
        来自聚合器/修复器的消息：{"schema_v":2, "metadata":..., "data":{...}}
        规范化为 CH 行。缺字段会兜底，bar_seq 必须存在；若无则置 1。
        """
        try:
            d = m["data"]
            return {
                "symbol": d["symbol"],
                "ts_start": int(d["ts_start"]),   # epoch秒
                "ts_end":   int(d["ts_end"]),
                "open":     float(d["open"]),
                "high":     float(d["high"]),
                "low":      float(d["low"]),
                "close":    float(d["close"]),
                "vol":      int(d.get("vol",0)),
                "vwap":     float(d.get("vwap", d["close"])),
                "rth":      int(1 if d.get("rth", True) else 0),
                "source":   str(d.get("source","AGG")),
                "calc_mode":str(d.get("calc_mode","realtime")),
                "late_count": int(d.get("late_count",0)),
                "bar_seq":  int(d.get("bar_seq", 1))
            }
        except Exception:
            # 严格保护：坏数据直接丢弃并计数
            self._metrics["rows_err"] += 1
            return None

    async def _maybe_flush(self):
        if len(self._batch) >= self.cfg.batch_max_rows or \
           (time.time() - self._last_flush) * 1000 >= self.cfg.batch_flush_ms:
            await self._flush()

    def _mk_insert_sql(self, rows: List[dict]) -> str:
        # JSONEachRow 批量写入；ts_start/ts_end 为 epoch秒，CH 会解析为 DateTime（UTC）
        # 注意：如果 CH 版本需要明确 toDateTime，请改用 VALUES + toDateTime() 显式转换。
        sql_head = f"INSERT INTO {self.cfg.ch_db}.{self.cfg.ch_table} FORMAT JSONEachRow\n"
        lines = []
        for r in rows:
            lines.append(json.dumps(r, separators=(",",":")))
        return sql_head + "\n".join(lines)

    async def _flush(self):
        if not self._batch:
            self._last_flush = time.time()
            return
        rows = self._batch; offsets = self._batch_offsets
        self._batch = []; self._batch_offsets = []
        sql = self._mk_insert_sql(rows)
        backoff = self.cfg.base_backoff_s
        for attempt in range(self.cfg.max_retry):
            try:
                resp = await self.client.post(f"/?database={self.cfg.ch_db}", content=sql)
                resp.raise_for_status()
                # 成功：提交所有 offset
                await self._commit_offsets(offsets)
                self._metrics["rows_ok"] += len(rows)
                self._metrics["flush"] += 1
                self._last_flush = time.time()
                return
            except Exception:
                self._metrics["retry"] += 1
                await asyncio.sleep(backoff)
                backoff *= 2
        # 最终失败：计错并放弃（at-least-once + ReplacingMergeTree 可撑住重复）
        self._metrics["rows_err"] += len(rows)
        self._last_flush = time.time()

    async def _commit_offsets(self, offsets: List[tuple]):
        # offsets: [(topic, partition, offset), ...] 取每个tp最大offset+1
        tp_max = {}
        for t,p,o in offsets:
            tp_max[(t,p)] = max(tp_max.get((t,p), -1), o)
        await self.consumer.commit({ self.consumer.assignment().__class__().tp(t,p): o+1
                                     for (t,p), o in tp_max.items() })

# ---- 启动样例 ----
async def main():
    cfg = CHWriterConfig(
        kafka_bootstrap=os.getenv("KAFKA_BOOTSTRAP","localhost:9092"),
        kafka_group=os.getenv("KAFKA_CHWRITER_GROUP","ch-writer-bar5s"),
        topics=(os.getenv("TOPIC_BAR","md.bar.5s.v2"), os.getenv("TOPIC_REPAIR","md.bar.5s.repair.v1")),
        ch_dsn=os.getenv("CH_DSN","http://localhost:8123"),
        ch_db=os.getenv("CH_DB","qsys"),
        ch_table=os.getenv("CH_TABLE","bar_5s_v2"),
        ch_user=os.getenv("CH_USER",""),
        ch_pass=os.getenv("CH_PASS",""),
        batch_max_rows=int(os.getenv("BATCH_MAX_ROWS","1000")),
        batch_flush_ms=int(os.getenv("BATCH_FLUSH_MS","300")),
        request_timeout_s=float(os.getenv("CH_TIMEOUT_S","2.0")),
    )
    w = CHWriter(cfg)
    await w.start()
    print("[CHWriter] online.")
    try:
        while True:
            await asyncio.sleep(10)
            # 这里可以打印指标或暴露 Prom：print(w._metrics)
    except KeyboardInterrupt:
        pass
    finally:
        await w.stop()

if __name__ == "__main__":
    asyncio.run(main())
```

## 3) 验收清单（5 分钟冒烟）

1. **先跑聚合器**（我们上一条已给）：确认 `md.bar.5s.v2` 有流。
2. **起 CH Writer**：看到 `[CHWriter] online.`；`kcat -C -t md.bar.5s.v2` 能看到消息。
3. **查表**：

   ```sql
   SELECT symbol, ts_start, close, bar_seq, source FROM qsys.bar_5s_v2 ORDER BY ts_start DESC LIMIT 5;
   ```

   能持续出现新行，`bar_seq` 单调上升。
4. **迟到修复**：把聚合器 `emit_buffer_ms` 暂调 10ms，观察 `md.bar.5s.repair.v1`；再查表，看到同 `(symbol, ts_start)` 的 `bar_seq` 被**更大版本覆盖**。
5. **压力与限速**：把 `batch_max_rows` 调大（如 5000），确保 1s 内不会多次小批写入；观察 `_metrics` 的 `flush/retry`。

## 4) 为什么优先做 CH Writer

* 它把**数据落地**，让策略引擎的 ClickHouse DAL 读到“统一、可回溯”的 5s/1m 数据；
* 有了 `bar_seq` 的**数值版本覆盖**，你后续做**回补**和**迟到修复**就天然安全；
* 生产属性（批量、重试、手动提交 offset）做好后，系统抗压和一致性就稳了。
