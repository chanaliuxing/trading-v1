我来拍板：**先把 MarketDataFacade（ClickHouse 读层）落地**。
这是策略引擎与追踪止损策略的“供血管道”，没有它，再好的执行内核也“巧妇难为无米之炊”。下面直接给你**可粘贴**的异步 Facade 实现，支持：

* 统一读取 **5 秒原始 Bar** 与 **任意 N 秒/1 分钟级聚合**（用 ClickHouse `toStartOfInterval` 聚合）
* 先“去重选最新版本”（用 `bar_seq` 做 `argMax`），再做 OHLC 聚合，**保证修复覆盖后读取正确**
* 每策略独立节拍拉取（5s/10s/60s…），**本地 TTL 缓存**避免打满 CH
* 支持 **RTH 过滤**，不足数据时返回空，策略端自动“硬止损兜底”
* 只用 `httpx` 调 CH HTTP 接口，无额外依赖，与你之前的写入栈完全兼容

# 二、异步 ClickHouse Facade（含 TTL 缓存与 5s→Ns 聚合）

```python
import time, asyncio, httpx, math
from typing import Dict, Tuple, List, Optional

def _now() -> float: return time.time()

class _TTLCache:
    """超轻量 TTL 缓存：key->(expire_ts, value)"""
    def __init__(self): self._m: Dict[Tuple, Tuple[float, object]] = {}
    def get(self, key):
        v = self._m.get(key)
        if not v: return None
        exp, val = v
        if _now() > exp:
            self._m.pop(key, None)
            return None
        return val
    def set(self, key, val, ttl_s: float):
        self._m[key] = (_now() + max(ttl_s, 0.1), val)

def _esc_symbol(sym: str) -> str:
    # ClickHouse HTTP 参数更安全（参见 _q），此函数仅兜底保护日志/拼接时用
    return sym.replace("'", "''")

class MarketDataFacadeCH:
    """
    ClickHouse 读取层：
      - 表：qsys.bar_5s_v2（ReplacingMergeTree(bar_seq)）
      - 先以 (symbol, ts_start) 维度用 argMax(..., bar_seq) 做“版本去重”，再按目标时间框聚合
    """
    def __init__(self, dsn: str = "http://localhost:8123", database: str = "qsys",
                 user: str = "", password: str = "", table: str = "bar_5s_v2",
                 rth_only_default: bool = True):
        self.base = dsn.rstrip("/")
        self.db = database
        self.user = user
        self.password = password
        self.table = table
        self.rth_only_default = rth_only_default
        self.http = httpx.AsyncClient(base_url=self.base, timeout=2.0,
                                      headers={"Content-Type":"text/plain; charset=utf-8"})
        self.cache = _TTLCache()

    async def close(self):
        await self.http.aclose()

    async def last_price(self, symbol: str) -> Optional[float]:
        """
        最近一根 5s bar 的收盘价（已去重挑最新版本）；
        若 5 分钟内无数据，返回 None。
        """
        sql = f"""
        WITH base AS (
            SELECT
              argMax(close, bar_seq) AS close,
              max(ts_start) AS max_ts
            FROM {self.db}.{self.table}
            WHERE symbol = {{sym:String}}
              AND ts_start >= toUInt32(now()) - 300
              {{rth_filter}}
        )
        SELECT close FROM base
        """
        params = {"sym": symbol}
        rth_filter = "AND rth = 1"  # 默认 RTH
        sql = sql.replace("{rth_filter}", rth_filter if self.rth_only_default else "")
        rows = await self._q(sql, params)
        return float(rows[0]["close"]) if rows else None

    async def bars(self, symbol: str, timeframe_s: int, lookback: int,
                   rth_only: Optional[bool] = None) -> List[Bar]:
        """
        读取最后 lookback 根 N 秒级 Bar。
          - timeframe_s == 5 → 直接取 5s 原表（已去重）
          - timeframe_s > 5 → 以 5s 为基，toStartOfInterval 聚合成 N 秒
        """
        rth = self.rth_only_default if rth_only is None else bool(rth_only)
        key = ("bars", symbol, timeframe_s, lookback, int(rth))
        cached = self.cache.get(key)
        if cached is not None:
            return cached

        # 估算需要的时间范围（多给 2 个桶做安全垫）
        horizon = (lookback + 2) * max(5, timeframe_s)
        sql_5s_window = f"ts_start >= toUInt32(now()) - {horizon}"

        if timeframe_s <= 5:
            # 直接 5s：先 per (ts_start) 选最新版本，再取最后 N 根
            sql = f"""
            SELECT ts_start, ts_end,
                   argMax(open,  bar_seq) AS o,
                   argMax(high,  bar_seq) AS h,
                   argMax(low,   bar_seq) AS l,
                   argMax(close, bar_seq) AS c,
                   argMax(vol,   bar_seq) AS v
            FROM {self.db}.{self.table}
            WHERE symbol = {{sym:String}}
              AND {sql_5s_window}
              {{rth_filter}}
            GROUP BY ts_start, ts_end
            ORDER BY ts_start DESC
            LIMIT {{lim:UInt32}}
            """
            params = {"sym": symbol, "lim": lookback}

            rows = await self._q(sql.replace("{rth_filter}", "AND rth = 1" if rth else ""), params)
            rows = list(reversed(rows))  # 还原升序
            out = [Bar(ts=int(r["ts_start"]), o=float(r["o"]), h=float(r["h"]),
                       l=float(r["l"]), c=float(r["c"]), v=float(r["v"])) for r in rows]
        else:
            # Ns 聚合：先以 (symbol, ts_start) 去重，再 toStartOfInterval 聚合
            sql = f"""
            WITH base AS (
              SELECT symbol, ts_start, ts_end,
                     argMax(open,  bar_seq) AS open,
                     argMax(high,  bar_seq) AS high,
                     argMax(low,   bar_seq) AS low,
                     argMax(close, bar_seq) AS close,
                     argMax(vol,   bar_seq) AS vol,
                     argMax(vwap,  bar_seq) AS vwap,
                     argMax(rth,   bar_seq) AS rth
              FROM {self.db}.{self.table}
              WHERE symbol = {{sym:String}}
                AND {sql_5s_window}
                {{rth_filter}}
              GROUP BY symbol, ts_start, ts_end
            )
            SELECT
              bucket AS ts_start,
              toUInt32(bucket + {{tf:UInt32}}) AS ts_end,
              argMin(open, ts_start)  AS o,
              max(high)               AS h,
              min(low)                AS l,
              argMax(close, ts_start) AS c,
              sum(vol)                AS v,
              -- vwap 可选：sum(vwap*vol)/sum(vol)
              -- 为简洁，这里不返回 vwap，策略侧通常用 c 即可
              count() AS cnt
            FROM (
              SELECT *, toStartOfInterval(toDateTime(ts_start), INTERVAL {{tf:UInt32}} SECOND) AS bucket
              FROM base
            )
            GROUP BY bucket
            ORDER BY bucket DESC
            LIMIT {{lim:UInt32}}
            """
            params = {"sym": symbol, "tf": timeframe_s, "lim": lookback}
            rows = await self._q(sql.replace("{rth_filter}", "AND rth = 1" if rth else ""), params)
            rows = list(reversed(rows))
            out = [Bar(ts=int(r["ts_start"]), o=float(r["o"]), h=float(r["h"]),
                       l=float(r["l"]), c=float(r["c"]), v=float(r["v"])) for r in rows]

        # TTL：按时间框的一半缓存，避免抖动；最短 1s
        ttl = max(1.0, timeframe_s * 0.5)
        self.cache.set(key, out, ttl)
        return out

    async def _q(self, sql: str, params: Dict) -> List[Dict]:
        """
        通过 ClickHouse HTTP 参数化执行（防注入）：
          - 在 SQL 中使用 {name:Type} 占位
          - 通过 URL 传递 param_name=value
        """
        # 组 URL 与参数
        url = f"/?database={self.db}&default_format=JSON"
        # 将 params 映射为 &param_x=... 形式
        qp = {f"param_{k}": v for k, v in params.items()}
        auth = (self.user, self.password) if self.user else None
        r = await self.http.post(url, params=qp, content=sql.encode("utf-8"), auth=auth)
        r.raise_for_status()
        j = r.json()
        return j.get("data", [])
```

## 四、几个工程要点（已替你踩坑）

1. **版本一致性**：
   所有读路径都先用 `argMax(..., bar_seq)` 在 (symbol, ts_start) 维度做“最新版本”聚合，**确保修复覆盖**后读到的就是*唯一*版本；再做更高层的 N 秒聚合。

2. **聚合正确性（OHLC）**：

   * open：`argMin(open, ts_start)`（区间内最早 ts_start 的 open）
   * high/low：`max / min`
   * close：`argMax(close, ts_start)`（区间内最晚 ts_start 的 close）
   * vol：`sum(vol)`
   * （可选）vwap：`sum(vwap*vol)/sum(vol)`（示例里留了注释位）

3. **范围估算**：
   读取最后 `lookback` 根 Ns Bar 时，用 `(lookback+2)*Ns` 秒的时间窗当作 WHERE 条件，避免全表扫描。
   如果你担心窗口不足，可以把 `+2` 增大到 `+4`。

4. **缓存策略**：
   采用半个时间框的 TTL（例如 5s bar 缓 2.5s，向上取 1s），对“策略引擎 5s 节拍”来说几乎无影响，但可以显著降低 CH QPS。

5. **RTH 过滤**：
   默认 `rth_only_default=True`，SQL 注入 `AND rth=1`；如需夜盘/盘前也看数据，策略侧可以 `bars(..., rth_only=False)`。

## 六、为什么这一步最重要

* 你的**秒级策略**与**追踪止损**都依赖“稳定、正确、可回溯”的 K 线序列；
* 通过“**去重后再聚合**”，彻底解决“修复覆盖但读到旧版本”的一致性问题；
* 轻量 HTTP + TTL 缓存，**部署成本最低**，又能顶住多策略并发读取；
* 与现有 CH Writer / Backfill / 执行内核完全解耦，**出问题也不拖累执行**。
