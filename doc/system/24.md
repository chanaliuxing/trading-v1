# 1) Kafka 适配器（异步、可多订阅）

```python
# ========= KafkaBusAdapter (aiokafka) =========
# 依赖: aiokafka>=0.8 ; Python 3.10+
import asyncio, json, contextlib, uuid
from typing import Dict, List, Optional
from aiokafka import AIOKafkaProducer, AIOKafkaConsumer

class KafkaBusAdapter:
    """
    与引擎的 Bus 接口一致：
      - subscribe(topic) -> asyncio.Queue  (每个订阅者拿到自己的队列)
      - publish(topic, key:str, value:dict)
    实现细节：
      - 每个 topic 一个 consumer task（同 group_id）
      - producer 复用一个实例
    """
    def __init__(self, bootstrap: str, group_id: str, loop: Optional[asyncio.AbstractEventLoop]=None):
        self.bootstrap = bootstrap
        self.group_id = group_id
        self.loop = loop or asyncio.get_event_loop()
        self._producer: Optional[AIOKafkaProducer] = None
        self._consumers: Dict[str, AIOKafkaConsumer] = {}
        self._consumer_tasks: Dict[str, asyncio.Task] = {}
        self._subscribers: Dict[str, List[asyncio.Queue]] = {}
        self._running = False

    async def start(self):
        if self._running: return
        self._producer = AIOKafkaProducer(
            loop=self.loop,
            bootstrap_servers=self.bootstrap,
            acks="all",
            linger_ms=5,
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda k: (k or "").encode("utf-8"),
            compression_type="lz4",
        )
        await self._producer.start()
        self._running = True

    async def stop(self):
        for t in list(self._consumer_tasks.values()):
            t.cancel()
        for topic, consumer in list(self._consumers.items()):
            with contextlib.suppress(Exception):
                await consumer.stop()
        self._consumers.clear()
        self._consumer_tasks.clear()
        if self._producer:
            with contextlib.suppress(Exception):
                await self._producer.stop()
        self._running = False

    def subscribe(self, topic: str) -> asyncio.Queue:
        """
        第一次订阅某个 topic 时，创建一个 consumer 和分发 task。
        之后的订阅者共享该 consumer（每个订阅独立队列）。
        """
        q = asyncio.Queue()
        self._subscribers.setdefault(topic, []).append(q)
        if topic not in self._consumers:
            consumer = AIOKafkaConsumer(
                topic,
                loop=self.loop,
                bootstrap_servers=self.bootstrap,
                group_id=self.group_id,
                enable_auto_commit=True,
                auto_offset_reset="latest",
                value_deserializer=lambda b: json.loads(b.decode("utf-8")),
                key_deserializer=lambda b: b.decode("utf-8") if b else "",
            )
            self._consumers[topic] = consumer
            self._consumer_tasks[topic] = asyncio.create_task(self._consume_loop(topic))
        return q

    async def _consume_loop(self, topic: str):
        consumer = self._consumers[topic]
        await consumer.start()
        try:
            async for msg in consumer:
                # 统一包装给引擎
                item = {"topic": topic, "key": msg.key, "value": msg.value, "partition": msg.partition, "offset": msg.offset}
                for q in self._subscribers.get(topic, []):
                    # 背压：队列默认无限，必要时可以在外层限流
                    await q.put(item)
        finally:
            await consumer.stop()

    async def publish(self, topic: str, key: str, value: dict):
        if not self._running:
            raise RuntimeError("KafkaBusAdapter not started")
        await self._producer.send_and_wait(topic, value=value, key=key or str(uuid.uuid4()))
```

# 2) ClickHouse 异步 DAL（HTTP 原生、带 RingCache + PREWHERE + 超时降级）

```python
# ========= ClickHouseDal (httpx AsyncClient) =========
# 依赖: httpx>=0.27 ; Python 3.10+
import asyncio, time, json, hashlib
from typing import Dict, List, Optional, Tuple
import httpx

class TTLCache:
    def __init__(self, ttl_sec: float, max_items: int = 1024):
        self.ttl = ttl_sec
        self.max = max_items
        self.data: Dict[str, Tuple[float, any]] = {}

    def _evict(self):
        if len(self.data) <= self.max: return
        # 粗暴淘汰：按时间排序丢前 5%
        items = sorted(self.data.items(), key=lambda kv: kv[1][0])
        for k,_ in items[: max(1, len(items)//20) ]:
            self.data.pop(k, None)

    def get(self, key: str):
        v = self.data.get(key)
        if not v: return None
        ts, val = v
        if time.time() - ts > self.ttl:
            self.data.pop(key, None)
            return None
        return val

    def set(self, key: str, val):
        self.data[key] = (time.time(), val)
        self._evict()

class ClickHouseDal:
    """
    读取 5s/1m bar 与分时特征的异步 DAL。
    - 使用 HTTP 接口（httpx）直接发 SQL，避免额外依赖；
    - PREWHERE 限定 symbol 和时间窗口，减扫描；
    - 带 TTL Cache（2s）与请求超时（默认 0.8s），超时降级为返回 None（引擎会跳过）。
    """
    def __init__(self, dsn: str, user: str = "", password: str = "", database: str = "qsys",
                 timeout_ms: int = 800, cache_ttl_sec: float = 2.0):
        # dsn 示例: "http://ch-host:8123"
        self.base = dsn.rstrip("/")
        self.auth = (user, password) if user else None
        self.db = database
        self.timeout = timeout_ms/1000.0
        self._client: Optional[httpx.AsyncClient] = None
        self.cache = TTLCache(ttl_sec=cache_ttl_sec)

    async def start(self):
        self._client = httpx.AsyncClient(base_url=self.base, timeout=self.timeout)

    async def stop(self):
        if self._client:
            await self._client.aclose()
            self._client = None

    # ---------- 公共查询器 ----------
    async def _query(self, sql: str, params: dict) -> Optional[List[dict]]:
        if not self._client:
            raise RuntimeError("ClickHouseDal not started")
        # 简单参数化：把 {name:val} 替换成 CH 的参数格式
        # 也可以使用 POST JSON，每家风格不同；这里选最通用的 SQL + params
        try:
            r = await self._client.post("/?database="+self.db, content=sql, params=params)
            r.raise_for_status()
            # 默认 TSV/CSV；这里改 JSONEachRow
            # 建议在 SQL 尾部加 " FORMAT JSONEachRow"
            lines = r.text.strip().splitlines()
            return [json.loads(line) for line in lines if line.strip()]
        except Exception:
            return None

    @staticmethod
    def _k(*parts) -> str:
        return hashlib.md5("|".join(map(str,parts)).encode()).hexdigest()

    # ---------- 接口：最近 N 根 ----------
    async def last_n(self, symbol: str, n: int, res: str = "5s") -> Optional[List[dict]]:
        """
        返回 [{ts_start, ts_end, open, high, low, close, vol, vwap}, ...] 升序
        """
        table = "bar_5s_v2" if res == "5s" else "bar_1m_v2"
        key = self._k("last_n", symbol, n, res)
        v = self.cache.get(key)
        if v is not None: return v

        # 近 n 根窗口
        if res == "5s":
            sql = f"""
SELECT
  symbol, ts_start, ts_end, open, high, low, close, vol, vwap
FROM qsys.{table}
PREWHERE symbol = {{sym:String}}
WHERE ts_start >= now() - INTERVAL {{win:Int32}} SECOND
ORDER BY ts_start ASC
FORMAT JSONEachRow
"""
            params = {"sym": symbol, "win": n*5 + 5}
        else:
            sql = f"""
SELECT symbol, ts_start, ts_end, open, high, low, close, vol
FROM qsys.{table}
PREWHERE symbol = {{sym:String}}
WHERE ts_start >= now() - INTERVAL {{win:Int32}} MINUTE
ORDER BY ts_start ASC
FORMAT JSONEachRow
"""
            params = {"sym": symbol, "win": n + 2}

        rows = await self._query(sql, params)
        if rows is None or len(rows) == 0:
            return None
        # 只保留最后 n 根（以防窗口略大）
        rows = rows[-n:] if len(rows) > n else rows
        self.cache.set(key, rows)
        return rows

    # ---------- 接口：特征（近窗口） ----------
    async def features_intraday(self, symbol: str, fields: List[str], window: str = "30m") -> Optional[List[dict]]:
        """
        返回 [{ts, field1:..., field2:...}, ...]
        window: '30m' / '10m' / '2h' ...
        """
        key = self._k("feat", symbol, ",".join(sorted(fields)), window)
        v = self.cache.get(key)
        if v is not None: return v

        cols = ", ".join(fields)
        sql = f"""
SELECT ts, {cols}
FROM qsys.feat_intraday_v2
PREWHERE symbol = {{sym:String}}
WHERE ts >= now() - toIntervalSecond({self._parse_window(window)})
ORDER BY ts ASC
FORMAT JSONEachRow
"""
        params = {"sym": symbol}
        rows = await self._query(sql, params)
        if rows is None:
            return None
        self.cache.set(key, rows)
        return rows

    # ---------- 接口：日快照 ----------
    async def daily_snapshot(self, symbols: List[str], asof: Optional[str] = None) -> Optional[List[dict]]:
        """
        返回 [{symbol, asof_date, atr_1d, vol_rank_20d, beta, ...}]
        """
        syms = ",".join([f"'{s}'" for s in symbols])
        cond = f"asof_date = toDate('{asof}')" if asof else "asof_date = today()"
        key = self._k("daily", syms, cond)
        v = self.cache.get(key)
        if v is not None: return v

        sql = f"""
SELECT *
FROM qsys.feat_daily_snapshot_v2
WHERE symbol IN ({syms}) AND {cond}
FORMAT JSONEachRow
"""
        rows = await self._query(sql, {})
        if rows is None: return None
        self.cache.set(key, rows)
        return rows

    @staticmethod
    def _parse_window(s: str) -> int:
        # '30m' -> 1800 ; '2h' -> 7200
        if s.endswith("m"): return int(s[:-1]) * 60
        if s.endswith("h"): return int(s[:-1]) * 3600
        if s.endswith("s"): return int(s[:-1])
        return int(s)
```

## 4) 关键对齐点（确保一步即跑）

* **接口对齐**：`KafkaBusAdapter.subscribe/publish` 与你引擎的 Bus 接口**完全一致**；把你原来的 `Bus` 替换成这里的 Kafka 版即可。
* **事件格式**：`md.bar.5s.v2` 的消息体需要是你聚合器产出的 JSON（我们统一用 `{"schema_v":2, "metadata":..., "data":{...}}`），引擎只会用到 `value["value"]["data"]` 内的 `symbol/ts_start/ts_end/close/...`。
* **引擎节奏**：每策略独立节拍（≥5s），`KafkaBusAdapter` 会把同一 topic 的消息送到**各自队列**，不会互相拖慢。
* **ClickHouse 读**：DAL 超时（默认 800ms）返回 `None`，引擎就会走 `ready/skip`，**不阻塞**、**不乱下单**。
