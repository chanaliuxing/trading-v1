收到！下面直接给你**“执行安全内核（Safety Kernel）”可运行骨架**：
覆盖我们定下的关键能力——**首填即挂 OCO**、**部分成交按比例调整**、**失败重试与巡检强撤**、**幽灵订单领养**、**Stop-MKT 兜底**、**速率限制（RPM）**、**幂等**、**指标导出**。
保持“**不生成文件**”，你只需把代码片段粘到你的服务中，替换/对接你的 IB 适配即可。

# 1) 事件与配置（与既有契约对齐）

* 输入：

  * `risk.decision.v2`（风控批准的下单请求，`data.order_type/side/qty/limit_px/exec_overrides/...`）
  * `exec.order.status.v2`（券商状态回报：Submitted/Filled/Cancelled/Inactive/...）
  * `exec.fill.v2`（成交回报：`order_id, qty, px, liquidity, ts`）

* 内核直接调用 **IB 适配**（接口在下方 `IBAdapter`）完成 `place/modify/cancel/query`。
  如你已有“独立 IB 服务 + 命令 Topic”，只要把适配器换成“Kafka 命令版适配器”即可，内核不变。

* 重要默认：

  * `place_rpm=10 / modify_rpm=30 / cancel_rpm=30`（可由环境变量覆盖）
  * 首笔成交 ≤1s 内**必须**挂出 OCO（SLO：`unprotected_window_seconds==0`）

# 2) 速率限制与通用工具

```python
import asyncio, time, json, uuid, os, contextlib, math
from dataclasses import dataclass, field
from typing import Dict, Optional, List, Tuple

# ============ 通用：令牌桶 ============
class TokenBucket:
    def __init__(self, rpm:int):
        self.capacity = max(1, rpm)
        self.refill_per_sec = self.capacity/60.0
        self.tokens = float(self.capacity)
        self.ts = time.time()

    def take(self, n:float=1.0) -> bool:
        now = time.time()
        elapsed = now - self.ts
        self.ts = now
        self.tokens = min(self.capacity, self.tokens + elapsed*self.refill_per_sec)
        if self.tokens >= n:
            self.tokens -= n
            return True
        return False

    async def wait(self, n:float=1.0):
        while not self.take(n):
            await asyncio.sleep(0.05)

def now_us(): return int(time.time()*1e6)

def idem(sid:str, sym:str) -> str:
    return f"ord:req:{sid}:{sym}:{uuid.uuid4()}"

def side_flip(side:str) -> str:
    s=side.upper()
    return "BUY" if s=="SELL" else "SELL"
```

```python
# ============ IB 适配接口 ============
@dataclass
class PlacedOrder:
    order_id: int
    order_ref: str
    symbol: str
    side: str         # BUY / SELL
    order_type: str   # LMT / STP / STP_LMT / MKT
    qty: float
    limit_px: Optional[float]=None
    stop_px: Optional[float]=None
    tif: str="DAY"
    oco_group: Optional[str]=None
    status: str="Submitted"   # PreSubmitted / Submitted / Filled / Cancelled / Inactive ...
    ts_us: int = field(default_factory=now_us)

class IBAdapter:
    async def place_limit(self, symbol:str, side:str, qty:float, px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def place_stop(self, symbol:str, side:str, qty:float, stop_px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def place_stop_limit(self, symbol:str, side:str, qty:float, stop_px:float, limit_px:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def place_market(self, symbol:str, side:str, qty:float, tif:str, order_ref:str, oco_group:Optional[str]=None) -> PlacedOrder: ...
    async def modify(self, order_id:int, **fields) -> bool: ...
    async def cancel(self, order_id:int) -> bool: ...
    async def open_orders(self) -> List[PlacedOrder]: ...
```

> 你可用 `ib_insync` 实现上述接口：`orderRef` 写 `order_ref`，OCO 用 `ocaGroup`/`ocaType`。
> 若你的 IB 层是**独立进程**，可把这些方法改为**向 Kafka “exec.order.cmd.v1” 发送命令**并等待回执。

# 5) 安全内核核心数据结构

```python
# ============ Plan/Position 状态 ============
@dataclass
class Position:
    qty: float = 0.0
    avg_px: float = 0.0
    def on_fill(self, side:str, fill_qty:float, fill_px:float):
        s = 1 if side.upper()=="BUY" else -1
        new_qty = self.qty + s*fill_qty
        if self.qty == 0 or (self.qty>0 and new_qty>0) or (self.qty<0 and new_qty<0):
            # 同向开/加仓：加权均价
            self.avg_px = (abs(self.qty)*self.avg_px + fill_qty*fill_px) / (abs(self.qty)+fill_qty) if abs(self.qty)+fill_qty>0 else 0.0
        elif (self.qty>0 and new_qty<0) or (self.qty<0 and new_qty>0):
            # 反转：余量以本次成交价为均价
            self.avg_px = fill_px
        elif new_qty == 0:
            self.avg_px = 0.0
        self.qty = new_qty

@dataclass
class Plan:
    strategy_id: str
    symbol: str
    decision_id: str
    idempotency_key: str
    tif: str
    exec_overrides: Dict
    entry_order_ids: List[int] = field(default_factory=list)
    oco_group: Optional[str] = None
    stop_order_id: Optional[int] = None
    tp_order_id: Optional[int] = None
    pos: Position = field(default_factory=Position)
    protection_attached: bool = False
    first_fill_ts_ms: Optional[int] = None
    last_activity_ts: float = field(default_factory=time.time)
    closed: bool = False
```

# 6) 安全内核主服务

```python
from aiokafka import AIOKafkaProducer, AIOKafkaConsumer

class SafetyKernelConfig:
    kafka_bootstrap = os.getenv("KAFKA_BOOTSTRAP","localhost:9092")
    group_id        = os.getenv("KERNEL_GROUP","exec-kernel-v1")
    topic_decision  = os.getenv("TOPIC_DECISION","risk.decision.v2")
    topic_status    = os.getenv("TOPIC_STATUS","exec.order.status.v2")
    topic_fill      = os.getenv("TOPIC_FILL","exec.fill.v2")
    metrics_port    = int(os.getenv("METRICS_PORT","9105"))
    place_rpm       = int(os.getenv("IB_PLACE_RPM","10"))
    modify_rpm      = int(os.getenv("IB_MODIFY_RPM","30"))
    cancel_rpm      = int(os.getenv("IB_CANCEL_RPM","30"))
    # 价格保护策略缺省（可由 exec_overrides 覆盖）
    hard_stop_pct   = float(os.getenv("HARD_STOP_PCT","0.90"))   # 多头：止损=avg_px*0.9；空头相反
    take_profit_bps = int(os.getenv("TAKE_PROFIT_BPS","40"))     # 40 bps
    use_stop_limit  = os.getenv("USE_STOP_LIMIT","false").lower()=="true"
    stop_limit_slip_bps = int(os.getenv("STOP_LIMIT_SLIP_BPS","20"))
    inspect_interval_s  = int(os.getenv("INSPECT_INTERVAL_S","5"))
    adopt_interval_s    = int(os.getenv("ADOPT_INTERVAL_S","60"))
    max_retry           = int(os.getenv("KERNEL_MAX_RETRY","3"))

class SafetyKernel:
    def __init__(self, adapter: IBAdapter, cfg: SafetyKernelConfig=SafetyKernelConfig()):
        self.cfg = cfg
        self.adapter = adapter
        self.producer: Optional[AIOKafkaProducer] = None
        self.c_decision: Optional[AIOKafkaConsumer] = None
        self.c_status: Optional[AIOKafkaConsumer] = None
        self.c_fill: Optional[AIOKafkaConsumer] = None
        # 状态
        self.plans: Dict[str, Plan] = {}     # decision_id -> Plan
        self.orders: Dict[int, str] = {}     # order_id -> decision_id
        # RPM 限速
        self.tb_place = TokenBucket(cfg.place_rpm)
        self.tb_modify= TokenBucket(cfg.modify_rpm)
        self.tb_cancel= TokenBucket(cfg.cancel_rpm)

    async def start(self):
        boot_metrics(self.cfg.metrics_port)
        # Kafka
        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.cfg.kafka_bootstrap,
            acks="all", linger_ms=5, compression_type="lz4",
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda k: k.encode("utf-8") if k else None
        )
        await self.producer.start()
        loop = asyncio.get_running_loop()
        self.c_decision = AIOKafkaConsumer(
            self.cfg.topic_decision, bootstrap_servers=self.cfg.kafka_bootstrap, group_id=self.cfg.group_id,
            value_deserializer=lambda b: json.loads(b.decode("utf-8")), key_deserializer=lambda b: b.decode("utf-8") if b else "",
            auto_offset_reset="latest", enable_auto_commit=True
        )
        self.c_status = AIOKafkaConsumer(
            self.cfg.topic_status, bootstrap_servers=self.cfg.kafka_bootstrap, group_id=self.cfg.group_id,
            value_deserializer=lambda b: json.loads(b.decode("utf-8")), key_deserializer=lambda b: b.decode("utf-8") if b else "",
            auto_offset_reset="latest", enable_auto_commit=True
        )
        self.c_fill = AIOKafkaConsumer(
            self.cfg.topic_fill, bootstrap_servers=self.cfg.kafka_bootstrap, group_id=self.cfg.group_id,
            value_deserializer=lambda b: json.loads(b.decode("utf-8")), key_deserializer=lambda b: b.decode("utf-8") if b else "",
            auto_offset_reset="latest", enable_auto_commit=True
        )
        await self.c_decision.start(); await self.c_status.start(); await self.c_fill.start()
        # 协程
        asyncio.create_task(self._loop_decision())
        asyncio.create_task(self._loop_status())
        asyncio.create_task(self._loop_fill())
        asyncio.create_task(self._inspector())
        asyncio.create_task(self._adopt_loop())

    async def stop(self):
        for c in (self.c_decision, self.c_status, self.c_fill):
            if c: with contextlib.suppress(Exception): await c.stop()
        if self.producer: with contextlib.suppress(Exception): await self.producer.stop()

    # ========== 消费：风控批准 ==========
    async def _loop_decision(self):
        async for msg in self.c_decision:
            try:
                d = msg.value["data"]; meta = msg.value.get("metadata",{})
                if msg.value.get("decision","APPROVED") != "APPROVED":
                    continue
                sid = d["strategy_id"]; sym = d["symbol"]; side = d["side"].upper()
                qty = float(d["qty"]); px = float(d.get("limit_px") or 0.0)
                tif = d.get("tif","DAY")
                overrides = d.get("exec_overrides",{})
                decision_id = meta.get("decision_id") or meta.get("idempotency_key") or str(uuid.uuid4())
                idem_key = meta.get("idempotency_key") or idem(sid, sym)

                # 幂等：已存在同 decision_id 的直接忽略
                if decision_id in self.plans: 
                    continue

                plan = Plan(strategy_id=sid, symbol=sym, decision_id=decision_id,
                            idempotency_key=idem_key, tif=tif, exec_overrides=overrides)
                self.plans[decision_id] = plan

                # 下达入场单
                await self.tb_place.wait()
                po = await self.adapter.place_limit(sym, side, qty, px, tif, order_ref=idem_key)
                MET["kern_place"].labels(kind="ENTRY").inc()
                self.orders[po.order_id] = decision_id
                plan.entry_order_ids.append(po.order_id)
                plan.last_activity_ts = time.time()
            except Exception:
                MET["kern_err"].labels(op="decision").inc()

    # ========== 消费：订单状态 ==========
    async def _loop_status(self):
        async for msg in self.c_status:
            try:
                s = msg.value["data"]  # 假定包含 order_id, status
                oid = int(s["order_id"]); status = s["status"]
                did = self.orders.get(oid)
                if not did: 
                    # 可能是新启动时的遗留状态，等待 adopt_loop
                    continue
                plan = self.plans.get(did)
                if not plan: continue
                plan.last_activity_ts = time.time()

                # 一侧 OCO 成交 → 取消另一侧 + 取消所有入场挂单 + 标记关闭
                if status.upper()=="FILLED":
                    # 等 fill 回来更新仓位后，inspector 会判断是否仍需保护
                    pass
                elif status.upper() in ("CANCELLED","INACTIVE"):
                    # 清理本地映射
                    pass
            except Exception:
                MET["kern_err"].labels(op="status").inc()

    # ========== 消费：成交 ==========
    async def _loop_fill(self):
        async for msg in self.c_fill:
            try:
                f = msg.value["data"]   # order_id, symbol, side, qty, price, ts_us
                oid = int(f["order_id"]); sym=f["symbol"]
                side=f["side"].upper(); fill_qty=float(f["qty"]); fill_px=float(f["price"])
                did = self.orders.get(oid)
                if not did:
                    continue
                plan = self.plans.get(did)
                if not plan: 
                    continue
                plan.pos.on_fill(side, fill_qty, fill_px)
                if plan.first_fill_ts_ms is None:
                    plan.first_fill_ts_ms = int((msg.value.get("metadata",{}).get("ts_us") or now_us())/1000)
                    # 首填后立即确保 OCO
                    await self._ensure_oco(plan)
                else:
                    # 部分成交：按比例调整 OCO 数量
                    await self._ensure_oco(plan)
                plan.last_activity_ts = time.time()
            except Exception:
                MET["kern_err"].labels(op="fill").inc()

    # ========== 保护：计算/挂载/调整 OCO ==========
    async def _ensure_oco(self, plan: Plan):
        """
        - 若无 OCO：首填后 1s 内必须挂出
        - 若已有 OCO：按 plan.pos.qty 调整数量/价格（追踪、或固定）
        - 若仓位已平：撤销所有保护与入场，标记 plan.closed
        """
        pos = plan.pos.qty
        if abs(pos) < 1e-8:
            # 平仓：撤销一切
            await self._cancel_all(plan)
            plan.protection_attached=False; plan.closed=True
            return

        # 计算方向与目标价格
        long = pos > 0
        avg = plan.pos.avg_px or 0.0
        ovr = plan.exec_overrides or {}
        hard_pct = float(ovr.get("hard_stop_pct", self.cfg.hard_stop_pct))   # e.g. 0.9
        tp_bps   = int(ovr.get("take_profit_bps", self.cfg.take_profit_bps)) # e.g. 40 bps

        # 期望止损/止盈价（可替换为 ATR/通道等模式，本骨架先给 FIX/MIXED 实现位）
        stop_px = avg*hard_pct if long else avg*(2-hard_pct)   # 空头对称：1/0.9≈1.111...近似 2-hard_pct
        tp_px   = avg*(1+tp_bps/1e4) if long else avg*(1-tp_bps/1e4)

        # 目标方向：OCO 两腿均与仓位方向相反（卖出平多/买入平空）
        oco_side = side_flip("BUY" if long else "SELL")  # 实际等价 "SELL" if long else "BUY"

        # 需要的保护数量 = |pos|
        need_qty = abs(pos)

        # 情况 A：未挂 → 新建 OCO
        if not plan.protection_attached or (plan.stop_order_id is None and plan.tp_order_id is None):
            plan.oco_group = plan.oco_group or f"OCO:{plan.decision_id}"
            await self.tb_place.wait()
            if self.cfg.use_stop_limit:
                # 止损限价 = stop_px*(1 ± slip)
                slip = tp_bps if "stop_limit_slip_bps" not in ovr else int(ovr["stop_limit_slip_bps"])
                lim_px = stop_px*(1 - slip/1e4) if long else stop_px*(1 + slip/1e4)
                so = await self.adapter.place_stop_limit(plan.symbol, oco_side, need_qty, stop_px, lim_px, plan.tif, order_ref=f"{plan.idempotency_key}:STP", oco_group=plan.oco_group)
            else:
                so = await self.adapter.place_stop(plan.symbol, oco_side, need_qty, stop_px, plan.tif, order_ref=f"{plan.idempotency_key}:STP", oco_group=plan.oco_group)
            to = await self.adapter.place_limit(plan.symbol, oco_side, need_qty, tp_px, plan.tif, order_ref=f"{plan.idempotency_key}:TP", oco_group=plan.oco_group)
            MET["kern_place"].labels(kind="STP").inc(); MET["kern_place"].labels(kind="TP").inc()
            plan.stop_order_id = so.order_id; self.orders[so.order_id]=plan.decision_id
            plan.tp_order_id = to.order_id;   self.orders[to.order_id]=plan.decision_id
            plan.protection_attached=True
            MET["kern_oco_attach"].inc()
            # 记录保护窗口
            if plan.first_fill_ts_ms:
                MET["kern_unprotected_window"].observe(max(0, time.time()*1000 - plan.first_fill_ts_ms))
            return

        # 情况 B：已有 OCO → 按仓位调整数量与（可选）价格
        # 这里只对数量做“必要调整”；价格追踪策略可以按你的模式扩展
        async def _adj(oid:int, new_qty:float):
            await self.tb_modify.wait()
            ok = await self.adapter.modify(oid, qty=new_qty)
            if ok: MET["kern_modify"].inc()
            else:  MET["kern_err"].labels(op="modify").inc()
        # 比对数量（示例中我们没有缓存 OID 的当前数量，真实实现应同步状态；这里仅演示调用）
        await _adj(plan.stop_order_id, need_qty)
        await _adj(plan.tp_order_id, need_qty)

    # ========== 取消全部（入场+OCO） ==========
    async def _cancel_all(self, plan:Plan):
        ids = list(plan.entry_order_ids)
        if plan.stop_order_id: ids.append(plan.stop_order_id)
        if plan.tp_order_id:   ids.append(plan.tp_order_id)
        for oid in ids:
            await self.tb_cancel.wait()
            ok = await self.adapter.cancel(oid)
            if ok: MET["kern_cancel"].inc()
            else:  MET["kern_err"].labels(op="cancel").inc()

    # ========== 巡检：强撤/补挂 ==========
    async def _inspector(self):
        while True:
            t0=time.time()
            try:
                for did, plan in list(self.plans.items()):
                    if plan.closed: 
                        continue
                    # 无保护且有仓位 → 立即补挂
                    if abs(plan.pos.qty)>1e-8 and not plan.protection_attached:
                        MET["kern_oco_missing"].set(1)
                        await self._ensure_oco(plan)
                    else:
                        MET["kern_oco_missing"].set(0)
                    # 无仓位但尚有入场挂单 → 强撤
                    if abs(plan.pos.qty)<1e-8 and (plan.entry_order_ids or plan.stop_order_id or plan.tp_order_id):
                        await self._cancel_all(plan)
                        plan.closed=True
                # 也可在此做：长时间未更新的订单 → 重试取消（防“幽灵单”）
            except Exception:
                MET["kern_err"].labels(op="inspect").inc()
            finally:
                MET["kern_inspect_ms"].observe((time.time()-t0)*1000.0)
                await asyncio.sleep(self.cfg.inspect_interval_s)

    # ========== 领养：启动后或定期把券商未识别订单纳入 ==========
    async def _adopt_loop(self):
        while True:
            try:
                await asyncio.sleep(self.cfg.adopt_interval_s)
                open_orders = await self.adapter.open_orders()
                for o in open_orders:
                    did = None
                    # 根据 order_ref 推断 decision_id（我们用 idempotency_key:STP/TP/ENTRY）
                    if o.order_ref and "ord:req:" in o.order_ref:
                        # 兼容 STP/TP 后缀
                        base = o.order_ref.split(":STP")[0].split(":TP")[0]
                        # 我们把 decision_id = idempotency_key（也可以单独字段传入）
                        did = base  # 简化：用 idem 作为 DID；若你有单独 decision_id，就查映射表
                    if not did:
                        # 推断失败：可按 symbol + 最近活动时间窗口 fuzzy adopt
                        MET["kern_adopted"].labels(reason="fuzzy_skip").inc()
                        continue
                    # 纳入计划
                    plan = self.plans.get(did)
                    if not plan:
                        # 创建影子计划，只做清理/保护用途
                        plan = Plan(strategy_id="ADOPT", symbol=o.symbol, decision_id=did, idempotency_key=did, tif="DAY", exec_overrides={})
                        self.plans[did]=plan
                    self.orders[o.order_id]=did
                    if o.order_type in ("LMT","MKT"): 
                        if o.order_id not in plan.entry_order_ids: plan.entry_order_ids.append(o.order_id)
                    elif o.order_type in ("STP","STP_LMT"):
                        plan.stop_order_id = o.order_id
                        plan.protection_attached=True
                    elif o.order_type=="LMT" and o.side in ("BUY","SELL") and o.oco_group:
                        plan.tp_order_id = o.order_id
                        plan.protection_attached=True
                    MET["kern_adopted"].labels(reason="order_ref").inc()
            except Exception:
                MET["kern_err"].labels(op="adopt").inc()
```

在极端情况下（反复修改/取消失败或状态异常），你可以随时调用：

```python
    async def emergency_flatten(self, plan:Plan):
        """紧急兜底：直接以市价反向平仓，然后强撤全部剩余订单"""
        if abs(plan.pos.qty) < 1e-8: 
            await self._cancel_all(plan); plan.closed=True; return
        o_side = side_flip("BUY" if plan.pos.qty>0 else "SELL")
        qty = abs(plan.pos.qty)
        try:
            await self.tb_place.wait()
            await self.adapter.place_market(plan.symbol, o_side, qty, plan.tif, order_ref=f"{plan.idempotency_key}:EMERGENCY")
            MET["kern_place"].labels(kind="EMERGENCY").inc()
        finally:
            await self._cancel_all(plan)
            plan.closed=True
```

> **红线策略**：
>
> * 发现 `first_fill_ts_ms` 后 **1s 内**没有成功挂出 OCO → 立即 `emergency_flatten()`（可在 `_ensure_oco` 中判断多次失败后触发）。
> * 订单“取消失败”重试 `KERNEL_MAX_RETRY` 次仍失败 → 触发 `emergency_flatten()`。

## 9) 对接要点（与你现有系统无缝衔接）

* **风控→执行**：`risk.decision.v2` 载荷里务必带：
  `metadata.idempotency_key`（引擎生成）、`metadata.decision_id`（可同 idem_key）、`data.exec_overrides`。
  内核按此创建 `Plan`，入场单 `order_ref=idempotency_key`，OCO 为 `idempotency_key:STP/TP`，`oco_group="OCO:{decision_id}"`。

* **状态/成交回流**：确保你的 IB 适配层已把 `exec.order.status.v2` 与 `exec.fill.v2` 推入 Kafka；
  字段至少包含：

  * `exec.order.status.v2`: `order_id, status, symbol, side, tif, ...`
  * `exec.fill.v2`: `order_id, symbol, side, qty, price, ts_us`

* **Stop-Limit 或 Stop-MKT**：通过环境变量 `USE_STOP_LIMIT=true/false` 控制；
  `STOP_LIMIT_SLIP_BPS` 控制止损限价的“让价”幅度，避免止损限价**成交不上**的风险。

* **RPM**：IB 对 `place/modify/cancel` 有节流；本内核默认 `10/30/30 rpm`，可按你网关设置调整。

* **幽灵订单**：启动后 `open_orders()` 会领养遗留订单（通过 `order_ref` 推断）；若你的历史遗留订单**没有 `orderRef`**，可把“模糊领养”逻辑加上品种+方向+时间窗口比对（这里只留了接口位）。
