## B) 编排器把 ID/文本向下传并落盘（便于审计与回滚）

**文件：`orchestrator.py`**
在 `_normalize_or_build_plan()` **保留 providers 中的** `execution_prompt_id/text` 不丢失；
在 `LLMTraderController.run_with_plan(plan)` 之前，把文本（若存在）**落盘为文件**，并通过 **ENV 或 CLI** 传给下游（建议 ENV 简单稳妥）。

**补丁要点（示意）：**

```python
# orchestrator.py
def _normalize_or_build_plan(self, session: str, plan: dict, mem: dict) -> dict:
    providers = (plan or {}).get("providers") or []
    # 确保 execution_prompt_id/text 被保留（不做任何过滤）
    # ... 其余 clamp/默认值逻辑保持不变
    return {"session": session, "providers": providers, "llm_trader_flags": flags}

def _prepare_prompt_overrides(self, plan: dict) -> dict:
    """
    将每个 provider 的 execution_prompt_text（如有）落盘为 runs_dir/<date>/prompts/{prov}_{ts}.txt
    返回 {index: {"PROMPT_FILE": "...", "PROMPT_ID": "..."}}
    """
    out = {}
    providers = (plan or {}).get("providers") or []
    out_dir = self.runs_dir / datetime.date.today().isoformat() / "prompts"
    out_dir.mkdir(parents=True, exist_ok=True)
    ts = self._ny_now_iso().replace(":", "-")
    for i, p in enumerate(providers):
        eid = p.get("execution_prompt_id")
        etx = p.get("execution_prompt_text")
        entry = {}
        if eid: entry["PROMPT_ID"] = str(eid)
        if etx:
            fp = out_dir / f"p{i}_{p.get('name','prov')}_{ts}.txt"
            fp.write_text(str(etx), encoding="utf-8")
            entry["PROMPT_FILE"] = str(fp)
        if entry:
            out[i] = entry
    return out
```

在 `morning_sequence()/intraday_sequence()` 完成 `plan = self._normalize_or_build_plan(...)` 后，添加：

```python
# prompt_builder_universal.py
class UniversalPromptBuilder:
    def __init__(self, ..., prompt_id: str | None = None, prompt_text: str | None = None):
        self.prompt_id = prompt_id
        self.prompt_text = prompt_text
        self._library = self._load_prompt_library()

    def _load_prompt_library(self) -> dict:
        # 例如 ./prompts/prompt_library.json，结构：{ "v3_stable": "...", "v4_exp": "..." }
        try:
            return json.loads(Path("prompts/prompt_library.json").read_text("utf-8"))
        except Exception:
            return {}

    def build_execution_prompt(self, context: dict) -> str:
        # 1) 外部文本覆盖（需白名单/审核开关，见下节安全）
        if self.prompt_text:
            return self._inject_context(self.prompt_text, context)

        # 2) 外部 ID → 安全库
        if self.prompt_id and self.prompt_id in self._library:
            return self._inject_context(self._library[self.prompt_id], context)

        # 3) 回落到默认内置模板（你原来的逻辑）
        return self._inject_context(self._default_template(), context)
```
