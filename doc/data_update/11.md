# 1) ClickHouse 表设计（追加到 `ch_init.sql` 或直接执行）

> 统一放在 `monitor` 库；全部幂等可重算。

```sql
-- 监控库
CREATE DATABASE IF NOT EXISTS monitor;

-- A) 日级缺口统计：按 symbol × 分辨率 × 交易日
CREATE TABLE IF NOT EXISTS monitor.gap_daily
(
  d                 Date,
  symbol            String,
  res               LowCardinality(String),   -- '1m','5m','15m',...
  expected_windows  UInt32,                   -- RTH 理论应有窗口数
  actual_windows    UInt32,                   -- bars_{res}_current 实际窗口数
  missing_windows   UInt32,
  missing_rate      Float64,
  computed_at       DateTime DEFAULT now()
)
ENGINE = ReplacingMergeTree(computed_at)
PARTITION BY d
ORDER BY (d, symbol, res);

-- B) 回补延迟明细：按每根 bar 记录近似延迟（秒）
-- 近似方法：latency_sec = toDateTime64(ver/1000,3) - ts_end  （ver来自消息metadata，足够做近似）
CREATE TABLE IF NOT EXISTS monitor.backfill_latency
(
  d             Date,
  symbol        String,
  res           LowCardinality(String),
  ts_start      DateTime,
  ts_end        DateTime,
  is_backfill   UInt8,
  source        LowCardinality(String),       -- 'REALTIME','IBKR','IBKR_RECON','AGG'...
  ver_ms        UInt64,
  latency_sec   Float64,
  computed_at   DateTime DEFAULT now()
)
ENGINE = ReplacingMergeTree(computed_at)
PARTITION BY d
ORDER BY (d, symbol, res, ts_start);

-- C) 盘口异常速览：日级（crossed/locked、负价等）
CREATE TABLE IF NOT EXISTS monitor.l1_quote_quality_daily
(
  d               Date,
  symbol          String,
  total_quotes    UInt64,
  crossed_locked  UInt64,                     -- ask < bid
  non_positive    UInt64,                     -- bid<=0 或 ask<=0
  bad_rate        Float64,
  computed_at     DateTime DEFAULT now()
)
ENGINE = ReplacingMergeTree(computed_at)
PARTITION BY d
ORDER BY (d, symbol);

-- D) （可选）隔离区：Writer 丢弃/清洗的数据可落这里供审计
CREATE TABLE IF NOT EXISTS monitor.quarantine_bars
(
  ts_ingest   DateTime DEFAULT now(),
  symbol      String,
  res         LowCardinality(String),
  ts_start    DateTime,
  reason      LowCardinality(String),
  payload     String
)
ENGINE = MergeTree
ORDER BY (ts_ingest, symbol, res, ts_start);
```

# 2) 日更脚本（新增 `services/monitor/dqc_job.py`）

> 这个脚本每天跑一次（建议 02:30 ET 后、离线对账完成后），写入上面三张监控表。
> 依赖：P2 的 `bars_{res}_current`、P5 的 `lake.bbo_1s`/`bbo_1s_features`。

```python
# services/monitor/dqc_job.py
from __future__ import annotations
from datetime import date, datetime, timedelta, time
from typing import Iterable, List, Tuple
from zoneinfo import ZoneInfo

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger

log = get_logger(__name__)
NY = ZoneInfo("America/New_York")

# 生成 [09:30, 16:00) 的理论时间轴窗口数
def expected_windows(res: str) -> int:
    step = {
        "1m": 60, "5m": 300, "15m": 900, "30m": 1800, "1h": 3600
    }[res]
    start_local = datetime.combine(date.today(), time(9,30), tzinfo=NY)
    end_local   = datetime.combine(date.today(), time(16,0), tzinfo=NY)
    seconds = int((end_local - start_local).total_seconds())
    return seconds // step

# 可扩：按需增加更多分辨率
RES_LIST = ("1m","5m","15m")

SQL_SYMBOLS = """
SELECT DISTINCT symbol
FROM trading.bars_5s
WHERE d = %(d)s
LIMIT 1000
"""

SQL_GAP_DAILY = """
INSERT INTO monitor.gap_daily
SELECT
  %(d)s AS d,
  s.symbol AS symbol,
  %(res)s AS res,
  %(expected)s AS expected_windows,
  cast(count(), 'UInt32') AS actual_windows,
  greatest(%(expected)s - cast(count(), 'UInt32'), 0) AS missing_windows,
  if(%(expected)s>0, (greatest(%(expected)s - cast(count(), 'UInt32'),0) / toFloat64(%(expected)s)), 1.0) AS missing_rate,
  now() AS computed_at
FROM %(table_current)s
RIGHT JOIN (SELECT %(d)s AS d, symbol FROM (SELECT DISTINCT symbol FROM trading.bars_5s WHERE d=%(d)s)) AS s
  ON (symbol = s.symbol AND d = %(d)s)
GROUP BY symbol
"""

SQL_BACKFILL_LATENCY = """
INSERT INTO monitor.backfill_latency
SELECT
  toDate(ts_start) AS d,
  symbol,
  %(res)s AS res,
  ts_start, ts_end,
  is_backfill,
  source,
  ver AS ver_ms,
  dateDiff('second', ts_end, toDateTime(ver/1000)) AS latency_sec,
  now() AS computed_at
FROM %(table_ibkr)s
WHERE d = %(d)s
"""

SQL_L1_QUALITY = """
INSERT INTO monitor.l1_quote_quality_daily
SELECT
  d,
  symbol,
  count() AS total_quotes,
  countIf(ask < bid) AS crossed_locked,
  countIf(ask <= 0 OR bid <= 0) AS non_positive,
  if(count()>0, (countIf(ask<bid) + countIf(ask<=0 OR bid<=0)) / toFloat64(count()), 0.0) AS bad_rate,
  now() AS computed_at
FROM lake.quotes_l1
WHERE d = %(d)s
GROUP BY d, symbol
"""

def run(target_day: date | None = None):
    d = target_day or (date.today() - timedelta(days=1))
    db = ClickHouseDB()

    # 动态发现 symbol 集（来自当天 5s，有交易的标的）
    symbols = [r["symbol"] for r in db.query(SQL_SYMBOLS, {"d": d})] or []
    if not symbols:
        log.warning("No symbols discovered for day", day=str(d))
        return

    # 缺口率：对各 res 写一行/标的
    for res in RES_LIST:
        expected = expected_windows(res)
        tbl_cur  = f"trading.bars_{res}_current"
        sql = SQL_GAP_DAILY % {
            "table_current": tbl_cur
        }
        db.query(sql, {"d": d, "res": res, "expected": expected})

        # 回补延迟（仅统计 IBKR 表，且包含 is_backfill 标记）
        tbl_ibkr = f"trading.bars_{res}_ibkr"
        if db.query(f"EXISTS TABLE {tbl_ibkr}")[0]["result"] == 1:
            sql2 = SQL_BACKFILL_LATENCY % {"table_ibkr": tbl_ibkr}
            db.query(sql2, {"d": d, "res": res})

    # 盘口异常日表
    db.query(SQL_L1_QUALITY, {"d": d})

    db.close()
    log.info("DQC job done", day=str(d))

if __name__ == "__main__":
    run()
```

**说明**

* `gap_daily`：按 RTH 理论窗口数与 `bars_{res}_current` 实际条数对比，得到缺口率。
* `backfill_latency`：对 `bars_{res}_ibkr` 估算近似延迟（ver 时间 vs bar ts_end），可筛选 `is_backfill=1` 看纯回补延迟。
* `l1_quote_quality_daily`：统计当天 crossed/locked 与非正价比例。
* 发现 symbol 用当天 `bars_5s`（有交易就会出现）。你也可以改为读取自定义“标的清单”。

# 4) 快速验收 SQL

```sql
-- A. 缺口率（近3日 TOP 10）
SELECT d, symbol, res, missing_rate, expected_windows, actual_windows
FROM monitor.gap_daily
WHERE d >= today()-3
ORDER BY missing_rate DESC
LIMIT 10;

-- B. 回补延迟（P50/P95）
SELECT
  d, res,
  quantileExact(0.5)(latency_sec) AS p50,
  quantileExact(0.95)(latency_sec) AS p95,
  countIf(is_backfill=1) AS backfilled
FROM monitor.backfill_latency
WHERE d >= today()-3
GROUP BY d, res
ORDER BY d DESC, res;

-- C. 盘口异常占比（近3日）
SELECT d, symbol, bad_rate, total_quotes
FROM monitor.l1_quote_quality_daily
WHERE d >= today()-3
ORDER BY bad_rate DESC
LIMIT 20;
```

# 5) Grafana/告警建议（简要）

* 面板1：**缺口率热力图**（维度：d×symbol，值：missing_rate，按 res 可切换）。
* 面板2：**回补延迟分布**（直方图/箱线图，p50/p95；筛 is_backfill=1）。
* 面板3：**盘口异常率**（bad_rate）；阈值告警：`bad_rate > 0.02` 或 `total_quotes < threshold`。
* 告警1：**缺口率 > 1%** 时发 Slack/飞书；
* 告警2：**回补延迟 P95 > 180s**（盘中）/**> 600s**（盘后）；
* 告警3：**某表当天增量停更 > N 分钟**（用 `max(ts_end)` 与当前时间比较即可）。
