## 1) ClickHouse DDL（追加到 `ch_init.sql` 或直接执行）

> 约定：统一用 `ReplacingMergeTree(ver)`，以 `ver`（消息时间戳）兜底去重/选最新；
> 分区按天，保留期默认 30 天（可调）。

```sql
-- 1) 数据库（容错创建）
CREATE DATABASE IF NOT EXISTS lake;

-- 2) 原始成交（Ticks / Trades）
CREATE TABLE IF NOT EXISTS lake.ticks_trades
(
  symbol           String,
  ts               DateTime64(6),           -- 事件时间（微秒）
  price            Float64,
  size             UInt32,
  exchange         LowCardinality(String) DEFAULT 'SMART',
  is_rth           UInt8 DEFAULT 1,
  source           LowCardinality(String) DEFAULT 'IBKR',
  ver              UInt64,                  -- 元数据时间（ms），用于ReplacingMergeTree选优
  d                Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (symbol, ts)
TTL d + INTERVAL 30 DAY;

-- 3) L1/BBO 报价（Quotes Level 1）
CREATE TABLE IF NOT EXISTS lake.quotes_l1
(
  symbol           String,
  ts               DateTime64(6),
  bid              Float64,
  ask              Float64,
  bid_size         UInt32,
  ask_size         UInt32,
  exchange         LowCardinality(String) DEFAULT 'SMART',
  is_rth           UInt8 DEFAULT 1,
  source           LowCardinality(String) DEFAULT 'IBKR',
  ver              UInt64,
  d                Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (symbol, ts)
TTL d + INTERVAL 30 DAY;

-- 4) 1秒BBO桥表（便于秒级盘口特征/聚合）
CREATE TABLE IF NOT EXISTS lake.bbo_1s
(
  symbol    String,
  ts        DateTime,                       -- 秒对齐
  bid       Float64,
  ask       Float64,
  bid_size  UInt32,
  ask_size  UInt32,
  ver       UInt64,
  d         Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (symbol, ts)
TTL d + INTERVAL 90 DAY;

-- 5) 从 quotes_l1 聚合生成 1秒BBO（取该秒内最新quote）
CREATE MATERIALIZED VIEW IF NOT EXISTS lake.mv_quotes_to_bbo_1s
TO lake.bbo_1s
AS
SELECT
  symbol,
  toStartOfSecond(ts) AS ts,
  argMax(bid, ver)      AS bid,
  argMax(ask, ver)      AS ask,
  argMax(bid_size, ver) AS bid_size,
  argMax(ask_size, ver) AS ask_size,
  max(ver)              AS ver,
  toDate(ts)            AS d
FROM lake.quotes_l1
GROUP BY symbol, ts;
```

> 说明
>
> * `ver` = 生产端/消息 `metadata.ts_us // 1000`，保证回补/迟到覆盖更旧数据。
> * 1s BBO 用 `argMax(..., ver)` 取该秒内**最新**的 bid/ask。
> * 后续策略可直接从 `lake.bbo_1s` 拉中间价、spread、imbalance 等秒级特征。

## 2) 新服务：`services/market/writer/lake_writer.py`

> 这是一个**通用 Lake Writer**，支持 `mode=quotes|trades` 两种输入格式，消费 Kafka → 写 ClickHouse（`lake.*`）。

```python
# services/market/writer/lake_writer.py
import asyncio
import json
from typing import Optional, Dict, Any, List
from datetime import datetime, timezone

from aiokafka import AIOKafkaConsumer
from aiokafka.structs import ConsumerRecord

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

logger = get_logger(__name__)

def _us_to_dt64(us: Optional[int]):
    if not us: return None
    # ClickHouse DateTime64(6) expects python datetime; CH client will map precision
    return datetime.fromtimestamp(us / 1_000_000, tz=timezone.utc).replace(tzinfo=None)

class LakeWriter:
    """
    Generic Kafka→ClickHouse writer for lake.{quotes_l1|ticks_trades}.
    Message schema (recommended):
      { "data": { ...fields... }, "metadata": { "ts_us": <int> }, "source": "IBKR" }
    """
    def __init__(
        self,
        *,
        topic: str,
        table: str,
        mode: str,                # "quotes" or "trades"
        group_id: str,
        batch_size: int = 500,
        batch_timeout_sec: float = 2.0,
        consumer_kwargs: Optional[Dict[str, Any]] = None,
    ):
        assert mode in ("quotes", "trades")
        self.topic = topic
        self.table = table
        self.mode = mode
        self.group_id = group_id
        self.batch_size = batch_size
        self.batch_timeout_sec = batch_timeout_sec

        kafka_cfg = config.kafka_config
        kwargs = dict(
            bootstrap_servers=kafka_cfg["bootstrap_servers"],
            group_id=group_id,
            auto_offset_reset="earliest",
            enable_auto_commit=True,
            value_deserializer=lambda m: json.loads(m.decode("utf-8")),
        )
        if consumer_kwargs:
            kwargs.update(consumer_kwargs)

        self.consumer = AIOKafkaConsumer(self.topic, **kwargs)
        self.db = ClickHouseDB()
        self._running = False
        self._batch: List[List[Any]] = []

        if self.mode == "quotes":
            self.cols = ("symbol","ts","bid","ask","bid_size","ask_size","exchange","is_rth","source","ver","d")
        else:
            self.cols = ("symbol","ts","price","size","exchange","is_rth","source","ver","d")

    async def start(self):
        await self.db.init_async_client()
        await self.consumer.start()
        self._running = True
        asyncio.create_task(self._consume())
        asyncio.create_task(self._flush_timer())
        logger.info("LakeWriter started", topic=self.topic, table=self.table, mode=self.mode)

    async def stop(self):
        self._running = False
        try:
            if self._batch:
                await self._flush()
        except Exception as e:
            logger.warning("Flush on stop failed", error=str(e))
        await self.consumer.stop()
        await self.db.close_async()
        logger.info("LakeWriter stopped", topic=self.topic)

    async def _consume(self):
        try:
            async for rec in self.consumer:
                row = self._to_row(rec)
                if row is None:
                    continue
                self._batch.append(row)
                if len(self._batch) >= self.batch_size:
                    await self._flush()
        except asyncio.CancelledError:
            pass
        except Exception as e:
            logger.error("Consume error", error=str(e))
            raise

    async def _flush_timer(self):
        while self._running:
            await asyncio.sleep(self.batch_timeout_sec)
            if self._batch:
                try:
                    await self._flush()
                except Exception as e:
                    logger.error("Flush error", error=str(e))

    async def _flush(self):
        batch = self._batch
        self._batch = []
        if not batch:
            return
        await self.db.insert_async(self.table, batch, list(self.cols))
        logger.info("Inserted", table=self.table, rows=len(batch))

    def _to_row(self, rec: ConsumerRecord):
        try:
            payload = rec.value
            data = payload.get("data", {}) or {}
            meta = payload.get("metadata", {}) or {}
            source = payload.get("source") or data.get("source") or "IBKR"

            # Common fields
            symbol = data.get("symbol")
            ts_us = data.get("ts_us") or data.get("ts")  # prefer microseconds
            ts = _us_to_dt64(ts_us)
            ver = int((meta.get("ts_us") or ts_us or 0) // 1000)
            is_rth = int(data.get("is_rth", 1))
            exchange = data.get("exchange", "SMART")

            if not symbol or ts is None:
                return None

            if self.mode == "quotes":
                bid = data.get("bid"); ask = data.get("ask")
                bid_size = int(data.get("bid_size", 0)); ask_size = int(data.get("ask_size", 0))
                return [
                    symbol, ts,
                    float(bid) if bid is not None else None,
                    float(ask) if ask is not None else None,
                    bid_size, ask_size,
                    str(exchange), is_rth, str(source), ver, ts.date()
                ]
            else:
                price = data.get("price"); size = int(data.get("size", 0))
                if price is None or size <= 0:
                    return None
                return [
                    symbol, ts,
                    float(price), size,
                    str(exchange), is_rth, str(source), ver, ts.date()
                ]
        except Exception as e:
            logger.warning("Drop malformed lake message", error=str(e))
            return None
```

## 3) `run_system.py` 集成（增量片段）

> 和 P1 的 bars writer 一样，增加一段读取 `config.lake_writers` 的启动/关闭逻辑。

**导入**

```python
from services.market.writer.lake_writer import LakeWriter
from libs.config import config
```

**启动处**

```python
self._lake_writers = []
for w in (getattr(config, "lake_writers", []) or []):
    if not w.get("enabled", False):
        continue
    writer = LakeWriter(
        topic=w["topic"],
        table=w["table"],
        mode=w["mode"],                        # "quotes" or "trades"
        group_id=w.get("group_id", f"lake_writer_{w['mode']}"),
        batch_size=int(w.get("batch_size", 500)),
        batch_timeout_sec=float(w.get("batch_timeout_sec", 2.0)),
    )
    await writer.start()
    self._lake_writers.append(writer)
```

**停止处**

```python
for w in getattr(self, "_lake_writers", []):
    try:
        await w.stop()
    except Exception as e:
        logger.warning("Lake writer stop failed", error=str(e))
```

## 5) 验收步骤（可直接复制）

**A. 基础检查**

```sql
SHOW TABLES FROM lake;

-- DDL 生效
DESCRIBE TABLE lake.quotes_l1;
DESCRIBE TABLE lake.ticks_trades;
DESCRIBE TABLE lake.bbo_1s;
```

**B. 开 quotes writer，观测入库与 1s 聚合**

1. 将 `lake_writers.quotes_l1.enabled: true`，重启服务；
2. 观察写入：

```sql
SELECT count(), min(ts), max(ts)
FROM lake.quotes_l1
WHERE d = today();

-- 1秒桥表是否自动出数
SELECT count(), min(ts), max(ts)
FROM lake.bbo_1s
WHERE d = today();
```

**C. 盘口质量快检**

```sql
-- 异常价过滤（bid<=ask），可做质检视图/告警
SELECT count() AS bad_quotes
FROM lake.quotes_l1
WHERE bid > ask AND d = today();

-- 计算秒级中间价与spread（bps）
SELECT
  symbol, ts,
  (bid+ask)/2 AS mid,
  (ask-bid)/((bid+ask)/2)*1e4 AS spread_bps
FROM lake.bbo_1s
WHERE d = today()
ORDER BY ts DESC
LIMIT 20;
```

**D. 开 ticks writer（如有成交流）**

```sql
-- 开启后应持续增长
SELECT count(), min(ts), max(ts)
FROM lake.ticks_trades
WHERE d = today();
```

## 6) 回退

* 把 `config.yml` 的 `lake_writers.*.enabled` 改回 `false`，重启即可；
* 如需暂停聚合，可 `DETACH TABLE lake.mv_quotes_to_bbo_1s`（不删数据）；
* `lake.*` 是独立库，不会影响 bars/feature 链路。

---

## 7) 给策略/特征的直接好处

* **统一盘口入口**：`lake.bbo_1s` 可直接给特征脚本计算 `mid/imbalance/spread_bps/microprice` 等；
* **交易成本表化**：基于 `bbo_1s` 或 `quotes_l1` 的分钟槽 P95 spread，可生成 `market_cost_profile` 表，策略下单门槛/滑点模型直接查；
* **回测真源**：需要复盘盘口行为，直接 `SELECT * FROM lake.quotes_l1 WHERE symbol=? AND ts BETWEEN ? AND ?` 即可，无需再“还原”过程。
