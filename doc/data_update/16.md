棒，给你一套**T-1 执行对账（IBKR → Kafka → ClickHouse）**最小闭环：

* **脚本**：连 IBKR 拉 T-1 执行，生成**确定性 fill_id**，统一发 `orders.fills.v1`；
* **幂等**：`fill_id = UUIDv5(permId:execId)`，重复重放不产生重复；
* **映射**：用 `trading.ibkr_order_map` 把 `permId → order_id`；找不到就用**可重复的替代 order_id**；
* **审计/汇总**：落一张**对账汇总表**（每天差额、遗漏/多余计数）；
* **开关**：完全“加法”，按 cron 夜里跑，随时停。

```sql
-- B) 日级对账汇总

(

  symbol          String,
  ib_fills        UInt32,
  our_fills       UInt32,
  ib_qty          Int64,
  our_qty         Int64,
  ib_gross        Float64,   -- 成交额（∑qty*price，买为+，卖为-）
  our_gross       Float64,
  fee_ib          Float64,
  fee_our         Float64,
  missing_count   UInt32,    -- IB 有而我们没有
  extra_count     UInt32,    -- 我们有而 IB 没有
  ok              UInt8,     -- 1=通过（数量与成交额都在容差内）

)

PARTITION BY d_local
ORDER BY (d_local, account, symbol);
```

# 2) 脚本：T-1 执行拉取与对账

**新建** `services/reconciliation/ibkr_exec_recon.py`

```python
# services/reconciliation/ibkr_exec_recon.py
from __future__ import annotations
import argparse, asyncio, json, time, uuid, hashlib
from dataclasses import dataclass
from datetime import datetime, date, timedelta, timezone
from typing import Dict, List, Tuple, Optional
from zoneinfo import ZoneInfo

import pandas as pd
from aiokafka import AIOKafkaProducer
from ib_insync import IB, util, Stock, ExecutionFilter  # pip install ib_insync

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

log = get_logger(__name__)
NY = ZoneInfo("America/New_York")

# ---------- helpers ----------
def to_utc(dt_local: datetime) -> datetime:
    if dt_local.tzinfo is None:
        dt_local = dt_local.replace(tzinfo=NY)
    return dt_local.astimezone(timezone.utc)

def ny_day_range(d_local: date) -> tuple[datetime, datetime]:
    s = datetime(d_local.year, d_local.month, d_local.day, 0, 0, 0, tzinfo=NY)
    e = s + timedelta(days=1)
    return s, e

def uuid5_perm_exec(perm_id: int, exec_id: str) -> str:
    # 生成稳定的 UUID：同一 permId+execId 恒定不变（幂等）
    return str(uuid.uuid5(uuid.NAMESPACE_URL, f"ibkr:{perm_id}:{exec_id}"))

def uuid5_perm(perm_id: int) -> str:
    return str(uuid.uuid5(uuid.NAMESPACE_URL, f"ibkr-perm:{perm_id}"))

def side_sign(side_str: str) -> int:
    return +1 if (side_str or "").upper() == "BOT" else -1

# ---------- main ----------
@dataclass
class Cfg:
    host: str
    port: int
    client_id: int
    account: str
    topic_fills: str = "orders.fills.v1"
    raw_table: str = "monitor.ibkr_exec_raw"
    recon_table: str = "monitor.broker_recon_daily"
    write_raw: bool = True
    rth_only: bool = False         # 如需仅 RTH 可在过滤时添加 9:30-16:00
    fee_default: float = 0.0
    tol_qty: int = 0               # 数量容差
    tol_amt: float = 1.0           # 成交额容差（美元）

class IBKRExecReconciler:
    def __init__(self, cfg: Cfg):
        self.cfg = cfg
        self.ib = IB()
        self.db = ClickHouseDB()
        self.producer = AIOKafkaProducer(
            bootstrap_servers=config.kafka_config["bootstrap_servers"],
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8"),
            linger_ms=5, retries=3
        )

    async def start(self):
        await self.producer.start()
        await self.ib.connectAsync(self.cfg.host, self.cfg.port, clientId=self.cfg.client_id, readonly=True)
        log.info("Connected to IBKR", connected=self.ib.isConnected())

    async def stop(self):
        try: await self.producer.stop()
        except Exception: pass
        try: self.ib.disconnect()
        except Exception: pass

    def _fetch_execs(self, d_local: date) -> pd.DataFrame:
        # IBKR 的 ExecutionFilter 不直接支持日期区间；常见做法：不带时间过滤拉当天所有，再本地筛选
        ef = ExecutionFilter(acctCode=self.cfg.account)
        execs = self.ib.reqExecutions(ef)  # 阻塞但很快；ib_insync 会在当前线程调 IB API
        if not execs:
            return pd.DataFrame(columns=[
                "execId","permId","conId","symbol","side","shares","price","time","exchange","commission"
            ])
        # 关联合同与佣金
        rows = []
        for ex, cr in execs.items():  # ib_insync 返回 {Execution: CommissionReport|None}
            rows.append({
                "execId": ex.execId,
                "permId": ex.permId,
                "conId": ex.conId,
                "symbol": ex.contract.symbol,
                "side": ex.side,                 # BOT/SLD
                "shares": int(float(ex.shares)), # IBKR 某些账户 shares 是小数形式字符串
                "price": float(ex.price),
                "time": util.parseIBDatetime(ex.time),  # naive，本地时区字符串
                "exchange": ex.exchange,
                "commission": float(getattr(cr, "commission", self.cfg.fee_default) or 0.0)
            })
        df = pd.DataFrame(rows)
        if df.empty: return df
        # 过滤到指定本地交易日
        s, e = ny_day_range(d_local)
        # ex.time 是 naive 的本地时间字符串，parse 后 naive；按 NY 解释
        df["ts_local"] = pd.to_datetime(df["time"]).dt.tz_localize(NY, nonexistent="shift_forward", ambiguous="NaT")
        df["ts_utc"] = df["ts_local"].dt.tz_convert(timezone.utc)
        df = df[(df["ts_local"] >= s) & (df["ts_local"] < e)]
        # 可选仅 RTH
        if self.cfg.rth_only:
            rth_s = datetime(d_local.year, d_local.month, d_local.day, 9, 30, tzinfo=NY)
            rth_e = datetime(d_local.year, d_local.month, d_local.day, 16, 0, tzinfo=NY)
            df = df[(df["ts_local"] >= rth_s) & (df["ts_local"] < rth_e)]
        return df

    def _map_order_ids(self, df: pd.DataFrame) -> pd.DataFrame:
        if df.empty: return df
        perms = list({int(x) for x in df["permId"].unique() if pd.notna(x)})
        if not perms: 
            df["order_id"] = None
            return df
        rows = self.db.query("""
            SELECT ib_perm_id, any(order_id) AS order_id
            FROM trading.ibkr_order_map
            WHERE ib_perm_id IN %(perms)s
            GROUP BY ib_perm_id
        """, {"perms": perms}) or []
        mp = {int(r["ib_perm_id"]): r["order_id"] for r in rows}
        df["order_id"] = df["permId"].map(mp)
        # 没映射到的，生成稳定替代 order_id（保证幂等）
        df.loc[df["order_id"].isna(), "order_id"] = df[df["order_id"].isna()]["permId"].apply(lambda p: uuid5_perm(int(p)))
        return df

    async def _emit_raw(self, d_local: date, df: pd.DataFrame):
        if not self.cfg.write_raw or df.empty: return
        # 直接 INSERT（不走 Kafka，减少复杂度；可改为 generic writer）
        recs = []
        ts_us = int(time.time()*1_000_000)
        ver = int(ts_us//1000)
        for _, r in df.iterrows():
            recs.append({
                "d_local": d_local, "account": self.cfg.account,
                "exec_id": r["execId"], "perm_id": int(r["permId"]), "con_id": int(r["conId"]),
                "symbol": r["symbol"], "side": r["side"], "qty": int(r["shares"]),
                "price": float(r["price"]), "fee": float(r["commission"]),
                "exchange": r["exchange"], "ts_local": r["ts_local"].to_pydatetime().replace(tzinfo=None),
                "ts_utc": r["ts_utc"].to_pydatetime().replace(tzinfo=None),
                "payload_json": json.dumps({
                    "raw_time": str(r["time"]), "exchange": r["exchange"]
                }), "ver": ver
            })
        self.db.insert(self.cfg.raw_table, recs, list(recs[0].keys()))

    async def _emit_fills_kafka(self, df: pd.DataFrame):
        if df.empty: return
        for _, r in df.iterrows():
            fill_id = uuid5_perm_exec(int(r["permId"]), str(r["execId"]))
            side = side_sign(r["side"])
            payload = {
                "data": {
                    "fill_id": fill_id,
                    "order_id": str(r["order_id"]),
                    "ts": r["ts_utc"].to_pydatetime().replace(tzinfo=None),
                    "symbol": r["symbol"],
                    "side": int(side),
                    "qty": int(r["shares"]),
                    "price": float(r["price"]),
                    "fee": float(r["commission"]),
                    "venue": r["exchange"] or "IB",
                    "liquidity": "T"
                },
                "metadata": {"ts_us": int(time.time()*1_000_000)},
                "source": "IBKR_RECON"
            }
            await self.producer.send(self.cfg.topic_fills, key=r["symbol"], value=payload)

    def _reconcile(self, d_local: date, df_ib: pd.DataFrame):
        # our side: trading.fills 当日（含所有 source）
        rows = self.db.query("""
            SELECT symbol,
                   count() AS our_fills,
                   sumIf(sign*qty, side>0) - sumIf(sign*qty, side<0) AS our_qty_signed,  -- 不用；下面直接求总和
                   sum(sign*qty*price) AS our_gross,     -- 买为+，卖为-
                   sum(fee) AS fee_our
            FROM
            (
                SELECT symbol, qty, price, fee, (side) AS sign
                FROM trading.fills
                WHERE d = %(d)s
            )
            GROUP BY symbol
        """, {"d": d_local}) or []
        d_our = {r["symbol"]: r for r in rows}

        res = []
        for sym, grp in df_ib.groupby("symbol"):
            ib_qty = int((grp["shares"] * grp["side"].map(lambda s: +1 if s=="BOT" else -1)).sum())
            ib_gross = float((grp["shares"] * grp["price"] * grp["side"].map(lambda s: +1 if s=="BOT" else -1)).sum())
            fee_ib = float(grp["commission"].sum())
            ib_fills = int(len(grp))

            our = d_our.get(sym, {"our_fills":0,"our_gross":0.0,"fee_our":0.0})
            our_fills = int(our.get("our_fills",0))
            our_gross = float(our.get("our_gross",0.0))
            fee_our = float(our.get("fee_our",0.0))

            # 粗略数量：把买卖数量绝对值累加比较（也可按 signed qty 对齐）
            our_qty_rows = self.db.query("""
                SELECT sum(if(side>0, qty, -qty)) AS q
                FROM trading.fills WHERE d=%(d)s AND symbol=%(s)s
            """, {"d": d_local, "s": sym}) or [{"q":0}]
            our_qty = int(our_qty_rows[0]["q"] or 0)

            missing = max(ib_fills - our_fills, 0)
            extra   = max(our_fills - ib_fills, 0)

            ok_qty = abs(ib_qty - our_qty) <= self.cfg.tol_qty
            ok_amt = abs(ib_gross - our_gross) <= self.cfg.tol_amt
            ok = 1 if (ok_qty and ok_amt) else 0
            note = f"Δqty={ib_qty-our_qty}, Δamt={ib_gross-our_gross:.2f}"

            res.append({
                "d_local": d_local, "account": self.cfg.account, "symbol": sym,
                "ib_fills": ib_fills, "our_fills": our_fills,
                "ib_qty": ib_qty, "our_qty": our_qty,
                "ib_gross": ib_gross, "our_gross": our_gross,
                "fee_ib": fee_ib, "fee_our": fee_our,
                "missing_count": missing, "extra_count": extra,
                "ok": ok, "note": note
            })
        if res:
            self.db.insert(self.cfg.recon_table, res, list(res[0].keys()))
        log.info("recon summary written", rows=len(res))

    async def run_once(self, d_local: date):
        await self.start()
        try:
            df = self._fetch_execs(d_local)
            if df.empty:
                log.warning("No IBKR executions", day=str(d_local))
                return
            df = self._map_order_ids(df)
            # 原始落库（可选）
            await self._emit_raw(d_local, df)
            # 回放到 fills 主题（幂等）
            await self._emit_fills_kafka(df)
            # 汇总对账
            self._reconcile(d_local, df)
        finally:
            await self.stop()

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--day", help="NY local date YYYY-MM-DD; default=yesterday", default=None)
    ap.add_argument("--rth-only", action="store_true", default=False)
    ap.add_argument("--write-raw", action="store_true", default=True)
    ap.add_argument("--tol-qty", type=int, default=0)
    ap.add_argument("--tol-amt", type=float, default=1.0)
    args = ap.parse_args()

    br = getattr(config, "broker", {}).get("ibkr", {})
    target_day = date.fromisoformat(args.day) if args.day else (datetime.now(tz=NY).date() - timedelta(days=1))
    cfg = Cfg(
        host=br.get("host","127.0.0.1"),
        port=int(br.get("port",7497)),
        client_id=int(br.get("client_id",42)),
        account=br.get("account","DUXXXXXXX"),
        rth_only=bool(args.rth_only),
        write_raw=bool(args.write_raw),
        tol_qty=int(args.tol_qty),
        tol_amt=float(args.tol_amt)
    )
    recon = IBKRExecReconciler(cfg)
    asyncio.get_event_loop().run_until_complete(recon.run_once(target_day))
```

**要点**

* 幂等：`fill_id = uuid5(permId:execId)`，即使多次跑/重放不会产生重复行；
* 归因：`order_id` 优先从 `trading.ibkr_order_map` 映射；若缺失，用 `uuid5(permId)` 生成稳定替代 `order_id`（不破坏表约束）；
* 兼容 P10/P11：写入 `orders.fills.v1` → `trading.fills`，PNL/持仓自动更新；
* 留痕：`monitor.ibkr_exec_raw` 保存原始执行；
* 对账：`monitor.broker_recon_daily` 一行/标的的差额与通过标记。

# 4) 快速验收 SQL

```sql
-- 原始执行是否落库
SELECT * FROM monitor.ibkr_exec_raw
WHERE d_local = yesterday()
ORDER BY ts_local DESC LIMIT 10;

-- fills 是否幂等写入（重复跑行数不变）
SELECT count() FROM trading.fills WHERE d = yesterday();

-- 对账结果
SELECT * FROM monitor.broker_recon_daily
WHERE d_local = yesterday()
ORDER BY ok ASC, symbol
LIMIT 50;

-- 逐笔定位缺失（IB 有我们没有）
WITH ib AS (
  SELECT symbol, exec_id FROM monitor.ibkr_exec_raw WHERE d_local=yesterday()
),
ours AS (
  SELECT symbol, toString(fill_id) AS exec_id FROM trading.fills WHERE d=yesterday()
)
SELECT ib.symbol, ib.exec_id
FROM ib LEFT JOIN ours USING(symbol, exec_id)
WHERE ours.exec_id IS NULL
LIMIT 50;
```

> 上面最后一个查询依赖你是否决定把 `exec_id` 也存到 `trading.fills`。若没有，可以把 `uuid5(permId:execId)` 的值也放进 `monitor.ibkr_exec_raw` 里对齐匹配。

# 5) 交易员视角的安全阀

* **只读拉取**：该脚本只读 IBKR、写自家 Kafka/CH，不会在券商侧下单；
* **RTH 过滤**：默认 `--rth-only`，避免盘前/盘后杂质；
* **容差**：`--tol-amt` 与 `--tol-qty` 容错阈值，避免因小额费用/四舍五入爆红；
* **幂等回放**：重复执行不会产生重复 fills。
