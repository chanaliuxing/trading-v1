# 1) 新增统一读层：`libs/data_reader.py`（覆盖/替换你当前的同名文件）

> 这个模块把所有表名细节都藏起来：
>
> * `5s/15s` → 仍读 `trading.bars_5s / bars_15s`；
> * `>=1m` → 读 `trading.bars_{res}_current`；
> * 特征 → 读 `trading.features_{res}`；
> * 成本曲线 → 读 `lake.market_cost_profile_minute_vw`。
>   返回 **pandas.DataFrame**，索引统一为 **UTC ts_start**。

```python
# libs/data_reader.py
from __future__ import annotations
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from typing import List, Optional, Dict, Any, Sequence, Tuple, Union
import pandas as pd

from libs.db.clickhouse import ClickHouseDB

# 分辨率→表名映射；>=1m 统一走 bars_{res}_current 视图
_RES_TO_TABLE = {
    "5s": "trading.bars_5s",
    "15s": "trading.bars_15s",
}
def _bars_table_for(res: str) -> str:
    res = res.lower()
    if res in _RES_TO_TABLE:
        return _RES_TO_TABLE[res]
    return f"trading.bars_{res}_current"  # >=1m

def _to_list(x):
    if x is None: return []
    return x if isinstance(x, (list, tuple)) else [x]

@dataclass
class MarketDataReader:
    db: ClickHouseDB

    # -------- Bars --------
    def bars(
        self,
        symbols: Union[str, Sequence[str]],
        res: str,
        start: Union[str, datetime, None] = None,
        end: Union[str, datetime, None] = None,
        columns: Optional[Sequence[str]] = None,
        limit: Optional[int] = None,
    ) -> pd.DataFrame:
        """
        读取 bars（统一口径）：
          - 5s/15s 读 trading.bars_5s / bars_15s
          - >=1m 读 trading.bars_{res}_current
        返回：MultiIndex [symbol, ts_start]，列默认含常见字段
        """
        table = _bars_table_for(res)
        syms = [s.upper() for s in _to_list(symbols)]
        cols = _to_list(columns) or [
            "symbol","ts_start","ts_end","open","high","low","close","vwap","volume","trades"
        ]
        where = ["symbol IN %(syms)s"]
        params: Dict[str, Any] = {"syms": syms}

        if start is not None:
            where.append("ts_start >= %(start)s")
            params["start"] = _to_dt(start)
        if end is not None:
            where.append("ts_start <  %(end)s")
            params["end"] = _to_dt(end)

        sql = f"""
        SELECT {", ".join(cols)}
        FROM {table}
        WHERE {" AND ".join(where)}
        ORDER BY symbol, ts_start
        """
        if limit:
            sql += " LIMIT %(limit)s"
            params["limit"] = int(limit)

        rows = self.db.query(sql, params) or []
        df = pd.DataFrame(rows, columns=cols)
        if df.empty:
            return df
        df["ts_start"] = pd.to_datetime(df["ts_start"], utc=True).dt.tz_convert(None)  # naive UTC
        df.set_index(["symbol","ts_start"], inplace=True)
        return df.sort_index()

    # -------- Features --------
    def features(
        self,
        symbols: Union[str, Sequence[str]],
        res: str,
        start: Union[str, datetime, None] = None,
        end: Union[str, datetime, None] = None,
        columns: Optional[Sequence[str]] = None,
    ) -> pd.DataFrame:
        """
        读取特征（表：trading.features_{res}）
        返回：MultiIndex [symbol, ts]，列按给定/默认返回
        """
        table = f"trading.features_{res.lower()}"
        syms = [s.upper() for s in _to_list(symbols)]
        cols = _to_list(columns) or ["symbol","ts","atr_14","rsi_14","vwap_day","ema_12","ema_26","macd","bb_mid","bb_up","bb_lo","rvol_20","minute_of_day"]
        where = ["symbol IN %(syms)s"]
        params: Dict[str, Any] = {"syms": syms}
        if start is not None:
            where.append("ts >= %(start)s")
            params["start"] = _to_dt(start)
        if end is not None:
            where.append("ts <  %(end)s")
            params["end"] = _to_dt(end)
        sql = f"""
        SELECT {", ".join(cols)}
        FROM {table}
        WHERE {" AND ".join(where)}
        ORDER BY symbol, ts
        """
        rows = self.db.query(sql, params) or []
        df = pd.DataFrame(rows, columns=cols)
        if df.empty:
            return df
        df["ts"] = pd.to_datetime(df["ts"], utc=True).dt.tz_convert(None)
        df.set_index(["symbol","ts"], inplace=True)
        return df.sort_index()

    # -------- Cost curve (minute-of-day P95 spread) --------
    def cost_profile_minute(
        self,
        symbols: Union[str, Sequence[str]],
    ) -> pd.DataFrame:
        """
        读取 lake.market_cost_profile_minute_vw（近30天分钟槽成本曲线）
        返回：index=[symbol, minute_of_day]
        """
        syms = [s.upper() for s in _to_list(symbols)]
        sql = """
        SELECT symbol, minute_of_day, p95_spread_bps_30d, avg_spread_bps_30d
        FROM lake.market_cost_profile_minute_vw
        WHERE symbol IN %(syms)s
        ORDER BY symbol, minute_of_day
        """
        rows = self.db.query(sql, {"syms": syms}) or []
        df = pd.DataFrame(rows, columns=["symbol","minute_of_day","p95_spread_bps_30d","avg_spread_bps_30d"])
        if df.empty:
            return df
        df.set_index(["symbol","minute_of_day"], inplace=True)
        return df.sort_index()

# -------- helpers --------
def _to_dt(x: Union[str, datetime]) -> datetime:
    if isinstance(x, datetime):
        if x.tzinfo is None:
            return x.replace(tzinfo=timezone.utc)
        return x.astimezone(timezone.utc)
    # string
    return pd.to_datetime(x, utc=True).to_pydatetime()
```

> 这个文件直接放到 `libs/data_reader.py`，**替换**你当前那个不完整版本。
> 唯一依赖是你现有的 `libs.db.clickhouse.ClickHouseDB` 的 `.query()`；索引和列名与你的表完全对齐。

# 4) 验收清单（策略视角）

1. **一致性**

   * 用 `MarketDataReader.bars("TSLA","1m",...)` 与你旧代码读 `trading.bars_1m` 对比近 3 日，收盘后吻合（差异=IBKR 直写的新鲜度更好）；
2. **新鲜度**

   ```sql
   SELECT symbol, max(ts_end) FROM trading.bars_1m_current WHERE d=today() GROUP BY symbol ORDER BY 2 DESC LIMIT 10;
   ```

   应大致与实时一致；
3. **功能完整性**

   * 依赖的特征列（ATR/RSI/BB/MACD/VWAP/RVOL 等）能直接从 `features_{res}` 读到；
4. **回测脚本**

   * `load_hist_multi.py / load_historical_data.py` 如有直接查询 bars 的语句，替换为 `MarketDataReader` 的调用（或在这些脚本内部 import 使用）。
