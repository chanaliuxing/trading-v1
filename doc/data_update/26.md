# 1) 新脚本｜`tools/suggest_params.py`

功能：

* 拉近 N 天数据，按我们定的口径计算每只票的 **p95_of_p95**（30s-P95 的分位）与 **imb95**；
* 用 ClickHouse 网格查询评估各档 `alpha` 的**放行率**，挑离目标最近的一档；
* 产出 `tools/params.suggest.yml` 与 `tools/params_report.md`（含和当前生效参数的 diff）。

> 依赖：你已有 `libs.db.clickhouse.ClickHouseDB`；需要 `pyyaml`。

```python
# tools/suggest_params.py
from __future__ import annotations
import argparse, math, os
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from typing import Dict, List, Tuple, Any
import yaml

from libs.db.clickhouse import ClickHouseDB

ALPHA_GRID = [0.004, 0.006, 0.008, 0.010, 0.012]

@dataclass
class SuggestCfg:
    lookback_days: int = 14
    target_pass: float = 0.75     # 目标放行率
    base_thr: float = 0.55
    min_minutes: int = 500        # 至少覆盖分钟数，低于此不给建议
    spread_min: float = 5.0
    spread_max: float = 12.0
    imb_min: float = 0.50
    imb_max: float = 0.80

def _time_range(days: int) -> Tuple[datetime, datetime]:
    ed = datetime.now(timezone.utc)
    st = ed - timedelta(days=days)
    return st, ed

def fetch_micro_distributions(db: ClickHouseDB, st: datetime, ed: datetime, symbol_regex: str) -> List[Dict[str, Any]]:
    # 每分钟计算一次 30s P95 与末秒 imbalance，跨分钟再取 p95_of_p95 与 imb95
    q = f"""
    WITH base AS (
      SELECT symbol, toStartOfMinute(ts) AS m,
             quantileExact(0.95)(spread_bps) AS p95_30s,
             anyLast(imbalance) AS last_imb
      FROM lake.bbo_1s_features
      WHERE ts >= %(st)s AND ts < %(ed)s AND match(symbol, %(sym)s)
      GROUP BY symbol, m
    )
    SELECT
      symbol,
      quantileExact(0.95)(p95_30s) AS p95_of_p95,
      quantileExact(0.95)(abs(last_imb)) AS imb95,
      count() AS minutes
    FROM base
    GROUP BY symbol
    ORDER BY p95_of_p95 DESC
    """
    return db.query(q, {"st": st, "ed": ed, "sym": symbol_regex}) or []

def fetch_alpha_pass_grid(db: ClickHouseDB, st: datetime, ed: datetime, symbol_regex: str, base_thr: float) -> List[Dict[str, Any]]:
    q = f"""
    WITH feat AS (
      SELECT symbol, toStartOfMinute(ts) AS m,
             quantileExact(0.95)(spread_bps) AS p95_30s
      FROM lake.bbo_1s_features
      WHERE ts >= %(st)s AND ts < %(ed)s AND match(symbol, %(sym)s)
      GROUP BY symbol, m
    ),
    sig AS (
      SELECT symbol, toStartOfMinute(ts) AS m, avg(prob) AS prob
      FROM trading.signals_1m
      WHERE ts >= %(st)s AND ts < %(ed)s AND match(symbol, %(sym)s)
      GROUP BY symbol, m
    ),
    grid AS (
      SELECT arrayJoin({ALPHA_GRID}) AS alpha
    )
    SELECT s.symbol, g.alpha,
           avg( s.prob >= (%(base)s + g.alpha * greatest(coalesce(f.p95_30s,0),0)) ) AS pass_rate
    FROM sig s
    LEFT JOIN feat f ON s.symbol=f.symbol AND s.m=f.m
    CROSS JOIN grid g
    GROUP BY s.symbol, g.alpha
    ORDER BY s.symbol, g.alpha
    """
    return db.query(q, {"st": st, "ed": ed, "sym": symbol_regex, "base": base_thr}) or []

def fetch_current_params(db: ClickHouseDB, symbol_regex: str) -> Dict[str, Dict[str, Any]]:
    q = """
    SELECT symbol, alpha_per_bps, base_threshold, spread_bps_max, imbalance_abs_max,
           oc_mode, oc_cool_sec, oc_max_pos_pct, enabled
    FROM config.symbol_risk_params_current
    WHERE match(symbol, %(sym)s)
    """
    rows = db.query(q, {"sym": symbol_regex}) or []
    return {r["symbol"]: r for r in rows}

def pick_alpha(rows: List[Dict[str, Any]], target: float) -> Dict[str, float]:
    # rows: [{symbol, alpha, pass_rate}, ...] → {symbol: alpha_best}
    by_sym: Dict[str, List[Tuple[float, float]]] = {}
    for r in rows:
        by_sym.setdefault(r["symbol"], []).append((float(r["alpha"]), float(r["pass_rate"] or 0.0)))
    out: Dict[str, float] = {}
    for sym, arr in by_sym.items():
        # 选与 target 差值最小的 alpha；如出现空洞，则回退 0.008
        best = min(arr, key=lambda x: abs(x[1] - target)) if arr else (0.008, target)
        out[sym] = float(best[0])
    return out

def clamp(x: float, lo: float, hi: float) -> float:
    return max(lo, min(hi, x))

def build_suggestions(cfg: SuggestCfg,
                      micro_rows: List[Dict[str, Any]],
                      alpha_map: Dict[str, float],
                      current: Dict[str, Dict[str, Any]]) -> Tuple[List[Dict[str, Any]], str]:
    items: List[Dict[str, Any]] = []
    report_lines = []
    report_lines.append("| Symbol | minutes | p95_of_p95 | imb95 | spread_suggest | imb_suggest | alpha_suggest | current_spread | current_imb | current_alpha | change |")
    report_lines.append("|---|---:|---:|---:|---:|---:|---:|---:|---:|---:|---|")

    for r in micro_rows:
        sym = r["symbol"]
        minutes = int(r["minutes"])
        if minutes < cfg.min_minutes:
            continue
        p95_of_p95 = float(r["p95_of_p95"] or 0.0)
        imb95 = float(r["imb95"] or 0.0)
        spread_suggest = clamp(round(p95_of_p95 + 1.0, 2), cfg.spread_min, cfg.spread_max)
        imb_suggest = clamp(round(max(0.55, imb95), 2), cfg.imb_min, cfg.imb_max)
        alpha_suggest = float(alpha_map.get(sym, 0.008))

        cur = current.get(sym) or current.get("*") or {}
        cur_spread = float(cur.get("spread_bps_max", 8.0))
        cur_imb = float(cur.get("imbalance_abs_max", 0.60))
        cur_alpha = float(cur.get("alpha_per_bps", 0.008))

        changed = []
        if abs(spread_suggest - cur_spread) >= 0.5: changed.append("spread")
        if abs(imb_suggest - cur_imb) >= 0.03:     changed.append("imb")
        if abs(alpha_suggest - cur_alpha) >= 0.002:changed.append("alpha")

        items.append({
            "symbol": sym,
            "alpha_per_bps": alpha_suggest,
            "base_threshold": cfg.base_thr,
            "spread_bps_max": spread_suggest,
            "imbalance_abs_max": imb_suggest,
            "oc_mode": cur.get("oc_mode", "throttle"),
            "oc_cool_sec": cur.get("oc_cool_sec", 20.0),
            "oc_max_pos_pct": cur.get("oc_max_pos_pct", 0.20),
            "enabled": cur.get("enabled", 1),
            "note": f"auto-suggest {datetime.now(timezone.utc).isoformat(timespec='seconds')}"
        })

        report_lines.append(
            f"| {sym} | {minutes} | {p95_of_p95:.2f} | {imb95:.2f} | {spread_suggest:.2f} | {imb_suggest:.2f} | {alpha_suggest:.3f} | "
            f"{cur_spread:.2f} | {cur_imb:.2f} | {cur_alpha:.3f} | {','.join(changed) if changed else '-'} |"
        )

    report_md = "\n".join(report_lines)
    return items, report_md

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--lookback-days", type=int, default=14)
    ap.add_argument("--target-pass", type=float, default=0.75)
    ap.add_argument("--base-thr", type=float, default=0.55)
    ap.add_argument("--symbol-regex", default=".*")
    ap.add_argument("--out-yaml", default="tools/params.suggest.yml")
    ap.add_argument("--out-report", default="tools/params_report.md")
    args = ap.parse_args()

    cfg = SuggestCfg(lookback_days=args.lookback_days, target_pass=args.target_pass, base_thr=args.base_thr)
    db = ClickHouseDB()
    st, ed = _time_range(cfg.lookback_days)

    micro = fetch_micro_distributions(db, st, ed, args.symbol_regex)
    grid = fetch_alpha_pass_grid(db, st, ed, args.symbol_regex, cfg.base_thr)
    alpha_map = pick_alpha(grid, cfg.target_pass)
    current = fetch_current_params(db, args.symbol_regex)

    items, report_md = build_suggestions(cfg, micro, alpha_map, current)

    # YAML：封一层 items，兼容 push 脚本
    ydoc = {"items": items}
    os.makedirs(os.path.dirname(args.out_yaml), exist_ok=True)
    with open(args.out_yaml, "w", encoding="utf-8") as f:
        yaml.safe_dump(ydoc, f, allow_unicode=True, sort_keys=False)

    with open(args.out_report, "w", encoding="utf-8") as f:
        f.write("# Auto Suggested Risk Params\n\n")
        f.write(f"- Lookback: {cfg.lookback_days}d, Target pass: {cfg.target_pass}, Base threshold: {cfg.base_thr}\n\n")
        f.write(report_md + "\n")

    print(f"Wrote {args.out_yaml} and {args.out_report}")

if __name__ == "__main__":
    main()
```

# 2) Makefile 目标（本地一键跑）

```makefile
.PHONY: suggest-params
SREGEX ?= .*
suggest-params:
	@python -m tools.suggest_params --symbol-regex "$(SREGEX)" --lookback-days 14 --target-pass 0.75 --base-thr 0.55
```

用法：

```bash
make suggest-params             # 全票池
make suggest-params SREGEX='^(TSLA|NVDA)$'  # 只跑TSLA、NVDA
```

# 3) GitHub Actions：每日生成建议并自动开 PR（不直接生效）

新文件：`.github/workflows/suggest-params.yml`

* 每天 **06:30 UTC**（≈ 02:30 ET）跑
* 产出 `tools/params.suggest.yml` 与 `tools/params_report.md`
* 使用 `peter-evans/create-pull-request` 自动创建 PR，带比对报告

```yaml
name: Suggest Risk Params

on:
  schedule:
    - cron: "30 6 * * 2-6"   # Tue–Sat 06:30 UTC ≈ Tue–Sat 02:30 ET
  workflow_dispatch:
    inputs:
      symbol_regex:
        description: "Regex of symbols to include"
        required: false
        default: ".*"
      lookback_days:
        description: "Lookback days"
        required: false
        default: "14"
      target_pass:
        description: "Target pass rate (0-1)"
        required: false
        default: "0.75"

jobs:
  suggest:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.11" }
      - name: Install deps
        run: |
          python -m pip install -U pip
          pip install -r requirements.txt
      - name: Generate suggestions
        env:
          CLICKHOUSE_HOST: ${{ secrets.CLICKHOUSE_HOST }}
          CLICKHOUSE_USER: ${{ secrets.CLICKHOUSE_USER }}
          CLICKHOUSE_PASSWORD: ${{ secrets.CLICKHOUSE_PASSWORD }}
          CLICKHOUSE_DB: ${{ secrets.CLICKHOUSE_DB }}
        run: |
          SYM="${{ github.event.inputs.symbol_regex || '.*' }}"
          LBD="${{ github.event.inputs.lookback_days || '14' }}"
          TAR="${{ github.event.inputs.target_pass || '0.75' }}"
          python -m tools.suggest_params --symbol-regex "$SYM" --lookback-days "$LBD" --target-pass "$TAR"
      - name: Create Pull Request
        uses: peter-evans/create-pull-request@v6
        with:
          commit-message: "chore(params): auto-suggest risk params"
          title: "Auto-suggest risk params"
          body: |
            This PR proposes updated risk parameters based on the last run.

            See report: `tools/params_report.md`

            - Lookback: 14d
            - Target pass rate: 0.75
            - Base threshold: 0.55
          branch: "auto/params-suggest"
          delete-branch: true
```

> 你需要把 ClickHouse 连接信息配置到 **repo secrets**；脚本用你现有的 `ClickHouseDB`，确保它会从环境变量读取（若没有，可在 `libs.db.clickhouse` 里支持从 env 注入）。

# 4) 合并后如何“生效”

* PR 合并后，你已有的 **推参工作流**（或手工 `make param-push`）会把 `tools/params.yml` 推到 Kafka；
* 建议把 CI 里推参的源文件指向 `tools/params.yml`，而建议脚本输出 `tools/params.suggest.yml`。你可以**人工比对/拷贝**或**把推参 workflow 改成读取 `params.suggest.yml`**（需审批保护）。
