# 1) 新增：通用参数存储（CH 热读 + 60s 缓存）

**文件**：`libs/param_store.py`（新建）

```python
# libs/param_store.py
from __future__ import annotations
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from typing import Optional, Dict, Tuple

from libs.db.clickhouse import ClickHouseDB  # 你现有的封装
from libs.logger import get_logger

log = get_logger(__name__)

@dataclass(frozen=True)
class RiskParams:
    alpha_per_bps: float = 0.008
    base_threshold: float = 0.55
    spread_bps_max: float = 8.0
    imbalance_abs_max: float = 0.60
    oc_mode: str = "throttle"        # throttle | block | none
    oc_cool_sec: float = 20.0
    oc_max_pos_pct: float = 0.20     # 相对正常上限的百分比(0~1)

class RiskParamStore:
    """
    从 ClickHouse 视图 config.symbol_risk_params_current 读取每个 symbol 的生效参数，
    优先精确匹配，其次 '*' 全局。带 TTL 缓存，默认 60s。
    """
    def __init__(self, ttl_sec: int = 60):
        self.ttl = timedelta(seconds=ttl_sec)
        self.db = ClickHouseDB()
        self._cache: Dict[str, Tuple[datetime, RiskParams]] = {}

    def invalidate(self, symbol: Optional[str] = None):
        if symbol is None:
            self._cache.clear()
        else:
            self._cache.pop(symbol, None)

    def get(self, symbol: str) -> RiskParams:
        now = datetime.now(timezone.utc)
        ent = self._cache.get(symbol)
        if ent and now - ent[0] < self.ttl:
            return ent[1]
        # 读视图；精确匹配优先，退回 '*' 行
        rows = self.db.query("""
            SELECT alpha_per_bps, base_threshold, spread_bps_max, imbalance_abs_max,
                   oc_mode, oc_cool_sec, oc_max_pos_pct
            FROM config.symbol_risk_params_current
            WHERE symbol IN (%(s)s, '*')
            ORDER BY (symbol = '*') ASC   -- 精确匹配优先
            LIMIT 1
        """, {"s": symbol}) or []
        if not rows:
            rp = RiskParams()  # 全部默认
        else:
            r = rows[0]
            rp = RiskParams(
                alpha_per_bps=float(r.get("alpha_per_bps", 0.008)),
                base_threshold=float(r.get("base_threshold", 0.55)),
                spread_bps_max=float(r.get("spread_bps_max", 8.0)),
                imbalance_abs_max=float(r.get("imbalance_abs_max", 0.60)),
                oc_mode=str(r.get("oc_mode", "throttle") or "throttle"),
                oc_cool_sec=float(r.get("oc_cool_sec", 20.0)),
                oc_max_pos_pct=float(r.get("oc_max_pos_pct", 0.20)),
            )
        self._cache[symbol] = (now, rp)
        return rp
```

> 如需把 TTL 改短做压测，构造 `RiskParamStore(ttl_sec=10)` 即可；也可以配合 Kafka（监听 `config.symbol_risk_params.v1`）在接收消息时 `invalidate(symbol)` 立刻热更——先走 TTL 方案就够用。

```python
def _dynamic_threshold(self, sym: str, tgt):
    rp = self.param_store.get(sym)  # NEW
    base = rp.base_threshold
    alpha = rp.alpha_per_bps
    lookback = int(self.dyn.get("lookback_min", 20))
    tbl = str(self.dyn.get("source_table","lake.bbo_1s_features"))
    start = tgt - timedelta(minutes=lookback)
    rows = self.db.query(f"""
        SELECT toFloat64(quantileExact(0.95)(spread_bps)) AS p95
        FROM {tbl}
        WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s
    """, {"s": sym, "st": start, "ed": tgt}) or [{"p95": 0.0}]
    p95 = float(rows[0]["p95"] or 0.0)
    thr = base + alpha * p95
    thr = max(float(self.dyn.get("min", 0.52)), min(thr, float(self.dyn.get("max", 0.80))))
    return thr
```

### 2.3 盘口异常检查（替换 `_micro_check` 判定线）

```python
def _micro_check(self, sym: str, tgt):
    cfg = self.risk.get("microstructure", {})
    if not cfg.get("enabled", False):
        return None
    rp = self.param_store.get(sym)  # NEW
    spread_max = rp.spread_bps_max
    imb_max = rp.imbalance_abs_max

    tbl = cfg.get("source_table", "lake.bbo_1s_features")
    ql1 = cfg.get("quotes_l1_table", "lake.quotes_l1")
    look = int(cfg.get("lookback_sec", 30))
    start = tgt - timedelta(seconds=look)

    rows = self.db.query(f"""
        SELECT quantileExact(0.95)(spread_bps) AS p95_spread,
               anyLast(imbalance) AS last_imb
        FROM {tbl}
        WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s
    """, {"s": sym, "st": start, "ed": tgt}) or [{"p95_spread":0.0,"last_imb":0.0}]
    p95 = float(rows[0]["p95_spread"] or 0.0)
    last_imb = float(rows[0]["last_imb"] or 0.0)

    if p95 > float(spread_max):
        return {"action": "BLOCK", "reason": "spread_p95_exceed", "metric": p95, "threshold": float(spread_max)}
    if abs(last_imb) > float(imb_max):
        return {"action": "BLOCK", "reason": "imbalance_exceed", "metric": abs(last_imb), "threshold": float(imb_max)}

    if self.risk.get("microstructure", {}).get("crossed_locked_block", True):
        crossed = int((self.db.query(f"""
            SELECT count() AS c FROM {ql1}
            WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s AND ask < bid
            LIMIT 1
        """, {"s": sym, "st": start, "ed": tgt}) or [{"c":0}])[0]["c"])
        if crossed > 0:
            return {"action": "BLOCK", "reason": "crossed_locked", "metric": 1.0, "threshold": 0.0}
    return None
```

### 2.4 开/收盘窗口（替换 `_risk_gate_executor` 的 OC 部分）

```python
def _risk_gate_executor(self, sym, tgt):
    key = (sym, tgt)
    if key in self._rg_cache: return self._rg_cache[key]
    rp = self.param_store.get(sym)  # NEW
    decision = {"allow": True, "action": None, "cool_sec": self.cfg.cool_sec, "max_pos_qty": self.cfg.max_pos_qty}

    # OC window
    oc_cfg = self.risk.get("open_close", {})
    if oc_cfg.get("enabled", False) and rp.oc_mode != "none":
        t_local = tgt.astimezone(NY).time()
        def in_range(s,e):
            sh,sm = map(int,s.split(":")); eh,em = map(int,e.split(":"))
            from datetime import time as T
            return T(sh,sm) <= t_local < T(eh,em)

        is_open  = in_range(oc_cfg.get("open_start","09:30"),  oc_cfg.get("open_end","09:35"))
        is_close = in_range(oc_cfg.get("close_start","15:55"), oc_cfg.get("close_end","16:00"))
        if is_open or is_close:
            if rp.oc_mode == "block":
                decision.update({"allow": False, "action": "BLOCK", "reason": "oc_window_block"})
                self._rg_cache[key] = decision; return decision
            if rp.oc_mode == "throttle":
                decision.update({
                    "action": "THROTTLE",
                    "reason": "oc_window_throttle",
                    "cool_sec": float(rp.oc_cool_sec),
                    "max_pos_qty": max(1, int(self.cfg.max_pos_qty * float(rp.oc_max_pos_pct)))
                })

    # microstructure
    mc = self._micro_check(sym, tgt)
    if mc:
        decision.update({"allow": False, **mc})

    self._rg_cache[key] = decision
    return decision
```

> 其它调用点保持不变（我们之前已经在 `run()` 里把决策结果应用到了冷却/仓位上限并写入监控）。

# 4) 推参工具：YAML/CSV → Kafka → CH（幂等）

**文件**：`tools/push_symbol_params.py`（新建）

```python
# tools/push_symbol_params.py
from __future__ import annotations
import argparse, csv, json, time
from datetime import datetime, timezone
from typing import List, Dict, Any
import yaml  # pip install pyyaml
from aiokafka import AIOKafkaProducer
import asyncio

# 你现有的配置/日志
from libs.config import config
from libs.logger import get_logger

TOPIC = "config.symbol_risk_params.v1"
FIELDS = ["symbol","alpha_per_bps","base_threshold","spread_bps_max","imbalance_abs_max",
          "oc_mode","oc_cool_sec","oc_max_pos_pct","enabled","effective_from","note","ver"]

log = get_logger(__name__)

def _normalize_row(row: Dict[str, Any], ef_default: str | None) -> Dict[str, Any]:
    # 字段清洗/默认
    out = {k: row.get(k) for k in FIELDS}
    out["symbol"] = str(out.get("symbol","*")).upper()
    out["alpha_per_bps"] = float(out.get("alpha_per_bps", 0.008))
    out["base_threshold"] = float(out.get("base_threshold", 0.55))
    out["spread_bps_max"] = float(out.get("spread_bps_max", 8.0))
    out["imbalance_abs_max"] = float(out.get("imbalance_abs_max", 0.60))
    out["oc_mode"] = str(out.get("oc_mode", "throttle"))
    out["oc_cool_sec"] = float(out.get("oc_cool_sec", 20.0))
    out["oc_max_pos_pct"] = float(out.get("oc_max_pos_pct", 0.20))
    out["enabled"] = int(out.get("enabled", 1))
    ef = out.get("effective_from") or ef_default or datetime.now(timezone.utc).isoformat()
    out["effective_from"] = ef
    out["note"] = str(out.get("note",""))
    out["ver"] = int(time.time() * 1_000)  # metadata.ts_us // 1000
    return out

def load_yaml(path: str) -> List[Dict[str, Any]]:
    doc = yaml.safe_load(open(path, "r", encoding="utf-8"))
    if isinstance(doc, dict) and "items" in doc:
        return list(doc["items"])
    if isinstance(doc, list):
        return list(doc)
    raise ValueError("YAML must be a list or {items: [...]}")

def load_csv(path: str) -> List[Dict[str, Any]]:
    rows = []
    with open(path, "r", encoding="utf-8") as f:
        for r in csv.DictReader(f):
            rows.append(r)
    return rows

async def push(rows: List[Dict[str, Any]], dry_run=False):
    if dry_run:
        for r in rows:
            print(json.dumps({"data": r, "metadata": {"ts_us": int(time.time()*1_000_000)}, "source":"PARAM_PUSH"}, ensure_ascii=False))
        return

    producer = AIOKafkaProducer(
        bootstrap_servers=config.kafka_config["bootstrap_servers"],
        value_serializer=lambda v: json.dumps(v).encode("utf-8"),
        key_serializer=lambda v: v.encode("utf-8"),
        linger_ms=5, retries=3
    )
    await producer.start()
    try:
        for r in rows:
            msg = {"data": r, "metadata": {"ts_us": int(time.time()*1_000_000)}, "source":"PARAM_PUSH"}
            key = r["symbol"]
            await producer.send(TOPIC, key=key, value=msg)
            log.info("pushed", symbol=r["symbol"], spread_bps_max=r["spread_bps_max"], imbalance_abs_max=r["imbalance_abs_max"])
    finally:
        await producer.stop()

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--yaml", help="YAML file of rows")
    ap.add_argument("--csv", help="CSV file of rows")
    ap.add_argument("--effective-from", help="override effective_from for all rows, ISO8601")
    ap.add_argument("--dry-run", action="store_true", default=False)
    args = ap.parse_args()
    if not args.yaml and not args.csv:
        ap.error("must provide --yaml or --csv")

    raw_rows = load_yaml(args.yaml) if args.yaml else load_csv(args.csv)
    rows = [_normalize_row(r, args.effective_from) for r in raw_rows]
    asyncio.get_event_loop().run_until_complete(push(rows, dry_run=args.dry_run))

if __name__ == "__main__":
    main()
```
