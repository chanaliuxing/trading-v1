# 0) 设计要点（与你现状贴合）

* **数据来源**：统一读 `bars_{res}_current`（P2）+ `features_{res}`（已有），用我们在 P6 的 `MarketDataReader`。
* **标签**（默认）：下一根收益 `fwd_ret_1 = ln(close[t+1]/close[t])`；

  * 二分类：`fwd_ret_1 > 0` → 1，否则 0；
  * 回归：直接预测 `fwd_ret_1`（可切换）。
* **版本号**：`data_version=YYYYMMDD.HHMM`（UTC）；训练产物与评估全面写入 ClickHouse 的 `ml` 库，保证**可追溯/可回滚**。
* **模型**：默认 `sklearn`（避免依赖地狱）

  * 分类：`LogisticRegression(max_iter=200)`；
  * 回归：`RandomForestRegressor`（轻量鲁棒）。
* **回测**：长/空平替简单规则（默认 long/flat），成本来自 `lake.market_cost_profile_minute_vw`（P5/P6）。
* **登记**：把模型、数据版本、指标写入 CH；产物用 `joblib` 落地+URI登记（本地/对象存储路径均可）。

```sql
-- ML 登记库
CREATE DATABASE IF NOT EXISTS ml;

-- A) 数据集版本登记
CREATE TABLE IF NOT EXISTS ml.dataset_versions
(
  data_version   String,
  created_at     DateTime DEFAULT now(),
  res            LowCardinality(String),
  symbols        Array(String),
  start_date     Date,
  end_date       Date,
  rows           UInt64,
  features       Array(String),
  label_name     LowCardinality(String),
  note           String DEFAULT ''
)
ENGINE = ReplacingMergeTree(created_at)
PARTITION BY toYYYYMM(created_at)
ORDER BY (data_version, res);

-- B) 模型注册表（支持回滚/切换阶段）
CREATE TABLE IF NOT EXISTS ml.models_registry
(
  model_name   LowCardinality(String),
  version      String,                             -- 同 data_version 或额外模型版本
  created_at   DateTime DEFAULT now(),
  res          LowCardinality(String),
  data_version String,
  algo         LowCardinality(String),
  params_json  String,
  uri          String,                             -- 模型文件URI（file:// 或 s3://）
  stage        LowCardinality(String) DEFAULT 'Staging',  -- Staging/Production/Archived
  auc          Nullable(Float64),
  accuracy     Nullable(Float64),
  f1           Nullable(Float64),
  rmse         Nullable(Float64),
  sharpe       Nullable(Float64),
  cagr         Nullable(Float64),
  max_dd       Nullable(Float64)
)
ENGINE = ReplacingMergeTree(created_at)
PARTITION BY toYYYYMM(created_at)
ORDER BY (model_name, version);

-- C) 运行记录（流水账）
CREATE TABLE IF NOT EXISTS ml.run_history
(
  run_id      UUID,
  started_at  DateTime,
  finished_at DateTime,
  step        LowCardinality(String),   -- make_dataset/train/backtest/register
  ok          UInt8,
  message     String,
  payload     String
)
ENGINE = MergeTree
ORDER BY (started_at, run_id)
PARTITION BY toYYYYMM(started_at);
```

```python
# libs/ml/registry.py
from __future__ import annotations
import json, uuid, time
from datetime import datetime
from typing import Any, Dict, List, Optional
from libs.db.clickhouse import ClickHouseDB

class MLRegistry:
    def __init__(self):
        self.db = ClickHouseDB()

    # 运行流水
    def run_log(self, step: str, ok: bool, message: str = "", payload: Dict[str, Any] | None = None):
        rid = str(uuid.uuid4())
        now = datetime.utcnow()
        self.db.query("""
            INSERT INTO ml.run_history (run_id, started_at, finished_at, step, ok, message, payload)
            VALUES (%(rid)s, %(t)s, %(t)s, %(step)s, %(ok)s, %(msg)s, %(pl)s)
        """, {"rid": rid, "t": now, "step": step, "ok": int(ok), "msg": message, "pl": json.dumps(payload or {})})

    # 数据集登记
    def register_dataset(self, *, data_version: str, res: str, symbols: List[str],
                         start_date, end_date, rows: int, features: List[str], label_name: str, note: str = ""):
        self.db.query("""
            INSERT INTO ml.dataset_versions
            (data_version, res, symbols, start_date, end_date, rows, features, label_name, note)
            VALUES (%(dv)s,%(res)s,%(syms)s,%(start)s,%(end)s,%(rows)s,%(feats)s,%(label)s,%(note)s)
        """, {"dv": data_version, "res": res, "syms": symbols, "start": start_date, "end": end_date,
              "rows": int(rows), "feats": features, "label": label_name, "note": note})

    # 模型登记
    def register_model(self, *, model_name: str, version: str, res: str, data_version: str,
                       algo: str, params: Dict[str, Any], uri: str, stage: str,
                       metrics: Dict[str, Any]):
        self.db.query("""
            INSERT INTO ml.models_registry
            (model_name, version, res, data_version, algo, params_json, uri, stage,
             auc, accuracy, f1, rmse, sharpe, cagr, max_dd)
            VALUES (%(mn)s,%(ver)s,%(res)s,%(dv)s,%(algo)s,%(pjs)s,%(uri)s,%(stage)s,
                    %(auc)s,%(acc)s,%(f1)s,%(rmse)s,%(sharpe)s,%(cagr)s,%(mdd)s)
        """, {"mn": model_name, "ver": version, "res": res, "dv": data_version, "algo": algo,
              "pjs": json.dumps(params or {}), "uri": uri, "stage": stage,
              "auc": metrics.get("auc"), "acc": metrics.get("accuracy"),
              "f1": metrics.get("f1"), "rmse": metrics.get("rmse"),
              "sharpe": metrics.get("sharpe"), "cagr": metrics.get("cagr"), "mdd": metrics.get("max_dd")})
```

# 3) 代码：制备数据（**新建** `services/ml/pipelines/make_dataset.py`）

```python
# services/ml/pipelines/make_dataset.py
from __future__ import annotations
import argparse, os, json
from datetime import datetime, timedelta, timezone
import pandas as pd
from libs.db.clickhouse import ClickHouseDB
from libs.data_reader import MarketDataReader
from libs.ml.registry import MLRegistry

def now_version() -> str:
    return datetime.utcnow().strftime("%Y%m%d.%H%M")

def make_dataset(res: str, symbols: list[str], lookback_days: int,
                 label_type: str = "bin", horizon: int = 1,
                 out_dir: str = "artifacts/ml") -> dict:
    os.makedirs(out_dir, exist_ok=True)
    db = ClickHouseDB()
    md = MarketDataReader(db)
    end = datetime.utcnow().replace(tzinfo=timezone.utc)
    start = end - timedelta(days=int(lookback_days))
    # 拉 bars+features
    bars = md.bars(symbols, res, start=start, end=end)
    feats = md.features(symbols, res, start=start, end=end)
    if bars.empty or feats.empty:
        raise RuntimeError("No bars/features data")
    df = bars.join(feats, how="left")
    # 生成标签：下一根收益
    df = df.reset_index().rename(columns={"ts_start":"ts"})
    df.sort_values(["symbol","ts"], inplace=True)
    df["close_shift"] = df.groupby("symbol")["close"].shift(-horizon)
    df["fwd_ret_1"] = (df["close_shift"]/df["close"]).apply(lambda x: float("nan") if pd.isna(x) or x<=0 else __import__("math").log(x))
    if label_type == "bin":
        df["label"] = (df["fwd_ret_1"] > 0).astype("int8")
    else:
        df["label"] = df["fwd_ret_1"]
    # 清洗与选列
    keep_cols = ["symbol","ts","open","high","low","close","volume","label"]
    feature_cols = [c for c in df.columns if c not in set(keep_cols+["close_shift","fwd_ret_1"])]
    # 简单空值处理
    df[feature_cols] = df[feature_cols].fillna(method="ffill").fillna(method="bfill")
    df = df.dropna(subset=["label"])
    # 划分 train/val/test（按日期自然切分：70/15/15）
    ts_min, ts_max = df["ts"].min(), df["ts"].max()
    dt_range = (ts_max - ts_min).days + 1
    t1 = ts_min + timedelta(days=int(dt_range*0.7))
    t2 = ts_min + timedelta(days=int(dt_range*0.85))
    df["split"] = pd.cut(df["ts"], [-1e18, t1, t2, 1e18], labels=["train","val","test"])
    # 输出
    dv = now_version()
    fpath = os.path.join(out_dir, f"dataset_{res}_{dv}.parquet")
    cols_out = ["symbol","ts","label"] + feature_cols
    df[cols_out].to_parquet(fpath, index=False)
    info = {
        "data_version": dv,
        "res": res,
        "symbols": symbols,
        "start_date": pd.to_datetime(ts_min).date().isoformat(),
        "end_date": pd.to_datetime(ts_max).date().isoformat(),
        "rows": int(len(df)),
        "features": feature_cols,
        "label_name": f"fwd_ret_{horizon}_{label_type}"
    }
    # 登记数据集
    MLRegistry().register_dataset(
        data_version=dv, res=res, symbols=symbols,
        start_date=pd.to_datetime(ts_min).date(),
        end_date=pd.to_datetime(ts_max).date(),
        rows=len(df), features=feature_cols, label_name=info["label_name"], note=""
    )
    return {"path": fpath, **info}

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--res", default="1m")
    ap.add_argument("--symbols", default="TSLA,NVDA,AMD,MSFT,AMZN")
    ap.add_argument("--lookback-days", type=int, default=180)
    ap.add_argument("--label-type", choices=["bin","reg"], default="bin")
    ap.add_argument("--horizon", type=int, default=1)
    ap.add_argument("--out-dir", default="artifacts/ml")
    args = ap.parse_args()
    out = make_dataset(args.res, [s.strip().upper() for s in args.symbols.split(",")],
                       args.lookback_days, args.label_type, args.horizon, args.out_dir)
    print(json.dumps(out, indent=2))
```

# 4) 代码：训练（**新建** `services/ml/pipelines/train_model.py`）

```python
# services/ml/pipelines/train_model.py
from __future__ import annotations
import argparse, os, json
import joblib
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestRegressor
from sklearn.pipeline import Pipeline
from sklearn.metrics import roc_auc_score, accuracy_score, f1_score, mean_squared_error
from libs.ml.registry import MLRegistry

def train(dataset_path: str, model_name: str = "clf_lr", label_type: str = "bin",
          out_dir: str = "artifacts/ml") -> dict:
    os.makedirs(out_dir, exist_ok=True)
    df = pd.read_parquet(dataset_path)
    feats = [c for c in df.columns if c not in ("symbol","ts","label","split")]
    train_df = df[df["split"]=="train"]
    val_df   = df[df["split"]=="val"]
    X_tr, y_tr = train_df[feats].values, train_df["label"].values
    X_va, y_va = val_df[feats].values, val_df["label"].values

    if label_type == "bin":
        pipe = Pipeline([
            ("scaler", StandardScaler(with_mean=False)),
            ("clf", LogisticRegression(max_iter=200, n_jobs=None))
        ])
        pipe.fit(X_tr, y_tr)
        proba = pipe.predict_proba(X_va)[:,1]
        pred  = (proba >= 0.5).astype(int)
        metrics = {
            "auc": float(roc_auc_score(y_va, proba)),
            "accuracy": float(accuracy_score(y_va, pred)),
            "f1": float(f1_score(y_va, pred)),
            "rmse": None
        }
        algo = "LogisticRegression"
    else:
        pipe = Pipeline([
            ("rf", RandomForestRegressor(n_estimators=200, n_jobs=-1, random_state=42))
        ])
        pipe.fit(X_tr, y_tr)
        pred  = pipe.predict(X_va)
        rmse  = mean_squared_error(y_va, pred, squared=False)
        metrics = {"auc": None, "accuracy": None, "f1": None, "rmse": float(rmse)}
        algo = "RandomForestRegressor"

    # 保存模型
    dv = dataset_path.split("_")[-1].replace(".parquet","")
    ver = dv
    fpath = os.path.join(out_dir, f"{model_name}_{ver}.joblib")
    joblib.dump({"model": pipe, "features": feats, "label_type": label_type}, fpath)

    # 登记模型（训练阶段，仅填充分类/回归指标）
    MLRegistry().register_model(
        model_name=model_name, version=ver, res="1m", data_version=dv,
        algo=algo, params=pipe.get_params(), uri=f"file://{os.path.abspath(fpath)}",
        stage="Staging", metrics={"auc":metrics["auc"], "accuracy":metrics["accuracy"], "f1":metrics["f1"],
                                  "rmse":metrics["rmse"], "sharpe":None, "cagr":None, "max_dd":None}
    )
    return {"version": ver, "uri": fpath, "metrics": metrics}

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--dataset", required=True)
    ap.add_argument("--model-name", default="clf_lr")
    ap.add_argument("--label-type", choices=["bin","reg"], default="bin")
    ap.add_argument("--out-dir", default="artifacts/ml")
    args = ap.parse_args()
    out = train(args.dataset, args.model_name, args.label_type, args.out_dir)
    print(json.dumps(out, indent=2))
```

# 5) 代码：回测（**新建** `services/ml/pipelines/backtest_model.py`）

```python
# services/ml/pipelines/backtest_model.py
from __future__ import annotations
import argparse, json, math, os
import joblib
import pandas as pd
from datetime import timedelta
from libs.ml.registry import MLRegistry

def compute_metrics(equity: pd.Series):
    # 年化/夏普/最大回撤（按 1min bar 粒度，252*6.5*60 ≈ 98280）
    r = equity.pct_change().dropna()
    if r.empty:
        return {"sharpe": None, "cagr": None, "max_dd": None}
    minutes_per_year = 252*390
    ann_mu = r.mean()*minutes_per_year
    ann_sigma = r.std()*(minutes_per_year**0.5)
    sharpe = None if ann_sigma==0 else float(ann_mu/ann_sigma)
    cagr = float((equity.iloc[-1]/equity.iloc[0])**(minutes_per_year/len(equity)) - 1) if len(equity)>0 else None
    # 最大回撤
    roll_max = equity.cummax()
    dd = (equity/roll_max - 1.0).min()
    return {"sharpe": sharpe, "cagr": cagr, "max_dd": float(dd)}

def backtest(dataset_path: str, model_uri: str, label_type: str = "bin",
             threshold: float = 0.5, cost_bps: float = 1.0,
             start_split: str = "test", out_dir: str = "artifacts/ml") -> dict:
    os.makedirs(out_dir, exist_ok=True)
    df = pd.read_parquet(dataset_path)
    df = df[df["split"]==start_split].copy()
    feats = [c for c in df.columns if c not in ("symbol","ts","label","split")]
    bundle = joblib.load(model_uri.replace("file://",""))
    pipe = bundle["model"]
    if label_type == "bin":
        proba = pipe.predict_proba(df[feats].values)[:,1]
        df["sig"] = (proba >= threshold).astype(int)   # long/flat
        ret = df.groupby("symbol")["label"].apply(lambda x: x.shift(0))  # fwd_ret_1
        df["ret"] = ret.values
    else:
        pred = pipe.predict(df[feats].values)
        df["sig"] = (pred > 0).astype(int)
        df["ret"] = df["label"]

    # 交易成本（bps → 返回率扣减）
    cost = cost_bps / 1e4
    # 每根产生换手（sig变化→执行一次，简单化）
    df["turnover"] = df.groupby("symbol")["sig"].diff().abs().fillna(df["sig"].abs())
    df["gross"] = df["sig"] * df["ret"]
    df["net"] = df["gross"] - df["turnover"]*cost

    # 聚合为组合曲线（等权）
    ts_order = df["ts"].sort_values().unique()
    curve = []
    equity = 1.0
    for ts in ts_order:
        slice_ = df[df["ts"]==ts]
        if len(slice_)==0: 
            curve.append((ts, equity)); continue
        equity *= float(1.0 + slice_["net"].mean())
        curve.append((ts, equity))
    eq = pd.Series([x[1] for x in curve], index=pd.to_datetime([x[0] for x in curve]))
    m = compute_metrics(eq)

    # 登记补充回测指标到模型表（同 version）
    dv = dataset_path.split("_")[-1].replace(".parquet","")
    ver = dv
    MLRegistry().register_model(
        model_name=os.path.basename(model_uri).split(".joblib")[0].split("_")[0],
        version=ver, res="1m", data_version=dv, algo=bundle["model"].__class__.__name__,
        params={}, uri=model_uri, stage="Staging",
        metrics={"auc":None,"accuracy":None,"f1":None,"rmse":None,"sharpe":m["sharpe"],"cagr":m["cagr"],"max_dd":m["max_dd"]}
    )

    out = {"equity_end": float(eq.iloc[-1]), **m}
    # 保存回测曲线
    eq_path = os.path.join(out_dir, f"equity_{ver}.csv")
    eq.to_csv(eq_path, header=["equity"])
    return {"metrics": out, "equity_path": eq_path}

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--dataset", required=True)
    ap.add_argument("--model-uri", required=True)
    ap.add_argument("--label-type", choices=["bin","reg"], default="bin")
    ap.add_argument("--threshold", type=float, default=0.5)
    ap.add_argument("--cost-bps", type=float, default=1.0)
    ap.add_argument("--start-split", choices=["val","test"], default="test")
    ap.add_argument("--out-dir", default="artifacts/ml")
    args = ap.parse_args()
    out = backtest(args.dataset, args.model_uri, args.label_type, args.threshold, args.cost_bps, args.start_split, args.out_dir)
    print(json.dumps(out, indent=2))
```

* **回滚**：将 `models_registry` 中某个历史版本的 `stage` 更新为 `Production`，并在你的在线推理/信号产线上仅加载 `stage='Production'` 的 URI。
