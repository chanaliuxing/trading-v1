太好，继续 **P11｜PNL 聚合与看板（纸上/影子交易闭环）**。这一步把 **orders/fills/positions → 分钟级 PNL → 日级 PNL** 做成“可落地”的最小闭环，仍坚持你的总原则：**写入优先走 Kafka → ClickHouse（可直写兜底）**、幂等覆盖、随时灰度/回退。

# 1) ClickHouse DDL（追加到 `ch_init.sql`，只做加法）

```sql
-- 分钟级 PNL 明细（按 ts 分钟对齐，UTC）
CREATE TABLE IF NOT EXISTS trading.pnl_minute
(
  account        LowCardinality(String) DEFAULT 'paper',
  symbol         String,
  ts             DateTime,                    -- 分钟对齐（UTC），通常等于 bars_1m.ts_start
  pos_qty        Int64,                       -- 该分钟结束时持仓
  pos_avg        Float64,                     -- 该分钟结束时持仓均价（无持仓=0）
  mtm_price      Float64,                     -- 标记价（默认 bars_1m_current.close）
  fees_cum       Float64,                     -- 累计费用
  realized_cum   Float64,                     -- 累计已实现 PnL
  unrealized     Float64,                     -- (mtm - pos_avg)*pos_qty
  total_pnl      Float64,                     -- realized_cum - fees_cum + unrealized
  trades_in_min  UInt32,                      -- 该分钟内成交笔数
  ver            UInt64,                      -- metadata.ts_us // 1000
  d              Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (account, symbol, ts);

-- 日级 PNL 汇总（按美东交易日）
CREATE TABLE IF NOT EXISTS trading.pnl_daily
(
  d_local        Date,                        -- America/New_York 的交易日
  account        LowCardinality(String) DEFAULT 'paper',
  symbol         String,
  realized       Float64,                     -- 当日已实现 PnL（含费）
  unrealized_eod Float64,                     -- 收盘的未实现（持仓×(收盘-均价)）
  fees           Float64,
  trades         UInt32,
  turn_qty       UInt64,                      -- 当日换手量（|买|+|卖|）
  pos_qty_eod    Int64,
  mtm_close      Float64,                     -- 当日收盘标记价
  ver            UInt64,
  d              Date DEFAULT d_local
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (d_local, account, symbol);
```

> 说明
>
> * 分钟标记价 `mtm_price` 默认用 `trading.bars_1m_current.close`（与信号周期一致，避免“看见未来”的秒级偏差）。
> * 幂等：重复写同一 `(account, symbol, ts)` 以更大 `ver` 覆盖。
> * `d_local` 用 **America/New_York** 的日历，更贴近交易员视角。

# 3) 批处理：构建分钟级 PNL（**新建** `services/analytics/build_intraday_pnl.py`）

> 默认 **Kafka 模式** 产出到 `pnl.minute.v1`，统一由 `GenericCHWriter` 落库；如需简化，可传 `--mode direct` 直接 INSERT ClickHouse（兜底）。

```python
# services/analytics/build_intraday_pnl.py
from __future__ import annotations
import argparse, asyncio, json, time
from dataclasses import dataclass
from datetime import date, datetime, timedelta, time as dtime, timezone
from typing import Dict, List, Tuple
from zoneinfo import ZoneInfo

import pandas as pd
from aiokafka import AIOKafkaProducer

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

log = get_logger(__name__)
NY = ZoneInfo("America/New_York")

@dataclass
class RunCfg:
    day: date
    account: str = "paper"
    res: str = "1m"
    topic: str = "pnl.minute.v1"
    mode: str = "kafka"  # kafka | direct

def rth_minutes(d_local: date) -> List[datetime]:
    start = datetime.combine(d_local, dtime(9,30), tzinfo=NY).astimezone(timezone.utc)
    end   = datetime.combine(d_local, dtime(16,0), tzinfo=NY).astimezone(timezone.utc)
    t = start
    acc = []
    while t < end:
        acc.append(t.replace(second=0, microsecond=0))
        t += timedelta(minutes=1)
    return acc

def load_inputs(db: ClickHouseDB, d_local: date, account: str) -> Tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    # 当日 fills（纸上经纪）
    fills = db.query("""
        SELECT order_id, ts, symbol, side, qty, price, fee
        FROM trading.fills
        WHERE d = %(d)s
    """, {"d": d_local})
    df_fills = pd.DataFrame(fills) if fills else pd.DataFrame(columns=["order_id","ts","symbol","side","qty","price","fee"])
    if not df_fills.empty:
        df_fills["ts"] = pd.to_datetime(df_fills["ts"], utc=True)

    # 当日 1m 收盘价（标记价）
    bars = db.query("""
        SELECT symbol, ts_start AS ts, close AS px
        FROM trading.bars_1m_current
        WHERE d = %(d)s
    """, {"d": d_local})
    df_mtm = pd.DataFrame(bars) if bars else pd.DataFrame(columns=["symbol","ts","px"])
    if not df_mtm.empty:
        df_mtm["ts"] = pd.to_datetime(df_mtm["ts"], utc=True)

    # 前一日收盘后的尾仓（如有）
    yday = d_local - timedelta(days=1)
    pos = db.query("""
        SELECT symbol, qty, avg_price
        FROM trading.positions_intraday
        WHERE d = %(d)s
    """, {"d": yday})
    df_prev = pd.DataFrame(pos) if pos else pd.DataFrame(columns=["symbol","qty","avg_price"])

    return df_fills, df_mtm, df_prev

def roll_positions(df_fills: pd.DataFrame, df_mtm: pd.DataFrame, df_prev: pd.DataFrame,
                   mins: List[datetime], account: str) -> pd.DataFrame:
    symbols = sorted(set(df_mtm["symbol"].unique()) | set(df_fills["symbol"].unique()))
    records = []
    ts_us = int(time.time() * 1_000_000)

    for sym in symbols:
        # init from yesterday's EOD
        qty = 0; avg = 0.0; realized = 0.0; fees_cum = 0.0
        if not df_prev.empty and sym in df_prev["symbol"].values:
            r = df_prev[df_prev["symbol"]==sym].iloc[0]
            qty = int(r["qty"] or 0); avg = float(r["avg_price"] or 0.0)

        f = df_fills[df_fills["symbol"]==sym].sort_values("ts")
        m = df_mtm[df_mtm["symbol"]==sym].set_index("ts").sort_index()

        for t in mins:
            # apply fills within this minute
            trades = f[(f["ts"] >= t) & (f["ts"] < t + timedelta(minutes=1))]
            trades_in_min = int(len(trades))
            if trades_in_min:
                for _, row in trades.iterrows():
                    side = int(row["side"]); q = int(row["qty"]); px = float(row["price"]); fee = float(row["fee"] or 0)
                    fees_cum += fee
                    if side > 0:  # buy
                        new_qty = qty + q
                        avg = (avg*qty + px*q)/new_qty if new_qty != 0 else 0.0
                        qty = new_qty
                    else:         # sell
                        sell = q
                        realized += (px - avg) * sell
                        qty -= sell
                        if qty == 0: avg = 0.0

            # mark
            try:
                px = float(m.loc[t, "px"])
            except Exception:
                # 无该分钟 bar：用上一分钟价格或 0
                px = float(m["px"].iloc[m.index.searchsorted(t)-1]) if len(m) and m.index.searchsorted(t)>0 else 0.0

            unrealized = (px - avg) * qty
            total = realized - fees_cum + unrealized

            records.append({
                "account": account,
                "symbol": sym,
                "ts": t.replace(second=0, microsecond=0),
                "pos_qty": int(qty),
                "pos_avg": float(avg),
                "mtm_price": float(px),
                "fees_cum": float(fees_cum),
                "realized_cum": float(realized),
                "unrealized": float(unrealized),
                "total_pnl": float(total),
                "trades_in_min": trades_in_min,
                "ver": int(ts_us//1000)
            })
    return pd.DataFrame(records)

async def produce_kafka(df: pd.DataFrame, topic: str):
    producer = AIOKafkaProducer(
        bootstrap_servers=config.kafka_config["bootstrap_servers"],
        value_serializer=lambda v: json.dumps(v).encode("utf-8"),
        key_serializer=lambda v: v.encode("utf-8"),
        linger_ms=5, retries=3
    )
    await producer.start()
    try:
        for _, r in df.iterrows():
            payload = {"data": r.to_dict(), "metadata": {"ts_us": int(time.time()*1_000_000)}, "source": "PNL_JOB"}
            await producer.send(topic, key=r["symbol"], value=payload)
    finally:
        await producer.stop()

def insert_direct(db: ClickHouseDB, df: pd.DataFrame):
    cols = ["account","symbol","ts","pos_qty","pos_avg","mtm_price","fees_cum","realized_cum","unrealized","total_pnl","trades_in_min","ver"]
    db.insert("trading.pnl_minute", df[cols].to_dict("records"), cols)

def run(day: date, mode: str, account: str):
    db = ClickHouseDB()
    mins = rth_minutes(day)
    df_fills, df_mtm, df_prev = load_inputs(db, day, account)
    if df_mtm.empty:
        log.warning("No 1m bars for day", day=str(day))
        return
    df = roll_positions(df_fills, df_mtm, df_prev, mins, account)
    if df.empty:
        log.info("No symbols to build pnl", day=str(day)); return
    if mode == "kafka":
        asyncio.get_event_loop().run_until_complete(produce_kafka(df, "pnl.minute.v1"))
    else:
        insert_direct(db, df)
    log.info("pnl_minute done", rows=len(df))

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--day", default=None, help="YYYY-MM-DD (local NY); default=yesterday")
    ap.add_argument("--mode", choices=["kafka","direct"], default="kafka")
    ap.add_argument("--account", default="paper")
    args = ap.parse_args()
    d = date.fromisoformat(args.day) if args.day else (datetime.now(tz=NY).date() - timedelta(days=1))
    run(d, args.mode, args.account)
```

# 4) 批处理：生成日级 PNL（**新建** `services/analytics/build_daily_pnl.py`）

```python
# services/analytics/build_daily_pnl.py
from __future__ import annotations
import argparse, asyncio, json, time
from datetime import date, datetime, timedelta
from zoneinfo import ZoneInfo
import pandas as pd
from aiokafka import AIOKafkaProducer

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

log = get_logger(__name__)
NY = ZoneInfo("America/New_York")

def load_pnl_minute(db: ClickHouseDB, d_local: date, account: str) -> pd.DataFrame:
    rows = db.query("""
        SELECT *
        FROM trading.pnl_minute
        WHERE d = %(d)s AND account=%(a)s
    """, {"d": d_local, "a": account})
    df = pd.DataFrame(rows) if rows else pd.DataFrame()
    if not df.empty:
        df["ts"] = pd.to_datetime(df["ts"], utc=True)
    return df

async def produce_daily(df: pd.DataFrame):
    prod = AIOKafkaProducer(
        bootstrap_servers=config.kafka_config["bootstrap_servers"],
        value_serializer=lambda v: json.dumps(v).encode("utf-8"),
        key_serializer=lambda v: v.encode("utf-8"),
    )
    await prod.start()
    try:
        for _, r in df.iterrows():
            payload = {"data": r.to_dict(), "metadata": {"ts_us": int(time.time()*1_000_000)}, "source": "PNL_JOB"}
            await prod.send("pnl.daily.v1", key=f"{r['account']}:{r['symbol']}", value=payload)
    finally:
        await prod.stop()

def run(day: date, mode: str, account: str):
    db = ClickHouseDB()
    dfm = load_pnl_minute(db, day, account)
    if dfm.empty:
        log.warning("No pnl_minute for day", day=str(day)); return
    # 聚合：最后一刻定义 EOD
    g = dfm.groupby(["account","symbol"])
    res = []
    ts_us = int(time.time()*1_000_000)
    for (acc, sym), grp in g:
        grp = grp.sort_values("ts")
        realized = float(grp["realized_cum"].iloc[-1] - grp["fees_cum"].iloc[-1])
        fees = float(grp["fees_cum"].iloc[-1])
        pos_qty_eod = int(grp["pos_qty"].iloc[-1])
        mtm_close = float(grp["mtm_price"].iloc[-1])
        pos_avg = float(grp["pos_avg"].iloc[-1])
        unrealized_eod = (mtm_close - pos_avg) * pos_qty_eod
        trades = int(grp["trades_in_min"].sum())
        turn_qty = int(
            # 以分钟净变动估算换手（也可直接统计 fills）
            abs(grp["pos_qty"].diff().fillna(grp["pos_qty"]).astype(int)).sum()
        )
        res.append({
            "d_local": day, "account": acc, "symbol": sym,
            "realized": realized, "unrealized_eod": float(unrealized_eod),
            "fees": fees, "trades": trades, "turn_qty": turn_qty,
            "pos_qty_eod": pos_qty_eod, "mtm_close": mtm_close,
            "ver": int(ts_us//1000)
        })
    dfd = pd.DataFrame(res)
    if mode == "kafka":
        asyncio.get_event_loop().run_until_complete(produce_daily(dfd))
    else:
        cols = ["d_local","account","symbol","realized","unrealized_eod","fees","trades","turn_qty","pos_qty_eod","mtm_close","ver"]
        db.insert("trading.pnl_daily", dfd[cols].to_dict("records"), cols)
    log.info("pnl_daily done", rows=len(dfd))

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--day", default=None, help="YYYY-MM-DD NY; default=yesterday")
    ap.add_argument("--mode", choices=["kafka","direct"], default="kafka")
    ap.add_argument("--account", default="paper")
    args = ap.parse_args()
    d = date.fromisoformat(args.day) if args.day else (datetime.now(tz=NY).date() - timedelta(days=1))
    run(d, args.mode, args.account)
```
