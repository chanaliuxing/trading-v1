# 1) ClickHouse：成本明细表 + 视图（追加到 `ch_init.sql`）

```sql
-- A) 日内分钟粒度的成本明细（由 bbo_1s_features 批量生成）
CREATE TABLE IF NOT EXISTS lake.market_cost_intraday_minute
(
  symbol           String,
  d                Date,                 -- 交易日（来自 ts）
  minute_of_day    UInt16,               -- 0..1439
  p95_spread_bps   Nullable(Float64),
  p50_spread_bps   Nullable(Float64),
  avg_spread_bps   Nullable(Float64),
  samples          UInt32,               -- 该分钟内有效样本数(秒)
  updated_at       DateTime DEFAULT now()
)
ENGINE = ReplacingMergeTree(updated_at)
PARTITION BY d
ORDER BY (symbol, d, minute_of_day)
TTL d + INTERVAL 365 DAY;

-- B) 30 天滚动“分钟槽成本曲线”（快速给策略用）
-- 说明：用日级明细近似聚合，速度快；若要精确也给了 raw 视图（见 C）。
CREATE OR REPLACE VIEW lake.market_cost_profile_minute_vw AS
SELECT
  symbol,
  minute_of_day,
  quantileTDigest(0.95)(p95_spread_bps) AS p95_spread_bps_30d,   -- 近似
  avg(avg_spread_bps)                   AS avg_spread_bps_30d,
  count()                               AS days
FROM lake.market_cost_intraday_minute
WHERE d >= today() - 30
GROUP BY symbol, minute_of_day;

-- C) 精确但更重的原始视图（直接从秒级特征算 30 天曲线）
CREATE OR REPLACE VIEW lake.market_cost_profile_minute_raw_vw AS
SELECT
  symbol,
  (toHour(ts)*60 + toMinute(ts)) AS minute_of_day,
  quantileTDigestIf(0.95)(spread_bps, is_valid=1) AS p95_spread_bps_30d,
  avgIf(spread_bps, is_valid=1)                   AS avg_spread_bps_30d,
  countIf(is_valid=1)                             AS samples
FROM lake.bbo_1s_features
WHERE d >= today() - 30
GROUP BY symbol, minute_of_day;
```

> 用法建议：策略在线查询优先用 **`market_cost_profile_minute_vw`**（快）；回测或校准可对照 **`*_raw_vw`**（精）。

# 2) 每日批处理脚本（新增：`services/analytics/build_market_cost_profile.py`）

```python
# services/analytics/build_market_cost_profile.py
from datetime import date, timedelta
from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger

log = get_logger(__name__)

SQL_INSERT_INTRADAY = """
INSERT INTO lake.market_cost_intraday_minute
SELECT
  symbol,
  toDate(ts) AS d,
  (toHour(ts)*60 + toMinute(ts)) AS minute_of_day,
  quantileTDigestIf(0.95)(spread_bps, is_valid=1) AS p95_spread_bps,
  quantileTDigestIf(0.50)(spread_bps, is_valid=1) AS p50_spread_bps,
  avgIf(spread_bps, is_valid=1)                   AS avg_spread_bps,
  countIf(is_valid=1)                              AS samples,
  now()                                            AS updated_at
FROM lake.bbo_1s_features
WHERE d = %(day)s
GROUP BY symbol, d, minute_of_day
"""

def run(day: date | None = None):
    d = day or (date.today() - timedelta(days=1))
    db = ClickHouseDB()
    try:
        log.info("building market_cost_intraday_minute", day=str(d))
        # 有的封装没有 execute，可以用 query 代替
        try:
            db.execute(SQL_INSERT_INTRADAY, {"day": d})
        except AttributeError:
            db.query(SQL_INSERT_INTRADAY, {"day": d})
        log.info("done", day=str(d))
    finally:
        try:
            db.close()
        except Exception:
            pass

if __name__ == "__main__":
    run()
```

* 依赖：P5 已落地 `lake.bbo_1s_features`（MV 自动产出）。
* 执行时机：**P4 对账跑完后**（建议 02:20 ET 之后），按天增量写入。
* 幂等：`ReplacingMergeTree(updated_at)`，重复跑会被最新覆盖。

# 4) 给策略的查询示例（直接套用）

```sql
-- 某标的、某分钟槽（本地时区换算为 NY 分钟槽）：
SELECT p95_spread_bps_30d, avg_spread_bps_30d
FROM lake.market_cost_profile_minute_vw
WHERE symbol = 'TSLA' AND minute_of_day = 9*60 + 35;

-- 取整天曲线（可画门槛曲线）
SELECT minute_of_day, p95_spread_bps_30d
FROM lake.market_cost_profile_minute_vw
WHERE symbol = 'TSLA'
ORDER BY minute_of_day;
```

> 你也可以把 **`p95_spread_bps_30d`** 作为下单门槛/滑点保护的基准：
> `min_tick + k * p95_spread_bps_30d`（k≈1.0~1.5）——高波动时自动抬高门槛。

# 5) 验收 SQL

```sql
-- 昨日是否生成
SELECT * FROM lake.market_cost_intraday_minute
WHERE d = yesterday()
ORDER BY symbol, minute_of_day
LIMIT 20;

-- 视图可用性
SELECT symbol, minute_of_day, p95_spread_bps_30d, avg_spread_bps_30d
FROM lake.market_cost_profile_minute_vw
WHERE symbol IN ('TSLA','NVDA')
ORDER BY symbol, minute_of_day
LIMIT 10;

-- 原始/近似对照（误差应很小）
WITH 570 AS m   -- 09:30
SELECT
  v.symbol,
  v.p95_spread_bps_30d  AS approx,
  r.p95_spread_bps_30d  AS exact
FROM
  (SELECT symbol, p95_spread_bps_30d FROM lake.market_cost_profile_minute_vw WHERE minute_of_day=m) v
JOIN
  (SELECT symbol, p95_spread_bps_30d FROM lake.market_cost_profile_minute_raw_vw WHERE minute_of_day=m) r
USING symbol
ORDER BY symbol
LIMIT 20;
```
