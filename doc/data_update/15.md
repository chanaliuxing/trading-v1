# 0) 上线前必做清单（交易员&工程双视角）

* 账户：准备 **IBKR TWS/Gateway**，确认 **host/port/clientId/account**；先连接 **Paper Trading** 账户。
* 交易品种：本版先支持 **US 股票**（`SMART/USD`）。期权/期货后续加。
* 交易时段：默认只在 **RTH**（美东 09:30–16:00）下单。
* 风控：逐步打开白名单（symbols、max_qty、max_notional、max_orders/min），**阈值>纸上更保守**。
* Kill-Switch：任何异常可**一键熔断**（Kafka Topic/配置开关）。
* 价差/成本：建议阈值用 **P95 spread**（我们在 P5 做过），先把 `threshold` 调高（如 0.58–0.62）。

```sql
-- 订单与 IBKR 的映射（回溯排查 & 对账）
CREATE TABLE IF NOT EXISTS trading.ibkr_order_map
(
  order_id      UUID,                     -- 我方订单ID
  ib_order_id   Int64,                    -- IBKR 临时 orderId
  ib_perm_id    Int64,                    -- IBKR 永久 permId（核心主键）
  client_id     Int32,
  con_id        Int32,
  symbol        String,
  ts            DateTime,
  ver           UInt64,
  d             Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (d, symbol, order_id);

-- （可选）订单状态历史（细粒度观察状态迁移）
CREATE TABLE IF NOT EXISTS trading.order_status_history
(
  order_id      UUID,
  status        LowCardinality(String),   -- Submitted/PreSubmitted/Filled/Cancelled/Rejected 等
  filled        Float64,
  remaining     Float64,
  avg_fill_price Float64,
  last_liquidity LowCardinality(String) DEFAULT '',
  ts            DateTime,
  ver           UInt64,
  d             Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (d, order_id, ts);
```

## 4.2 IBKR 实现（基于 `ib_insync`，异步，带风控/熔断/状态回写）

`services/trading/brokers/ibkr_broker.py`

```python
from __future__ import annotations
import asyncio, json, time, uuid
from dataclasses import dataclass
from datetime import datetime, timezone
from typing import Dict, Optional
from aiokafka import AIOKafkaConsumer, AIOKafkaProducer
from ib_insync import IB, Stock, MarketOrder, LimitOrder, util  # pip install ib_insync

from libs.config import config
from libs.logger import get_logger

log = get_logger(__name__)

@dataclass
class IbkrCfg:
    host: str
    port: int
    client_id: int
    account: str
    exchange: str = "SMART"
    currency: str = "USD"
    rth_only: bool = True
    what_if: bool = True
    max_orders_per_min: int = 30
    max_qty_per_order: int = 100
    max_notional_per_symbol: float = 50000.0
    allowed_symbols: Optional[list[str]] = None
    kill_switch_topic: str = "risk.kill.v1"

class IBKRBroker:
    def __init__(self, cfg: IbkrCfg):
        self.cfg = cfg
        self.ib = IB()
        self.kafka = config.kafka_config["bootstrap_servers"]
        self.consumer = AIOKafkaConsumer(
            "orders.intent.v1",
            bootstrap_servers=self.kafka,
            group_id="ibkr_broker.v1",
            auto_offset_reset="latest",
            enable_auto_commit=True,
            value_deserializer=lambda m: json.loads(m.decode("utf-8")),
        )
        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.kafka,
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8"),
            linger_ms=5, retries=3
        )
        self._kill = False
        self._rate_bucket = []   # timestamps of last sends per minute
        self._pos: Dict[str, int] = {}  # 可换为从 CH 读取

    async def start(self):
        await self.consumer.start()
        await self.producer.start()
        await self._connect_ib()
        asyncio.create_task(self._run())
        asyncio.create_task(self._watch_kill())
        log.info("IBKRBroker started", host=self.cfg.host, port=self.cfg.port)

    async def stop(self):
        try: await self.producer.stop()
        except Exception: pass
        try: await self.consumer.stop()
        except Exception: pass
        try: self.ib.disconnect()
        except Exception: pass

    async def _connect_ib(self):
        # ib_insync 提供 connectAsync（非阻塞）
        await self.ib.connectAsync(self.cfg.host, self.cfg.port, clientId=self.cfg.client_id, readonly=False)
        # 订阅错误/断线事件
        self.ib.errorEvent += lambda reqId, code, msg, advancedOrderRejectJson: log.error(
            "IBKR error", reqId=reqId, code=code, msg=msg)
        log.info("Connected to IBKR", isConnected=self.ib.isConnected())

    async def _watch_kill(self):
        consumer = AIOKafkaConsumer(
            self.cfg.kill_switch_topic,
            bootstrap_servers=self.kafka,
            group_id="ibkr_kill_switch.v1",
            auto_offset_reset="latest",
            enable_auto_commit=True,
            value_deserializer=lambda m: json.loads(m.decode("utf-8")),
        )
        await consumer.start()
        try:
            async for rec in consumer:
                v = rec.value
                if v.get("kill", False):
                    self._kill = True
                    log.warning("KILL SWITCH ACTIVATED!")
                else:
                    self._kill = False
                    log.warning("KILL SWITCH DEACTIVATED.")
        finally:
            await consumer.stop()

    def _risk_gate(self, od: dict) -> tuple[bool, str]:
        sym = od.get("symbol","").upper()
        qty = int(od.get("qty",0))
        side = int(od.get("side",0))
        if self._kill: return False, "kill_switch_on"
        if self.cfg.allowed_symbols and sym not in self.cfg.allowed_symbols:
            return False, "symbol_not_allowed"
        if qty <= 0 or abs(qty) > self.cfg.max_qty_per_order:
            return False, "qty_limit"
        # 节流：每分钟最多 N 单
        now = time.time()
        self._rate_bucket = [t for t in self._rate_bucket if now - t < 60]
        if len(self._rate_bucket) >= self.cfg.max_orders_per_min:
            return False, "rate_limit"
        # 名义金额上限（用 signals 附带价格或最近价，保守用 limit_price / 不得已用 1e6 停单）
        lim = od.get("limit_price")
        if lim:
            notional = abs(qty * float(lim))
            if notional > self.cfg.max_notional_per_symbol:
                return False, "notional_limit"
        return True, ""

    async def _run(self):
        try:
            async for rec in self.consumer:
                msg = rec.value
                d = msg.get("data", {})
                if d.get("account") and d.get("account") != self.cfg.account:
                    continue
                ok, reason = self._risk_gate(d)
                if not ok:
                    await self._emit_status(d, "Rejected", reason)
                    continue
                await self._place_order(d)
        except asyncio.CancelledError:
            pass

    async def _place_order(self, od: dict):
        sym = od["symbol"].upper()
        side = int(od["side"])   # +1 buy, -1 sell
        qty = int(od["qty"])
        # 合约/订单
        contract = Stock(sym, self.cfg.exchange, self.cfg.currency)
        order = MarketOrder("BUY" if side>0 else "SELL", qty)
        order.account = self.cfg.account
        order.outsideRth = not self.cfg.rth_only
        order.whatIf = bool(self.cfg.what_if)
        # 下单
        trade = self.ib.placeOrder(contract, order)
        self._rate_bucket.append(time.time())
        # 订单映射（回写）
        ts_us = int(time.time()*1_000_000)
        map_msg = {
            "data": {
                "order_id": od["order_id"],
                "ib_order_id": trade.order.orderId or -1,
                "ib_perm_id": trade.order.permId or -1,
                "client_id": self.cfg.client_id,
                "con_id": trade.contract.conId or 0,
                "symbol": sym,
                "ts": datetime.utcnow().replace(tzinfo=timezone.utc),
            },
            "metadata": {"ts_us": ts_us}, "source": "IBKR"
        }
        await self.producer.send("orders.ibkr.map.v1", key=od["order_id"], value=map_msg)
        # 状态/成交监听
        def onStatus(trade_):
            st = trade_.orderStatus
            payload = {
                "data": {
                    **{k: od.get(k) for k in ["order_id","ts","symbol","side","qty","order_type","limit_price","signal_prob","threshold","model_name","model_version","data_version","res","account","reason"]},
                    "status": st.status
                },
                "metadata": {"ts_us": int(time.time()*1_000_000)},
                "source": "IBKR"
            }
            # 直接重发到 orders.intent.v1（表用 ReplacingMergeTree 接受新状态覆盖）
            asyncio.create_task(self.producer.send("orders.intent.v1", key=od["symbol"], value=payload))
            # 历史表
            hist = {
                "data": {
                    "order_id": od["order_id"],
                    "status": st.status,
                    "filled": st.filled or 0.0,
                    "remaining": st.remaining or 0.0,
                    "avg_fill_price": st.avgFillPrice or 0.0,
                    "last_liquidity": getattr(st, "lastLiquidity", "") or "",
                    "ts": datetime.utcnow().replace(tzinfo=timezone.utc),
                },
                "metadata": {"ts_us": int(time.time()*1_000_000)},
                "source": "IBKR"
            }
            asyncio.create_task(self.producer.send("orders.status.history.v1", key=od["order_id"], value=hist))

        def onFill(trade_, fill):
            ex = fill.execution
            price = float(fill.price)
            qty_f = int(ex.shares) if hasattr(ex, "shares") else int(qty)  # 以回报为准
            fee = float(getattr(fill, "commission", 0.0) or 0.0)
            payload = {
                "data": {
                    "fill_id": str(uuid.uuid4()),
                    "order_id": od["order_id"],
                    "ts": datetime.utcnow().replace(tzinfo=timezone.utc),
                    "symbol": sym,
                    "side": side,
                    "qty": qty_f,
                    "price": price,
                    "fee": fee,
                    "venue": ex.exchange or "IB",
                    "liquidity": "T"
                },
                "metadata": {"ts_us": int(time.time()*1_000_000)},
                "source": "IBKR"
            }
            asyncio.create_task(self.producer.send("orders.fills.v1", key=sym, value=payload))

        trade.statusEvent += onStatus
        trade.fillEvent += onFill
        # 提前发一次 Submitted（冪等会被后续状态覆盖）
        await self._emit_status(od, "Submitted", "")

    async def _emit_status(self, od: dict, status: str, reason: str):
        payload = {
            "data": {
                **{k: od.get(k) for k in ["order_id","ts","symbol","side","qty","order_type","limit_price","signal_prob","threshold","model_name","model_version","data_version","res","account","reason"]},
                "status": status
            },
            "metadata": {"ts_us": int(time.time()*1_000_000)},
            "source": "IBKR"
        }
        await self.producer.send("orders.intent.v1", key=od["symbol"], value=payload)
```

# 6) 灰度步骤（稳妥上实盘）

1. **保持 `what_if=true`**：打开 `broker.mode=ibkr` + `ibkr.enabled=true`，此时不会真实成交；

   * 观察：`orders.status_history` 会出现 `PreSubmitted/Submitted` + whatIf 回执；无 `fills`。
2. **单标的白名单**：`allowed_symbols: ["TSLA"]`，设置 `threshold` 比纸上更高（如 0.60）。
3. **小单**：`max_qty_per_order=10`、`max_notional_per_symbol=5000`；
4. **RTH**：先只在 RTH 跑 30–60 分钟；
5. **打开真实成交**：改 `what_if=false`；
6. **扩大标的**：逐步增加 `allowed_symbols`；
7. **扩大限额**：逐步放宽 `max_qty_per_order / max_notional_per_symbol / max_orders_per_min`。

# 10) 交易员级别的保护条款（强烈建议保持开启）

* **阈值门槛**：`threshold ≥ 0.58` 起步，动态联动 `P95 spread`（可把门槛做成 `base+α*P95`）。
* **开盘/收盘窗口抑制**：09:30–09:35 与 15:55–16:00 先不下单或限价更保守。
* **极端行情熔断**：`imbalance` 或 `spread_bps` 超阈停止下单。
* **异常特征熔断**：当分钟特征缺失或延迟时不下单（P9 已实现仅对完整行发信号）。
