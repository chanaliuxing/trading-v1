## 1) 新版 `ch_writer.py`（直接替换）

> 路径保持不变：`services/market/writer/ch_writer.py`
> 兼容你现有封装：`libs.db.clickhouse.ClickHouseDB.insert_async(table, rows, column_names)`

```python
# services/market/writer/ch_writer.py
"""ClickHouse Bar Writer (parameterized)
- Consumes bar messages from Kafka (aiokafka)
- Writes to ClickHouse using ClickHouseDB.insert_async()
- Idempotent via ReplacingMergeTree(ver)
- Minimal assumptions about message schema (data/metadata keys)
"""
import asyncio
import json
from typing import List, Dict, Any, Optional, Tuple
from datetime import datetime, timezone

from aiokafka import AIOKafkaConsumer
from aiokafka.structs import ConsumerRecord

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

logger = get_logger(__name__)

def _us_to_dt(us: Optional[int]) -> Optional[datetime]:
    if us is None:
        return None
    try:
        return datetime.fromtimestamp(us / 1_000_000, tz=timezone.utc).replace(tzinfo=None)
    except Exception:
        return None

class ClickHouseBarWriter:
    """Generic Kafka→ClickHouse writer for bar messages."""

    # Column order compatible with existing bars_* tables in your ch_init.sql
    COLS: Tuple[str, ...] = (
        "symbol",
        # bar_sec is DEFAULTed inside CH tables, do not send
        "ts_start",
        "ts_end",
        "open",
        "high",
        "low",
        "close",
        "vwap",
        "volume",
        "trades",
        "bid_last",
        "ask_last",
        "mid_last",
        "ver",
        "is_backfill",
        "source",
    )

    def __init__(
        self,
        *,
        topic: str,
        table_name: str,
        group_id: str = "ch_writer_default",
        batch_size: int = 200,
        batch_timeout_sec: float = 2.0,
        consumer_kwargs: Optional[Dict[str, Any]] = None,
    ) -> None:
        self.topic = topic
        self.table_name = table_name
        self.group_id = group_id
        self.batch_size = int(batch_size)
        self.batch_timeout_sec = float(batch_timeout_sec)

        kafka_cfg = config.kafka_config
        kwargs = dict(
            bootstrap_servers=kafka_cfg["bootstrap_servers"],
            group_id=group_id,
            auto_offset_reset="earliest",
            enable_auto_commit=True,
            value_deserializer=lambda m: json.loads(m.decode("utf-8")),
        )
        if consumer_kwargs:
            kwargs.update(consumer_kwargs)

        self.consumer = AIOKafkaConsumer(self.topic, **kwargs)
        self.db = ClickHouseDB()
        self._running = False
        self._batch: List[List[Any]] = []
        self._dedup_keys: set = set()  # (symbol, ts_start)
        self._bg_tasks: List[asyncio.Task] = []

        logger.info("ClickHouseBarWriter initialized", topic=self.topic, table=self.table_name, group=self.group_id)

    # ---------------- Lifecycle ----------------

    async def start(self) -> None:
        await self.db.init_async_client()
        await self.consumer.start()
        self._running = True
        self._bg_tasks.append(asyncio.create_task(self._batch_flusher(), name=f"ch_writer_flush:{self.topic}"))
        self._bg_tasks.append(asyncio.create_task(self._consume_loop(), name=f"ch_writer_consume:{self.topic}"))
        logger.info("ClickHouseBarWriter started", topic=self.topic)

    async def stop(self) -> None:
        self._running = False
        try:
            if self._batch:
                await self._flush_batch()
        except Exception as exc:
            logger.warning("Flush on stop failed", error=str(exc))
        for t in self._bg_tasks:
            t.cancel()
        await asyncio.gather(*self._bg_tasks, return_exceptions=True)
        self._bg_tasks.clear()
        await self.consumer.stop()
        await self.db.close_async()
        logger.info("ClickHouseBarWriter stopped", topic=self.topic)

    # ---------------- Kafka consume & batch flush ----------------

    async def _consume_loop(self) -> None:
        try:
            async for msg in self.consumer:
                row = self._record_to_row(msg)
                if row is None:
                    continue
                key = (row[0], row[1])  # (symbol, ts_start)
                if key in self._dedup_keys:
                    continue
                self._dedup_keys.add(key)
                self._batch.append(row)
                if len(self._batch) >= self.batch_size:
                    await self._flush_batch()
        except asyncio.CancelledError:
            logger.info("Consume loop cancelled", topic=self.topic)
        except Exception as exc:
            logger.error("Consume loop error", topic=self.topic, error=str(exc))
            raise

    async def _batch_flusher(self) -> None:
        while self._running:
            try:
                await asyncio.sleep(self.batch_timeout_sec)
                if self._batch:
                    await self._flush_batch()
            except asyncio.CancelledError:
                break
            except Exception as exc:
                logger.error("Batch flusher error", topic=self.topic, error=str(exc))

    async def _flush_batch(self) -> None:
        batch = self._batch
        self._batch = []
        self._dedup_keys.clear()
        if not batch:
            return
        try:
            await self.db.insert_async(self.table_name, batch, list(self.COLS))
            logger.info("Inserted", table=self.table_name, rows=len(batch))
        except Exception as exc:
            logger.error("Insert batch failed", table=self.table_name, rows=len(batch), error=str(exc))

    # ---------------- Record normalization ----------------

    def _record_to_row(self, msg: ConsumerRecord) -> Optional[List[Any]]:
        try:
            payload = msg.value
            meta = payload.get("metadata", {}) or {}
            data = payload.get("data", {}) or {}

            symbol = data.get("symbol")
            ts_start = _us_to_dt(data.get("ts_start_us"))
            ts_end = _us_to_dt(data.get("ts_end_us"))
            open_ = data.get("open")
            high = data.get("high")
            low = data.get("low")
            close = data.get("close")
            vwap = data.get("vwap")
            volume = data.get("volume")
            trades = data.get("trades", 0)
            bid_last = data.get("bid_last")
            ask_last = data.get("ask_last")
            mid_last = data.get("mid_last")
            is_backfill = int(data.get("is_backfill", 0))
            source = payload.get("source") or data.get("source") or meta.get("producer") or "REALTIME"

            ts_meta_us = meta.get("ts_us") or meta.get("ts") or 0
            ver = int((ts_meta_us or 0) // 1000)  # us→ms

            if not symbol or ts_start is None or ts_end is None:
                logger.warning("Drop invalid message", reason="missing symbol/ts", topic=self.topic)
                return None

            return [
                symbol,
                ts_start,
                ts_end,
                float(open_) if open_ is not None else None,
                float(high) if high is not None else None,
                float(low) if low is not None else None,
                float(close) if close is not None else None,
                float(vwap) if vwap is not None else None,
                float(volume) if volume is not None else 0.0,
                int(trades) if trades is not None else 0,
                float(bid_last) if bid_last is not None else None,
                float(ask_last) if ask_last is not None else None,
                float(mid_last) if mid_last is not None else None,
                ver,
                is_backfill,
                str(source),
            ]
        except Exception as exc:
            logger.error("Failed to parse record", topic=self.topic, error=str(exc))
            return None
```

**说明**

* 不改动你的 DB 封装与 Kafka 依赖：继续用 `aiokafka` 与 `ClickHouseDB.insert_async`。
* `ver` 来自消息 `metadata.ts_us // 1000`，保证回补更新顺序优先级。
* `bar_sec/created_at` 走表默认值；列集合与你现有 `bars_5s` 等表完全兼容。

## 2) `run_system.py` 增量片段（启动多实例）

> 在系统启动处（你原来加载服务的地方）加入“按配置启动 Writer”的逻辑；在优雅退出时关闭。

**新增导入**

```python
from services.market.writer.ch_writer import ClickHouseBarWriter
from libs.config import config
```

**辅助：取 writers 配置（兼容不同 config 结构）**

```python
def _get_writers_cfg():
    # 兼容 config 字段或 dict 形式
    for attr in ("writers",):
        if hasattr(config, attr):
            val = getattr(config, attr)
            if val: return val
    try:
        raw = getattr(config, "raw", None) or getattr(config, "data", None)
        if raw and "writers" in raw: return raw["writers"]
    except Exception:
        pass
    return []
```

**在启动流程中加入：**

```python
self._ch_writers = []

writers_cfg = _get_writers_cfg()
if writers_cfg:
    logger.info("Starting ClickHouse Writers...")
    for w in writers_cfg:
        if not w.get("enabled", False):
            continue
        topic = w["topic"]
        table = w["table"]
        group = w.get("group_id", f"ch_writer_{table}")
        bs = int(w.get("batch_size", 200))
        bt = float(w.get("batch_timeout_sec", 2.0))
        writer = ClickHouseBarWriter(
            topic=topic, table_name=table, group_id=group,
            batch_size=bs, batch_timeout_sec=bt,
        )
        await writer.start()
        self._ch_writers.append(writer)
    logger.info("   ✅ Writers started", count=len(self._ch_writers))
else:
    logger.info("No writers configured; skipping ClickHouse writers")
```

**在优雅退出/stop() 中加入：**

```python
# stop writers
if getattr(self, "_ch_writers", None):
    for w in self._ch_writers:
        try:
            await w.stop()
        except Exception as e:
            logger.warning("Writer stop failed", error=str(e))
```

> 这样做**不影响**你现在的其它服务启动顺序；`writers` 全部由配置控制开/关，回退=把 `enabled: false` 改回即可。

## 5) 常见问题/边界处理

* **消息缺字段**：`symbol/ts_*` 缺失会自动丢弃并打日志（不会阻断）；
* **重复/乱序**：批内轻去重 + CH 端 `ReplacingMergeTree(ver)`；
* **回补覆盖**：回补消息里有 `is_backfill=1`；`ver` 取自 `metadata.ts_us`，回补到达晚于实时 → 自动覆盖旧值；
* **时间单位**：消息用微秒（us），Writer 转为 `DateTime`（UTC）入库；
* **安全回退**：全部由 `config.yml` 控制开关；不动已有 5s 链路时，现网零风险。
