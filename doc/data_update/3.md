好嘞，进入 **P3｜盘中自动小回补（Catchup Service）**。这一波我们做三件事：

1. 新增常驻服务 `catchup_service.py`：自动扫描 **今天+昨天** 的理论时间轴 vs 实际数据，找缺口 → 发送“**小回补请求**”；
2. 配置一个 **专属的 1m BackfillService 实例**：只监听 1m 的回补请求（不吃 5s 的 gap 事件），把 IBKR 拉回的 1m bar 发到 `md.bar.1m.ibkr`；
3. 把 **1m Writer 打开**（P1 已实现多实例）以消费 `md.bar.1m.ibkr` 并落库到 `trading.bars_1m_ibkr`。

整个流程仍然是统一链路：**Catchup → Kafka(request) → BackfillService(IBKR) → Kafka(bar) → ClickHouse**，与现网 5s 互不干扰。

# 1) 新增文件：`services/market/catchup_service.py`

> 每隔 `interval_sec`（默认 120s）扫描 `trading.bars_1m_current` 的缺口；
> 发现缺口后，按连续区间切片（不超过 `max_range_minutes`，默认 60 分钟），向 **1m 回补请求主题** 发消息：`md.backfill.request.1m.v1`；
> BackfillService 收到后向 IBKR 拉 `bar_size="1 min"`，再写回 `md.bar.1m.ibkr`。

```python
# services/market/catchup_service.py
"""
CatchupService
--------------
Periodic intraday/yesterday gap scanner that enqueues small backfill jobs to Kafka.

- Reads from ClickHouse `trading.bars_{res}_current` (views created in P2)
- For each symbol & day (today / yesterday), builds the theoretical RTH timeline and
  diffs against existing bars. Missing windows are coalesced into short ranges
  (<= max_range_minutes) and sent to BackfillService via request topic.
- BackfillService (1m) must be started with request_topic="md.backfill.request.1m.v1",
  bar_topic="md.bar.1m.ibkr", default_bar_size="1 min".
"""
from __future__ import annotations

import asyncio
import json
from dataclasses import dataclass
from datetime import datetime, timedelta, time, timezone
from typing import Iterable, List, Dict, Tuple, Optional

from aiokafka import AIOKafkaProducer
from zoneinfo import ZoneInfo

from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

logger = get_logger(__name__)

NY_TZ = ZoneInfo(getattr(config, "timezone", "America/New_York"))

def _parse_bar_size_to_seconds(bar_size: str) -> int:
    s = bar_size.strip().lower()
    if s in ("5 sec", "5 secs", "5 second", "5 seconds"):
        return 5
    if s in ("1 min", "1 mins", "1 minute", "1 minutes"):
        return 60
    if s in ("5 min", "5 mins", "5 minutes"):
        return 300
    if s in ("15 min", "15 mins", "15 minutes"):
        return 900
    if s in ("30 min", "30 mins", "30 minutes"):
        return 1800
    if s in ("1 hour", "1 hr", "60 min", "60 mins"):
        return 3600
    raise ValueError(f"Unsupported bar_size: {bar_size}")

def _res_token(bar_size: str) -> str:
    sec = _parse_bar_size_to_seconds(bar_size)
    if sec == 5: return "5s"
    if sec == 60: return "1m"
    if sec == 300: return "5m"
    if sec == 900: return "15m"
    if sec == 1800: return "30m"
    if sec == 3600: return "1h"
    return f"{sec}s"

def _rth_bounds(d: datetime) -> Tuple[datetime, datetime]:
    # RTH 09:30–16:00 America/New_York -> UTC naive
    local_day = d.astimezone(NY_TZ).date()
    start_local = datetime.combine(local_day, time(9, 30), tzinfo=NY_TZ)
    end_local   = datetime.combine(local_day, time(16, 0), tzinfo=NY_TZ)
    return (start_local.astimezone(timezone.utc).replace(tzinfo=None),
            end_local.astimezone(timezone.utc).replace(tzinfo=None))

def _daterange_days(days: Iterable[int]) -> List[datetime]:
    # 0=today, -1=yesterday
    now = datetime.now(timezone.utc).replace(tzinfo=None)
    today = now.date()
    return [datetime.combine(today + timedelta(days=o), time(0,0,0)) for o in days]

def _chunk_ranges(indices: List[datetime], step: timedelta, max_range: timedelta):
    if not indices: return []
    shards = []
    cur_start = indices[0]
    prev = indices[0]
    max_end = cur_start + max_range
    for ts in indices[1:]:
        if (ts - prev) == step and ts < max_end:
            prev = ts
            continue
        shards.append((cur_start, prev + step))
        cur_start, prev, max_end = ts, ts, ts + max_range
    shards.append((cur_start, prev + step))
    return shards

@dataclass
class CatchupConfig:
    enabled: bool = True
    interval_sec: int = 120          # 扫描频率
    bar_size: str = "1 min"          # 目前只开 1m；后续可开 5m/15m
    request_topic: str = "md.backfill.request.1m.v1"
    use_rth: bool = True
    lag_grace_sec: int = 30          # 离“现在”留出缓冲，避免刚生成的最后一根被误判缺口
    max_range_minutes: int = 60      # 小回补仅处理 <= 60 分钟的缺口
    symbols: List[str] = None        # 不配置则从 5s 当天动态发现

class CatchupService:
    def __init__(self, cfg: Optional[CatchupConfig] = None) -> None:
        self.cfg = cfg or CatchupConfig()
        self.db = ClickHouseDB()
        kafka_cfg = getattr(config, "kafka_config", {}) or {}
        self.bootstrap = kafka_cfg.get("bootstrap_servers")
        self.producer: Optional[AIOKafkaProducer] = None
        self._running = False

        self.bar_sec = _parse_bar_size_to_seconds(self.cfg.bar_size)
        self.res_token = _res_token(self.cfg.bar_size)
        self.view_name = f"trading.bars_{self.res_token}_current"

    async def start(self) -> None:
        logger.info("CatchupService starting...", request_topic=self.cfg.request_topic, bar_size=self.cfg.bar_size)
        if not self.bootstrap:
            raise RuntimeError("Kafka bootstrap_servers missing in config")
        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.bootstrap,
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8") if isinstance(v, str) else v,
            linger_ms=5,
            retries=3,
        )
        await self.producer.start()
        self._running = True
        asyncio.create_task(self._loop(), name=f"catchup_scan_{self.res_token}")

    async def stop(self) -> None:
        self._running = False
        if self.producer:
            await self.producer.stop()
        await self.db.close_async()
        logger.info("CatchupService stopped")

    async def _loop(self) -> None:
        await self.db.init_async_client()
        while self._running and self.cfg.enabled:
            try:
                symbols = self.cfg.symbols or self._discover_symbols()
                await self._scan(symbols)
            except Exception as exc:
                logger.error("Catchup loop error", error=str(exc))
            await asyncio.sleep(self.cfg.interval_sec)

    def _discover_symbols(self, limit: int = 200) -> List[str]:
        # 从 5s 表按当天动态发现 symbol（保守上限 200）
        sql = """
        SELECT DISTINCT symbol
        FROM trading.bars_5s
        WHERE d = today()
        LIMIT %(limit)s
        """
        try:
            rows = self.db.query(sql, {"limit": limit})
            return [r["symbol"] for r in rows]
        except Exception as exc:
            logger.warning("Discover symbols failed; fallback empty", error=str(exc))
            return []

    async def _scan(self, symbols: List[str]) -> None:
        days = _daterange_days([0, -1])
        step = timedelta(seconds=self.bar_sec)
        max_range = timedelta(minutes=self.cfg.max_range_minutes)
        lag_cutoff = datetime.now(timezone.utc).replace(tzinfo=None) - timedelta(seconds=self.cfg.lag_grace_sec)

        for day0 in days:
            rth_start, rth_end = _rth_bounds(day0)
            # 今日不要扫到“未来”，留出缓冲
            if day0.date() == datetime.utcnow().date():
                rth_end = min(rth_end, lag_cutoff)
            if rth_start >= rth_end:
                continue

            for symbol in symbols:
                missing_ts = await self._missing_windows(symbol, rth_start, rth_end, step)
                if not missing_ts:
                    continue
                shards = _chunk_ranges(missing_ts, step, max_range)
                for (start_dt, end_dt) in shards:
                    await self._enqueue(symbol, start_dt, end_dt)

    async def _missing_windows(
        self,
        symbol: str,
        rth_start: datetime,
        rth_end: datetime,
        step: timedelta,
    ) -> List[datetime]:
        # 读 current 视图，拿已存窗口
        sql = f"""
        SELECT ts_start
        FROM {self.view_name}
        WHERE symbol = %(symbol)s
          AND ts_start >= %(start)s
          AND ts_start <  %(end)s
        ORDER BY ts_start
        """
        rows = self.db.query(sql, {"symbol": symbol, "start": rth_start, "end": rth_end})
        have = {r["ts_start"] for r in rows}

        out = []
        cur = rth_start
        while cur < rth_end:
            if cur not in have:
                out.append(cur)
            cur += step
        return out

    async def _enqueue(self, symbol: str, start_dt: datetime, end_dt: datetime) -> None:
        if not self.producer:
            return
        payload = {
            "data": {
                "symbol": symbol,
                "from_ts": int(start_dt.replace(tzinfo=timezone.utc).timestamp()),
                "to_ts": int(end_dt.replace(tzinfo=timezone.utc).timestamp()),
                "use_rth": bool(self.cfg.use_rth),
                "bar_size": self.cfg.bar_size,
                "reason": "catchup",
            }
        }
        try:
            await self.producer.send(self.cfg.request_topic, key=symbol, value=payload)
            logger.info("Catchup enqueued",
                        symbol=symbol,
                        from_ts=payload["data"]["from_ts"],
                        to_ts=payload["data"]["to_ts"],
                        bar_size=self.cfg.bar_size,
                        topic=self.cfg.request_topic)
        except Exception as exc:
            logger.error("Catchup enqueue failed", symbol=symbol, error=str(exc))
```

# 3) 验收清单（可复制执行）

**A. 影子~实战通路**

1. 确认 1m Writer 已启用并可入库：

   ```sql
   SELECT count() FROM trading.bars_1m_ibkr WHERE d = today();
   ```
2. 启动 Backfill 1m + Catchup，观察日志有 `Catchup enqueued`/`Emit bar`。
3. CH 里逐步出现今日/昨日的 1m 回补行（`is_backfill=1, source='backfill'` 或 `'IBKR_RECON'` 视你的 backfill 标注）。

**B. 缺口扫描正确性**

```sql
-- 应该只针对 RTH 范围，且不会扫到“未来”
SELECT min(ts_start), max(ts_start) FROM trading.bars_1m_current WHERE d = today();

-- 对单个 symbol 抽查缺口是否被填平（2 分钟内）
SELECT ts_start
FROM trading.bars_1m_current
WHERE symbol='TSLA' AND d=today()
ORDER BY ts_start;
```

**C. 不串线**

* 确认 1m Backfill 的 `gap_topic` 是不存在的主题（避免吃 5s 的 gap）；
* 5s Backfill 的 `bar_topic` 仍是 `md.bar.5s.v2`；1m 的 `bar_topic` 是 `md.bar.1m.ibkr`。

# 4) 交易员视角：保护条款（默认内置）

* **开盘 3 分钟**内的数据最易抖动，`lag_grace_sec` 默认 30s，还可在 Catchup 里把首 2 根 1m 加白名单（如需，我再补 09:30/09:31 的“延时回补”策略）。
* **只补短缺口**（≤60 分钟），超长的交给 P4 的离线对账，避免触发 IBKR pacing。
* **useRTH=true**，避免把 AH 混进来导致窗口错位。
