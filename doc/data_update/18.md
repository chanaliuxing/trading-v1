  action    LowCardinality(String),     -- BLOCK / THROTTLE

  metric    Nullable(Float64),          -- 触发值（如 spread_p95）
  threshold Nullable(Float64),
  ver       UInt64,
  d         Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (d, symbol, ts);

### 3.2 开/收窗口判断（新增）

```python
def _in_open_close_window(self, ts_utc: datetime) -> Optional[tuple[str,str]]:
    # 返回 ("open"|"close", "block"|"throttle"|None)
    if not self.risk.get("open_close", {}).get("enabled", False):
        return None
    t_local = ts_utc.astimezone(NY).time()
    oc = self.risk["open_close"]
    def in_range(start, end):
        s_h, s_m = map(int, start.split(":")); e_h, e_m = map(int, end.split(":"))
        from datetime import time as T
        return T(s_h, s_m) <= t_local < T(e_h, e_m)
    if in_range(oc.get("open_start","09:30"), oc.get("open_end","09:35")):
        return ("open", oc.get("mode","throttle"))
    if in_range(oc.get("close_start","15:55"), oc.get("close_end","16:00")):
        return ("close", oc.get("mode","throttle"))
    return None
```

### 3.3 盘口异常检查（新增）

```python
def _micro_check(self, sym: str, tgt: datetime) -> Optional[dict]:
    cfg = self.risk.get("microstructure", {})
    if not cfg.get("enabled", False):
        return None
    tbl = cfg.get("source_table", "lake.bbo_1s_features")
    ql1 = cfg.get("quotes_l1_table", "lake.quotes_l1")
    look = int(cfg.get("lookback_sec", 30))
    start = tgt - timedelta(seconds=look)

    # 过去 look 秒的 P95 spread_bps 与最后一秒的 imbalance
    rows = self.db.query(f"""
        SELECT
          quantileExact(0.95)(spread_bps) AS p95_spread,
          anyLast(imbalance) AS last_imb
        FROM {tbl}
        WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s
    """, {"s": sym, "st": start, "ed": tgt}) or [{"p95_spread": None, "last_imb": None}]
    p95 = float(rows[0]["p95_spread"] or 0.0)
    last_imb = float(rows[0]["last_imb"] or 0.0)

    crossed = 0
    if cfg.get("crossed_locked_block", True):
        crossed = int((self.db.query(f"""
            SELECT count() AS c
            FROM {ql1}
            WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s AND ask < bid
            LIMIT 1
        """, {"s": sym, "st": start, "ed": tgt}) or [{"c":0}])[0]["c"])

    # 判定
    if cfg.get("use_p95", True) and p95 > float(cfg.get("spread_bps_max", 8.0)):
        return {"action": "BLOCK", "reason": "spread_p95_exceed", "metric": p95, "threshold": float(cfg["spread_bps_max"])}
    if abs(last_imb) > float(cfg.get("imbalance_abs_max", 0.60)):
        return {"action": "BLOCK", "reason": "imbalance_exceed", "metric": abs(last_imb), "threshold": float(cfg["imbalance_abs_max"])}
    if crossed > 0:
        return {"action": "BLOCK", "reason": "crossed_locked", "metric": 1.0, "threshold": 0.0}
    return None
```

### 3.4 风控统一入口（新增）

```python
def _risk_gate_executor(self, sym: str, tgt: datetime) -> dict:
    # 缓存按 (sym, minute) 粒度
    key = (sym, tgt)
    if key in self._rg_cache:
        return self._rg_cache[key]
    decision = {"allow": True, "action": None, "cool_sec": self.cfg.cool_sec, "max_pos_qty": self.cfg.max_pos_qty}

    # 先看开/收窗口
    oc = self._in_open_close_window(tgt)
    if oc:
        oc_name, mode = oc
        if mode == "block":
            decision.update({"allow": False, "action": "BLOCK", "reason": f"{oc_name}_window_block"})
            self._rg_cache[key] = decision; return decision
        elif mode == "throttle":
            th = self.risk["open_close"].get("throttle", {})
            decision.update({
                "action": "THROTTLE",
                "reason": f"{oc_name}_window_throttle",
                "cool_sec": float(th.get("cool_sec", self.cfg.cool_sec)),
                "max_pos_qty": int(th.get("max_pos_qty", self.cfg.max_pos_qty))
            })

    # 再看盘口异常
    mc = self._micro_check(sym, tgt)
    if mc:
        decision.update({"allow": False, **mc})

    self._rg_cache[key] = decision
    return decision

def _emit_guard_event(self, sym: str, stage: str, guard: str, action: str, reason: str, metric: float | None, threshold: float | None, tgt: datetime):
    try:
        ts_us = int(time.time()*1_000_000)
        self.db.insert("monitor.risk_guard_events", [{
            "ts": tgt.replace(tzinfo=None), "symbol": sym, "stage": stage, "guard": guard,
            "action": action, "reason": reason, "metric": metric, "threshold": threshold,
            "ver": int(ts_us//1000)
        }], ["ts","symbol","stage","guard","action","reason","metric","threshold","ver"])
    except Exception:
        pass
```

```python
def _in_open_close_window(self, ts_utc: datetime) -> Optional[str]:
    oc = self.risk.get("open_close", {})
    if not oc.get("enabled", False): return None
    t_local = ts_utc.astimezone(NY).time()
    s_open, e_open = oc.get("open_start","09:30"), oc.get("open_end","09:35")
    s_close, e_close = oc.get("close_start","15:55"), oc.get("close_end","16:00")
    def in_range(s,e):
        sh,sm = map(int,s.split(":")); eh,em = map(int,e.split(":"))
        from datetime import time as T
        return T(sh,sm) <= t_local < T(eh,em)
    if in_range(s_open,e_open): return "open"
    if in_range(s_close,e_close): return "close"
    return None

def _micro_block(self, sym: str, now_utc: datetime) -> Optional[str]:
    cfg = self.risk.get("microstructure", {})
    if not cfg.get("enabled", False): return None
    tbl = cfg.get("source_table", "lake.bbo_1s_features")
    ql1 = cfg.get("quotes_l1_table", "lake.quotes_l1")
    look = int(cfg.get("lookback_sec", 30))
    start = now_utc - timedelta(seconds=look)
    rows = self.db.query(f"""
        SELECT quantileExact(0.95)(spread_bps) AS p95_spread, anyLast(imbalance) AS last_imb
        FROM {tbl}
        WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s
    """, {"s": sym, "st": start, "ed": now_utc}) or [{"p95_spread": None, "last_imb": None}]
    p95 = float(rows[0]["p95_spread"] or 0.0)
    last_imb = float(rows[0]["last_imb"] or 0.0)

    if p95 > float(cfg.get("spread_bps_max", 8.0)):
        return f"spread_p95_exceed:{p95:.3f}"
    if abs(last_imb) > float(cfg.get("imbalance_abs_max", 0.60)):
        return f"imbalance_exceed:{abs(last_imb):.3f}"
    if cfg.get("crossed_locked_block", True):
        crossed = int((self.db.query(f"""
            SELECT count() AS c FROM {ql1}
            WHERE symbol=%(s)s AND ts >= %(st)s AND ts < %(ed)s AND ask < bid
            LIMIT 1
        """, {"s": sym, "st": start, "ed": now_utc}) or [{"c":0}])[0]["c"])
        if crossed > 0:
            return "crossed_locked"
    return None
```
