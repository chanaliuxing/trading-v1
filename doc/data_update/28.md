# 2) `services/market/catchup_service.py`

```python
from __future__ import annotations
import asyncio, json, time
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from typing import Iterable, List, Tuple, Dict
from aiokafka import AIOKafkaProducer
from zoneinfo import ZoneInfo

from libs.config import config
from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger

log = get_logger(__name__)
NY = ZoneInfo("America/New_York")

BACKFILL_TOPIC = "md.backfill.request.v1"   # 若你现网不同，改这里

@dataclass
class GapSlice:
    start: datetime
    end: datetime
    windows: int

def _rth_span(d_local) -> Tuple[datetime, datetime]:
    # 最简 RTH：09:30–16:00，若你有交易日历，替换这里
    s = datetime(d_local.year, d_local.month, d_local.day, 9, 30, tzinfo=NY)
    e = datetime(d_local.year, d_local.month, d_local.day, 16, 0, tzinfo=NY)
    return (s.astimezone(timezone.utc), e.astimezone(timezone.utc))

def _expected_windows(res: str, day_local) -> List[datetime]:
    s,e = _rth_span(day_local)
    step = {"1m":60, "5m":300}.get(res)
    assert step, f"unsupported res {res}"
    out = []
    t = s
    while t < e:
        out.append(t)
        t += timedelta(seconds=step)
    return out

def _compact_missing(ts_list: List[datetime], res: str, max_span_min: int) -> List[GapSlice]:
    # 输入：缺失窗口的 ts_start 列表 → 合并成连续段（<= max_span_min）
    if not ts_list: return []
    step = {"1m":60, "5m":300}[res]
    slices = []
    run_start = ts_list[0]; prev = ts_list[0]; cnt = 1
    for t in ts_list[1:]:
        if (t - prev).total_seconds() <= step + 1e-6:
            cnt += 1; prev = t; continue
        # 断裂
        dur_min = cnt * step // 60
        if dur_min <= max_span_min:
            slices.append(GapSlice(run_start, prev + timedelta(seconds=step), cnt))
        run_start, prev, cnt = t, t, 1
    dur_min = cnt * step // 60
    if dur_min <= max_span_min:
        slices.append(GapSlice(run_start, prev + timedelta(seconds=step), cnt))
    return slices

class CatchupService:
    def __init__(self):
        self.cfg = config.catchup
        self.db = ClickHouseDB()
        self.bootstrap = config.kafka_config["bootstrap_servers"]
        self.producer: AIOKafkaProducer | None = None

    async def start(self):
        if not self.cfg.get("enabled", True):
            log.info("catchup disabled"); return
        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.bootstrap,
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8"),
            linger_ms=5, retries=3, acks="all"
        )
        await self.producer.start()
        asyncio.create_task(self._loop())
        log.info("catchup started")

    async def stop(self):
        if self.producer:
            await self.producer.stop()
            self.producer = None

    def _symbols(self) -> List[str]:
        s_cfg = self.cfg.get("symbols", {})
        if s_cfg.get("list"): return [s.upper() for s in s_cfg["list"]]
        if s_cfg.get("query"):
            rows = self.db.query(s_cfg["query"]) or []
            return [r["symbol"] for r in rows]
        # 兜底：从 1m current 抓最近 symbol
        rows = self.db.query("SELECT DISTINCT symbol FROM trading.bars_1m_current WHERE d>=today()-3 LIMIT 1000") or []
        return [r["symbol"] for r in rows]

    def _existing_windows(self, sym: str, res: str, st: datetime, ed: datetime) -> set[datetime]:
        tbl = f"trading.bars_{res}_current"
        rows = self.db.query(f"""
            SELECT ts_start FROM {tbl}
            WHERE symbol=%(s)s AND ts_start >= %(st)s AND ts_start < %(ed)s
        """, {"s": sym, "st": st, "ed": ed}) or []
        return {r["ts_start"] for r in rows}

    def _emit_audit(self, sym: str, res: str, sli: GapSlice):
        ts_us = int(time.time()*1_000_000)
        self.db.insert("monitor.catchup_tasks", [{
            "ts": datetime.utcnow().replace(tzinfo=None),
            "symbol": sym, "res": res,
            "start_ts": sli.start.replace(tzinfo=None),
            "end_ts": sli.end.replace(tzinfo=None),
            "windows": int(sli.windows),
            "ver": int(ts_us//1000)
        }], ["ts","symbol","res","start_ts","end_ts","windows","ver"])

    async def _enqueue(self, sym: str, res: str, start: datetime, end: datetime):
        assert self.producer
        payload = {
            "data": {
                "symbol": sym,
                "res": res,
                "start": start.isoformat().replace("+00:00","Z"),
                "end": end.isoformat().replace("+00:00","Z"),
                "use_rth": True,
                "is_backfill": 1,
                "source": "CATCHUP"
            },
            "metadata": {"ts_us": int(time.time()*1_000_000)}
        }
        await self.producer.send_and_wait(BACKFILL_TOPIC, key=sym, value=payload)
        log.info("catchup.enqueued", symbol=sym, res=res, start=str(start), end=str(end))

    async def _once(self):
        syms = self._symbols()
        now = datetime.now(timezone.utc)
        today_local = now.astimezone(NY).date()
        yday_local = today_local - timedelta(days=1)
        res_list = list(self.cfg.get("resolutions", ["1m"]))
        max_small = int(self.cfg.get("short_gap_max_min", 60))

        for res in res_list:
            # 今天窗口
            st_today, _ = _rth_span(datetime(today_local.year,today_local.month,today_local.day,tzinfo=NY))
            st_today_look = max(st_today, now - timedelta(minutes=int(self.cfg.get("window_today_min", 120))))
            exp_today = [t for t in _expected_windows(res, datetime(today_local.year,today_local.month,today_local.day,tzinfo=NY)) if st_today_look <= t < now]
            existed = self._existing_windows_batch(syms, res, st_today_look, now)
            for sym in syms:
                miss = [t for t in exp_today if t not in existed.get(sym, set())]
                for sli in _compact_missing(miss, res, max_small):
                    await self._enqueue(sym, res, sli.start, sli.end)
                    self._emit_audit(sym, res, sli)

            # 昨天窗口（全 RTH）
            exp_yday = _expected_windows(res, datetime(yday_local.year,yday_local.month,yday_local.day,tzinfo=NY))
            st_y, ed_y = exp_yday[0], exp_yday[-1] + timedelta(seconds={"1m":60,"5m":300}[res])
            existed_y = self._existing_windows_batch(syms, res, st_y, ed_y)
            for sym in syms:
                miss = [t for t in exp_yday if t not in existed_y.get(sym, set())]
                for sli in _compact_missing(miss, res, max_small):
                    await self._enqueue(sym, res, sli.start, sli.end)
                    self._emit_audit(sym, res, sli)

    def _existing_windows_batch(self, syms: List[str], res: str, st: datetime, ed: datetime) -> Dict[str, set]:
        tbl = f"trading.bars_{res}_current"
        rows = self.db.query(f"""
            SELECT symbol, ts_start
            FROM {tbl}
            WHERE symbol IN %(syms)s AND ts_start >= %(st)s AND ts_start < %(ed)s
        """, {"syms": tuple(syms), "st": st, "ed": ed}) or []
        out: Dict[str,set] = {}
        for r in rows:
            out.setdefault(r["symbol"], set()).add(r["ts_start"])
        return out

    async def _loop(self):
        sec = int(self.cfg.get("poll_every_sec", 300))
        while True:
            try:
                await self._once()
            except Exception as e:
                log.warning("catchup.once.error", err=str(e))
            await asyncio.sleep(sec)

async def main():
    svc = CatchupService()
    await svc.start()
    # 常驻
    while True:
        await asyncio.sleep(3600)

if __name__ == "__main__":
    asyncio.run(main())
```

# 3) `services/reconciliation/offline_data_reconciler.py`

> 按我们的“减少重复 IBKR 逻辑”的决定：**大回补也发回 `md.backfill.request.v1`**，但把片段切到小时级，并打上 `source=IBKR_RECON`，优先级略低于盘中 gap。

```python
from __future__ import annotations
import asyncio, json, time, math
from datetime import datetime, timedelta, timezone, date
from typing import List, Tuple, Dict
from aiokafka import AIOKafkaProducer
from zoneinfo import ZoneInfo

from libs.config import config
from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger

log = get_logger(__name__)
NY = ZoneInfo("America/New_York")
TOPIC = "md.backfill.request.v1"

def rth_utc(d_local: date) -> Tuple[datetime, datetime]:
    s = datetime(d_local.year, d_local.month, d_local.day, 9, 30, tzinfo=NY).astimezone(timezone.utc)
    e = datetime(d_local.year, d_local.month, d_local.day, 16, 0, tzinfo=NY).astimezone(timezone.utc)
    return s, e

def expected_ts_starts(res: str, d_local: date) -> List[datetime]:
    s,e = rth_utc(d_local)
    step = {"1m":60,"5m":300}[res]
    t = s; out=[]
    while t < e:
        out.append(t); t += timedelta(seconds=step)
    return out

class OfflineReconciler:
    def __init__(self):
        self.cfg = config.offline_reconciler
        self.db = ClickHouseDB()
        self.bootstrap = config.kafka_config["bootstrap_servers"]
        self.sem = asyncio.Semaphore(int(self.cfg.get("max_concurrency",4)))
        self.producer: AIOKafkaProducer | None = None

    def _symbols(self) -> List[str]:
        q = self.cfg.get("symbols_query")
        rows = self.db.query(q) or []
        return [r["symbol"] for r in rows]

    def _have_windows(self, sym: str, res: str, st: datetime, ed: datetime) -> set[datetime]:
        tbl = f"trading.bars_{res}_current"
        rows = self.db.query(f"""
            SELECT ts_start FROM {tbl}
            WHERE symbol=%(s)s AND ts_start >= %(st)s AND ts_start < %(ed)s
        """, {"s": sym, "st": st, "ed": ed}) or []
        return {r["ts_start"] for r in rows}

    async def _emit(self, sym: str, res: str, start: datetime, end: datetime, trade_d: date, shard_min: int, windows: int):
        assert self.producer
        payload = {
            "data": {
                "symbol": sym, "res": res,
                "start": start.isoformat().replace("+00:00","Z"),
                "end": end.isoformat().replace("+00:00","Z"),
                "use_rth": True, "is_backfill": 1, "source": "IBKR_RECON"
            },
            "metadata": {"ts_us": int(time.time()*1_000_000)}
        }
        await self.producer.send_and_wait(TOPIC, key=sym, value=payload)
        # 审计
        ts_us = int(time.time()*1_000_000)
        self.db.insert("monitor.recon_audit", [{
            "ts": datetime.utcnow().replace(tzinfo=None),
            "trade_date": trade_d, "symbol": sym, "res": res,
            "start_ts": start.replace(tzinfo=None), "end_ts": end.replace(tzinfo=None),
            "shard_min": shard_min, "windows": windows,
            "status": "ENQUEUED", "err": "",
            "ver": int(ts_us//1000)
        }], ["ts","trade_date","symbol","res","start_ts","end_ts","shard_min","windows","status","err","ver"])
        log.info("recon.enqueued", symbol=sym, res=res, start=str(start), end=str(end))

    async def _recon_day_symbol(self, sym: str, res: str, d_local: date):
        st, ed = rth_utc(d_local)
        exp = expected_ts_starts(res, d_local)
        have = self._have_windows(sym, res, st, ed)
        missing = [t for t in exp if t not in have]
        if not missing: return
        # 合并成小时级片段
        step = {"1m":60,"5m":300}[res]
        shard = int(self.cfg.get("shard_minutes", 60))
        # 依次从 st 开始切片
        cur = missing[0]; run = [cur]
        for t in missing[1:]:
            if (t - run[-1]).total_seconds() <= step + 1e-6:
                run.append(t)
            else:
                await self._emit(sym, res, run[0], run[-1] + timedelta(seconds=step), d_local, shard, len(run))
                run = [t]
        await self._emit(sym, res, run[0], run[-1] + timedelta(seconds=step), d_local, shard, len(run))

    async def run(self):
        self.producer = AIOKafkaProducer(
            bootstrap_servers=self.bootstrap,
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8"),
            linger_ms=10, retries=5, acks="all"
        )
        await self.producer.start()
        try:
            syms = self._symbols()
            days = int(self.cfg.get("run_days", 5))
            res_list = list(self.cfg.get("resolutions", ["1m"]))
            today_local = datetime.now(NY).date()
            dates = [today_local - timedelta(days=i+1) for i in range(days)]  # 跳过今天，做历史日
            for d in dates:
                for sym in syms:
                    for res in res_list:
                        async with self.sem:
                            try:
                                await self._recon_day_symbol(sym, res, d)
                            except Exception as e:
                                log.warning("recon.error", symbol=sym, res=res, day=str(d), err=str(e))
        finally:
            await self.producer.stop()

async def main():
    svc = OfflineReconciler()
    await svc.run()

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```

```python
from __future__ import annotations
import asyncio, json, time
from typing import Dict, Any, List
from aiokafka import AIOKafkaConsumer, AIOKafkaProducer

from libs.config import config
from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger

log = get_logger(__name__)

class LakeWriter:
    def __init__(self, topic: str, table: str, key_field: str, columns: List[str]):
        self.topic = topic
        self.table = table
        self.key_field = key_field
        self.columns = columns
        self.bootstrap = config.kafka_config["bootstrap_servers"]
        self.db = ClickHouseDB()
        self.consumer: AIOKafkaConsumer | None = None
        self.dlq: AIOKafkaProducer | None = None

    async def start(self):
        self.consumer = AIOKafkaConsumer(
            self.topic,
            bootstrap_servers=self.bootstrap,
            group_id=f"lake_writer.{self.table}",
            enable_auto_commit=False,
            auto_offset_reset="latest",
            value_deserializer=lambda b: json.loads(b.decode("utf-8")),
            key_deserializer=lambda b: b.decode("utf-8") if b else None,
            max_poll_records=500
        )
        self.dlq = AIOKafkaProducer(
            bootstrap_servers=self.bootstrap,
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8"),
            linger_ms=5, retries=3
        )
        await self.consumer.start(); await self.dlq.start()
        log.info("lake_writer started", topic=self.topic, table=self.table)

        try:
            while True:
                batch = await self.consumer.getmany(timeout_ms=1000, max_records=1000)
                rows = []
                to_commit = []
                for tp, records in batch.items():
                    for rec in records:
                        try:
                            payload = rec.value
                            data = payload.get("data", payload)
                            # —— 字段映射（如不同请在此处调整）——
                            row = self._map_row(data)
                            rows.append(row)
                            to_commit.append(rec)
                        except Exception as e:
                            key = rec.key or ""
                            await self.dlq.send(f"{self.topic}.dlq", key=key, value={"error": str(e), "raw": rec.value})
                if rows:
                    self.db.insert(self.table, rows, self.columns)
                if to_commit:
                    await self.consumer.commit()
        finally:
            await self.consumer.stop(); await self.dlq.stop()

    def _map_row(self, d: Dict[str, Any]) -> Dict[str, Any]:
        # quotes_l1
        if self.table.endswith("quotes_l1"):
            return {
                "ts":     d.get("ts") or d.get("timestamp"),
                "symbol": d["symbol"],
                "bid": float(d.get("bid", d.get("bid_price", 0.0))),
                "bid_size": float(d.get("bid_size", d.get("bsize", 0.0))),
                "ask": float(d.get("ask", d.get("ask_price", 0.0))),
                "ask_size": float(d.get("ask_size", d.get("asize", 0.0))),
                "ver": int(d.get("ver", int(time.time()*1_000))),
                "source": d.get("source", "REALTIME")
            }
        # ticks_trades
        if self.table.endswith("ticks_trades"):
            return {
                "ts":     d.get("ts") or d.get("timestamp"),
                "symbol": d["symbol"],
                "price": float(d["price"]),
                "size": float(d.get("size", d.get("volume", 0.0))),
                "exchange": d.get("exchange",""),
                "ver": int(d.get("ver", int(time.time()*1_000))),
                "source": d.get("source", "REALTIME")
            }
        # 其它 lake 表可在此扩展
        raise ValueError(f"unknown table mapping: {self.table}")

async def main():
    tasks = []
    for w in config.lake_writers:
        if not w.get("enabled", True): continue
        srv = LakeWriter(
            topic=w["topic"],
            table=w["table"],
            key_field=w.get("key_field","symbol"),
            columns=w["columns"]
        )
        tasks.append(asyncio.create_task(srv.start(), name=f"lake_writer.{w['name']}"))
    if not tasks:
        return
    await asyncio.gather(*tasks)

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```

```python
from __future__ import annotations
import argparse, asyncio, json, time
from datetime import datetime, timezone
from aiokafka import AIOKafkaConsumer, AIOKafkaProducer

from libs.config import config
from libs.logger import get_logger

log = get_logger(__name__)

async def replay(src: str, dst: str, start_ts: float | None, end_ts: float | None, limit: int, rps: float):
    bs = config.kafka_config["bootstrap_servers"]
    cons = AIOKafkaConsumer(
        src, bootstrap_servers=bs, group_id=None,                 # 独立读取
        enable_auto_commit=False, auto_offset_reset="earliest",
        value_deserializer=lambda b: json.loads(b.decode("utf-8")),
        key_deserializer=lambda b: b.decode("utf-8") if b else None,
        consumer_timeout_ms=5000
    )
    prod = AIOKafkaProducer(
        bootstrap_servers=bs,
        value_serializer=lambda v: json.dumps(v).encode("utf-8"),
        key_serializer=lambda v: v.encode("utf-8"),
        linger_ms=5, retries=5
    )
    await cons.start(); await prod.start()
    try:
        sent = 0
        pace = 1.0/ max(rps, 0.1)
        while True:
            batch = await cons.getmany(timeout_ms=1000, max_records=1000)
            if not batch: break
            for tp, recs in batch.items():
                for rec in recs:
                    ts_ms = rec.timestamp
                    if start_ts and ts_ms/1000.0 < start_ts: continue
                    if end_ts and ts_ms/1000.0 > end_ts: continue
                    await prod.send_and_wait(dst, key=(rec.key or ""), value=rec.value)
                    sent += 1
                    if sent % int(rps) == 0:
                        await asyncio.sleep(pace)
                    if limit and sent >= limit: break
                if limit and sent >= limit: break
            if limit and sent >= limit: break
        log.info("replay.done", src=src, dst=dst, sent=sent)
        # 审计
        from libs.db.clickhouse import ClickHouseDB
        db = ClickHouseDB()
        ts_us = int(time.time()*1_000_000)
        db.insert("monitor.replay_audit", [{
            "ts": datetime.utcnow().replace(tzinfo=None),
            "src_topic": src, "dst_topic": dst,
            "key": "", "count": sent,
            "note": f"start={start_ts}, end={end_ts}, limit={limit}",
            "ver": int(ts_us//1000)
        }], ["ts","src_topic","dst_topic","key","count","note","ver"])
    finally:
        await cons.stop(); await prod.stop()

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--src", required=True, help="source DLQ topic")
    ap.add_argument("--dst", help="dest topic; 默认去掉 .dlq 后缀")
    ap.add_argument("--start", help="ISO time or epoch seconds", default=None)
    ap.add_argument("--end", help="ISO time or epoch seconds", default=None)
    ap.add_argument("--limit", type=int, default=0)
    ap.add_argument("--rps", type=float, default=200.0)
    args = ap.parse_args()

    dst = args.dst or (args.src[:-4] if args.src.endswith(".dlq") else args.src + ".replayed")
    def parse_ts(x):
        if not x: return None
        try:
            return float(x)
        except:
            return datetime.fromisoformat(x.replace("Z","+00:00")).timestamp()
    start_ts = parse_ts(args.start); end_ts = parse_ts(args.end)
    asyncio.get_event_loop().run_until_complete(replay(args.src, dst, start_ts, end_ts, args.limit, args.rps))

if __name__ == "__main__":
    main()
```

# 6) 最小化部署（systemd / 定时）

## systemd：常驻 `catchup_service` / `lake_writer`

`/etc/systemd/system/catchup.service`

```ini
[Unit]
Description=Catchup Service (small backfills)
After=network.target

[Service]
User=trader
WorkingDirectory=/opt/yourrepo
ExecStart=/usr/bin/python -m services.market.catchup_service
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
```

`/etc/systemd/system/lake_writer.service`

```ini
[Unit]
Description=Lake Writer (quotes & ticks)
After=network.target

[Service]
User=trader
WorkingDirectory=/opt/yourrepo
ExecStart=/usr/bin/python -m services.market.lake_writer
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
```

启用：

```bash
sudo systemctl daemon-reload
sudo systemctl enable --now catchup.service lake_writer.service
```

## systemd 定时：离线大回补（每日 02:00 ET）

`/etc/systemd/system/offline_recon.service`

```ini
[Unit]
Description=Offline Data Reconciler (daily)
After=network.target

[Service]
User=trader
WorkingDirectory=/opt/yourrepo
ExecStart=/usr/bin/python -m services.reconciliation.offline_data_reconciler
```

`/etc/systemd/system/offline_recon.timer`

```ini
[Unit]
Description=Run Offline Reconciler at 02:00 ET daily

[Timer]
OnCalendar=*-*-* 07:00:00 UTC     # ≈ 02:00 ET（夏令时略差一小时，按需换成 cron+TZ）
Persistent=true

[Install]
WantedBy=timers.target
```

启用：

```bash
sudo systemctl daemon-reload
sudo systemctl enable --now offline_recon.timer
```
