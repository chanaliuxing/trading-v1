# 目标

* 每分钟（或你配置的周期）读取 **最新 `features_{res}`**，用 **`ml.models_registry` 中 `stage='Production'`** 的模型做预测；
* 产出**概率/信号**，统一发 Kafka `signals.{res}.v1`；
* 用一个**通用 Writer** 落库到 `trading.signals_{res}`（`ReplacingMergeTree(ver)`）；
* 零侵入策略：需要时策略直接查库或订阅 `signals` 主题即可。

## 1) ClickHouse：signals 表（追加到 `ch_init.sql`）

```sql
CREATE TABLE IF NOT EXISTS trading.signals_1m
(
  symbol         String,
  ts             DateTime,                         -- 对齐 features_1m.ts（UTC）
  prob           Nullable(Float64),
  signal         UInt8,                            -- 0/1（或 -1/0/1 未来再扩）
  threshold      Float64,
  model_name     LowCardinality(String),
  model_version  String,
  data_version   String,
  res            LowCardinality(String) DEFAULT '1m',
  source         LowCardinality(String) DEFAULT 'ML_ONLINE',
  ver            UInt64,                           -- metadata.ts_us//1000
  d              Date DEFAULT toDate(ts)
)
ENGINE = ReplacingMergeTree(ver)
PARTITION BY d
ORDER BY (symbol, ts);
```

> 说明：信号是“**late-write 覆盖**”，迟到/回补仍以更大 `ver` 覆盖；你的查询可直接 `SELECT * FROM trading.signals_1m WHERE d=today()`。

## 2) 通用 ClickHouse Writer（Kafka→CH）

> 新增一个通用 Writer（和我们做 `lake_writer` 的风格一致），按列名把 `payload["data"]` 写入任意表。

**新文件**：`services/market/writer/generic_writer.py`

```python
from __future__ import annotations
import asyncio, json
from typing import Optional, Dict, Any, List
from aiokafka import AIOKafkaConsumer
from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

log = get_logger(__name__)

class GenericCHWriter:
    """
    通用Kafka→ClickHouse写入器：
      - 读取 JSON 消息：{ "data": {...}, "metadata": {"ts_us": ...}, "source": "..." }
      - 将 data 中的字段按 self.columns 顺序写入 table
      - 自动补 ver（来自 metadata.ts_us//1000）
    """
    def __init__(self, *, topic: str, table: str, columns: List[str], group_id: str,
                 batch_size: int = 500, batch_timeout_sec: float = 2.0):
        self.topic = topic
        self.table = table
        self.columns = columns
        self.group_id = group_id
        self.batch_size = batch_size
        self.batch_timeout_sec = batch_timeout_sec
        self.kafka = config.kafka_config["bootstrap_servers"]
        self.consumer = AIOKafkaConsumer(
            self.topic,
            bootstrap_servers=self.kafka,
            group_id=self.group_id,
            auto_offset_reset="earliest",
            enable_auto_commit=True,
            value_deserializer=lambda m: json.loads(m.decode("utf-8")),
        )
        self.db = ClickHouseDB()
        self._running = False
        self._batch: List[list] = []

    async def start(self):
        await self.db.init_async_client()
        await self.consumer.start()
        self._running = True
        asyncio.create_task(self._consume())
        asyncio.create_task(self._flush_timer())
        log.info("GenericCHWriter started", topic=self.topic, table=self.table)

    async def stop(self):
        self._running = False
        try:
            if self._batch: await self._flush()
        finally:
            await self.consumer.stop()
            await self.db.close_async()
            log.info("GenericCHWriter stopped", topic=self.topic)

    async def _consume(self):
        try:
            async for rec in self.consumer:
                row = self._to_row(rec.value)
                if row is not None:
                    self._batch.append(row)
                    if len(self._batch) >= self.batch_size:
                        await self._flush()
        except asyncio.CancelledError:
            pass
        except Exception as e:
            log.error("GenericCHWriter consume error", error=str(e))
            raise

    async def _flush_timer(self):
        while self._running:
            await asyncio.sleep(self.batch_timeout_sec)
            if self._batch:
                try: await self._flush()
                except Exception as e: log.error("GenericCHWriter flush error", error=str(e))

    def _to_row(self, payload: Dict[str, Any]):
        try:
            data = payload.get("data", {}) or {}
            meta = payload.get("metadata", {}) or {}
            ver = int((meta.get("ts_us") or 0)//1000)
            row = []
            for c in self.columns:
                if c == "ver":
                    row.append(ver)
                else:
                    row.append(data.get(c))
            return row
        except Exception as e:
            log.warning("drop malformed payload", error=str(e))
            return None

    async def _flush(self):
        batch = self._batch; self._batch = []
        await self.db.insert_async(self.table, batch, self.columns)
        log.info("Inserted", table=self.table, rows=len(batch))
```

## 3) 在线推理服务

**新文件**：`services/ml/online/inference_service.py`

```python
from __future__ import annotations
import asyncio, json, os, time
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from typing import List, Optional, Dict, Any
import joblib
import pandas as pd
from aiokafka import AIOKafkaProducer
from libs.db.clickhouse import ClickHouseDB
from libs.logger import get_logger
from libs.config import config

log = get_logger(__name__)

@dataclass
class InferenceConfig:
    enabled: bool = True
    res: str = "1m"
    symbols: List[str] = None                    # 为空则动态发现
    feature_table: str = "trading.features_1m"
    prod_query: str = """
        SELECT uri, model_name, version, data_version
        FROM ml.models_registry
        WHERE stage='Production' AND res='1m'
        ORDER BY created_at DESC LIMIT 1
    """
    threshold: float = 0.5
    loop_sec: float = 5.0                        # 轮询频率
    align_delay_sec: float = 2.0                 # 分钟对齐后延迟，等特征落表
    kafka_topic: str = "signals.1m.v1"

class InferenceService:
    def __init__(self, cfg: InferenceConfig):
        self.cfg = cfg
        self.db = ClickHouseDB()
        self.producer = AIOKafkaProducer(
            bootstrap_servers=config.kafka_config["bootstrap_servers"],
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
            key_serializer=lambda v: v.encode("utf-8"),
            linger_ms=5, retries=3
        )
        self.bundle = None
        self.model = None
        self.features: List[str] = []
        self.model_name = ""
        self.model_ver = ""
        self.data_version = ""
        self.last_ts_per_symbol: Dict[str, datetime] = {}

    async def start(self):
        await self.producer.start()
        asyncio.create_task(self._loop())
        log.info("InferenceService started", res=self.cfg.res, topic=self.cfg.kafka_topic)

    async def stop(self):
        try: await self.producer.stop()
        except Exception: pass
        log.info("InferenceService stopped")

    def _load_prod_model(self):
        row = (self.db.query(self.cfg.prod_query) or [None])[0]
        if not row:
            raise RuntimeError("No Production model in ml.models_registry")
        uri = row["uri"]
        path = uri.replace("file://","")
        self.bundle = joblib.load(path)
        self.model = self.bundle["model"]
        self.features = self.bundle.get("features") or []
        self.model_name = row["model_name"]; self.model_ver = row["version"]; self.data_version = row["data_version"]
        log.info("Loaded model", model=self.model_name, version=self.model_ver, features=len(self.features))

    def _discover_symbols(self, d: datetime) -> List[str]:
        sql = "SELECT DISTINCT symbol FROM trading.bars_5s WHERE d = %(d)s LIMIT 500"
        rows = self.db.query(sql, {"d": d.date()})
        return [r["symbol"] for r in (rows or [])]

    def _target_ts(self, now: datetime) -> datetime:
        # 对齐到上一个整分钟结束（UTC）
        t = now.replace(second=0, microsecond=0)
        return t

    def _fetch_feature_slice(self, symbols: List[str], ts: datetime) -> pd.DataFrame:
        sql = f"""
        SELECT *
        FROM {self.cfg.feature_table}
        WHERE symbol IN %(syms)s AND ts = %(ts)s
        ORDER BY symbol
        """
        rows = self.db.query(sql, {"syms": symbols, "ts": ts})
        df = pd.DataFrame(rows)
        if df.empty: return df
        df.set_index("symbol", inplace=True)
        return df

    async def _loop(self):
        self._load_prod_model()
        while True:
            try:
                now = datetime.utcnow().replace(tzinfo=timezone.utc)
                # 等到分钟对齐 + 延迟
                tgt = self._target_ts(now) - timedelta(seconds=self.cfg.align_delay_sec)
                if self.cfg.symbols:
                    symbols = self.cfg.symbols
                else:
                    symbols = self._discover_symbols(now)

                if symbols:
                    df = self._fetch_feature_slice(symbols, tgt)
                    if not df.empty:
                        # 仅对“新时刻”做推理（每 symbol 自己节流）
                        work = []
                        for sym in symbols:
                            last = self.last_ts_per_symbol.get(sym)
                            if last == tgt: continue
                            if sym in df.index:
                                work.append(sym)
                        if work:
                            X = df.loc[work, self.features].fillna(method="ffill").fillna(method="bfill")
                            # 分类：输出 proba；回归：直接输出 pred（此处按二分类处理）
                            proba = self.model.predict_proba(X.values)[:,1]
                            msgs = []
                            ts_us = int(time.time() * 1_000_000)
                            for sym, p in zip(work, proba):
                                sig = 1 if p >= self.cfg.threshold else 0
                                payload = {
                                    "data": {
                                        "symbol": sym,
                                        "ts": tgt,                           # CH DateTime（秒）
                                        "prob": float(p),
                                        "signal": int(sig),
                                        "threshold": float(self.cfg.threshold),
                                        "model_name": self.model_name,
                                        "model_version": self.model_ver,
                                        "data_version": self.data_version,
                                        "res": self.cfg.res
                                    },
                                    "metadata": {"ts_us": ts_us},
                                    "source": "ML_ONLINE"
                                }
                                msgs.append(payload)
                                self.last_ts_per_symbol[sym] = tgt
                            # 批量发送
                            for m in msgs:
                                await self.producer.send(self.cfg.kafka_topic, key=m["data"]["symbol"], value=m)
                            log.info("Emitted signals", ts=str(tgt), symbols=len(work))
                await asyncio.sleep(self.cfg.loop_sec)
            except Exception as e:
                log.error("Inference loop error", error=str(e))
                await asyncio.sleep(self.cfg.loop_sec)
```

## 4) 配置（`config.yml` 追加）

```yaml
# 在线推理服务
inference:
  enabled: false            # 先关，灰度再开
  res: "1m"
  symbols: []               # 留空=动态发现
  threshold: 0.55           # 初始略保守
  align_delay_sec: 2.0
  loop_sec: 5.0
  kafka_topic: "signals.1m.v1"

# 通用 Writer（Kafka→CH）：把 signals 写进 trading.signals_1m
generic_writers:
  - name: signals_1m_writer
    enabled: false
    topic: "signals.1m.v1"
    table: "trading.signals_1m"
    group_id: "signals_1m_writer.v1"
    columns: ["symbol","ts","prob","signal","threshold","model_name","model_version","data_version","res","source","ver"]  # 顺序要与表列对应；ver自动填充
    batch_size: 500
    batch_timeout_sec: 1.0
```

> 你也可以把 `columns` 简化为和表一一对应（除了 `d`，由表默认生成）。

## 5) `run_system.py` 集成（增量片段）

**导入并启动/停止：**

```python
# imports
from services.market.writer.generic_writer import GenericCHWriter
from services.ml.online.inference_service import InferenceService, InferenceConfig
from libs.config import config

# start()
self._generic_writers = []
for w in (getattr(config, "generic_writers", []) or []):
    if not w.get("enabled", False): continue
    writer = GenericCHWriter(
        topic=w["topic"],
        table=w["table"],
        columns=w["columns"],
        group_id=w.get("group_id", f"generic_writer_{w['name']}"),
        batch_size=int(w.get("batch_size", 500)),
        batch_timeout_sec=float(w.get("batch_timeout_sec", 2.0)),
    )
    await writer.start()
    self._generic_writers.append(writer)

inf_cfg_raw = getattr(config, "inference", {}) or {}
if inf_cfg_raw.get("enabled", False):
    self._inference = InferenceService(InferenceConfig(
        enabled=True,
        res=str(inf_cfg_raw.get("res","1m")),
        symbols=list(inf_cfg_raw.get("symbols") or []),
        threshold=float(inf_cfg_raw.get("threshold", 0.55)),
        align_delay_sec=float(inf_cfg_raw.get("align_delay_sec", 2.0)),
        loop_sec=float(inf_cfg_raw.get("loop_sec", 5.0)),
        kafka_topic=str(inf_cfg_raw.get("kafka_topic","signals.1m.v1")),
    ))
    await self._inference.start()

# stop()
for w in getattr(self, "_generic_writers", []):
    try: await w.stop()
    except Exception: pass
if getattr(self, "_inference", None):
    try: await self._inference.stop()
    except Exception: pass
```

## 6) 启用灰度与验收

**灰度步骤：**

1. 执行 DDL 创建 `trading.signals_1m`；
2. 在 `config.yml` 打开 `generic_writers.signals_1m_writer.enabled: true` 并重启（先只启 Writer，不启推理）；
3. 打开 `inference.enabled: true`，保留 `symbols: ["TSLA"]` 做单标的影子；
4. 验证通过再扩大标的集。

**验收 SQL：**

```sql
-- 是否在写入
SELECT * FROM trading.signals_1m WHERE d = today() ORDER BY ts DESC LIMIT 20;

-- 与 features 对齐（应一一对应同一 ts）
SELECT count() FROM trading.features_1m WHERE d=today();
SELECT count() FROM trading.signals_1m  WHERE d=today();

-- 最新时刻每标的一条
SELECT symbol, argMax(prob, ts) AS prob, argMax(signal, ts) AS sig, max(ts) AS last_ts
FROM trading.signals_1m
WHERE d = today()
GROUP BY symbol
ORDER BY last_ts DESC
LIMIT 20;
```

## 7) 交易员视角的保护条款

* **开盘初期抖动**：`align_delay_sec=2.0` 可适当加大到 3–5 秒；
* **门槛**：先用 `threshold=0.55` 保守一点；
* **缺特征不下信号**：代码只对 `ts` 有完整行才发；
* **模型热更新**：切换 `stage` 为 `Production` 后，重启 Inference（或加一个定时 reload，我可以再补）。

## 8) 回退

* 把 `inference.enabled: false` & `generic_writers.signals_1m_writer.enabled: false`，重启；
* 表/主题保留不影响现网；已写入的行由 `ReplacingMergeTree(ver)` 保障不会污染（至多多一列影子数据）。
