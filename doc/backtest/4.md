下面是**接入 Optuna 的 `alpha_loop.py` 完整版本**（直接可用，已把 Bayesian Optimization 嵌入进主循环）。
要点：

* 新增 `provider="optuna"`：使用 Optuna 直接在**连续/离散/类别**空间里采样参数，每个 trial 会跑一遍 Walk-Forward（固定参），返回目标分数（默认 **IS** 上的目标，亦可改为 **OOS**）。
* 支持从 `hpo.grid` **自动推断搜索空间**（数值→区间；类别→choices），或通过 `hpo.optuna_space` **显式指定**。
* 目标函数可用 **metric**（`sharpe|...|calmar`）或 **formula**（如 `"0.7*sharpe + 0.3*sortino - 2*max_drawdown"`），与你已有的 HPO 一致。
* CLI：`--provider optuna` + 在 `alpha_config_example.json` 里加 `"optuna": {"trials": 50, "target": "is", "direction": "maximize"}`（见代码注释）。

> 依赖：`pip install optuna`（无则抛出友好错误）。

# -*- coding: utf-8 -*-
"""
alpha_loop.py (Optuna-enabled)
------------------------------
End-to-end "Chain-of-Alpha" skeleton:
- LLM/ML propose grid (原逻辑仍保留)
- 或使用 Optuna 在连续/离散空间里进行贝叶斯优化（新增）
- 每次评估运行 Walk-Forward（固定一组参数），返回 metric/formula 分数

CLI:
  python alpha_loop.py --config /path/to/alpha_config.json \
    --gens 3 --out_dir ./alpha_out --k-top 8 --provider optuna

Config (alpha_config_example.json) 需新增/可选:
  "optuna": {
    "trials": 50,
    "target": "is",              # "is" or "oos" （建议 is 防止泄漏）
    "direction": "maximize",     # or "minimize"
    "sampler": "tpe",            # tpe | cmaes | nsgaii(多目标就别用了)...
    "widen": 0.10                # 自动推断空间时在原 grid 极值外扩比例
  }
  # 可显式定义搜索空间（优先级更高）：
  "hpo": {
    ...
    "optuna_space": {
      "A1_thr_sigma": {"type":"float","low":0.6,"high":1.4,"log":false},
      "B1_rvol_min":  {"type":"int","low":1,"high":3,"step":1},
      "SomeFlag":     {"type":"categorical","choices":[0,1]}
    }
  }
"""

from __future__ import annotations
import os, sys, json, argparse, random, math, ast
from typing import Any, Dict, List, Tuple
from copy import deepcopy
from pathlib import Path

from wf_hpo import WalkForwardHPO, make_folds  # 复用已有WF逻辑（含排序诺/欧米伽/卡玛尔等派生指标）

# ---------- Utils ----------

def _ensure_dir(p: str) -> None:
    Path(p).mkdir(parents=True, exist_ok=True)

def _load_json(path: str) -> Dict[str, Any]:
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def _dump_json(obj: Any, path: str) -> None:
    with open(path, "w", encoding="utf-8") as f:
        json.dump(obj, f, indent=2, ensure_ascii=False)

def _topk_rows(rows: List[Dict[str, Any]], k: int, key: str) -> List[Dict[str, Any]]:
    rows2 = [r for r in rows if r.get(key) is not None]
    rows2.sort(key=lambda r: r.get(key, 0.0), reverse=True)
    return rows2[:k]

# ---------- Safe formula eval (与 wf_hpo 同风格) ----------

_ALLOWED = {
    ast.Expression, ast.BinOp, ast.UnaryOp, ast.Num, ast.Load, ast.Name,
    ast.Add, ast.Sub, ast.Mult, ast.Div, ast.Pow, ast.USub, ast.UAdd,
    ast.Mod, ast.Constant, ast.FloorDiv
}
def _safe_eval_formula(expr: str, ctx: Dict[str, float]) -> float:
    tree = ast.parse(expr, mode='eval')
    for node in ast.walk(tree):
        if type(node) not in _ALLOWED:
            raise ValueError(f"Unsupported in formula: {type(node).__name__}")
        if isinstance(node, ast.Name) and node.id not in ctx:
            raise ValueError(f"Unknown name: {node.id}")
    return float(eval(compile(tree, "<formula>", "eval"), {"__builtins__": {}}, ctx))

def _llm_propose_grid(context: Dict[str, Any], model: str = "gpt-4o-mini") -> Tuple[Dict[str, List[Any]], str]:
    if not _llm_available():
        return ({}, "LLM unavailable; fallback.")
    try:
        from llm_client import ask_for_grid
        return ask_for_grid(context, model=model)
    except Exception as e:
        return ({}, f"LLM error: {e}")

def _ml_available() -> bool:

        return False

def _ml_suggest(rows: List[Dict[str, Any]], target_key: str, n: int = 16) -> Dict[str, List[Any]]:
    import numpy as np
    import pandas as pd
    try:
        from sklearn.ensemble import RandomForestRegressor
    except Exception:
        return {}
    if not rows:
        return {}
    df = pd.DataFrame(rows)
    try:
        pexp = df["params"].apply(lambda s: json.loads(s) if isinstance(s, str) else (s or {}))
    except Exception:
        return {}
    P = pd.json_normalize(pexp)
    y = df[target_key].astype(float)

    X = P.copy()
    for c in X.columns:
        if not np.issubdtype(X[c].dtype, np.number):
            X[c] = X[c].astype("category").cat.codes

    rf = RandomForestRegressor(n_estimators=200, random_state=42)
    rf.fit(X.fillna(0), y)

    top = _topk_rows(rows, k=min(20, len(rows)), key=target_key)
    rng = np.random.default_rng(42)
    cand = []
    for _ in range(n * 3):
        row = {}
        for c in X.columns:
            col = X[c]
            if np.issubdtype(col.dtype, np.number):
                mu, sd = float(col.mean()), float(col.std() or 1.0)
                row[c] = float(rng.normal(mu, sd))
            else:
                ch = list(col.dropna().unique())
                row[c] = rng.choice(ch) if ch else 0
        cand.append(row)
    C = pd.DataFrame(cand)
    preds = rf.predict(C.fillna(0))
    C["pred"] = preds
    C = C.sort_values("pred", ascending=False).head(n)

    grid: Dict[str, List[Any]] = {}
    for _, r in C.iterrows():
        for k, v in r.drop(labels=["pred"]).items():
            grid.setdefault(k, [])
            if v not in grid[k]:
                grid[k].append(v)
    for k in grid:
        grid[k] = grid[k][:5]
    return grid

# ---------- Optuna integration ----------

def _optuna_available() -> bool:
    try:
        import optuna  # noqa
        return True
    except Exception:
        return False

def _is_number(x) -> bool:
    try:
        float(x)
        return True
    except Exception:
        return False

def _infer_space_from_grid(grid: Dict[str, List[Any]], widen: float = 0.10) -> Dict[str, Dict[str, Any]]:
    """
    从现有 hpo.grid 推断搜索空间：
    - 数值型：取 min/max 并在两侧各外扩 widen 比例（避免把最优卡在边界）
    - 整型 vs 浮点：看原值是否全为整数
    - 类别型：使用 categorical choices
    返回 {param: {"type": "...", ...}}
    """
    space: Dict[str, Dict[str, Any]] = {}
    for k, vals in grid.items():
        if not isinstance(vals, list) or len(vals) == 0:
            continue
        if all(_is_number(v) for v in vals):
            nums = [float(v) for v in vals]
            lo, hi = min(nums), max(nums)
            span = hi - lo
            if span <= 0:
                span = max(abs(lo), 1.0)
            lo2 = lo - span * widen
            hi2 = hi + span * widen
            # 判断是否整数集合
            is_int = all(abs(v - int(v)) < 1e-12 for v in nums)
            if is_int:
                space[k] = {"type":"int", "low": int(math.floor(lo2)), "high": int(math.ceil(hi2)), "step": 1}
            else:
                space[k] = {"type":"float", "low": float(lo2), "high": float(hi2), "log": False}
        else:
            # categorical
            uniq = []
            for v in vals:
                if v not in uniq:
                    uniq.append(v)
            space[k] = {"type":"categorical", "choices": uniq[:16]}
    return space

def _suggest_from_space(trial, space: Dict[str, Dict[str, Any]]) -> Dict[str, Any]:
    params = {}
    for k, spec in space.items():
        t = spec.get("type")
        if t == "float":
            params[k] = trial.suggest_float(k, spec["low"], spec["high"], log=bool(spec.get("log", False)))
        elif t == "int":
            step = spec.get("step")
            if step and step > 1:
                # Optuna 没有 step 的 int，做个取整
                raw = trial.suggest_int(k, spec["low"], spec["high"])
                base = spec["low"]
                snapped = base + int(round((raw - base) / step)) * step
                params[k] = max(spec["low"], min(spec["high"], snapped))
            else:
                params[k] = trial.suggest_int(k, spec["low"], spec["high"])
        elif t == "categorical":
            params[k] = trial.suggest_categorical(k, spec["choices"])
        else:
            raise ValueError(f"Unknown space type for {k}: {t}")
    return params

def _score_from_rows_or_summary(rows: List[Dict[str, Any]], summary: Dict[str, Any], objective: Dict[str, Any], target: str) -> float:
    """
    target: "is" or "oos"
    - 若是 is：对 rows 中 phase==IS 的行按 objective 计算分数并取均值
    - 若是 oos：直接使用 summary（oos_summary）里相应指标
    支持 formula（安全求值）与 metric。
    """
    metric_key = (objective or {}).get("metric", "sharpe")
    formula = (objective or {}).get("formula")

    def _ctx_from_row(r: Dict[str, Any]) -> Dict[str, float]:
        return {
            "sharpe": float(r.get("sharpe", 0.0)),
            "total_return": float(r.get("total_return", 0.0)),
            "max_drawdown": float(r.get("max_drawdown", 0.0)),
            "win_rate": float(r.get("win_rate", 0.0)),
            "profit_factor": float(r.get("profit_factor", 0.0)),
            "trades": float(r.get("trades", 0.0)),
            "sortino": float(r.get("sortino", 0.0)),
            "omega": float(r.get("omega", 0.0)),
            "calmar": float(r.get("calmar", 0.0)),
        }

    if target == "is":
        is_rows = [r for r in rows if r.get("phase") == "IS"]
        if not is_rows:
            return 0.0
        vals = []
        for r in is_rows:
            if formula:
                vals.append(_safe_eval_formula(formula, _ctx_from_row(r)))
            else:
                vals.append(float(r.get(metric_key, 0.0)))
        return float(sum(vals)/len(vals))
    else:
        # OOS 汇总
        if formula:
            return _safe_eval_formula(formula, {
                "sharpe": float(summary.get("oos_avg_sharpe", 0.0)),
                "total_return": float(summary.get("oos_avg_total_return", 0.0)),
                "max_drawdown": float(summary.get("oos_avg_max_drawdown", 0.0)),
                "win_rate": float(summary.get("oos_avg_win_rate", 0.0)),
                "profit_factor": float(summary.get("oos_avg_profit_factor", 0.0)),
                "trades": float(summary.get("oos_trades", 0.0)),
                "sortino": float(summary.get("oos_avg_sortino", 0.0)),
                "omega": float(summary.get("oos_avg_omega", 0.0)),
                "calmar": float(summary.get("oos_avg_calmar", 0.0)),
            })
        # metric
        key_map = {
            "sharpe": "oos_avg_sharpe",
            "total_return": "oos_avg_total_return",
            "max_drawdown": "oos_avg_max_drawdown",
            "win_rate": "oos_avg_win_rate",
            "profit_factor": "oos_avg_profit_factor",
            "sortino": "oos_avg_sortino",
            "omega": "oos_avg_omega",
            "calmar": "oos_avg_calmar"
        }
        return float(summary.get(key_map.get(metric_key, "oos_avg_sharpe"), 0.0))

def _run_optuna(cfg: Dict[str, Any], base_hpo_cfg: Dict[str, Any]) -> Tuple[Dict[str, Any], Dict[str, Any]]:
    """
    返回 (best_params, study_summary)
    - 每个 trial：用建议参数构建单点 grid，跑完整的 WalkForwardHPO（IS+OOS），
      返回 IS/OOS 的目标分数（由 cfg["optuna"]["target"] 决定，建议 "is"）
    """
    if not _optuna_available():
        raise RuntimeError("Optuna not installed. Please `pip install optuna`.")

    import optuna

    opt_conf = cfg.get("optuna", {}) or {}
    trials = int(opt_conf.get("trials", 50))
    target = opt_conf.get("target", "is")  # "is" or "oos"
    direction = opt_conf.get("direction", "maximize")
    sampler_name = opt_conf.get("sampler", "tpe").lower()
    widen = float(opt_conf.get("widen", 0.10))

    # 1) 构建搜索空间（优先 optuna_space，否则从 grid 推断）
    hpo_grid = deepcopy(base_hpo_cfg.get("grid", {}))
    overrides = (base_hpo_cfg.get("optuna_space") or {})
    if overrides:
        space = overrides
    else:
        space = _infer_space_from_grid(hpo_grid, widen=widen)

    # 2) 准备基础 hpo_cfg 字典（保留 folds、objective、broker、数据等）
    base_cfg = deepcopy(base_hpo_cfg)

    # 3) 选择采样器
    if sampler_name == "cmaes":
        sampler = optuna.samplers.CmaEsSampler()
    elif sampler_name == "tpe":
        sampler = optuna.samplers.TPESampler(seed=42)
    else:
        sampler = optuna.samplers.TPESampler(seed=42)

    study = optuna.create_study(direction=direction, sampler=sampler)

    def objective(trial: "optuna.trial.Trial") -> float:
        # 3.1 采样参数
        params = _suggest_from_space(trial, space)  # dict: param -> value

        # 3.2 用“单点 grid”跑一遍 WalkForward（IS+OOS）
        cfg1 = deepcopy(base_cfg)
        cfg1["grid"] = {k: [v] for k, v in params.items()}  # 单点
        runner = WalkForwardHPO(cfg1)
        result = runner.run()

        # 3.3 计算 trial 分数（IS 或 OOS）
        score = _score_from_rows_or_summary(
            rows=result.get("rows", []),
            summary=result.get("oos_summary", {}),
            objective=cfg1.get("objective", {}),
            target=target
        )

        # 记录 Trial 附加信息（便于回溯）
        trial.set_user_attr("params", params)
        trial.set_user_attr("oos_summary", result.get("oos_summary", {}))
        return float(score)

    study.optimize(objective, n_trials=trials, show_progress_bar=False)

    best = study.best_trial
    best_params = best.user_attrs.get("params", {})
    study_summary = {
        "best_value": float(best.value),
        "best_params": best_params,
        "direction": direction,
        "target": target,
        "trials": trials,
        "oos_summary_of_best": best.user_attrs.get("oos_summary", {})
    }
    return best_params, study_summary

# ---------- 主循环（加入 provider=optuna） ----------

def run_loop(cfg: Dict[str, Any], out_dir: str, gens: int, k_top: int, provider: str) -> None:
    _ensure_dir(out_dir)
    base_hpo_cfg = cfg["hpo"]
    objective_key = base_hpo_cfg.get("objective", {}).get("metric", "sharpe")

    history: List[Dict[str, Any]] = []

    for g in range(1, gens+1):
        gen_dir = os.path.join(out_dir, f"gen_{g:02d}")
        _ensure_dir(gen_dir)

        # ---- A) 如选择 Optuna，则直接做一次贝叶斯优化，拿到 best_params ----
        if provider == "optuna":
            best_params, study_summary = _run_optuna(cfg, base_hpo_cfg)
            # 用最佳参数再跑一次（生成产物与图表）
            hpo_cfg = deepcopy(base_hpo_cfg)
            hpo_cfg["grid"] = {k: [v] for k, v in best_params.items()}  # 单点 grid
            runner = WalkForwardHPO(hpo_cfg)
            result = runner.run()
            artifacts = runner.export_tables(result, gen_dir)
            try:
                plots = runner.export_plots(result, gen_dir)
                artifacts.update(plots)
            except Exception as e:
                artifacts["plot_error"] = str(e)

            payload = {
                "gen": g,
                "mode": "optuna",
                "best_params": best_params,
                "study_summary": study_summary,
                "oos_summary": result.get("oos_summary", {}),
                "rows": result.get("rows", []),
                "artifacts": artifacts
            }
            _dump_json(payload, os.path.join(gen_dir, "gen_summary.json"))
            history.append(payload)
            # 下一代可选择在 best 附近收缩/扩张，再次做 Optuna（此处我们保持简单：每代独立搜索）
            continue

        # ---- B) 非 Optuna：保留原 LLM/ML/Heuristic 网格迭代 ----
        hpo_cfg = deepcopy(base_hpo_cfg)
        if history:
            prev = history[-1]
            is_rows = [r for r in prev.get("rows", []) if r.get("phase") == "IS"]
            top_is = _topk_rows(is_rows, k=min(k_top, len(is_rows)), key=objective_key)
            context = {
                "symbols": hpo_cfg["symbols"],
                "objective": hpo_cfg.get("objective", {}),
                "top_is": top_is,
                "base_grid": hpo_cfg["grid"],
                "notes": "Propose a JSON grid. Keep <=5 values each; numeric refine around winners; prune weak params."
            }
            new_grid = {}
            rationale = ""

            if provider in ("llm","hybrid"):
                grid_llm, rationale = _llm_propose_grid(context, model=cfg.get("llm", {}).get("model", "gpt-4o-mini"))
                if grid_llm:
                    new_grid = grid_llm
            if not new_grid and provider in ("ml","hybrid"):
                new_grid = _ml_suggest(is_rows, target_key=objective_key, n=cfg.get("ml", {}).get("n_points", 16))
            if not new_grid:
                # fallback：围绕 top K 的中位数扩展
                import statistics as st
                base = hpo_cfg["grid"]
                # 仅做很简洁的启发式：保留原 grid 即可，或你可以按之前实现做中位数策略
                new_grid = base
                rationale = rationale or "Fallback: keep base grid."

            merged = dict(hpo_cfg["grid"])
            merged.update(new_grid)
            hpo_cfg["grid"] = merged
            _dump_json({"rationale": rationale, "new_grid": new_grid}, os.path.join(gen_dir, "refine.json"))

        runner = WalkForwardHPO(hpo_cfg)
        result = runner.run()
        artifacts = runner.export_tables(result, gen_dir)
        try:
            plots = runner.export_plots(result, gen_dir)
            artifacts.update(plots)
        except Exception as e:
            artifacts["plot_error"] = str(e)

        payload = {
            "gen": g,
            "mode": provider,
            "hpo_cfg": hpo_cfg,
            "oos_summary": result.get("oos_summary", {}),
            "rows": result.get("rows", []),
            "artifacts": artifacts
        }
        _dump_json(payload, os.path.join(gen_dir, "gen_summary.json"))
        history.append(payload)

    # 索引
    idx = {"generations": [{"gen": p["gen"], "mode": p["mode"], "oos_summary": p.get("oos_summary", {}), "artifacts": p.get("artifacts", {})} for p in history]}
    _dump_json(idx, os.path.join(out_dir, "index.json"))

# ---------- CLI ----------

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--config", required=True, help="alpha loop config json")
    ap.add_argument("--out_dir", required=False, default="./alpha_out", help="output directory")
    ap.add_argument("--gens", type=int, default=3, help="number of generations")
    ap.add_argument("--k-top", type=int, default=8, help="top-K IS rows to guide refinement (non-optuna)")
    ap.add_argument("--provider", choices=["llm","ml","hybrid","optuna","none"], default="optuna", help="who proposes next params")
    args = ap.parse_args()

    cfg = _load_json(args.config)
    run_loop(cfg, args.out_dir, gens=args.gens, k_top=args.k_top, provider=args.provider)

if __name__ == "__main__":
    main()

**说明与建议**

* `target: "is"` 常用于**训练期**选择，避免把 OOS 当成目标（会泄漏）。
* 如果你确实想以**验证稳定性**为目标，可把 `target` 改为 `"oos"`（会显著更慢，因为每次都要跑完 OOS ）。
* 搜索空间：默认从 `hpo.grid` 推断；如果你对上下界/对数采样有更精确把握，请在 `hpo.optuna_space` 显式指定。
* 你也可以把 Optuna 作为“预搜索器”：先用 Optuna 找到 `best_params`，然后把它周围做微调网格给 `provider=hybrid` 的 LLM/ML 继续细化。
