我优先把**多标的严格估值（MTM）+ 真实交易账本（trade ledger）**做了——这是最关键的一环。原因很简单：不论你用 LLM 产出多少 Alpha、跑多少 HPO，如果盈亏与胜率统计建立在**单标的近似估值**或**配对撮合的粗略统计**上，结论会偏；一旦跨标的或组合级研究，误差会被放大。把这块打实，后面的并行扩展、目标函数扩展、可视化才有坚实地基。

已完成内容（可直接替换你当前版本）：

* **sim_adapter.py**

  * ✔️ **严格 MTM**：为每个 symbol 维护 `last_price`，组合净值 `equity = cash + Σ(qty_i * last_price_i)`，不再用“最后一个标的价格”近似。
  * ✔️ **交易账本**：对每次**减仓/平仓/反手**，记录一条 closed trade（时间、方向、数量、入场价、离场价、佣金、P&L）。
  * ✔️ 原有成交规则保持不变（MARKET / MKT_LIMITABLE / LIMIT / STOP，含跳空与盘中触发），佣金/滑点沿用。

  * ✔️ **每时间戳只记一次净值**：对同一 `ts` 的多个 symbol，处理完该时间戳的**所有**行后再写入 equity（避免多标的时序不一致）。

# -*- coding: utf-8 -*-
"""
Simulated Broker Adapter (vectorized-friendly)
----------------------------------------------
- Multi-symbol strict mark-to-market (per-symbol last price cache)
- Order types: MARKET / MKT_LIMITABLE / LIMIT / STOP
- Intrabar trigger logic with gap handling
- Slippage (bps), commission (per share + min)
- Positions, cash, realized PnL
- Trade ledger for closed trades (for accurate win-rate / profit factor)
"""

from __future__ import annotations
import math, uuid
from dataclasses import dataclass, field, asdict
from typing import Any, Dict, List, Optional, Callable
from datetime import datetime, timezone

# -------- Data structures --------

@dataclass
class Order:
    order_id: str
    decision_id: str
    ts: int                     # event time (ms)
    symbol: str
    side: str                   # "BUY" or "SELL"
    qty: float
    type: str                   # "MARKET" | "LIMIT" | "STOP" | "MKT_LIMITABLE"
    limit_price: Optional[float] = None
    stop_price: Optional[float] = None
    tif: str = "DAY"
    tag: Optional[str] = None
    status: str = "NEW"         # NEW, WORKING, FILLED, CANCELLED, REJECTED
    filled_qty: float = 0.0
    avg_fill_price: float = 0.0

@dataclass
class Fill:
    fill_id: str
    order_id: str
    ts: int
    symbol: str
    side: str
    qty: float
    price: float
    commission: float = 0.0
    slippage_cost: float = 0.0

@dataclass
class Position:
    symbol: str
    qty: float = 0.0
    avg_price: float = 0.0
    realized_pnl: float = 0.0

@dataclass
class Portfolio:
    cash: float = 100_000.0
    positions: Dict[str, Position] = field(default_factory=dict)
    realized_pnl: float = 0.0
    equity: float = 100_000.0
    last_mark_ts: Optional[int] = None

def _now_ms() -> int:
    return int(datetime.now(timezone.utc).timestamp() * 1000)

def _uid(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:10]}"

class SimAdapter:
    """
    Deterministic simulator with multi-symbol MTM and trade ledger.
    Price book is provided by bar features: keys 'open','high','low','close'.
    """

    def __init__(self,
                 initial_cash: float = 100_000.0,
                 commission_per_share: float = 0.005,
                 commission_min: float = 1.0,
                 slippage_bps: float = 1.0,
                 fill_ref: str = "open"  # 'open' or 'close'
                 ) -> None:
        self.portfolio = Portfolio(cash=initial_cash, equity=initial_cash)
        self.commission_per_share = commission_per_share
        self.commission_min = commission_min
        self.slippage_bps = slippage_bps
        self.fill_ref = fill_ref
        self._orders: Dict[str, Order] = {}
        self._working: Dict[str, Order] = {}
        self._on_fill_callbacks: List[Callable[[Fill], None]] = []
        self._on_status_callbacks: List[Callable[[Order], None]] = []

        # Per-symbol last price cache for strict MTM
        self._last_prices: Dict[str, float] = {}

        # Trade ledger for closed fragments
        # Each entry: {ts, symbol, side, qty, entry_price, exit_price, pnl, commission}
        self._closed_trades: List[Dict[str, float]] = []

    # --------- Callback wiring ---------

    def on_fill(self, fn: Callable[[Fill], None]) -> None:
        self._on_fill_callbacks.append(fn)

    def on_status(self, fn: Callable[[Order], None]) -> None:
        self._on_status_callbacks.append(fn)

    def _emit_status(self, order: Order) -> None:
        for fn in self._on_status_callbacks:
            try:
                fn(order)
            except Exception:
                pass

    def _emit_fill(self, fill: Fill) -> None:
        for fn in self._on_fill_callbacks:
            try:
                fn(fill)
            except Exception:
                pass

    # --------- Public API ---------

    def submit(self, order: Order) -> str:
        if order.qty <= 0:
            order.status = "REJECTED"
            self._emit_status(order)
            return order.order_id
        order.status = "WORKING"
        self._orders[order.order_id] = order
        self._working[order.order_id] = order
        self._emit_status(order)
        return order.order_id

    def cancel(self, order_id: str) -> None:
        o = self._orders.get(order_id)
        if not o or o.status in ("CANCELLED","FILLED"):
            return
        o.status = "CANCELLED"
        self._working.pop(order_id, None)
        self._emit_status(o)

    def cancel_all(self, symbol: Optional[str] = None) -> None:
        for oid, o in list(self._working.items()):
            if symbol and o.symbol != symbol:
                continue
            self.cancel(oid)

    # --------- Core mechanics ---------

    def on_bar(self, ts_ms: int, symbol: str, bar: Dict[str, float]) -> None:
        """
        Called once per bar per symbol. bar must include open/high/low/close.
        1) Try to execute orders
        2) Update last price for symbol
        3) Mark-to-market equity across ALL symbols
        """
        # 1) Try to execute market & triggered orders
        for oid, o in list(self._working.items()):
            if o.symbol != symbol:
                continue
            filled = False
            fill_price = None

            oref = bar.get(self.fill_ref, bar.get("open"))
            high = bar.get("high"); low = bar.get("low"); open_ = bar.get("open")

            if o.type == "MARKET":
                base = oref
                slip = base * (self.slippage_bps / 10_000.0)
                fill_price = base + (slip if o.side == "BUY" else -slip)
                filled = True

            elif o.type == "MKT_LIMITABLE":
                base = oref
                slip = base * (self.slippage_bps / 10_000.0)
                px = base + (slip if o.side == "BUY" else -slip)
                if o.limit_price is not None:
                    if o.side == "BUY":
                        px = min(px, o.limit_price)
                    else:
                        px = max(px, o.limit_price)
                fill_price = px
                filled = True

            elif o.type == "LIMIT":
                if o.side == "BUY":
                    if low is not None and o.limit_price is not None and low <= o.limit_price:
                        exec_price = min(open_, o.limit_price)
                        fill_price = exec_price
                        filled = True
                else:  # SELL
                    if high is not None and o.limit_price is not None and high >= o.limit_price:
                        exec_price = max(open_, o.limit_price)
                        fill_price = exec_price
                        filled = True

            elif o.type == "STOP":
                if o.side == "BUY":
                    if high is not None and o.stop_price is not None and high >= o.stop_price:
                        exec_price = max(open_, o.stop_price)
                        fill_price = exec_price
                        filled = True
                else:  # SELL
                    if low is not None and o.stop_price is not None and low <= o.stop_price:
                        exec_price = min(open_, o.stop_price)
                        fill_price = exec_price
                        filled = True

            if filled and fill_price is not None:
                self._fill_order(o, ts_ms, fill_price)

        # 2) Update last price for strict MTM
        mark_price = bar.get("close", bar.get("open"))
        if mark_price is not None:
            self._last_prices[symbol] = float(mark_price)

        # 3) Mark-to-market equity across ALL symbols (strict)
        self._mark_equity(ts_ms)

    # --------- Internals ---------

    def _commission(self, qty: float) -> float:
        fee = max(self.commission_min, abs(qty) * self.commission_per_share)
        return round(fee, 4)

    def _record_closed_fragment(self, ts_ms: int, symbol: str, closing_qty: float, entry_price: float, exit_price: float, side_long: bool, commission: float) -> float:
        """
        Record a closed fragment for ledger and return realized pnl of this fragment.
        """
        # side_long indicates the direction of the position being closed
        pnl = (exit_price - entry_price) * closing_qty if side_long else (entry_price - exit_price) * closing_qty
        self._closed_trades.append({
            "ts": ts_ms,
            "symbol": symbol,
            "qty": float(closing_qty),
            "entry_price": float(entry_price),
            "exit_price": float(exit_price),
            "side": "LONG" if side_long else "SHORT",
            "pnl": float(pnl - commission),
            "commission": float(commission)
        })
        return pnl

    def _fill_order(self, o: Order, ts_ms: int, price: float) -> None:
        qty_signed = o.qty if o.side == "BUY" else -o.qty
        pos = self.portfolio.positions.get(o.symbol) or Position(symbol=o.symbol)

        # If reducing/closing existing position, realize pnl fragment(s)
        realized = 0.0
        commission_alloc = self._commission(abs(qty_signed))
        if pos.qty != 0 and (pos.qty > 0) != (qty_signed > 0):
            # Closing direction
            closing_qty = min(abs(pos.qty), abs(qty_signed))
            # Attribute a proportional commission to the closing fragment
            close_comm = commission_alloc * (closing_qty / abs(qty_signed)) if abs(qty_signed) > 0 else 0.0
            realized_fragment = self._record_closed_fragment(
                ts_ms=ts_ms, symbol=o.symbol, closing_qty=closing_qty,
                entry_price=pos.avg_price, exit_price=price, side_long=(pos.qty > 0),
                commission=close_comm
            )
            realized += realized_fragment

        # Position arithmetic
        new_qty = pos.qty + qty_signed
        if new_qty == 0:
            pos.avg_price = 0.0
        else:
            # Same-side add -> average
            if (pos.qty >= 0 and qty_signed >= 0) or (pos.qty <= 0 and qty_signed <= 0):
                total_cost = pos.avg_price * abs(pos.qty) + price * abs(qty_signed)
                pos.avg_price = total_cost / (abs(pos.qty) + abs(qty_signed))
            # Flip handled by realized fragment above; avg_price becomes the residual side's price when non-zero

        pos.qty = new_qty
        pos.realized_pnl += realized
        self.portfolio.realized_pnl += realized

        # Cash and commission
        cash_delta = -price * qty_signed
        self.portfolio.cash += cash_delta - commission_alloc

        # Finalize order
        o.status = "FILLED"
        o.filled_qty = abs(qty_signed)
        o.avg_fill_price = price
        self._working.pop(o.order_id, None)

        fill = Fill(
            fill_id=_uid("fill"),
            order_id=o.order_id,
            ts=ts_ms,
            symbol=o.symbol,
            side=o.side,
            qty=abs(qty_signed),
            price=price,
            commission=commission_alloc,
            slippage_cost=0.0
        )
        self._emit_fill(fill)

    def _mark_equity(self, ts_ms: int) -> None:
        eq = self.portfolio.cash
        for sym, pos in self.portfolio.positions.items():
            last_px = self._last_prices.get(sym, pos.avg_price or 0.0)
            eq += pos.qty * last_px
        self.portfolio.equity = eq
        self.portfolio.last_mark_ts = ts_ms

    # --------- Helpers for engine ---------

    def build_order(self,
                    decision_id: str,
                    symbol: str,
                    side: str,
                    qty: float,
                    type_: str = "MARKET",
                    limit_price: float | None = None,
                    stop_price: float | None = None,
                    tag: str | None = None,
                    ts_ms: int | None = None
                    ) -> Order:
        return Order(
            order_id=_uid("ord"),
            decision_id=decision_id,
            ts=ts_ms or _now_ms(),
            symbol=symbol, side=side, qty=qty,
            type=type_, limit_price=limit_price, stop_price=stop_price, tag=tag
        )

    def snapshot(self) -> Dict[str, Any]:
        return {
            "cash": self.portfolio.cash,
            "equity": self.portfolio.equity,
            "realized_pnl": self.portfolio.realized_pnl,
            "positions": {k: asdict(v) for k, v in self.portfolio.positions.items()},
            "open_orders": {k: asdict(v) for k, v in self._working.items()},
            "last_prices": dict(self._last_prices),
            "closed_trades": len(self._closed_trades)
        }

    def get_trade_ledger(self) -> List[Dict[str, Any]]:
        return list(self._closed_trades)
''')

# -*- coding: utf-8 -*-
"""
Backtest Engine (vectorized bar-by-bar, async strategy API)
----------------------------------------------------------
- Loads historical features (ClickHouse or DataFrame)
- Calls strategy.on_features() per bar
- Routes intents -> SimAdapter orders
- Metrics from strict trade ledger (win rate, profit factor) + equity curve KPIs
- Appends equity once per timestamp (after processing ALL symbols of that ts)

Author: ChatGPT
"""

from __future__ import annotations
import os, math, importlib, asyncio, statistics, json
from dataclasses import dataclass, field, asdict
from typing import Any, Dict, List, Optional, Callable, Iterable, Tuple
from datetime import datetime, timezone
import pandas as pd

# Optional ClickHouse clients (lazy import)
def _get_clickhouse():
    try:
        import clickhouse_connect as ch  # pip install clickhouse-connect
        return ("connect", ch)
    except Exception:
        try:
            from clickhouse_driver import Client  # pip install clickhouse-driver
            return ("driver", Client)
        except Exception:
            return (None, None)

# ---- Engine Data Structures ----

@dataclass
class BacktestConfig:
    symbols: List[str]
    start: str                   # "YYYY-MM-DD HH:MM:SS"
    end: str                     # "
    timeframe: str = "1m"
    features_table: Optional[str] = None
    bars_table: Optional[str] = None
    clickhouse: Dict[str, Any] = field(default_factory=dict)
    initial_cash: float = 100_000.0
    commission_per_share: float = 0.005
    commission_min: float = 1.0
    slippage_bps: float = 1.0
    fill_ref: str = "open"
    strategy_path: str = "tsla_v7_2.TSLASimplifiedV7_2"
    strategy_params: Dict[str, Any] = field(default_factory=dict)
    seed: int = 42

@dataclass
class TradeRecord:
    ts: int
    symbol: str
    side: str
    qty: float
    price: float
    commission: float

@dataclass
class Metrics:
    total_return: float
    sharpe: float
    max_drawdown: float
    win_rate: float
    profit_factor: float
    trades: int

def _import_strategy_class(path: str):
    mod_name, cls_name = path.rsplit(".", 1)
    mod = importlib.import_module(mod_name)
    return getattr(mod, cls_name)

class _EngineShim:
    def __init__(self, outer):
        self._outer = outer
    async def submit_intent(self, strategy_id: str, decision) -> None:
        self._outer._pending_decisions.append((strategy_id, decision))

class BacktestEngine:
    def __init__(self, cfg: BacktestConfig) -> None:
        self.cfg = cfg
        from sim_adapter import SimAdapter
        self.sim = SimAdapter(
            initial_cash=cfg.initial_cash,
            commission_per_share=cfg.commission_per_share,
            commission_min=cfg.commission_min,
            slippage_bps=cfg.slippage_bps,
            fill_ref=cfg.fill_ref,
        )
        self._pending_decisions: List[Tuple[str, Any]] = []
        self._trades: List[TradeRecord] = []
        self._equity_curve: List[Tuple[int, float]] = []  # (ts_ms, equity)

        self.sim.on_fill(self._on_fill)

        StrategyCls = _import_strategy_class(cfg.strategy_path)
        params = {"extras": self.cfg.strategy_params or {}}
        self.strategy = StrategyCls(strategy_id="STRAT_1", engine=_EngineShim(self), params=params)

    def _on_fill(self, fill) -> None:
        self._trades.append(
            TradeRecord(ts=fill.ts, symbol=fill.symbol, side=fill.side,
                        qty=fill.qty, price=fill.price, commission=fill.commission)
        )

    def _features_table_name(self) -> str:
        if self.cfg.features_table:
            return self.cfg.features_table
        tf = self.cfg.timeframe
        return f"trading.features_{tf}"

    def _clickhouse_query(self) -> Optional[pd.DataFrame]:
        kind, client_or_mod = _get_clickhouse()
        if kind is None:
            return None
        start = self.cfg.start.replace("T"," ")
        end = self.cfg.end.replace("T"," ")
        tbl = self._features_table_name()
        syms = ",".join([f"'{s}'" for s in self.cfg.symbols])
        sql = f"""
            SELECT *
            FROM {tbl}
            WHERE symbol IN ({syms})
              AND ts >= toDateTime('{start}') AND ts < toDateTime('{end}')
            ORDER BY ts, symbol
        """
        if kind == "connect":
            conn = client_or_mod.get_client(**self.cfg.clickhouse)
            df = conn.query_df(sql)
            return df
        else:
            client = client_or_mod(**self.cfg.clickhouse)
            data = client.execute(sql)
            cols = [c[0] for c in client.execute("DESCRIBE TABLE " + tbl)]
            import pandas as pd
            return pd.DataFrame(data, columns=cols)

    def load_data(self, df: Optional[pd.DataFrame] = None) -> pd.DataFrame:
        if df is not None:
            out = df.copy()
        else:
            out = self._clickhouse_query()
            if out is None:
                raise RuntimeError("ClickHouse client not available and no DataFrame provided.")
        needed = {"ts","symbol","open","high","low","close"}
        missing = needed - set(out.columns)
        if missing:
            raise ValueError(f"Missing required columns: {missing}")
        if pd.api.types.is_datetime64_any_dtype(out["ts"]) is False:
            out["ts"] = pd.to_datetime(out["ts"], utc=True)
        out["ts_ms"] = (out["ts"].astype("int64") // 10**6).astype("int64")
        out = out.sort_values(["ts","symbol"]).reset_index(drop=True)
        return out

    @staticmethod
    def _max_drawdown(values: List[float]) -> float:
        peak = -math.inf
        mdd = 0.0
        for v in values:
            if v > peak:
                peak = v
            dd = (peak - v) / peak if peak != 0 else 0
            mdd = max(mdd, dd)
        return mdd

    @staticmethod
    def _sharpe(returns: List[float], risk_free: float = 0.0) -> float:
        if not returns:
            return 0.0
        avg = statistics.mean(returns)
        std = statistics.pstdev(returns) or 1e-9
        return (avg - risk_free) / std

    def _compute_metrics(self) -> Metrics:
        eq = [e for _, e in self._equity_curve]
        if not eq:
            return Metrics(0,0,0,0,0,0)
        rets = [(eq[i]-eq[i-1]) / max(eq[i-1],1e-9) for i in range(1, len(eq))]
        total_ret = (eq[-1] - eq[0]) / max(eq[0], 1e-9)
        sharpe = self._sharpe(rets)
        mdd = self._max_drawdown(eq)

        # Accurate trade stats from simulator ledger
        ledger = getattr(self.sim, "get_trade_ledger", lambda: [])()
        wins = sum(1 for t in ledger if t.get("pnl", 0.0) > 0)
        losses = sum(1 for t in ledger if t.get("pnl", 0.0) <= 0)
        gross_win = sum(max(t.get("pnl", 0.0), 0.0) for t in ledger)
        gross_loss = sum(max(-t.get("pnl", 0.0), 0.0) for t in ledger)

        win_rate = wins / max(wins+losses, 1) if (wins+losses) > 0 else 0.0
        profit_factor = (gross_win / max(gross_loss, 1e-9)) if gross_loss > 0 else float('inf')
        trades_n = len(ledger)

        return Metrics(
            total_return=float(total_ret),
            sharpe=float(sharpe),
            max_drawdown=float(mdd),
            win_rate=float(win_rate),
            profit_factor=float(profit_factor),
            trades=int(trades_n)
        )

    async def _run_async(self, df: pd.DataFrame) -> Metrics:
        if hasattr(self.strategy, "initialize") and asyncio.iscoroutinefunction(self.strategy.initialize):
            await self.strategy.initialize()

        # Iterate rows; append equity AFTER all symbols of same ts processed
        n = len(df)
        for i in range(n):
            row = df.iloc[i]
            ts_ms = int(row["ts_ms"])
            symbol = row["symbol"]
            bar = {
                "open": float(row["open"]),
                "high": float(row["high"]),
                "low": float(row["low"]),
                "close": float(row["close"]),
            }

            # 1) Update simulator with bar
            self.sim.on_bar(ts_ms, symbol, bar)

            # 2) Strategy callback with full features
            features = row.to_dict()
            if hasattr(self.strategy, "on_features"):
                await self.strategy.on_features(symbol, features)

            # 3) Drain decisions -> orders
            while self._pending_decisions:
                strat_id, decision = self._pending_decisions.pop(0)
                qty = float(getattr(decision, "target_qty", 0))
                side = getattr(decision, "side", "BUY")
                symbol_d = getattr(decision, "symbol", symbol)
                typ = getattr(decision, "order_type", "MARKET")
                limit_px = getattr(decision, "limit_price", None)
                stop_px = getattr(decision, "stop_price", None)
                decision_id = getattr(decision, "decision_id", f"DEC_{ts_ms}")
                order = self.sim.build_order(
                    decision_id=decision_id, symbol=symbol_d, side=side, qty=qty,
                    type_=typ, limit_price=limit_px, stop_price=stop_px, ts_ms=ts_ms
                )
                self.sim.submit(order)

            # 4) Append equity only when next ts changes (or last row)
            next_ts = int(df.iloc[i+1]["ts_ms"]) if i+1 < n else None
            if next_ts is None or next_ts != ts_ms:
                snap = self.sim.snapshot()
                self._equity_curve.append((ts_ms, float(snap["equity"])))

        return self._compute_metrics()

    def run(self, df: Optional[pd.DataFrame] = None) -> Dict[str, Any]:
        data = self.load_data(df=df)
        metrics = asyncio.run(self._run_async(data))
        return {
            "metrics": asdict(metrics),
            "trades": [asdict(t) for t in self._trades],

def _equity_to_returns(equity_curve: List[Tuple[int, float]]) -> List[float]:
    rets: List[float] = []
    if not equity_curve or len(equity_curve) < 2:
        return rets
    eq = [float(e) for _, e in equity_curve]
    for i in range(1, len(eq)):
        prev = eq[i-1] if eq[i-1] != 0 else 1e-9
        rets.append((eq[i]-eq[i-1]) / prev)
    return rets

def _sortino(returns: List[float]) -> float:
    if not returns:
        return 0.0
    neg = [r for r in returns if r < 0]
    down_stdev = (sum(r*r for r in neg)/len(neg))**0.5 if neg else 1e-9
    mean_r = sum(returns)/len(returns)
    return mean_r / down_stdev

def _omega(returns: List[float], tau: float = 0.0) -> float:
    if not returns:
        return 0.0
    gains = sum(max(r - tau, 0.0) for r in returns)
    losses = sum(max(tau - r, 0.0) for r in returns)
    return gains / (losses if losses > 0 else 1e-9)

def _calmar(total_return: float, max_drawdown: float) -> float:
    if max_drawdown <= 0:
        return float('inf')
    return total_return / max_drawdown

def _safe_eval_formula(expr: str, context: Dict[str, float]) -> float:
    tree = ast.parse(expr, mode='eval')

            raise ValueError(f"Unsafe or unsupported expression node: {type(node).__name__}")
        if isinstance(node, ast.Name) and node.id not in context:
            raise ValueError(f"Unknown name in formula: {node.id}")
    return float(eval(compile(tree, "<formula>", "eval"), {"__builtins__": {}}, context))

## 目标函数扩展（可直接在配置里用）

在 `objective` 段中现在支持：

* `metric`: `"sharpe" | "total_return" | "profit_factor" | "win_rate" | "sortino" | "omega" | "calmar"`
