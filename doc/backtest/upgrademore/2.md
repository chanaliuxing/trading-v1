# 1) `pipelines/lstm_oof_to_clickhouse.py`

**作用**：按时间折分 + 禁运生成**严格 OOF**预测 → 写入 `trading.oof_lstm_h5`（独立表）

> 依赖：`pip install clickhouse-driver torch pandas numpy`

```python
# pipelines/lstm_oof_to_clickhouse.py
from __future__ import annotations
import os, sys, time, math, argparse, subprocess
from dataclasses import dataclass
from typing import List, Tuple

import numpy as np
import pandas as pd
from clickhouse_driver import Client

import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader


# ===================== Dataset & Model =====================

class SeqFeatDataset(Dataset):
    """
    给定按时间排序的 DataFrame，提供 [seq_len, F] 的滑窗序列样本
    返回: x_seq, y, ts, symbol
    """
    def __init__(self, df: pd.DataFrame, seq_cols: List[str], y_col: str | None, seq_len: int):
        self.df = df.reset_index(drop=True)
        self.seq_cols = seq_cols
        self.y_col = y_col
        self.seq_len = seq_len
        self.idx = np.arange(seq_len - 1, len(df))
    def __len__(self): return len(self.idx)
    def __getitem__(self, i):
        j = self.idx[i]
        x = self.df.iloc[j - self.seq_len + 1: j + 1][self.seq_cols].values.astype(np.float32)
        y = np.float32(self.df.iloc[j][self.y_col]) if self.y_col else np.float32(0.0)
        ts = self.df.iloc[j]["ts"]; sym = self.df.iloc[j]["symbol"]
        return x, y, ts, sym

class LSTMHead(nn.Module):
    def __init__(self, in_dim: int, hidden: int = 64, num_layers: int = 1, dropout: float = 0.1):
        super().__init__()
        self.lstm = nn.LSTM(
            input_size=in_dim, hidden_size=hidden,
            num_layers=num_layers, batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0
        )
        self.head = nn.Sequential(nn.LayerNorm(hidden), nn.ReLU(), nn.Linear(hidden, 1))
    def forward(self, x):  # x: [B, T, F]
        out, _ = self.lstm(x)
        h = out[:, -1, :]
        y = self.head(h).squeeze(-1)
        return y


# ===================== Config =====================

@dataclass
class OOFConfig:
    ch_host: str = "localhost"
    ch_port: int = 9000
    ch_user: str = "default"
    ch_password: str = ""
    ch_db: str = "trading"

    # 读取序列的“原始特征表”（不是视图），例如 features_1m
    src_table: str = "features_1m"

    # 写入的 OOF 目标表
    oof_table: str = "oof_lstm_h5"

    symbol: str = "TSLA"
    start: str = "2024-01-01 00:00:00"
    end: str   = "2025-10-01 00:00:00"

    # 标签与序列
    label_col: str = "label_h5"  # 例如未来5m收益/方向
    seq_cols: Tuple[str, ...] = ("close","vwap","rsi_14","ema_9","atr_14")
    seq_len: int = 60
    horizon_min: int = 5

    # 训练
    batch: int = 512
    lr: float = 1e-3
    epochs: int = 2
    folds: int = 4
    embargo_min: int = 10
    hidden: int = 64
    num_layers: int = 1
    dropout: float = 0.1

    # 版本/元信息
    run_id: str | None = None
    model_ver: str = "lstm_h{hidden}_l{num_layers}_seq{seq_len}_e{epochs}"

    # reproducibility
    seed: int = 1337


# ===================== Helpers =====================

def ch_client(cfg: OOFConfig) -> Client:
    return Client(host=cfg.ch_host, port=cfg.ch_port, user=cfg.ch_user,
                  password=cfg.ch_password, database=cfg.ch_db,
                  settings={"use_numpy": True})

def load_frame(cfg: OOFConfig) -> pd.DataFrame:
    cols = ["ts","symbol", cfg.label_col, *cfg.seq_cols]
    q = f"""
    SELECT {", ".join(cols)}
    FROM {cfg.src_table}
    WHERE symbol = %(sym)s AND ts >= %(st)s AND ts < %(ed)s
    ORDER BY ts
    """
    cli = ch_client(cfg)
    data = cli.execute(q, {"sym": cfg.symbol, "st": cfg.start, "ed": cfg.end})
    df = pd.DataFrame(data, columns=cols)
    # 清洗
    df = df.dropna().reset_index(drop=True)
    return df

def time_folds(n_rows: int, n_splits: int) -> List[Tuple[int, int]]:
    # 按比例简单等分
    bounds = [int(i * n_rows / n_splits) for i in range(n_splits + 1)]
    return [(bounds[i], bounds[i+1]) for i in range(n_splits)]

def embargo_start(idx_end: int, embargo_steps: int, total: int) -> int:
    # idx_end（训练末尾）之后空出 embargo_steps，再开始验证
    start = min(total - 1, idx_end + embargo_steps)
    return start

def set_seed(seed: int):
    import random
    torch.manual_seed(seed); np.random.seed(seed); random.seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


# ===================== Train One Fold =====================

def train_fold(train_df: pd.DataFrame, val_df: pd.DataFrame, cfg: OOFConfig, device: str = "cpu") -> pd.DataFrame:
    model = LSTMHead(in_dim=len(cfg.seq_cols), hidden=cfg.hidden, num_layers=cfg.num_layers, dropout=cfg.dropout).to(device)
    opt = torch.optim.AdamW(model.parameters(), lr=cfg.lr)
    loss_fn = nn.MSELoss()

    ds_tr = SeqFeatDataset(train_df, list(cfg.seq_cols), cfg.label_col, cfg.seq_len)
    ds_va = SeqFeatDataset(val_df,   list(cfg.seq_cols), cfg.label_col, cfg.seq_len)
    if len(ds_tr) == 0 or len(ds_va) == 0:
        return pd.DataFrame(columns=["ts","symbol","pred","conf"])

    dl_tr = DataLoader(ds_tr, batch_size=cfg.batch, shuffle=False, num_workers=0)
    dl_va = DataLoader(ds_va, batch_size=cfg.batch, shuffle=False, num_workers=0)

    for _ in range(cfg.epochs):
        model.train()
        for x, y, _, _ in dl_tr:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            p = model(x)
            loss = loss_fn(p, y)
            loss.backward()
            opt.step()

    # 仅对验证段预测（严格 OOF）
    model.eval(); preds = []; ts_list = []; sym_list = []
    with torch.no_grad():
        for x, _, ts, sym in dl_va:
            x = x.to(device)
            p = model(x).cpu().numpy()
            preds.append(p); ts_list.extend(ts.tolist()); sym_list.extend(sym.tolist())

    if len(preds) == 0:
        return pd.DataFrame(columns=["ts","symbol","pred","conf"])

    pred = np.concatenate(preds, axis=0).astype(np.float32)
    # 简易置信度：|pred| 归一（生产可替换为 MC dropout 方差等）
    denom = np.percentile(np.abs(pred), 90) + 1e-6
    conf = np.clip(np.abs(pred) / denom, 0, 1).astype(np.float32)

    oof_df = pd.DataFrame({"ts": ts_list, "symbol": sym_list, "pred": pred, "conf": conf})
    return oof_df.sort_values("ts")


# ===================== Upsert to ClickHouse =====================

def upsert_oof(cfg: OOFConfig, oof_df: pd.DataFrame):
    if oof_df.empty:
        print("No OOF rows to write.")
        return
    cli = ch_client(cfg)

    # run_id & model_ver
    if cfg.run_id is None:
        git = subprocess.getoutput("git rev-parse --short HEAD") or "nogit"
        cfg.run_id = f"oof_{time.strftime('%Y%m%d_%H%M%S')}_{git}"
    model_ver = cfg.model_ver.format(hidden=cfg.hidden, num_layers=cfg.num_layers, seq_len=cfg.seq_len, epochs=cfg.epochs)

    # 先删除同 run_id 的旧记录（幂等）
    cli.execute(f"ALTER TABLE {cfg.oof_table} DELETE WHERE run_id = %(rid)s", {"rid": cfg.run_id})

    # 分批写入
    cols = ["ts","symbol","pred","conf","run_id","model_ver"]
    q = f"INSERT INTO {cfg.oof_table} (ts,symbol,pred,conf,run_id,model_ver) VALUES"
    batch = 100_000
    n = len(oof_df)
    for i in range(0, n, batch):
        part = oof_df.iloc[i:i+batch].copy()
        part["run_id"] = cfg.run_id
        part["model_ver"] = model_ver
        rows = list(part[cols].itertuples(index=False, name=None))
        cli.execute(q, rows)
    print(f"[OK] OOF written: {n} rows → {cfg.oof_table}, run_id={cfg.run_id}, model_ver={model_ver}")


# ===================== Main OOF Pipeline =====================

def run_oof_pipeline(cfg: OOFConfig):
    set_seed(cfg.seed)
    df = load_frame(cfg)
    if len(df) < cfg.seq_len + 200:
        print(f"[WARN] Not enough rows ({len(df)}) for seq_len={cfg.seq_len}.")
        return

    slices = time_folds(len(df), cfg.folds)
    embargo_steps = cfg.embargo_min  # 因为是 1m 频率，直接用“分钟”即“步数”；若是 5s 则换算

    all_oof = []
    for k, (a, b) in enumerate(slices):
        tr_df = df.iloc[:a].copy()
        va_start = embargo_start(a - 1, embargo_steps, len(df))
        va_df = df.iloc[max(va_start, a):b].copy()

        # 保护：训练/验证都必须要有足够序列长度
        if len(tr_df) < (cfg.seq_len + 50) or len(va_df) < (cfg.seq_len + 10):
            print(f"[SKIP] fold {k}: insufficient data (train={len(tr_df)}, valid={len(va_df)})")
            continue

        oof_k = train_fold(tr_df, va_df, cfg, device="cpu")
        if not oof_k.empty:
            all_oof.append(oof_k)
        print(f"[FOLD {k}] oof_rows={len(oof_k)}")

    if not all_oof:
        print("[WARN] No folds produced OOF.")
        return

    oof_df = pd.concat(all_oof, axis=0).drop_duplicates(subset=["ts","symbol"]).sort_values("ts")
    upsert_oof(cfg, oof_df)


# ===================== CLI =====================

def parse_args() -> OOFConfig:
    p = argparse.ArgumentParser("LSTM OOF → ClickHouse (strict, embargoed)")
    p.add_argument("--ch_host", default="localhost")
    p.add_argument("--ch_port", type=int, default=9000)
    p.add_argument("--ch_user", default="default")
    p.add_argument("--ch_password", default="")
    p.add_argument("--ch_db", default="trading")
    p.add_argument("--src_table", default="features_1m")
    p.add_argument("--oof_table", default="oof_lstm_h5")

    p.add_argument("--symbol", default="TSLA")
    p.add_argument("--start", default="2024-01-01 00:00:00")
    p.add_argument("--end",   default="2025-10-01 00:00:00")

    p.add_argument("--label_col", default="label_h5")
    p.add_argument("--seq_cols", default="close,vwap,rsi_14,ema_9,atr_14")
    p.add_argument("--seq_len", type=int, default=60)
    p.add_argument("--horizon_min", type=int, default=5)

    p.add_argument("--batch", type=int, default=512)
    p.add_argument("--lr", type=float, default=1e-3)
    p.add_argument("--epochs", type=int, default=2)
    p.add_argument("--folds", type=int, default=4)
    p.add_argument("--embargo_min", type=int, default=10)
    p.add_argument("--hidden", type=int, default=64)
    p.add_argument("--num_layers", type=int, default=1)
    p.add_argument("--dropout", type=float, default=0.1)

    p.add_argument("--run_id", default=None)
    p.add_argument("--model_ver", default="lstm_h{hidden}_l{num_layers}_seq{seq_len}_e{epochs}")
    p.add_argument("--seed", type=int, default=1337)

    args = p.parse_args()
    cfg = OOFConfig(
        ch_host=args.ch_host, ch_port=args.ch_port, ch_user=args.ch_user, ch_password=args.ch_password, ch_db=args.ch_db,
        src_table=args.src_table, oof_table=args.oof_table,
        symbol=args.symbol, start=args.start, end=args.end,
        label_col=args.label_col,
        seq_cols=tuple([c.strip() for c in args.seq_cols.split(",") if c.strip()]),
        seq_len=args.seq_len, horizon_min=args.horizon_min,
        batch=args.batch, lr=args.lr, epochs=args.epochs, folds=args.folds, embargo_min=args.embargo_min,
        hidden=args.hidden, num_layers=args.num_layers, dropout=args.dropout,
        run_id=args.run_id, model_ver=args.model_ver, seed=args.seed
    )
    return cfg


if __name__ == "__main__":
    cfg = parse_args()
    run_oof_pipeline(cfg)
```

# 2) `pipelines/train_xgb_with_oof.py`

**作用**：从视图 `trading.features_1m_with_oof` 读取（基础特征 + 最新 OOF），做时间序列交叉验证训练；可用 `--use_relay 0/1` 开关。

> 依赖：`pip install clickhouse-driver xgboost scikit-learn pandas numpy`

```python
# pipelines/train_xgb_with_oof.py
from __future__ import annotations
import argparse, json, hashlib, subprocess
from dataclasses import dataclass
from typing import Tuple, List

import numpy as np
import pandas as pd
from clickhouse_driver import Client

import xgboost as xgb
from sklearn.model_selection import TimeSeriesSplit
from sklearn.metrics import roc_auc_score, log_loss


@dataclass
class XGBRelayCfg:
    ch_host: str = "localhost"; ch_port: int = 9000
    ch_user: str = "default"; ch_password: str = ""
    ch_db: str = "trading"

    table_view: str = "features_1m_with_oof"
    symbol: str = "TSLA"
    start: str = "2024-06-01 00:00:00"
    end: str   = "2025-10-01 00:00:00"

    label_col: str = "label_h5"
    base_feats: Tuple[str, ...] = ("vwap","rsi_14","ema_9","atr_14","close")
    relay_feats: Tuple[str, ...] = ("lstm_oof_h5","lstm_oof_conf_h5")
    use_relay: int = 1

    n_splits: int = 5
    eta: float = 0.05
    max_depth: int = 5
    subsample: float = 0.8
    colsample_bytree: float = 0.8
    min_child_weight: float = 1.0
    num_boost_round: int = 300


def ch_client(cfg: XGBRelayCfg) -> Client:
    return Client(host=cfg.ch_host, port=cfg.ch_port, user=cfg.ch_user,
                  password=cfg.ch_password, database=cfg.ch_db,
                  settings={"use_numpy": True})

def fetch(cfg: XGBRelayCfg) -> pd.DataFrame:
    cols = ["ts","symbol", cfg.label_col, *cfg.base_feats]
    if cfg.use_relay:
        cols += list(cfg.relay_feats)
    q = f"""
    SELECT {", ".join(cols)}
    FROM {cfg.table_view}
    WHERE symbol=%(sym)s AND ts >= %(st)s AND ts < %(ed)s
    ORDER BY ts
    """
    data = ch_client(cfg).execute(q, {"sym": cfg.symbol, "st": cfg.start, "ed": cfg.end})
    df = pd.DataFrame(data, columns=cols).dropna()
    return df

def train_eval(cfg: XGBRelayCfg):
    df = fetch(cfg)
    feats = list(cfg.base_feats) + (list(cfg.relay_feats) if cfg.use_relay else [])
    X = df[feats].values
    y = (df[cfg.label_col].values > 0).astype(np.int32)

    params = dict(
        eta=cfg.eta, max_depth=cfg.max_depth, subsample=cfg.subsample, colsample_bytree=cfg.colsample_bytree,
        min_child_weight=cfg.min_child_weight,
        objective="binary:logistic", eval_metric="auc"
    )

    tscv = TimeSeriesSplit(n_splits=cfg.n_splits)
    aucs, lls = [], []
    for k, (tr, va) in enumerate(tscv.split(X)):
        dtr = xgb.DMatrix(X[tr], label=y[tr])
        dva = xgb.DMatrix(X[va], label=y[va])
        bst = xgb.train(params, dtr, num_boost_round=cfg.num_boost_round, evals=[(dva,"valid")], verbose_eval=False)
        p = bst.predict(dva)
        aucs.append(roc_auc_score(y[va], p))
        lls.append(log_loss(y[va], p, labels=[0,1]))
        print(f"[fold {k}] AUC={aucs[-1]:.4f}, LogLoss={lls[-1]:.4f}")

    print(f"[XGB] use_relay={cfg.use_relay} | AUC(mean)={np.mean(aucs):.4f} | LogLoss(mean)={np.mean(lls):.4f}")
    return np.mean(aucs), aucs

def parse_args() -> XGBRelayCfg:
    p = argparse.ArgumentParser("Train XGBoost with/without LSTM-OOF features")
    p.add_argument("--ch_host", default="localhost"); p.add_argument("--ch_port", type=int, default=9000)
    p.add_argument("--ch_user", default="default");   p.add_argument("--ch_password", default="")
    p.add_argument("--ch_db", default="trading")
    p.add_argument("--table_view", default="features_1m_with_oof")
    p.add_argument("--symbol", default="TSLA")
    p.add_argument("--start", default="2024-06-01 00:00:00")
    p.add_argument("--end",   default="2025-10-01 00:00:00")
    p.add_argument("--label_col", default="label_h5")
    p.add_argument("--base_feats", default="vwap,rsi_14,ema_9,atr_14,close")
    p.add_argument("--relay_feats", default="lstm_oof_h5,lstm_oof_conf_h5")
    p.add_argument("--use_relay", type=int, default=1)
    p.add_argument("--n_splits", type=int, default=5)
    p.add_argument("--eta", type=float, default=0.05)
    p.add_argument("--max_depth", type=int, default=5)
    p.add_argument("--subsample", type=float, default=0.8)
    p.add_argument("--colsample_bytree", type=float, default=0.8)
    p.add_argument("--min_child_weight", type=float, default=1.0)
    p.add_argument("--num_boost_round", type=int, default=300)
    args = p.parse_args()

    cfg = XGBRelayCfg(
        ch_host=args.ch_host, ch_port=args.ch_port, ch_user=args.ch_user, ch_password=args.ch_password, ch_db=args.ch_db,
        table_view=args.table_view, symbol=args.symbol, start=args.start, end=args.end,
        label_col=args.label_col,
        base_feats=tuple([c.strip() for c in args.base_feats.split(",") if c.strip()]),
        relay_feats=tuple([c.strip() for c in args.relay_feats.split(",") if c.strip()]),
        use_relay=args.use_relay,
        n_splits=args.n_splits,
        eta=args.eta, max_depth=args.max_depth, subsample=args.subsample,
        colsample_bytree=args.colsample_bytree, min_child_weight=args.min_child_weight,
        num_boost_round=args.num_boost_round
    )
    return cfg

if __name__ == "__main__":
    cfg = parse_args()
    train_eval(cfg)
```
