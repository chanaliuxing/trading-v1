```python
import os, json, argparse, glob

def list_generations(out_dir):
    gens = []
    for p in sorted(Path(out_dir).glob("gen_*")):
        if p.is_dir():
            gens.append(p.name)
    return gens

def eq_plot(curves, title):
    if not curves:
        st.info("No equity curves.")
        return
    fig, ax = plt.subplots()
    for payload in curves:
        curve = payload.get("equity_curve", [])
        if not curve: continue
        xs = [ts for ts, _ in curve]
        eq = [float(e) for _, e in curve]
        base = eq[0] if eq and eq[0] != 0 else 1e-9
        norm = [e/base for e in eq]
        ax.plot(xs, norm, label=f"Fold {payload.get('fold_id')}")
    ax.set_xlabel("ts (ms)"); ax.set_ylabel("Normalized Equity"); ax.set_title(title)
    ax.legend()
    st.pyplot(fig)

def dd_plot(curves, title):
    if not curves:
        st.info("No drawdown curves.")
        return
    fig, ax = plt.subplots()
    for payload in curves:
        curve = payload.get("equity_curve", [])
        if not curve: continue
        eq = [float(e) for _, e in curve]
        peak, dd = -1e18, []
        for e in eq:
            peak = max(peak, e)
            dd.append(0.0 if peak <= 0 else (peak - e) / peak)
        xs = [ts for ts, _ in curve]
        ax.plot(xs, dd, label=f"Fold {payload.get('fold_id')}")
    ax.set_xlabel("ts (ms)"); ax.set_ylabel("Drawdown"); ax.set_title(title)
    ax.legend()
    st.pyplot(fig)
```

```sql
CREATE TABLE IF NOT EXISTS alpha_runs (
  gen UInt16,
  mode String,
  stage String,                -- 'stage1' | 'stage2' | 'single'
  ts DateTime DEFAULT now(),
  oos_avg_sharpe Float64,
  oos_avg_total_return Float64,
  oos_avg_max_drawdown Float64,
  oos_avg_win_rate Float64,
  oos_avg_profit_factor Float64,
  oos_avg_sortino Float64,
  oos_avg_omega Float64,
  oos_avg_calmar Float64,
  oos_trades UInt32
) ENGINE = MergeTree ORDER BY (gen, stage, ts);

-- 行级结果（可选带来源标签）
CREATE TABLE IF NOT EXISTS alpha_rows (
  gen UInt16,
  phase LowCardinality(String),    -- 'IS' | 'OOS'
  fold_id UInt16,
  params String,                   -- 原始 JSON 字符串
  sharpe Float64,
  total_return Float64,
  max_drawdown Float64,
  win_rate Float64,
  profit_factor Float64,
  sortino Float64,
  omega Float64,
  calmar Float64,
  trades UInt32,
  origin_label LowCardinality(String) DEFAULT 'NA',
  ts DateTime DEFAULT now()
) ENGINE = MergeTree ORDER BY (gen, phase, fold_id, ts);

-- OOS 曲线（每折每点）
CREATE TABLE IF NOT EXISTS alpha_equity (
  gen UInt16,
  stage String,                    -- 'stage1' or 'stage2' or 'single'
  fold_id UInt16,
  ts_ms UInt64,
  equity Float64
) ENGINE = MergeTree ORDER BY (gen, stage, fold_id, ts_ms);
```

```sql
SELECT phase, origin_label, count() AS cnt
```

### 小结：怎么判断“他们干得怎么样”？

* **协同有效性**：`rows_with_origin.csv` + `proposer_contrib.png`（或 Grafana 柱状）：混合/两者共同（BOTH/MIXED）占比上升，同时 OOS 指标更好 → 协同确有贡献。
* **稳健性**：查看 **跨折 OOS 曲线/回撤**是否齐整、**OOS vs IS** 差距是否收敛。
* **收敛效率**：Optuna trials 的历史最佳曲线是否陡峭提升、参数敏感度是否集中到少数维度。
* **演进轨迹**：Grafana 折线（代次→OOS Sharpe/MaxDD）是否稳定上行。

```python
"""
alpha_loop.py — Two-Stage COOP + Optuna + ClickHouse Sink
---------------------------------------------------------
Stage 1 (COOP): LLM 与 ML 同时提出网格 → 合并/标注来源 → Walk-Forward HPO
Stage 2 (Optuna): 以 Stage 1 的冠军与合并网格为中心，构造邻域搜索空间 → 贝叶斯细化

def _topk_rows(rows: List[Dict[str, Any]], k: int, key: str) -> List[Dict[str, Any]]:

# ----- safe formula for scoring (与 wf_hpo 风格一致) -----
_ALLOWED = {
    ast.Expression, ast.BinOp, ast.UnaryOp, ast.Num, ast.Load, ast.Name,
    ast.Add, ast.Sub, ast.Mult, ast.Div, ast.Pow, ast.USub, ast.UAdd,
    ast.Mod, ast.Constant, ast.FloorDiv
}
def _safe_eval_formula(expr: str, ctx: Dict[str, float]) -> float:
    tree = ast.parse(expr, mode='eval')
    for node in ast.walk(tree):
        if type(node) not in _ALLOWED:
            raise ValueError(f"Unsupported in formula: {type(node).__name__}")
        if isinstance(node, ast.Name) and node.id not in ctx:
            raise ValueError(f"Unknown name: {node.id}")
    return float(eval(compile(tree, "<formula>", "eval"), {"__builtins__": {}}, ctx))

def _ml_suggest(rows: List[Dict[str, Any]], target_key: str, n: int = 24) -> Tuple[Dict[str, List[Any]], Dict[str, float]]:
    """
    返回 (grid, feature_importances)。feature_importances 是代理模型对“参数列”的重要度。
    """
    import numpy as np
    import pandas as pd
    try:
        from sklearn.ensemble import RandomForestRegressor
    except Exception:
        return {}, {}
    if not rows:
        return {}, {}
    df = pd.DataFrame(rows)
    try:
        pexp = df["params"].apply(lambda s: json.loads(s) if isinstance(s, str) else (s or {}))
    except Exception:
        return {}, {}
    P = pd.json_normalize(pexp)
    if P.empty:
        return {}, {}
    y = df[target_key].astype(float)

    X = P.copy()
    for c in X.columns:
        if not np.issubdtype(X[c].dtype, np.number):
            X[c] = X[c].astype("category").cat.codes

    rf = RandomForestRegressor(n_estimators=300, random_state=42)
    rf.fit(X.fillna(0), y)
    importances = {c: float(w) for c, w in zip(X.columns, rf.feature_importances_)}

    # 生成候选点（随机+靠近均值）
    rng = np.random.default_rng(42)
    cand = []
    for _ in range(n * 3):
        row = {}
        for c in X.columns:
            col = X[c]
            if np.issubdtype(col.dtype, np.number):
                mu, sd = float(col.mean()), float(col.std() or 1.0)
                row[c] = float(rng.normal(mu, sd))
            else:
                ch = list(col.dropna().unique())
                row[c] = rng.choice(ch) if ch else 0
        cand.append(row)
    C = pd.DataFrame(cand)
    preds = rf.predict(C.fillna(0))
    C["pred"] = preds
    C = C.sort_values("pred", ascending=False).head(n)

    grid: Dict[str, List[Any]] = {}
    for _, r in C.iterrows():
        for k, v in r.drop(labels=["pred"]).items():
            grid.setdefault(k, [])
            if v not in grid[k]:
                grid[k].append(v)
    for k in grid:
        grid[k] = grid[k][:5]
    return grid, importances


# ---------------- Grid merge & attribution ----------------

def _merge_grids(grid_base: Dict[str, List[Any]],
                 grid_llm: Dict[str, List[Any]],
                 grid_ml: Dict[str, List[Any]],
                 cap_per_param: int = 5) -> Tuple[Dict[str, List[Any]], Dict[str, Dict[Any, str]]]:
    """
    合并网格并记录来源：
      origin_map[param][value] = "LLM" | "ML" | "BOTH" | "BASE"
    合并策略：BASE ∪ LLM ∪ ML，去重后截断到 cap_per_param。
    """
    out: Dict[str, List[Any]] = {}
    origin: Dict[str, Dict[Any, str]] = {}

    all_keys = set(grid_base) | set(grid_llm) | set(grid_ml)
    for k in sorted(all_keys):
        vals = []
        src = {}

        def add(vals_in, label):
            for v in (vals_in or []):
                if v not in vals:
                    vals.append(v)
                    src[v] = label
                else:
                    if src.get(v) != label and src.get(v) in ("LLM", "ML"):
                        src[v] = "BOTH"

        add(grid_base.get(k), "BASE")
        add(grid_llm.get(k), "LLM")
        add(grid_ml.get(k), "ML")

        vals = vals[:cap_per_param]
        out[k] = vals
        origin[k] = {v: src[v] for v in vals}
    return out, origin

def _label_row_origin(row_params: Dict[str, Any], origin_map: Dict[str, Dict[Any, str]]) -> str:
    flags = set()
    for k, v in row_params.items():
        src = origin_map.get(k, {}).get(v)
        if src in ("LLM", "ML", "BOTH"):
            flags.add(src)
        else:
            flags.add("BASE")
    if flags == {"LLM"}: return "LLM"
    if flags == {"ML"}: return "ML"
    if flags == {"BASE"}: return "BASE"
    if "LLM" in flags and "ML" in flags and len(flags) == 2: return "MIXED"
    if "BOTH" in flags and len(flags) == 1: return "BOTH"
    return "MIXED"

def _optuna_available() -> bool:

def _score_from_rows_or_summary(rows: List[Dict[str, Any]], summary: Dict[str, Any], objective: Dict[str, Any], target: str) -> float:
    """
    target: "is" or "oos"
    - is：对 rows 中 phase==IS 的行按 objective 计算分数并取均值
    - oos：用 oos_summary 的聚合指标
    支持 formula 与 metric。
    """
    metric_key = (objective or {}).get("metric", "sharpe")
    formula = (objective or {}).get("formula")

    def _ctx_from_row(r: Dict[str, Any]) -> Dict[str, float]:
        return {
            "sharpe": float(r.get("sharpe", 0.0)),
            "total_return": float(r.get("total_return", 0.0)),
            "max_drawdown": float(r.get("max_drawdown", 0.0)),
            "win_rate": float(r.get("win_rate", 0.0)),
            "profit_factor": float(r.get("profit_factor", 0.0)),
            "trades": float(r.get("trades", 0.0)),
            "sortino": float(r.get("sortino", 0.0)),
            "omega": float(r.get("omega", 0.0)),
            "calmar": float(r.get("calmar", 0.0)),
        }

    if target == "is":
        is_rows = [r for r in rows if r.get("phase") == "IS"]
        if not is_rows:
            return 0.0
        vals = []
        for r in is_rows:
            if formula:
                vals.append(_safe_eval_formula(formula, _ctx_from_row(r)))
            else:
                vals.append(float(r.get(metric_key, 0.0)))
        return float(sum(vals)/len(vals))
    else:
        # OOS 汇总
        if formula:
            return _safe_eval_formula(formula, {
                "sharpe": float(summary.get("oos_avg_sharpe", 0.0)),
                "total_return": float(summary.get("oos_avg_total_return", 0.0)),
                "max_drawdown": float(summary.get("oos_avg_max_drawdown", 0.0)),
                "win_rate": float(summary.get("oos_avg_win_rate", 0.0)),
                "profit_factor": float(summary.get("oos_avg_profit_factor", 0.0)),
                "trades": float(summary.get("oos_trades", 0.0)),
                "sortino": float(summary.get("oos_avg_sortino", 0.0)),
                "omega": float(summary.get("oos_avg_omega", 0.0)),
                "calmar": float(summary.get("oos_avg_calmar", 0.0)),
            })
        key_map = {
            "sharpe": "oos_avg_sharpe",
            "total_return": "oos_avg_total_return",
            "max_drawdown": "oos_avg_max_drawdown",
            "win_rate": "oos_avg_win_rate",
            "profit_factor": "oos_avg_profit_factor",
            "sortino": "oos_avg_sortino",
            "omega": "oos_avg_omega",
            "calmar": "oos_avg_calmar"
        }
        return float(summary.get(key_map.get(metric_key, "oos_avg_sharpe"), 0.0))

def _run_optuna_with_space(base_hpo_cfg: Dict[str, Any], space: Dict[str, Dict[str, Any]], opt_conf: Dict[str, Any]) -> Tuple[Dict[str, Any], Dict[str, Any]]:
    """
    用显式 space 运行 Optuna：返回 (best_params, study_summary)
    """
    if not _optuna_available():
        raise RuntimeError("Optuna not installed. `pip install optuna`.")

    import optuna

    trials = int(opt_conf.get("trials", 40))
    target = opt_conf.get("target", "is")  # "is" or "oos"
    direction = opt_conf.get("direction", "maximize")
    sampler_name = opt_conf.get("sampler", "tpe").lower()

    if sampler_name == "cmaes":
        sampler = optuna.samplers.CmaEsSampler()
    else:
        sampler = optuna.samplers.TPESampler(seed=42)

    study = optuna.create_study(direction=direction, sampler=sampler)

    def _suggest(trial: "optuna.trial.Trial") -> Dict[str, Any]:
        params = {}
        for k, spec in space.items():
            t = spec.get("type")
            if t == "float":
                params[k] = trial.suggest_float(k, spec["low"], spec["high"], log=bool(spec.get("log", False)))
            elif t == "int":
                step = int(spec.get("step", 1))
                if step > 1:
                    raw = trial.suggest_int(k, int(spec["low"]), int(spec["high"]))
                    base = int(spec["low"])
                    snapped = base + int(round((raw - base) / step)) * step
                    params[k] = max(int(spec["low"]), min(int(spec["high"]), snapped))
                else:
                    params[k] = trial.suggest_int(k, int(spec["low"]), int(spec["high"]))
            elif t == "categorical":
                params[k] = trial.suggest_categorical(k, list(spec["choices"]))
            else:
                raise ValueError(f"Unknown space type for {k}: {t}")
        return params

    def objective(trial: "optuna.trial.Trial") -> float:
        params = _suggest(trial)
        cfg1 = deepcopy(base_hpo_cfg)
        cfg1["grid"] = {k: [v] for k, v in params.items()}
        runner = WalkForwardHPO(cfg1)
        result = runner.run()

        score = _score_from_rows_or_summary(
            rows=result.get("rows", []),
            summary=result.get("oos_summary", {}),
            objective=cfg1.get("objective", {}),
            target=target
        )
        trial.set_user_attr("params", params)
        trial.set_user_attr("oos_summary", result.get("oos_summary", {}))
        return float(score)

    study.optimize(objective, n_trials=trials, show_progress_bar=False)

    best = study.best_trial
    best_params = best.user_attrs.get("params", {})
    study_summary = {
        "best_value": float(best.value),
        "best_params": best_params,
        "direction": direction,
        "target": target,
        "trials": trials,
        "oos_summary_of_best": best.user_attrs.get("oos_summary", {})
    }
    return best_params, study_summary


# ---------------- Neighborhood space builder ----------------

def _is_number(x) -> bool:
    try:
        float(x)
        return True
    except Exception:
        return False

def _center(values: List[Any]) -> float:
    vals = [float(v) for v in values]
    vals.sort()
    n = len(vals)
    if n == 0: return 0.0
    return vals[n//2] if n % 2 == 1 else 0.5*(vals[n//2-1] + vals[n//2])

def _build_neighbor_space(merged_grid: Dict[str, List[Any]],
                          champions: List[Dict[str, Any]],
                          scale: float = 0.20,
                          clamp_to_merged: bool = True) -> Dict[str, Dict[str, Any]]:
    """
    基于合并网格 merged_grid 与冠军参数列表 champions（来自 best_params_by_fold），构造 Optuna 搜索空间
    """
    space: Dict[str, Dict[str, Any]] = {}
    champ_params = champions or []
    champ_by_key: Dict[str, List[Any]] = {}
    for c in champ_params:
        for k, v in (c or {}).items():
            champ_by_key.setdefault(k, []).append(v)

    for k, merged_vals in merged_grid.items():
        cvals = champ_by_key.get(k, [])
        if merged_vals and all(_is_number(v) for v in merged_vals):
            merged_nums = [float(v) for v in merged_vals]
            lo_m, hi_m = min(merged_nums), max(merged_nums)
            if cvals and all(_is_number(v) for v in cvals):
                ctr = _center(cvals)
                span = max(abs(ctr), (hi_m - lo_m) or 1.0)
                lo = ctr - scale * span
                hi = ctr + scale * span
            else:
                span = (hi_m - lo_m) or max(abs(lo_m), 1.0)
                lo = lo_m - scale * span
                hi = hi_m + scale * span
            if clamp_to_merged:
                lo = max(lo, lo_m - 2*abs(hi_m - lo_m))
                hi = min(hi, hi_m + 2*abs(hi_m - lo_m))
            is_int = all(abs(float(v) - int(float(v))) < 1e-9 for v in merged_nums)
            if is_int:
                space[k] = {"type":"int", "low": int(math.floor(lo)), "high": int(math.ceil(hi)), "step": 1}
            else:
                space[k] = {"type":"float", "low": float(lo), "high": float(hi), "log": False}
        else:
            choices = []
            for v in cvals:
                if v not in choices:
                    choices.append(v)
            for v in merged_vals or []:
                if v not in choices:
                    choices.append(v)
            space[k] = {"type":"categorical", "choices": choices[:16]}
    return space

def _ch_enabled(cfg: Dict[str, Any]) -> bool:
    return bool(cfg.get("sink_clickhouse", {}).get("enabled"))

def _ch_client(cfg: Dict[str, Any]):
    try:
        import clickhouse_connect  # type: ignore
    except Exception as e:
        raise RuntimeError("clickhouse-connect not installed. `pip install clickhouse-connect`") from e
    kw = cfg.get("sink_clickhouse", {})
    host = kw.get("host", "localhost")
    port = kw.get("port", 8123)
    username = kw.get("username", "default")
    password = kw.get("password", "")
    database = kw.get("database", "default")
    client = clickhouse_connect.get_client(host=host, port=port, username=username, password=password, database=database)
    return client

def _ch_ensure_tables(client) -> None:
    client.command("""
    CREATE TABLE IF NOT EXISTS alpha_runs (
      gen UInt16,
      mode String,
      stage String,
      ts DateTime DEFAULT now(),
      oos_avg_sharpe Float64,
      oos_avg_total_return Float64,
      oos_avg_max_drawdown Float64,
      oos_avg_win_rate Float64,
      oos_avg_profit_factor Float64,
      oos_avg_sortino Float64,
      oos_avg_omega Float64,
      oos_avg_calmar Float64,
      oos_trades UInt32
    ) ENGINE = MergeTree ORDER BY (gen, stage, ts)
    """)
    client.command("""
    CREATE TABLE IF NOT EXISTS alpha_rows (
      gen UInt16,
      phase LowCardinality(String),
      fold_id UInt16,
      params String,
      sharpe Float64,
      total_return Float64,
      max_drawdown Float64,
      win_rate Float64,
      profit_factor Float64,
      sortino Float64,
      omega Float64,
      calmar Float64,
      trades UInt32,
      origin_label LowCardinality(String) DEFAULT 'NA',
      ts DateTime DEFAULT now()
    ) ENGINE = MergeTree ORDER BY (gen, phase, fold_id, ts)
    """)
    client.command("""
    CREATE TABLE IF NOT EXISTS alpha_equity (
      gen UInt16,
      stage String,
      fold_id UInt16,
      ts_ms UInt64,
      equity Float64
    ) ENGINE = MergeTree ORDER BY (gen, stage, fold_id, ts_ms)
    """)

def _ch_insert_stage_runs(client, gen: int, mode: str, stage: str, oos_summary: Dict[str, Any]) -> None:
    row = [
        gen,
        mode,
        stage,
        float(oos_summary.get("oos_avg_sharpe", 0.0)),
        float(oos_summary.get("oos_avg_total_return", 0.0)),
        float(oos_summary.get("oos_avg_max_drawdown", 0.0)),
        float(oos_summary.get("oos_avg_win_rate", 0.0)),
        float(oos_summary.get("oos_avg_profit_factor", 0.0)),
        float(oos_summary.get("oos_avg_sortino", 0.0)),
        float(oos_summary.get("oos_avg_omega", 0.0)),
        float(oos_summary.get("oos_avg_calmar", 0.0)),
        int(oos_summary.get("oos_trades", 0)),
    ]
    client.insert(
        "alpha_runs",
        [row],
        column_names=[
            "gen","mode","stage",
            "oos_avg_sharpe","oos_avg_total_return","oos_avg_max_drawdown",

            "oos_avg_omega","oos_avg_calmar","oos_trades"
        ],
    )

def _ch_insert_stage_rows(client, gen: int, rows: List[Dict[str, Any]], origin_map: Dict[str, Dict[Any, str]] | None) -> None:
    payload = []
    for r in rows:
        params_raw = r.get("params")
        try:
            p = params_raw if isinstance(params_raw, str) else json.dumps(params_raw, ensure_ascii=False)
        except Exception:
            p = "{}"
        # 计算来源标签（若提供 origin_map）
        label = "NA"
        if origin_map:
            try:
                pdict = json.loads(params_raw) if isinstance(params_raw, str) else (params_raw or {})
                label = _label_row_origin(pdict or {}, origin_map)
            except Exception:
                label = "NA"
        payload.append([
            int(r.get("fold_id", 0)),
            str(r.get("phase", "IS")),
            p,
            float(r.get("sharpe", 0.0)),
            float(r.get("total_return", 0.0)),
            float(r.get("max_drawdown", 0.0)),
            float(r.get("win_rate", 0.0)),
            float(r.get("profit_factor", 0.0)),
            float(r.get("sortino", 0.0)),
            float(r.get("omega", 0.0)),
            float(r.get("calmar", 0.0)),
            int(r.get("trades", 0)),
            label
        ])
    if payload:
        # prepend gen per row
        rows_insert = [[gen] + row for row in payload]
        client.insert(
            "alpha_rows",
            rows_insert,
            column_names=[
                "gen","fold_id","phase","params","sharpe","total_return","max_drawdown",
                "win_rate","profit_factor","sortino","omega","calmar","trades","origin_label"
            ],
        )

def _ch_insert_stage_equity(client, gen: int, stage: str, oos_equity_curves: List[Dict[str, Any]]) -> None:
    eq_rows = []
    for payload in (oos_equity_curves or []):
        fold_id = int(payload.get("fold_id", 0))
        for ts_ms, eq in payload.get("equity_curve", []):
            eq_rows.append([gen, stage, fold_id, int(ts_ms), float(eq)])
    if eq_rows:
        client.insert(
            "alpha_equity",
            eq_rows,
            column_names=["gen","stage","fold_id","ts_ms","equity"],
        )

def _persist_stage_to_clickhouse(cfg: Dict[str, Any],
                                 gen: int,
                                 mode: str,
                                 stage: str,
                                 result: Dict[str, Any],
                                 origin_map: Dict[str, Dict[Any, str]] | None = None) -> None:
    """
    将某个 stage 的结果写入 ClickHouse：alpha_runs / alpha_rows / alpha_equity
    """
    if not _ch_enabled(cfg):
        return
    client = _ch_client(cfg)
    _ch_ensure_tables(client)

    # runs
    _ch_insert_stage_runs(client, gen, mode, stage, result.get("oos_summary", {}) or {})

    # rows
    _ch_insert_stage_rows(client, gen, result.get("rows", []), origin_map)

    # equity curves
    _ch_insert_stage_equity(client, gen, stage, result.get("oos_equity_curves", []))


# ---------------- 主循环（包含 coop_optuna 两阶段 + ClickHouse Sink） ----------------

def run_loop(cfg: Dict[str, Any], out_dir: str, gens: int, k_top: int, provider: str) -> None:
    _ensure_dir(out_dir)
    base_hpo_cfg = cfg["hpo"]
    objective_key = base_hpo_cfg.get("objective", {}).get("metric", "sharpe")

    history: List[Dict[str, Any]] = []

    for g in range(1, gens+1):
        gen_dir = os.path.join(out_dir, f"gen_{g:02d}")
        _ensure_dir(gen_dir)

        # ====== 两阶段：先 COOP 再 Optuna 细化 ======
        if provider == "coop_optuna":
            hpo_cfg = deepcopy(base_hpo_cfg)

            grid_base = deepcopy(hpo_cfg["grid"])
            prev_rows = history[-1]["rows"] if history else []
            top_is = _topk_rows([r for r in prev_rows if r.get("phase") == "IS"], k=min(k_top, len(prev_rows)), key=objective_key)

            context = {
                "symbols": hpo_cfg["symbols"],
                "objective": hpo_cfg.get("objective", {}),
                "top_is": top_is,
                "base_grid": grid_base,
                "notes": "Coop stage: propose refined grid (<=5 values/param), numeric around winners; prune weak."
            }

            # LLM 提案
            grid_llm, rationale = _llm_propose_grid(context, model=cfg.get("llm", {}).get("model", "gpt-4o-mini"))
            _dump_json({"llm_rationale": rationale, "llm_grid": grid_llm}, os.path.join(gen_dir, "llm_refine.json"))

            # ML 提案
            grid_ml, ml_importances = _ml_suggest(top_is if top_is else prev_rows, target_key=objective_key, n=cfg.get("ml", {}).get("n_points", 24))
            _dump_json({"ml_importances": ml_importances, "ml_grid": grid_ml}, os.path.join(gen_dir, "ml_refine.json"))

            # 合并 + 来源标注
            cap = int(cfg.get("coop", {}).get("cap_per_param", 5))
            merged_grid, origin_map = _merge_grids(grid_base, grid_llm, grid_ml, cap_per_param=cap)
            hpo_cfg["grid"] = merged_grid
            _dump_json({"merged_grid": merged_grid, "origin_map": origin_map}, os.path.join(gen_dir, "coop_merged.json"))

            # —— Stage 1：COOP 网格跑 Walk-Forward HPO
            runner1 = WalkForwardHPO(hpo_cfg)
            result1 = runner1.run()
            artifacts1 = runner1.export_tables(result1, gen_dir)
            try:
                plots1 = runner1.export_plots(result1, gen_dir)
                artifacts1.update(plots1)
            except Exception as e:
                artifacts1["plot_error"] = str(e)

            # 标注来源可视化 + ClickHouse 入库（stage1）
            try:
                extra1 = _export_origin_tables_and_plot(result1.get("rows", []), origin_map, gen_dir)
                artifacts1.update(extra1)
            except Exception:
                pass
            _persist_stage_to_clickhouse(cfg, gen=g, mode="coop_optuna", stage="stage1", result=result1, origin_map=origin_map)

            # —— 邻域：以冠军参数为中心构造 Optuna 搜索空间
            champions = [bp.get("params", {}) for bp in result1.get("best_params_by_fold", [])]
            neighbor_scale = float(cfg.get("optuna", {}).get("neighbor_scale", 0.20))
            space = _build_neighbor_space(merged_grid, champions, scale=neighbor_scale, clamp_to_merged=True)
            _dump_json({"optuna_neighbor_space": space}, os.path.join(gen_dir, "optuna_space.json"))

            # —— Stage 2：在邻域里 Optuna 细化（单点 Walk-Forward 评估）
            best_params, study_summary = _run_optuna_with_space(base_hpo_cfg, space, cfg.get("optuna", {}) or {})

            # 用最佳参数再跑一次生成最终产物
            hpo_final = deepcopy(base_hpo_cfg)
            hpo_final["grid"] = {k: [v] for k, v in best_params.items()}
            runner2 = WalkForwardHPO(hpo_final)
            result2 = runner2.run()
            artifacts2 = runner2.export_tables(result2, gen_dir)
            try:
                plots2 = runner2.export_plots(result2, gen_dir)
                artifacts2.update(plots2)
            except Exception as e:
                artifacts2["plot_error"] = str(e)

            # ClickHouse 入库（stage2）
            _persist_stage_to_clickhouse(cfg, gen=g, mode="coop_optuna", stage="stage2", result=result2, origin_map=None)

            payload = {
                "gen": g,
                "mode": "coop_optuna",
                "stage1": {
                    "oos_summary": result1.get("oos_summary", {}),
                    "artifacts": artifacts1
                },
                "stage2": {
                    "best_params": best_params,
                    "study_summary": study_summary,
                    "oos_summary": result2.get("oos_summary", {}),
                    "artifacts": artifacts2
                },
                "rows": result1.get("rows", [])  # 保留 Stage1 明细
            }
            _dump_json(payload, os.path.join(gen_dir, "gen_summary.json"))
            history.append(payload)
            continue

        # ====== 其他 provider（简化：直接跑一代 & 入库）======
        hpo_cfg = deepcopy(base_hpo_cfg)
        runner = WalkForwardHPO(hpo_cfg)
        result = runner.run()
        artifacts = runner.export_tables(result, gen_dir)
        try:
            plots = runner.export_plots(result, gen_dir)
            artifacts.update(plots)
        except Exception as e:
            artifacts["plot_error"] = str(e)
        payload = {
            "gen": g,
            "mode": provider,
            "oos_summary": result.get("oos_summary", {}),
            "rows": result.get("rows", []),
            "artifacts": artifacts
        }
        _dump_json(payload, os.path.join(gen_dir, "gen_summary.json"))
        history.append(payload)
        # 入库：作为 single 阶段
        _persist_stage_to_clickhouse(cfg, gen=g, mode=provider, stage="single", result=result, origin_map=None)

    # 顶层索引
    idx = {"generations": [
        {
            "gen": p["gen"],
            "mode": p["mode"],
            "stage1": p.get("stage1", {}),
            "stage2": p.get("stage2", {}),
            "oos_summary": p.get("oos_summary", {}),
            "artifacts": p.get("artifacts", {})
        } for p in history
    ]}
    _dump_json(idx, os.path.join(out_dir, "index.json"))
```
