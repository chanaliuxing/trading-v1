## P0（立刻补）：数据与回测可信度

1. **时点一致 & 泄漏防线**

   * 确保任何特征 `features.ts` ≤ 交易决策时刻；训练/验证之间加 **embargo**（时间隔离）。
   * 行业常见翻车点：复权处理、成分生存偏差、新闻/财报时间戳错位、盘后数据误用。

**落地代码：泄漏/embargo 工具（`leakage_guard.py`）**

```python
# leakage_guard.py
from __future__ import annotations
import pandas as pd

def enforce_point_in_time(df: pd.DataFrame, ts_col="ts", cutoff_col="feature_cutoff_ts") -> pd.DataFrame:
    """
    只保留 ts >= feature_cutoff_ts 的行；若没有 cutoff_col，默认放行。
    你可以在落库/抽取特征时把 'feature_cutoff_ts' 一并写入，指明该行能被使用的最早时刻。
    """
    if cutoff_col in df.columns:
        return df[df[ts_col] >= df[cutoff_col]].copy()
    return df

def apply_embargo(splits: list[tuple[pd.Timestamp, pd.Timestamp]], embargo: pd.Timedelta):
    """
    对 (IS/OOS) 划分应用时间隔离：在每个 IS 之后空出 embargo 再开始 OOS。
    splits: [(is_start, is_end, oos_start, oos_end)] -> 返回新元组
    """
    out = []
    for (is_s, is_e, oos_s, oos_e) in splits:
        new_oos_s = max(oos_s, is_e + embargo)
        out.append((is_s, is_e, new_oos_s, oos_e))
    return out
```

> 集成点：在 `BacktestEngine._load_data()` 之后、`WalkForwardHPO` 生成折分之后，各调用一次。

**落地代码：`sim_adapter.py` 增强片段（替换/补充到你的 `SimAdapter`）**

```python
# --- in services/backtest/sim_adapter.py ---
import math

class SimAdapter:
    def __init__(self, commission_per_share=0.005, commission_min=1.0,
                 slippage_bps=1.0, dynamic_spread_bps=0.0,
                 borrow_rate=0.0,  # 年化借券费率（做空或保证金融资）
                 fill_ref="open"):
        self.commission_per_share = commission_per_share
        self.commission_min = commission_min
        self.slippage_bps = slippage_bps
        self.dynamic_spread_bps = dynamic_spread_bps
        self.borrow_rate = borrow_rate
        self.fill_ref = fill_ref
        self.cash = 100_000.0
        self.positions = {}  # symbol -> qty, avg_price
        self.ledger = []     # fills/fees/pnl events

    def _price_with_slippage(self, ref_px: float, side: str, spread_bps_now: float = 0.0) -> float:
        bps = self.slippage_bps + spread_bps_now
        adj = ref_px * (bps / 1e4)
        return ref_px + (adj if side.upper()=="BUY" else -adj)

    def _commission(self, qty: int) -> float:
        return max(self.commission_min, abs(qty) * self.commission_per_share)

    def _borrow_fee_increment(self, notional: float, dt_days: float) -> float:
        # 连续复利近似：fee = notional * (rate/365)*dt
        return abs(notional) * (self.borrow_rate/365.0) * dt_days

    async def submit_order(self, order):
        """
        支持：
         - 市价：按 open/close 参考价 ± 滑点
         - 限价：若 ref_px 优于/等于限价则立即成交；否则挂单并尝试部分成交
        order: {symbol, side, qty, type: 'MKT'|'LMT', limit_price:float|None, ts:int, bar:{open,close,high,low,spread_bps?}}
        """
        sym = order["symbol"]; side = order["side"].upper(); qty = int(order["qty"])
        bar = order.get("bar", {})  # 当前K线
        ref_px = float(bar.get(self.fill_ref, bar.get("open") or bar.get("close")))
        spread_bps_now = float(bar.get("spread_bps", 0.0)) * self.dynamic_spread_bps
        if order.get("type","MKT") == "MKT":
            fill_px = self._price_with_slippage(ref_px, side, spread_bps_now)
            fill_qty = qty
        else:
            lmt = float(order["limit_price"])
            # 简易撮合：看能否触及；部分成交按区间比例
            hi, lo = float(bar.get("high", ref_px)), float(bar.get("low", ref_px))
            touched = (side=="BUY" and lo <= lmt) or (side=="SELL" and hi >= lmt)
            if not touched:
                return  # 留在挂单簿（简化：此处直接不成交）
            # 价格：向有利方向最小化冲击
            exec_px = min(lmt, ref_px) if side=="BUY" else max(lmt, ref_px)
            fill_px = self._price_with_slippage(exec_px, side, spread_bps_now)
            # 部分成交比例（区间越长，拍到的可能越大；此处线性近似）
            rng = max(1e-9, hi - lo)
            touch_dist = abs((lmt - ref_px) / rng)
            pct = max(0.2, min(1.0, 1.0 - touch_dist))
            fill_qty = max(1, int(qty * pct))

        fee = self._commission(fill_qty)
        notional = fill_px * fill_qty * (1 if side=="BUY" else -1)
        self.cash -= notional + fee  # 买入减少现金，卖出增加现金（fee 总是扣）
        pos = self.positions.get(sym, {"qty":0, "avg":0.0})
        if side=="BUY":
            new_qty = pos["qty"] + fill_qty
            pos["avg"] = (pos["avg"]*pos["qty"] + fill_px*fill_qty) / max(1, new_qty)
            pos["qty"] = new_qty
        else:
            # 卖出/做空（允许负仓）
            new_qty = pos["qty"] - fill_qty
            pos["qty"] = new_qty
            if pos["qty"] == 0:
                pos["avg"] = 0.0
        self.positions[sym] = pos
        self.ledger.append({"ts": order.get("ts"), "symbol": sym, "side": side, "qty": fill_qty, "price": fill_px, "fee": fee})

    async def on_market_data(self, features: dict):
        """逐bar滚动：按当前价 MTM，并计提借券费（若仓位为负或融资有额外规则）。"""
        sym = features["symbol"]; px = float(features.get("close", 0.0))
        ts = features.get("ts")
        for s, pos in list(self.positions.items()):
            if s != sym: continue
            # 计提借券费（简单按 1 bar ≈ dt_days）
            dt_days = 1/ (6.5*60) if features.get("bar_interval") == "1m" else 1/252
            notional = pos["qty"] * px
            fee = self._borrow_fee_increment(notional if pos["qty"]<0 else 0.0, dt_days)
            if fee:
                self.cash -= fee
                self.ledger.append({"ts": ts, "symbol": s, "side": "FEE", "qty": 0, "price": 0.0, "fee": fee})
```

3. **统计稳健性：PSR / Deflated Sharpe / PBO**

   * 把“看起来很高的 Sharpe”去水分，避免过拟合被误选。
   * 这些指标也能成为 **objective** 的变量（例如 `0.5*psr + 0.5*sortino - 2*mdd`）。

**落地代码：`metrics_extra.py`（在 `wf_hpo` 中调用并写入 rows & oos_summary）**

```python
# metrics_extra.py
import numpy as np

def probabilistic_sharpe_ratio(sharpe: float, n: int, ref: float = 0.0) -> float:
    """
    PSR (Bailey): P( SR > ref ); 近似：Z = (SR - ref) * sqrt(n - 1); PSR = Phi(Z)
    n 为独立样本数（用交易数或天数近似），ref 通常取 0 或 0.5
    """
    if n <= 1: return 0.0
    z = (sharpe - ref) * np.sqrt(max(1, n-1))
    from math import erf, sqrt
    return 0.5 * (1.0 + erf(z / np.sqrt(2)))

def deflated_sharpe(sharpe: float, n: int, trials: int) -> float:
    """
    DSR：考虑多重检验(trials)后的折扣；简化版：SR / (1 + log(trials))
    更严格版本可引入极值理论，出于篇幅此处给轻量近似。
    """
    if trials <= 1: return sharpe
    return sharpe / (1.0 + np.log(max(2, trials)))

def pbo_fraction(oos_sharpes: list[float]) -> float:
    """
    Probability of Backtest Overfitting：OOS 表现低于IS最优的概率近似度量。
    简化：统计折间 OOS Sharpe < 0 的比例。
    """
    if not oos_sharpes: return 0.0
    arr = np.array(oos_sharpes)
    return float(np.mean(arr < 0.0))
```

> 集成点：在 `WalkForwardHPO._aggregate_oos()` 里：
>
> * `oos_summary["psr"] = probabilistic_sharpe_ratio(oos_avg_sharpe, n_days)`
> * `oos_summary["dsr"] = deflated_sharpe(oos_avg_sharpe, trials=len(rows))`
> * `oos_summary["pbo"] = pbo_fraction([fold_sharpe...])`
>   并把 `psr/dsr/pbo` 同步加入 **objective 可用变量**。

```python
# grid_guard.py
from pydantic import BaseModel, Field, ValidationError, conlist, confloat, conint
from typing import Dict, List, Any

class GridModel(BaseModel):
    grid: Dict[str, conlist(Any, min_items=1, max_items=5)] = Field(default_factory=dict)

def sanitize_grid(obj: dict, bounds: dict[str, tuple[float,float]]|None=None) -> dict:
    try:
        m = GridModel(**obj)
    except ValidationError as e:
        # 直接抛错或回退为空
        return {}
    grid = {}
    for k, vals in m.grid.items():
        out = []
        for v in vals:
            if isinstance(v, (int,float)) and bounds and k in bounds:
                lo, hi = bounds[k]
                v = max(lo, min(hi, float(v)))
            out.append(v)
        grid[k] = out
    return grid
```

```python
from __future__ import annotations
import pandas as pd

def enforce_point_in_time(df: pd.DataFrame,
                          ts_col: str = "ts",
```

## 2) 稳健评估指标（PSR / DSR / PBO）

### 新增文件：`metrics_extra.py`

```python
# metrics_extra.py
from __future__ import annotations
import numpy as np
from math import erf, sqrt, log

def probabilistic_sharpe_ratio(sharpe: float, n: int, ref: float = 0.0) -> float:
    """
    PSR (Bailey) 近似：Z = (SR - ref) * sqrt(n - 1)；PSR = Phi(Z)
    n: 独立样本数（可用交易数、日数近似；别用 bar 数过大）
    """
    if n <= 1:
        return 0.0
    z = (sharpe - ref) * np.sqrt(max(1, n - 1))
    return 0.5 * (1.0 + erf(z / sqrt(2)))

def deflated_sharpe(sharpe: float, n: int, trials: int) -> float:
    """
    DSR：考虑多重检验的折扣，轻量近似：SR / (1 + log(trials))
    更严谨可用 EVT（篇幅所限，这里给稳健保守近似）
    """
    if trials <= 1:
        return sharpe
    return sharpe / (1.0 + log(max(2, trials)))

def pbo_fraction(oos_sharpes: list[float]) -> float:
    """
    概率式过拟合（简化版）：统计 OOS Sharpe < 0 的比例，作为保守 PBO proxy。
    """
    if not oos_sharpes:
        return 0.0
    arr = np.array(oos_sharpes, dtype=float)
    return float(np.mean(arr < 0.0))
```

## 5) 为什么先做这两块？

* **防泄漏与隔离**是“真金白银”的**护城河**：没它，再强的 LLM/ML 都是在“错的地基”上优化。

* 一旦这些打牢，你的 **LLM+ML+Optuna 两阶段**会更稳，Grafana/Streamlit 的趋势也更可信。

## 新增：`grid_guard.py`（完整可用）

```python
# grid_guard.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Any, Dict, List, Tuple, Optional
import math
import re
import json

Number = float | int

@dataclass
class GuardConfig:
    cap_per_param: int = 5
    # 基础约束
    bounds: Dict[str, Tuple[Number, Number]] = None         # {"ema_short":[3,50], ...}
    types: Dict[str, str] = None                             # {"ema_short":"int"|"float"|"cat"}
    choices: Dict[str, List[Any]] = None                     # {"side":["long","short","both"]}
    # 关系约束（形如 "ema_short < ema_long", "stop_loss_bps >= 0"）
    relations: List[str] = None
    # 条件约束（例如：use_filter==0 则删除某些参数）
    conditionals: List[Dict[str, Any]] = None                # [{"if":"use_filter==0","drop":["filter_thr","filter_len"]}]
    # 允许的参数白名单（可选）
    allow_params: Optional[List[str]] = None
    # 额外：将整数参数按 step 对齐
    steps: Dict[str, int] = None                             # {"atr_n": 1, "ema_short": 1}

def _coerce_type(v: Any, t: str) -> Any:
    if t == "int":
        try:
            return int(round(float(v)))
        except Exception:
            return None
    if t == "float":
        try:
            return float(v)
        except Exception:
            return None
    # cat: 原样
    return v

def _clip_bounds(v: Any, lo: Number, hi: Number, t: str) -> Any:
    try:
        x = float(v)
        x = max(lo, min(hi, x))
        if t == "int":
            return int(round(x))
        if t == "float":
            return float(x)
        return v
    except Exception:
        return None

def _apply_step(v: Any, step: int, t: str) -> Any:
    if t != "int" or step is None or step <= 1:
        return v
    try:
        return int(round(int(v) / step) * step)
    except Exception:
        return v

def _parse_rel(rel: str) -> Tuple[str, str, str]:
    # 支持: <, <=, >, >=, ==, !=
    m = re.match(r"^\s*([A-Za-z0-9_]+)\s*(<=|>=|<|>|==|!=)\s*([A-Za-z0-9_\.]+)\s*$", rel)
    if not m:
        raise ValueError(f"Bad relation: {rel}")
    return m.group(1), m.group(2), m.group(3)

def _relation_prune(grid: Dict[str, List[Any]], rels: List[str], types: Dict[str, str]) -> Tuple[Dict[str, List[Any]], Dict[str, Any]]:
    """
    用“边界剪枝”处理二元关系：比如 a < b
    我们把 a 的候选值裁掉 >= max(b)，把 b 的候选值裁掉 <= min(a) ；（保守近似，能显著减少非法组合）
    """
    report = {"relations_applied": [], "pruned": {}}
    g = {k: list(dict.fromkeys(v)) for k, v in grid.items()}  # 去重
    for expr in (rels or []):
        try:
            L, op, R = _parse_rel(expr)
        except Exception:
            continue
        if L not in g or R not in g:
            continue
        tL = (types or {}).get(L, "float")
        tR = (types or {}).get(R, "float")

        left_vals  = g[L][:]
        right_vals = g[R][:]

        # 仅对数值型关系做剪枝
        if tL in ("int","float") and tR in ("int","float"):
            try:
                lnums = sorted(float(x) for x in left_vals)
                rnums = sorted(float(x) for x in right_vals)
            except Exception:
                continue

            before = (len(left_vals), len(right_vals))

            def keep_left(x):
                if op == "<":  return x < max(rnums)
                if op == "<=": return x <= max(rnums)
                if op == ">":  return x > min(rnums)
                if op == ">=": return x >= min(rnums)
                if op == "==": return min(rnums) <= x <= max(rnums)
                if op == "!=": return True  # 边界剪枝难处理，全留给组合层
                return True

            def keep_right(y):
                if op == "<":  return y > min(lnums)
                if op == "<=": return y >= min(lnums)
                if op == ">":  return y < max(lnums)
                if op == ">=": return y <= max(lnums)
                if op == "==": return min(lnums) <= y <= max(lnums)
                if op == "!="": return True
                return True

            left_vals  = [v for v in left_vals  if keep_left (float(v))]
            right_vals = [v for v in right_vals if keep_right(float(v))]

            g[L] = left_vals
            g[R] = right_vals
            after = (len(left_vals), len(right_vals))
            report["relations_applied"].append(expr)
            if before != after:
                report["pruned"][expr] = {"before": before, "after": after}
        else:
            # 非数值关系先跳过（可以在 HPO 内部组合层过滤）
            continue
    return g, report

def _apply_conditionals(grid: Dict[str, List[Any]], conds: List[Dict[str, Any]]) -> Tuple[Dict[str, List[Any]], Dict[str, Any]]:
    """
    支持简单 'if': 'param==value' / 'param!=value'
    满足条件时 drop 指定参数
    """
    rep = {"conditionals": []}
    g = {k: list(v) for k, v in grid.items()}
    for c in (conds or []):
        expr = str(c.get("if","")).strip()
        drop = c.get("drop", [])
        m_eq = re.match(r"^\s*([A-Za-z0-9_]+)\s*==\s*([A-Za-z0-9_\.-]+)\s*$", expr)
        m_ne = re.match(r"^\s*([A-Za-z0-9_]+)\s*!=\s*([A-Za-z0-9_\.-]+)\s*$", expr)
        def _satisfied():
            if not m_eq and not m_ne: 
                return False
            p = (m_eq or m_ne).group(1)
            val = (m_eq or m_ne).group(2)
            # 只要 grid 中该参数可能取到该值，就认为条件“可能满足”，保守地执行 drop
            return p in g and any(str(x)==val for x in g[p])
        if _satisfied():
            for d in drop:
                if d in g:
                    del g[d]
            rep["conditionals"].append({"if": expr, "drop": drop})
    return g, rep

def sanitize_and_constrain_grid(
    grid: Dict[str, List[Any]],
    guard_cfg: Dict[str, Any] | None
) -> Tuple[Dict[str, List[Any]], Dict[str, Any]]:
    """
    入口：对 LLM/ML/合并网格 施加全套护栏
    返回 (sanitized_grid, report)
    """
    guard = GuardConfig(
        cap_per_param = (guard_cfg or {}).get("cap_per_param", 5),
        bounds        = (guard_cfg or {}).get("bounds") or {},
        types         = (guard_cfg or {}).get("types") or {},
        choices       = (guard_cfg or {}).get("choices") or {},
        relations     = (guard_cfg or {}).get("relations") or [],
        conditionals  = (guard_cfg or {}).get("conditionals") or [],
        allow_params  = (guard_cfg or {}).get("allow_params"),
        steps         = (guard_cfg or {}).get("steps") or {},
    )

    report: Dict[str, Any] = {"dropped": {}, "coerced": {}, "clipped": {}, "choices": {}, "steps": {}, "cap": guard.cap_per_param}

    # 1) 仅保留白名单（如配置）
    g = {}
    for k, vals in (grid or {}).items():
        if guard.allow_params and k not in guard.allow_params:
            report["dropped"].setdefault(k, []).extend(vals)
            continue
        g[k] = list(vals) if isinstance(vals, list) else [vals]

    # 2) 类型、边界、choices、step、去重与截断
    out: Dict[str, List[Any]] = {}
    for k, vals in g.items():
        t = guard.types.get(k, "float" if k not in (guard.choices or {}) else "cat")
        lo, hi = None, None
        if k in (guard.bounds or {}):
            lo, hi = guard.bounds[k]
        cho = (guard.choices or {}).get(k)
        step = (guard.steps or {}).get(k)

        cleaned = []
        for v in vals:
            vv = _coerce_type(v, t)
            if vv is None:
                report["dropped"].setdefault(k, []).append(v)
                continue
            if cho is not None:
                if vv not in cho:
                    report["dropped"].setdefault(k, []).append(vv)
                    continue
                report["choices"].setdefault(k, []).append(vv)
            if lo is not None and hi is not None and t in ("int","float"):
                v_before = vv
                vv = _clip_bounds(vv, lo, hi, t)
                if vv is None: 
                    report["dropped"].setdefault(k, []).append(v_before)
                    continue
                if v_before != vv:
                    report["clipped"].setdefault(k, []).append({"from": v_before, "to": vv})
            if step is not None:
                v_before = vv
                vv = _apply_step(vv, step, t)
                if v_before != vv:
                    report["steps"].setdefault(k, []).append({"from": v_before, "to": vv})
            cleaned.append(vv)

        # 去重并截断
        uniq = []
        for x in cleaned:
            if x not in uniq:
                uniq.append(x)
        out[k] = uniq[:guard.cap_per_param]

    # 3) 条件与关系约束剪枝
    out, rep_cnd = _apply_conditionals(out, guard.conditionals)
    report.update(rep_cnd)

    out, rep_rel = _relation_prune(out, guard.relations or [], guard.types or {})
    report.update(rep_rel)

    return out, report
```

### 这样做的直接收益

* **稳定性**：LLM 和 ML 的“野值/脏值”不再污染搜索；网格规模可控；

* **可审计**：每代生成三份 `*_guard.json`，明确记录**丢了哪些值/为什么丢**；
* **可扩展**：以后你若加新的参数，只要在 `guard.bounds/types/relations` 里补一条即可。
