```python
# services/backtest/sim_adapter.py
from __future__ import annotations
from typing import Dict, Any, Optional
import time
import math

class _NullOverlay:
    def adjust_order(self, symbol: str, side: str, qty: int, ref_px: float) -> int:
        return qty
    def on_fill(self, symbol: str, side: str, qty: int, price: float):  # 可选
        pass
    @property
    def halted(self) -> bool:
        return False
    @property
    def state(self):
        class S: peak_nav = 0.0; halted = False
        return S()

class SimAdapter:
    """
    生产级成本模型（供回测/HPO使用）：
      - 手续费：每股 + 最低费用
      - 滑点：固定 bps + 动态 spread bps
      - 部分成交：受参与率cap与bar成交量限制
      - 借券费用：对空头按年化费率分摊到bar
      - Overlay：下单前裁剪；成交后回调
    """
    def __init__(
        self,
        starting_cash: float = 1_000_000.0,
        commission_per_share: float = 0.005,
        commission_min: float = 1.0,
        slippage_bps: float = 1.0,
        dynamic_spread_bps: float = 0.0,
        borrow_rate: float = 0.0,            # 年化（例如 0.08 = 8%）
        participation_cap: float = 0.1,      # 单bar最大参与率（相对bar volume）
        fill_ref: str = "open",              # "open" | "close"
        overlay: Optional[object] = None,
        slo: Optional[object] = None,
    ):
        self.cash = float(starting_cash)
        self.positions: Dict[str, Dict[str, float]] = {}  # symbol -> {qty, avg_cost, last_px, last_ts}
        self.ledger: list[dict] = []
        self._last_features: Dict[str, dict] = {}
        self.commission_per_share = float(commission_per_share)
        self.commission_min = float(commission_min)
        self.slippage_bps = float(slippage_bps)
        self.dynamic_spread_bps = float(dynamic_spread_bps)
        self.borrow_rate = float(borrow_rate)
        self.participation_cap = float(participation_cap)
        self.fill_ref = fill_ref
        self.overlay = overlay or _NullOverlay()
        self.slo = slo

    # ---------- 市场数据推进 ----------
    async def on_market_data(self, features: dict):
        sym = str(features.get("symbol"))
        ts = int(features.get("ts", 0))
        self._last_features[sym] = features
        px = float(features.get("close", features.get("open", 0.0)))
        if sym not in self.positions:
            self.positions[sym] = {"qty": 0.0, "avg_cost": 0.0, "last_px": px, "last_ts": ts}
            return
        pos = self.positions[sym]
        # 借券费（空头）
        dt_sec = max(0, (ts - int(pos.get("last_ts", ts))) / 1000.0)
        if dt_sec > 0 and self.borrow_rate > 0 and pos["qty"] < 0:
            # 连续复利近似：费率 * 价值 * 时间
            notional = abs(pos["qty"]) * pos.get("last_px", px)
            borrow_fee = notional * (self.borrow_rate) * (dt_sec / (365*24*3600))
            self.cash -= borrow_fee
            self.ledger.append({"ts": ts, "symbol": sym, "side": "BORROW", "qty": 0, "price": 0.0, "fee": float(borrow_fee)})
        pos["last_px"] = px
        pos["last_ts"] = ts

    # ---------- 下单/成交 ----------
    async def submit_order(self, order: dict):
        """
        order: {symbol, side(BUY/SELL), qty, type(MKT/LMT), limit_price?, ts?}
        """
        if getattr(self.overlay, "halted", False):
            if self.slo:
                try: self.slo.record_order_failure()
                except Exception: pass
            return

        sym = str(order["symbol"])
        side = str(order["side"]).upper()
        req_qty = int(order.get("qty", 0))
        if req_qty <= 0:
            if self.slo:
                try: self.slo.record_order_failure()
                except Exception: pass
            return

        # 参考价格
        f = self._last_features.get(sym) or {}
        ref_px = float(f.get(self.fill_ref, f.get("close", f.get("open", 0.0))))
        if ref_px <= 0.0:
            if self.slo:
                try: self.slo.record_order_failure()
                except Exception: pass
            return

        # Overlay 裁剪
        adj_qty = req_qty
        if hasattr(self.overlay, "adjust_order"):
            try:
                adj_qty = int(self.overlay.adjust_order(sym, side, req_qty, ref_px))
            except Exception:
                adj_qty = req_qty
        if adj_qty <= 0:
            if self.slo:
                try: self.slo.record_order_failure()
                except Exception: pass
            return

        # bar 内可成交量 & 参与率
        bar_vol = float(f.get("volume", 0.0))
        cap_qty = int(math.floor(self.participation_cap * bar_vol)) if bar_vol > 0 else adj_qty
        fill_qty = max(0, min(adj_qty, cap_qty))
        if fill_qty <= 0:
            if self.slo:
                try: self.slo.record_order_failure()
                except Exception: pass
            return

        # 限价检查
        typ = str(order.get("type", "MKT")).upper()
        lim_ok = True
        if typ == "LMT":
            lp = float(order.get("limit_price", 0.0))
            lim_ok = (ref_px <= lp + 1e-9) if side == "BUY" else (ref_px >= lp - 1e-9)
            if not lim_ok:
                if self.slo:
                    try: self.slo.record_order_failure()
                    except Exception: pass
                return

        # 成交价（ref ± (slippage + dynamic_spread)）
        slip_bps = self.slippage_bps + self.dynamic_spread_bps
        slip = ref_px * (slip_bps / 1e4)
        fill_px = ref_px + (slip if side == "BUY" else -slip)

        # 手续费
        fee = max(self.commission_min, self.commission_per_share * fill_qty)

        # 现金/持仓更新
        signed_qty = fill_qty if side == "BUY" else -fill_qty
        pos = self.positions.setdefault(sym, {"qty": 0.0, "avg_cost": 0.0, "last_px": ref_px, "last_ts": int(order.get("ts", time.time()*1000))})
        new_qty = pos["qty"] + signed_qty

        # 平/反手部分的实现盈亏
        realized = 0.0
        if pos["qty"] * signed_qty < 0:  # 有对冲
            closed = min(abs(pos["qty"]), abs(signed_qty)) * (1 if abs(pos["qty"]) <= abs(signed_qty) else -1)
            # 根据平均成本计算平仓盈亏
            pnl_per = (pos["avg_cost"] - fill_px) if pos["qty"] < 0 else (fill_px - pos["avg_cost"])
            realized = abs(closed) * pnl_per

        # 新的均价
        if (pos["qty"] >= 0 and signed_qty > 0) or (pos["qty"] <= 0 and signed_qty < 0):
            # 加仓
            tot_shares = abs(pos["qty"]) + abs(signed_qty)
            if tot_shares > 0:
                pos["avg_cost"] = (abs(pos["qty"])*pos["avg_cost"] + abs(signed_qty)*fill_px) / tot_shares
        else:
            # 减仓或反手
            if new_qty == 0:
                pos["avg_cost"] = 0.0  # 清空
            # 反手后的均价在下一次加仓时更新

        pos["qty"] = new_qty
        pos["last_px"] = fill_px
        pos["last_ts"] = int(order.get("ts", time.time()*1000))

        cash_delta = -fill_px * signed_qty - fee  # 买：现金减少；卖：现金增加（这里 signed_qty 对卖为负）
        self.cash += cash_delta
        ts_now = int(order.get("ts", time.time()*1000))
        self.ledger.append({"ts": ts_now, "symbol": sym, "side": side, "qty": fill_qty, "price": float(fill_px), "fee": float(fee)})
        # SLO
        if self.slo:
            try: self.slo.record_slippage_bps(ref_px, fill_px, side)
            except Exception: pass
        # Overlay 回调
        if hasattr(self.overlay, "on_fill"):
            try: self.overlay.on_fill(sym, side, fill_qty, float(fill_px))
            except Exception:
                pass

    # ---------- 账户视图 ----------
    def current_nav(self, px_map: Optional[Dict[str, float]] = None) -> float:
        nav = self.cash
        for sym, pos in self.positions.items():
            px = pos.get("last_px", 0.0)
            if px_map and sym in px_map:
                px = float(px_map[sym])
            nav += pos["qty"] * px
        return float(nav)
```

        else:
            path = self.cfg.get("csv")
            if path and path.endswith(".parquet"):
                df = pd.read_parquet(path)
            else:
                df = pd.read_csv(path)
        # 时间/排序
        df["ts"] = pd.to_datetime(df["ts"], utc=False, errors="coerce")
        df = df.sort_values(["ts"]).reset_index(drop=True)
        # leakage guard
        if "feature_cutoff_ts" in df.columns:
            df = enforce_point_in_time(df, ts_col="ts", cutoff_col="feature_cutoff_ts")
        return df

    # --- 引擎提供给策略的提交接口（调用 SimAdapter） ---
    async def submit_intent(self, strategy_id: str, decision):
        sym = getattr(decision, "symbol", None) or decision.get("symbol")
        side = getattr(decision, "side", None) or decision.get("side")
        qty  = int(getattr(decision, "target_qty", 0) or decision.get("target_qty", 0))
        typ  = getattr(decision, "order_type", "MKT") if hasattr(decision, "order_type") else decision.get("order_type","MKT")
        lmt  = getattr(decision, "limit_price", None) if hasattr(decision, "limit_price") else decision.get("limit_price")
        await self.sim.submit_order({"symbol": sym, "side": side, "qty": qty, "type": typ, "limit_price": lmt, "ts": int(time.time()*1000)})

    async def run(self):
        df = self._load_data()
        # 初始化策略
        if hasattr(self.strategy, "initialize"):
            await self.strategy.initialize()

        # 主循环：逐bar推进
        for _, row in df.iterrows():
            feat = row.to_dict()
            await self.sim.on_market_data(feat)
            # 注意：如果你使用了实时推理（InferenceManager），在这里写入 ml_signal 再调策略
            await self.strategy.on_features(str(feat.get("symbol")), feat)

        self._generate_report(df)

    # --- 绩效汇总（简化版；你可换成 ClickHouse 落库+图表） ---
    def _generate_report(self, df: pd.DataFrame):
        nav = self.sim.current_nav()
        # 简要统计
        fills = self.sim.ledger
        total_fee = sum(x.get("fee",0.0) for x in fills)
        trades = sum(1 for x in fills if x.get("qty",0)>0)
        print(f"[BT] NAV={nav:.2f} Trades={trades} Fee={total_fee:.2f}")

```python
# utils/repro.py
from __future__ import annotations
import json, hashlib, subprocess

def git_commit_sha() -> str:
    try:
        return subprocess.check_output(["git","rev-parse","HEAD"], stderr=subprocess.DEVNULL, text=True).strip()
    except Exception:
        return "unknown"

def config_sha256(cfg: dict) -> str:
    s = json.dumps(cfg, sort_keys=True, ensure_ascii=False, separators=(",",":"))
    return hashlib.sha256(s.encode("utf-8")).hexdigest()

def dataframe_snapshot(df, cols=("ts","symbol")) -> str:
    try:
        import pandas as pd
        h = hashlib.sha256()
        for c in cols:
            if c in df.columns:
                v = df[c].astype(str).values
                h.update(("/".join([v[0], v[-1]])).encode("utf-8"))
        h.update(str(tuple(sorted(df.columns))).encode("utf-8"))
        return h.hexdigest()
    except Exception:
        return "unknown"
```
