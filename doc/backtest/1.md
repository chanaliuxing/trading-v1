# -*- coding: utf-8 -*-
"""
Simulated Broker Adapter (vectorized-friendly)
----------------------------------------------
Drop-in adapter for backtesting/paper trading. Handles:
- Market / marketable-limit / limit / stop
- Intrabar trigger logic using bar high/low
- Slippage (bps or absolute), commission, fee minimums
- Positions, cash, realized/unrealized PnL
- Callbacks for order status / fills

Author: ChatGPT
"""

from __future__ import annotations
import math, uuid
from dataclasses import dataclass, field, asdict
from typing import Any, Dict, List, Optional, Callable
from datetime import datetime, timezone

# -------- Data structures --------

@dataclass
class Order:
    order_id: str
    decision_id: str
    ts: int                     # event time (ms)
    symbol: str
    side: str                   # "BUY" or "SELL"
    qty: float
    type: str                   # "MARKET" | "LIMIT" | "STOP" | "MKT_LIMITABLE"
    limit_price: Optional[float] = None
    stop_price: Optional[float] = None
    tif: str = "DAY"
    tag: Optional[str] = None
    status: str = "NEW"         # NEW, WORKING, FILLED, CANCELLED, REJECTED
    filled_qty: float = 0.0
    avg_fill_price: float = 0.0

@dataclass
class Fill:
    fill_id: str
    order_id: str
    ts: int
    symbol: str
    side: str
    qty: float
    price: float
    commission: float = 0.0
    slippage_cost: float = 0.0

@dataclass
class Position:
    symbol: str
    qty: float = 0.0
    avg_price: float = 0.0
    realized_pnl: float = 0.0

@dataclass
class Portfolio:
    cash: float = 100_000.0
    positions: Dict[str, Position] = field(default_factory=dict)
    realized_pnl: float = 0.0
    equity: float = 100_000.0
    last_mark_ts: Optional[int] = None

def _now_ms() -> int:
    return int(datetime.now(timezone.utc).timestamp() * 1000)

def _uid(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:10]}"

class SimAdapter:
    """
    Simple, deterministic simulator. Price book is provided by bar features:
    expects keys: 'open','high','low','close' for the current bar.

    Fill semantics:
    - MARKET: filled immediately at reference ('open' by default) + slippage
    - MKT_LIMITABLE: same as MARKET but caps price at provided limit_price
    - LIMIT (BUY): fills if low <= limit_price; price = min(limit_price, open) if opening gap-down,
                   otherwise at limit_price (or intrabar better price)
      LIMIT (SELL): fills if high >= limit_price analogously
    - STOP (BUY): triggers if high >= stop_price; fill at max(stop_price, open) for gap-ups
      STOP (SELL): triggers if low <= stop_price similarly
    """

    def __init__(self,
                 initial_cash: float = 100_000.0,
                 commission_per_share: float = 0.005,
                 commission_min: float = 1.0,
                 slippage_bps: float = 1.0,
                 fill_ref: str = "open"  # 'open' or 'close'
                 ) -> None:
        self.portfolio = Portfolio(cash=initial_cash, equity=initial_cash)
        self.commission_per_share = commission_per_share
        self.commission_min = commission_min
        self.slippage_bps = slippage_bps
        self.fill_ref = fill_ref
        self._orders: Dict[str, Order] = {}
        self._working: Dict[str, Order] = {}
        self._on_fill_callbacks: List[Callable[[Fill], None]] = []
        self._on_status_callbacks: List[Callable[[Order], None]] = []

    # --------- Callback wiring ---------

    def on_fill(self, fn: Callable[[Fill], None]) -> None:
        self._on_fill_callbacks.append(fn)

    def on_status(self, fn: Callable[[Order], None]) -> None:
        self._on_status_callbacks.append(fn)

    def _emit_status(self, order: Order) -> None:
        for fn in self._on_status_callbacks:
            try:
                fn(order)
            except Exception:
                pass

    def _emit_fill(self, fill: Fill) -> None:
        for fn in self._on_fill_callbacks:
            try:
                fn(fill)
            except Exception:
                pass

    # --------- Public API ---------

    def submit(self, order: Order) -> str:
        if order.qty <= 0:
            order.status = "REJECTED"
            self._emit_status(order)
            return order.order_id
        order.status = "WORKING"
        self._orders[order.order_id] = order
        self._working[order.order_id] = order
        self._emit_status(order)
        return order.order_id

    def cancel(self, order_id: str) -> None:
        o = self._orders.get(order_id)
        if not o or o.status in ("CANCELLED","FILLED"):
            return
        o.status = "CANCELLED"
        self._working.pop(order_id, None)
        self._emit_status(o)

    def cancel_all(self, symbol: Optional[str] = None) -> None:
        for oid, o in list(self._working.items()):
            if symbol and o.symbol != symbol:
                continue
            self.cancel(oid)

    # --------- Core mechanics ---------

    def on_bar(self, ts_ms: int, symbol: str, bar: Dict[str, float]) -> None:
        """
        Called once per bar per symbol. bar must include open/high/low/close.
        Also re-marks equity.
        """
        # 1) Try to execute market & triggered orders
        for oid, o in list(self._working.items()):
            if o.symbol != symbol:
                continue
            filled = False
            fill_price = None

            oref = bar.get(self.fill_ref, bar.get("open"))
            high = bar.get("high"); low = bar.get("low"); open_ = bar.get("open")

            if o.type == "MARKET":
                base = oref
                slip = base * (self.slippage_bps / 10_000.0)
                fill_price = base + (slip if o.side == "BUY" else -slip)
                filled = True

            elif o.type == "MKT_LIMITABLE":
                base = oref
                slip = base * (self.slippage_bps / 10_000.0)
                px = base + (slip if o.side == "BUY" else -slip)
                # cap against limit
                if o.limit_price is not None:
                    if o.side == "BUY":
                        px = min(px, o.limit_price)
                    else:
                        px = max(px, o.limit_price)
                fill_price = px
                filled = True

            elif o.type == "LIMIT":
                if o.side == "BUY":
                    if low is not None and o.limit_price is not None and low <= o.limit_price:
                        # If gap-down open below limit, fill at min(open, limit)
                        exec_price = min(open_, o.limit_price)
                        fill_price = exec_price
                        filled = True
                else:  # SELL
                    if high is not None and o.limit_price is not None and high >= o.limit_price:
                        exec_price = max(open_, o.limit_price)
                        fill_price = exec_price
                        filled = True

            elif o.type == "STOP":
                if o.side == "BUY":
                    if high is not None and o.stop_price is not None and high >= o.stop_price:
                        exec_price = max(open_, o.stop_price)
                        fill_price = exec_price
                        filled = True
                else:  # SELL
                    if low is not None and o.stop_price is not None and low <= o.stop_price:
                        exec_price = min(open_, o.stop_price)
                        fill_price = exec_price
                        filled = True

            if filled and fill_price is not None:
                self._fill_order(o, ts_ms, fill_price)

        # 2) Mark-to-market equity
        self._mark_equity(ts_ms, symbol, bar.get("close", bar.get("open")))

    # --------- Internals ---------

    def _commission(self, qty: float) -> float:
        fee = max(self.commission_min, abs(qty) * self.commission_per_share)
        return round(fee, 4)

    def _fill_order(self, o: Order, ts_ms: int, price: float) -> None:
        qty = o.qty if o.side == "BUY" else -o.qty
        pos = self.portfolio.positions.get(o.symbol) or Position(symbol=o.symbol)

        # Realized PnL if closing/flip
        realized = 0.0
        if pos.qty != 0 and (pos.qty > 0) != (qty > 0):
            # Closing some or all
            closing_qty = min(abs(pos.qty), abs(qty)) * (1 if pos.qty > 0 else -1)
            pnl = (price - pos.avg_price) * closing_qty * (1 if pos.qty > 0 else -1)
            realized += pnl

        # Position update
        new_qty = pos.qty + qty
        if new_qty == 0:
            pos.avg_price = 0.0
        else:
            if (pos.qty >= 0 and qty >= 0) or (pos.qty <= 0 and qty <= 0):
                # Adding to same direction -> average
                total_cost = pos.avg_price * abs(pos.qty) + price * abs(qty)
                pos.avg_price = total_cost / (abs(pos.qty) + abs(qty))
            # else flip handled via realized pnl above
        pos.qty = new_qty
        pos.realized_pnl += realized
        self.portfolio.realized_pnl += realized

        # Cash: pay for buy, receive for sell
        cash_delta = -price * qty  # qty signed
        fee = self._commission(abs(qty))
        self.portfolio.cash += cash_delta - fee

        # Finalize order
        o.status = "FILLED"
        o.filled_qty = abs(qty)
        o.avg_fill_price = price
        self._working.pop(o.order_id, None)

        fill = Fill(
            fill_id=_uid("fill"),
            order_id=o.order_id,
            ts=ts_ms,
            symbol=o.symbol,
            side=o.side,
            qty=abs(qty),
            price=price,
            commission=fee,
            slippage_cost=0.0
        )
        self._emit_fill(fill)

    def _mark_equity(self, ts_ms: int, symbol: str, mark_price: float) -> None:
        eq = self.portfolio.cash
        for pos in self.portfolio.positions.values():
            eq += pos.qty * mark_price  # assume single symbol MTM by last price
        self.portfolio.equity = eq
        self.portfolio.last_mark_ts = ts_ms

    # --------- Helpers for engine ---------

    def build_order(self,
                    decision_id: str,
                    symbol: str,
                    side: str,
                    qty: float,
                    type_: str = "MARKET",
                    limit_price: float | None = None,
                    stop_price: float | None = None,
                    tag: str | None = None,
                    ts_ms: int | None = None
                    ) -> Order:
        return Order(
            order_id=_uid("ord"),
            decision_id=decision_id,
            ts=ts_ms or _now_ms(),
            symbol=symbol, side=side, qty=qty,
            type=type_, limit_price=limit_price, stop_price=stop_price, tag=tag
        )

    def snapshot(self) -> Dict[str, Any]:
        return {
            "cash": self.portfolio.cash,
            "equity": self.portfolio.equity,
            "realized_pnl": self.portfolio.realized_pnl,
            "positions": {k: asdict(v) for k, v in self.portfolio.positions.items()},
            "open_orders": {k: asdict(v) for k, v in self._working.items()},
        }
'''

# -*- coding: utf-8 -*-
"""
Backtest Engine (vectorized bar-by-bar, async strategy API)
----------------------------------------------------------
- Loads historical features (ClickHouse or DataFrame)
- Calls strategy.on_features() per bar
- Routes intents -> SimAdapter orders
- Computes KPIs: total return, Sharpe, max drawdown, win rate, profit factor
- Supports one or multiple symbols (per-symbol bar iteration)

Author: ChatGPT
"""

from __future__ import annotations
import os, math, importlib, asyncio, statistics, json
from dataclasses import dataclass, field, asdict
from typing import Any, Dict, List, Optional, Callable, Iterable, Tuple

# Optional ClickHouse clients (lazy import)
def _get_clickhouse():
    try:
        import clickhouse_connect as ch  # pip install clickhouse-connect
        return ("connect", ch)
    except Exception:
        try:
            from clickhouse_driver import Client  # pip install clickhouse-driver
            return ("driver", Client)
        except Exception:
            return (None, None)

# ---- Engine Data Structures ----

@dataclass
class BacktestConfig:
    symbols: List[str]
    start: str                   # "YYYY-MM-DD HH:MM:SS"
    end: str                     # "
    timeframe: str = "1m"        # aligns with features table (features_1m, features_5m ...)
    features_table: Optional[str] = None  # override e.g. "trading.features_1m"
    bars_table: Optional[str] = None      # optional raw bars (for validation)
    clickhouse: Dict[str, Any] = field(default_factory=dict)  # host, port, user, password, database
    initial_cash: float = 100_000.0
    commission_per_share: float = 0.005
    commission_min: float = 1.0
    slippage_bps: float = 1.0
    fill_ref: str = "open"
    strategy_path: str = "tsla_v7_2.TSLASimplifiedV7_2"
    strategy_params: Dict[str, Any] = field(default_factory=dict)  # passed to strategy
    seed: int = 42

@dataclass
class TradeRecord:
    ts: int
    symbol: str
    side: str
    qty: float
    price: float
    commission: float

@dataclass
class Metrics:
    total_return: float
    sharpe: float
    max_drawdown: float
    win_rate: float
    profit_factor: float
    trades: int

# ---- Import Strategy Base ----
def _import_strategy_class(path: str):
    mod_name, cls_name = path.rsplit(".", 1)
    mod = importlib.import_module(mod_name)
    return getattr(mod, cls_name)

# ---- Decision envelope adapter ----
class _EngineShim:
    """
    Minimal engine surface for StrategyAlgorithm to call submit_intent().
    Strategy should call: await engine.submit_intent(strategy_id, DecisionEnvelope)
    """
    def __init__(self, outer):
        self._outer = outer

    async def submit_intent(self, strategy_id: str, decision) -> None:
        self._outer._pending_decisions.append((strategy_id, decision))

# ---- Backtest Engine ----
class BacktestEngine:
    def __init__(self, cfg: BacktestConfig) -> None:
        self.cfg = cfg
        from sim_adapter import SimAdapter
        self.sim = SimAdapter(
            initial_cash=cfg.initial_cash,
            commission_per_share=cfg.commission_per_share,
            commission_min=cfg.commission_min,
            slippage_bps=cfg.slippage_bps,
            fill_ref=cfg.fill_ref,
        )
        self._pending_decisions: List[Tuple[str, Any]] = []
        self._trades: List[TradeRecord] = []
        self._equity_curve: List[Tuple[int, float]] = []  # (ts_ms, equity)

        # Wire sim callbacks -> record trades & equity
        self.sim.on_fill(self._on_fill)

        # Strategy
        StrategyCls = _import_strategy_class(cfg.strategy_path)
        # Strategy expects (strategy_id, engine, params)
        params = {
            "extras": self.cfg.strategy_params or {}
        }
        self.strategy = StrategyCls(strategy_id="STRAT_1", engine=_EngineShim(self), params=params)

    # --- Sim callbacks ---
    def _on_fill(self, fill) -> None:
        self._trades.append(
            TradeRecord(ts=fill.ts, symbol=fill.symbol, side=fill.side,
                        qty=fill.qty, price=fill.price, commission=fill.commission)
        )

    # --- Data Loading ---
    def _features_table_name(self) -> str:
        if self.cfg.features_table:
            return self.cfg.features_table
        tf = self.cfg.timeframe
        return f"trading.features_{tf}"

    def _clickhouse_query(self) -> Optional[pd.DataFrame]:
        kind, client_or_mod = _get_clickhouse()
        if kind is None:
            return None
        start = self.cfg.start.replace("T"," ")
        end = self.cfg.end.replace("T"," ")
        tbl = self._features_table_name()
        syms = ",".join([f\"'{s}'\" for s in self.cfg.symbols])
        sql = f\"\"\"
            SELECT *
            FROM {tbl}
            WHERE symbol IN ({syms})
              AND ts >= toDateTime('{start}') AND ts < toDateTime('{end}')
            ORDER BY symbol, ts
        \"\"\"
        if kind == "connect":
            conn = client_or_mod.get_client(**self.cfg.clickhouse)
            df = conn.query_df(sql)
            return df
        else:
            client = client_or_mod(**self.cfg.clickhouse)
            data = client.execute(sql)
            cols = [c[0] for c in client.execute(\"DESCRIBE TABLE \" + tbl)]
            import pandas as pd
            return pd.DataFrame(data, columns=cols)

    def load_data(self, df: Optional[pd.DataFrame] = None) -> pd.DataFrame:
        if df is not None:
            out = df.copy()
        else:
            out = self._clickhouse_query()
            if out is None:
                raise RuntimeError(\"ClickHouse client not available and no DataFrame provided.\")
        # Sanity columns
        needed = {\"ts\",\"symbol\",\"open\",\"high\",\"low\",\"close\"}
        missing = needed - set(out.columns)
        if missing:
            raise ValueError(f\"Missing required columns: {missing}\")
        # Build ms timestamp
        if pd.api.types.is_datetime64_any_dtype(out[\"ts\"]) is False:
            out[\"ts\"] = pd.to_datetime(out[\"ts\"], utc=True)
        out[\"ts_ms\"] = (out[\"ts\"].astype(\"int64\") // 10**6).astype(\"int64\")
        return out

    # --- Metrics ---
    @staticmethod
    def _max_drawdown(values: List[float]) -> float:
        peak = -math.inf
        mdd = 0.0
        for v in values:
            if v > peak:
                peak = v
            dd = (peak - v) / peak if peak != 0 else 0
            mdd = max(mdd, dd)
        return mdd

    @staticmethod
    def _sharpe(returns: List[float], risk_free: float = 0.0) -> float:
        if not returns:
            return 0.0
        avg = statistics.mean(returns)
        std = statistics.pstdev(returns) or 1e-9
        return (avg - risk_free) / std

    def _compute_metrics(self) -> Metrics:
        eq = [e for _, e in self._equity_curve]
        if not eq:
            return Metrics(0,0,0,0,0,0)
        rets = []
        for i in range(1, len(eq)):
            r = (eq[i] - eq[i-1]) / max(eq[i-1], 1e-9)
            rets.append(r)
        total_ret = (eq[-1] - eq[0]) / max(eq[0], 1e-9)
        sharpe = self._sharpe(rets)
        mdd = self._max_drawdown(eq)

        # Trade stats
        wins = 0; losses = 0; gross_win = 0.0; gross_loss = 0.0
        # Very rough: treat each pair of opposite-side fills as a round-trip
        # (In practice, use order/position-aware PnL attribution.)
        # Here we approximate using realized PnL from simulator's portfolio.
        # So compute win-rate by checking realized PnL deltas after each fill.
        realized_series = []
        cum = 0.0
        from sim_adapter import SimAdapter
        # not importing portfolio directly; we approximate by deltas recorded from fills via engine snapshot
        # Fallback: use fill price deltas per symbol which is less accurate.
        # For now, compute based on cumulative realized from sim snapshot when we appended equity.
        # -> We'll compute wins via trade pairs:
        for i in range(0, len(self._trades), 2):
            if i+1 < len(self._trades):
                a, b = self._trades[i], self._trades[i+1]
                if a.side == "BUY":
                    pnl = (b.price - a.price) * a.qty - (a.commission + b.commission)
                else:
                    pnl = (a.price - b.price) * a.qty - (a.commission + b.commission)
                if pnl > 0: wins += 1; gross_win += pnl
                else: losses += 1; gross_loss += -pnl

        trades_n = len(self._trades)
        win_rate = wins / max(wins+losses, 1) if (wins+losses) > 0 else 0.0
        profit_factor = (gross_win / max(gross_loss, 1e-9)) if gross_loss > 0 else float('inf')

        return Metrics(
            total_return=total_ret,
            sharpe=sharpe,
            max_drawdown=mdd,
            win_rate=win_rate,
            profit_factor=profit_factor,
            trades=trades_n
        )

    # --- Run Loop ---
    async def _run_async(self, df: pd.DataFrame) -> Metrics:
        # Initialize strategy
        if hasattr(self.strategy, "initialize") and asyncio.iscoroutinefunction(self.strategy.initialize):
            await self.strategy.initialize()

        # Iterate by (ts, symbol)
        df = df.sort_values([\"ts\",\"symbol\"]).reset_index(drop=True)
        last_ts = None

        for _, row in df.iterrows():
            ts_ms = int(row[\"ts_ms\"])
            symbol = row[\"symbol\"]
            bar = {
                \"open\": float(row[\"open\"]),
                \"high\": float(row[\"high\"]),
                \"low\": float(row[\"low\"]),
                \"close\": float(row[\"close\"]),
            }

            # 1) Update simulator with bar (exec working orders + mark-to-market)
            self.sim.on_bar(ts_ms, symbol, bar)

            # 2) Call strategy with full features dict
            features = row.to_dict()
            if hasattr(self.strategy, "on_features"):
                await self.strategy.on_features(symbol, features)

            # 3) If strategy submitted intents, convert to orders and send
            while self._pending_decisions:
                strat_id, decision = self._pending_decisions.pop(0)
                # Expect DecisionEnvelope-like object
                # Required: decision_id, symbol, side, target_qty (+ optional price fields)
                qty = float(getattr(decision, "target_qty", 0))
                side = getattr(decision, "side", "BUY")
                symbol_d = getattr(decision, "symbol", symbol)
                typ = getattr(decision, "order_type", "MARKET")
                limit_px = getattr(decision, "limit_price", None)
                stop_px = getattr(decision, "stop_price", None)
                decision_id = getattr(decision, "decision_id", f\"DEC_{ts_ms}\")

                order = self.sim.build_order(
                    decision_id=decision_id,
                    symbol=symbol_d, side=side, qty=qty,
                    type_=typ, limit_price=limit_px, stop_price=stop_px,
                    ts_ms=ts_ms
                )
                self.sim.submit(order)

            # 4) Append equity snapshot once per timestamp
            if last_ts != ts_ms:
                snap = self.sim.snapshot()
                self._equity_curve.append((ts_ms, float(snap[\"equity\"])))
                last_ts = ts_ms

        return self._compute_metrics()

    def run(self, df: Optional[pd.DataFrame] = None) -> Dict[str, Any]:
        data = self.load_data(df=df)
        metrics = asyncio.run(self._run_async(data))
        return {
            \"metrics\": asdict(metrics),
            \"trades\": [asdict(t) for t in self._trades],
            \"equity_curve\": self._equity_curve,
            \"final_snapshot\": self.sim.snapshot(),
        }

# -*- coding: utf-8 -*-
"""
wf_hpo.py
---------
Purged walk-forward cross-validation + grid search HPO scaffold
for the BacktestEngine defined in backtest_engine.py.

Design goals
- Deterministic, filesystem-free core (you pass a config dict or JSON)
- Works with 1+ folds; each fold:
  * Grid-search on in-sample (IS) to pick best params by objective
  * Purge gap between IS and OOS to reduce leakage
  * Evaluate chosen params on out-of-sample (OOS)
- Outputs: detailed per-run rows + summary aggregates

Author: ChatGPT
"""

from __future__ import annotations
import os, sys, math, json, itertools, statistics, time
from dataclasses import dataclass, field, asdict
from typing import Any, Dict, List, Optional, Iterable, Tuple
from datetime import datetime, timedelta, timezone

# Import engine from current working dir
from backtest_engine import BacktestConfig, BacktestEngine

# ------------------ Utilities ------------------

def _dt(s: str) -> datetime:
    # Accept "YYYY-MM-DD HH:MM:SS" or "YYYY-MM-DDTHH:MM:SS"
    s2 = s.replace("T", " ")
    if len(s2) == 10:
        s2 += " 00:00:00"
    return datetime.fromisoformat(s2)

def _fmt_dt(dt: datetime) -> str:
    return dt.strftime("%Y-%m-%d %H:%M:%S")

def _product_dict(grid: Dict[str, Iterable[Any]]) -> List[Dict[str, Any]]:
    keys = list(grid.keys())
    vals = [list(v) for v in grid.values()]
    combos = []
    for tup in itertools.product(*vals):
        combos.append({k: v for k, v in zip(keys, tup)})
    return combos

def _jsonify(obj: Any) -> str:
    try:
        return json.dumps(obj, ensure_ascii=False, sort_keys=True)
    except TypeError:
        return str(obj)

# ------------------ Data Models ------------------

@dataclass
class Fold:
    fold_id: int
    is_start: str
    is_end: str
    oos_start: str
    oos_end: str
    purge_days: int = 0

@dataclass
class Objective:
    metric: str = "sharpe"      # "sharpe" | "total_return" | "profit_factor" | "win_rate"
    maximize: bool = True
    min_trades: int = 0         # filter candidates
    max_mdd: Optional[float] = None  # e.g. 0.25 (25%)

# ------------------ Core Runner ------------------

class WalkForwardHPO:
    def __init__(self, cfg: Dict[str, Any]) -> None:
        """
        cfg expects keys:
          symbols, timeframe, features_table,
          clickhouse (optional): {host, port, user, password, database}
          strategy_path, strategy_base_extras (dict),
          broker: {initial_cash, commission_per_share, commission_min, slippage_bps, fill_ref},
          grid: {param: [values...]}, 
          folds: [Fold-like dicts],
          objective: {metric, maximize, min_trades, max_mdd}
        """
        self.cfg = cfg
        self.symbols = cfg["symbols"]
        self.timeframe = cfg.get("timeframe", "1m")
        self.features_table = cfg.get("features_table")
        self.clickhouse = cfg.get("clickhouse", {})
        self.strategy_path = cfg.get("strategy_path", "tsla_v7_2.TSLASimplifiedV7_2")
        self.strategy_base_extras = cfg.get("strategy_base_extras", {}) or {}

        b = cfg.get("broker", {}) or {}
        self.broker = {
            "initial_cash": b.get("initial_cash", 100_000.0),
            "commission_per_share": b.get("commission_per_share", 0.005),
            "commission_min": b.get("commission_min", 1.0),
            "slippage_bps": b.get("slippage_bps", 1.0),
            "fill_ref": b.get("fill_ref", "open"),
        }

        self.grid = cfg["grid"]
        self.folds = [Fold(**f) for f in cfg["folds"]]
        obj = cfg.get("objective", {}) or {}
        self.objective = Objective(**obj)

    # ----- engine helper -----
    def _run_engine(self, start: str, end: str, strategy_params: Dict[str, Any]) -> Dict[str, Any]:
        bt_cfg = BacktestConfig(
            symbols=self.symbols,
            start=start, end=end,
            timeframe=self.timeframe,
            features_table=self.features_table,
            clickhouse=self.clickhouse,
            initial_cash=self.broker["initial_cash"],
            commission_per_share=self.broker["commission_per_share"],
            commission_min=self.broker["commission_min"],
            slippage_bps=self.broker["slippage_bps"],
            fill_ref=self.broker["fill_ref"],
            strategy_path=self.strategy_path,
            strategy_params=strategy_params
        )
        engine = BacktestEngine(bt_cfg)
        result = engine.run()
        return result

    # ----- objective scoring -----
    def _score(self, metrics: Dict[str, Any]) -> Optional[float]:
        # constraints
        if self.objective.min_trades and metrics.get("trades", 0) < self.objective.min_trades:
            return None
        if self.objective.max_mdd is not None and metrics.get("max_drawdown", 0.0) > self.objective.max_mdd:
            return None
        val = float(metrics.get(self.objective.metric, 0.0))
        return val if self.objective.maximize else -val

    # ----- main loop -----
    def run(self) -> Dict[str, Any]:
        grid_list = _product_dict(self.grid)
        all_rows: List[Dict[str, Any]] = []
        oos_rows: List[Dict[str, Any]] = []
        best_params_by_fold: List[Dict[str, Any]] = []

        for fold in self.folds:
            # 1) grid on IS
            scores: List[Tuple[float, Dict[str, Any], Dict[str, Any]]] = []
            for params in grid_list:
                extras = {
                    **self.strategy_base_extras,
                    "strategy_config": {
                        **self.strategy_base_extras.get("strategy_config", {}),
                        **params
                    }
                }
                res = self._run_engine(fold.is_start, fold.is_end, extras)
                row = {
                    "fold_id": fold.fold_id,
                    "phase": "IS",
                    "start": fold.is_start,
                    "end": fold.is_end,
                    "params": _jsonify(params),
                    **res["metrics"]
                }
                all_rows.append(row)
                score = self._score(res["metrics"])
                if score is not None:
                    scores.append((score, params, res["metrics"]))

            # 2) select best
            if not scores:
                # nothing passed constraints; fallback to max raw metric
                candidates = [(self._score(r["metrics"]) or float("-inf"), json.loads(r["params"]), r["metrics"])
                              for r in all_rows if r["fold_id"] == fold.fold_id and r["phase"] == "IS"]
                if not candidates:
                    raise RuntimeError(f"Fold {fold.fold_id}: no IS candidates at all.")
                scores = candidates
            scores.sort(key=lambda x: x[0], reverse=True)
            best_score, best_params, best_is_metrics = scores[0]
            best_params_by_fold.append({"fold_id": fold.fold_id, "params": best_params, "is_metrics": best_is_metrics})

            # 3) purge gap then OOS eval
            if fold.purge_days and fold.is_end < fold.oos_start:
                # Here we trust user provided oos_start is already purged;
                # this placeholder keeps API explicit.
                pass
            extras = {
                **self.strategy_base_extras,
                "strategy_config": {
                    **self.strategy_base_extras.get("strategy_config", {}),
                    **best_params
                }
            }
            oos_res = self._run_engine(fold.oos_start, fold.oos_end, extras)
            row = {
                "fold_id": fold.fold_id,
                "phase": "OOS",
                "start": fold.oos_start,
                "end": fold.oos_end,
                "params": _jsonify(best_params),
                **oos_res["metrics"]
            }
            all_rows.append(row)
            oos_rows.append(row)

        # 4) aggregate OOS
        if oos_rows:
            # weight by number of equity points as proxy for duration
            weights = []
            for r in oos_rows:
                # simple: assume equal weights if not available
                weights.append(1.0)
            wsum = sum(weights) or 1.0
            def wavg(key: str) -> float:
                return sum(w * float(r.get(key, 0.0)) for w, r in zip(weights, oos_rows)) / wsum
            summary = {
                "oos_avg_total_return": wavg("total_return"),
                "oos_avg_sharpe": wavg("sharpe"),
                "oos_avg_max_drawdown": wavg("max_drawdown"),
                "oos_avg_win_rate": wavg("win_rate"),
                "oos_avg_profit_factor": wavg("profit_factor"),
                "oos_trades": sum(int(r.get("trades", 0)) for r in oos_rows),
            }
        else:
            summary = {}

        return {
            "rows": all_rows,
            "oos_summary": summary,
            "best_params_by_fold": best_params_by_fold,
        }
''')

* **网格搜索（IS）→ 最优参数 → OOS 验证**：严格“只看未来之外”的流程，减少泄漏。
* **约束过滤**：先按 `min_trades / max_mdd` 筛选，再按目标指标（默认 Sharpe）排序选优。

wf_hpo_new = dedent(r'''
# -*- coding: utf-8 -*-
"""
wf_hpo.py
---------
Purged walk-forward cross-validation + grid search HPO scaffold
for the BacktestEngine defined in backtest_engine.py.

Enhancements:
- make_folds(): auto rolling-window fold generator (IS/OOS/GAP/STEP days)
- Optional parallel grid evaluation via ProcessPoolExecutor
- Export helpers for CSV/Parquet leaderboards

Author: ChatGPT
"""

from __future__ import annotations
import os, sys, math, json, itertools, statistics, time
from dataclasses import dataclass, field, asdict
from typing import Any, Dict, List, Optional, Iterable, Tuple
from datetime import datetime, timedelta, timezone
from concurrent.futures import ProcessPoolExecutor, as_completed

import pandas as pd

# Import engine from current working dir
from backtest_engine import BacktestConfig, BacktestEngine

# ------------------ Utilities ------------------

def _dt(s: str) -> datetime:
    # Accept "YYYY-MM-DD HH:MM:SS" or "YYYY-MM-DDTHH:MM:SS"
    s2 = s.replace("T", " ")
    if len(s2) == 10:
        s2 += " 00:00:00"
    return datetime.fromisoformat(s2)

def _fmt_dt(dt: datetime) -> str:
    return dt.strftime("%Y-%m-%d %H:%M:%S")

def _product_dict(grid: Dict[str, Iterable[Any]]) -> List[Dict[str, Any]]:
    keys = list(grid.keys())
    vals = [list(v) for v in grid.values()]
    combos = []
    for tup in itertools.product(*vals):
        combos.append({k: v for k, v in zip(keys, tup)})
    return combos

def _jsonify(obj: Any) -> str:
    try:
        return json.dumps(obj, ensure_ascii=False, sort_keys=True)
    except TypeError:
        return str(obj)

def _ensure_dir(path: str) -> None:
    os.makedirs(path, exist_ok=True)

# ------------------ Fold Generator ------------------

def make_folds(start: str, end: str, *, is_days: int, oos_days: int, gap_days: int = 0, step_days: int | None = None) -> List[Dict[str, Any]]:
    """
    Build rolling time windows:
      [IS start, IS end] --(gap)--> [OOS start, OOS end]
    Windows slide forward by step_days (default = oos_days).
    """
    start_dt = _dt(start)
    end_dt = _dt(end)
    if step_days is None or step_days <= 0:
        step_days = oos_days
    folds = []
    fid = 1
    cursor = start_dt
    while True:
        is_start = cursor
        is_end = is_start + timedelta(days=is_days)
        oos_start = is_end + timedelta(days=gap_days)
        oos_end = oos_start + timedelta(days=oos_days)
        if oos_end > end_dt:
            break
        folds.append({
            "fold_id": fid,
            "is_start": _fmt_dt(is_start),
            "is_end": _fmt_dt(is_end),
            "oos_start": _fmt_dt(oos_start),
            "oos_end": _fmt_dt(oos_end),
            "purge_days": gap_days
        })
        fid += 1
        cursor = cursor + timedelta(days=step_days)
    return folds

# ------------------ Data Models ------------------

@dataclass
class Fold:
    fold_id: int
    is_start: str
    is_end: str
    oos_start: str
    oos_end: str
    purge_days: int = 0

@dataclass
class Objective:
    metric: str = "sharpe"      # "sharpe" | "total_return" | "profit_factor" | "win_rate"
    maximize: bool = True
    min_trades: int = 0         # filter candidates
    max_mdd: Optional[float] = None  # e.g. 0.25 (25%)

# ------------------ Engine helper (process-safe) ------------------

def _run_engine_external(bt_cfg_dict: Dict[str, Any], start: str, end: str, strategy_params: Dict[str, Any]) -> Dict[str, Any]:
    cfg = BacktestConfig(**bt_cfg_dict)
    cfg.start = start
    cfg.end = end
    cfg.strategy_params = strategy_params
    eng = BacktestEngine(cfg)
    return eng.run()

# ------------------ Core Runner ------------------

class WalkForwardHPO:
    def __init__(self, cfg: Dict[str, Any]) -> None:
        """
        cfg expects keys:
          symbols, timeframe, features_table,
          clickhouse (optional): {host, port, user, password, database}
          strategy_path, strategy_base_extras (dict),
          broker: {initial_cash, commission_per_share, commission_min, slippage_bps, fill_ref},
          grid: {param: [values...]}, 
          folds: [Fold-like dicts], OR
          fold_generator: {start, end, is_days, oos_days, gap_days, step_days}
          objective: {metric, maximize, min_trades, max_mdd}
          parallel: {workers: int}
        """
        self.cfg = cfg
        self.symbols = cfg["symbols"]
        self.timeframe = cfg.get("timeframe", "1m")
        self.features_table = cfg.get("features_table")
        self.clickhouse = cfg.get("clickhouse", {})
        self.strategy_path = cfg.get("strategy_path", "tsla_v7_2.TSLASimplifiedV7_2")
        self.strategy_base_extras = cfg.get("strategy_base_extras", {}) or {}

        b = cfg.get("broker", {}) or {}
        self.broker = {
            "initial_cash": b.get("initial_cash", 100_000.0),
            "commission_per_share": b.get("commission_per_share", 0.005),
            "commission_min": b.get("commission_min", 1.0),
            "slippage_bps": b.get("slippage_bps", 1.0),
            "fill_ref": b.get("fill_ref", "open"),
        }

        self.grid = cfg["grid"]
        if "folds" in cfg and cfg["folds"]:
            self.folds = [Fold(**f) for f in cfg["folds"]]
        elif "fold_generator" in cfg:
            gen = cfg["fold_generator"]
            folds = make_folds(
                gen["start"], gen["end"],
                is_days=int(gen["is_days"]),
                oos_days=int(gen["oos_days"]),
                gap_days=int(gen.get("gap_days", 0)),
                step_days=int(gen.get("step_days", gen["oos_days"]))
            )
            self.folds = [Fold(**f) for f in folds]
        else:
            raise ValueError("Must provide 'folds' or 'fold_generator'.")

        obj = cfg.get("objective", {}) or {}
        self.objective = Objective(**obj)

        par = cfg.get("parallel", {}) or {}
        self.workers = int(par.get("workers", 1))

        # Build base bt_cfg once (process-safe dict)
        self._bt_cfg_dict = asdict(BacktestConfig(
            symbols=self.symbols,
            start="1970-01-01 00:00:00",
            end="1970-01-02 00:00:00",
            timeframe=self.timeframe,
            features_table=self.features_table,
            clickhouse=self.clickhouse,
            initial_cash=self.broker["initial_cash"],
            commission_per_share=self.broker["commission_per_share"],
            commission_min=self.broker["commission_min"],
            slippage_bps=self.broker["slippage_bps"],
            fill_ref=self.broker["fill_ref"],
            strategy_path=self.strategy_path,
            strategy_params=self.strategy_base_extras
        ))

    # ----- objective scoring -----
    def _score(self, metrics: Dict[str, Any]) -> Optional[float]:
        # constraints
        if self.objective.min_trades and metrics.get("trades", 0) < self.objective.min_trades:
            return None
        if self.objective.max_mdd is not None and metrics.get("max_drawdown", 0.0) > self.objective.max_mdd:
            return None
        val = float(metrics.get(self.objective.metric, 0.0))
        return val if self.objective.maximize else -val

    # ----- IS Grid (serial/parallel) -----
    def _grid_eval(self, fold: Fold, grid_list: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        rows: List[Dict[str, Any]] = []
        tasks = []
        if self.workers > 1:
            with ProcessPoolExecutor(max_workers=self.workers) as ex:
                for params in grid_list:
                    extras = {
                        **self.strategy_base_extras,
                        "strategy_config": {
                            **self.strategy_base_extras.get("strategy_config", {}),
                            **params
                        }
                    }
                    tasks.append(ex.submit(_run_engine_external, self._bt_cfg_dict, fold.is_start, fold.is_end, extras))
                for params, fut in zip(grid_list, as_completed(tasks)):
                    res = fut.result()
                    row = {
                        "fold_id": fold.fold_id,
                        "phase": "IS",
                        "start": fold.is_start,
                        "end": fold.is_end,
                        "params": _jsonify(params),
                        **res["metrics"]
                    }
                    rows.append(row)
        else:
            for params in grid_list:
                extras = {
                    **self.strategy_base_extras,
                    "strategy_config": {
                        **self.strategy_base_extras.get("strategy_config", {}),
                        **params
                    }
                }
                res = _run_engine_external(self._bt_cfg_dict, fold.is_start, fold.is_end, extras)
                row = {
                    "fold_id": fold.fold_id,
                    "phase": "IS",
                    "start": fold.is_start,
                    "end": fold.is_end,
                    "params": _jsonify(params),
                    **res["metrics"]
                }
                rows.append(row)
        return rows

    def _eval_oos(self, fold: Fold, best_params: Dict[str, Any]) -> Dict[str, Any]:
        extras = {
            **self.strategy_base_extras,
            "strategy_config": {
                **self.strategy_base_extras.get("strategy_config", {}),
                **best_params
            }
        }
        res = _run_engine_external(self._bt_cfg_dict, fold.oos_start, fold.oos_end, extras)
        row = {
            "fold_id": fold.fold_id,
            "phase": "OOS",
            "start": fold.oos_start,
            "end": fold.oos_end,
            "params": _jsonify(best_params),
            **res["metrics"]
        }
        return row

    # ----- main loop -----
    def run(self) -> Dict[str, Any]:
        grid_list = _product_dict(self.grid)
        all_rows: List[Dict[str, Any]] = []
        oos_rows: List[Dict[str, Any]] = []
        best_params_by_fold: List[Dict[str, Any]] = []

        for fold in self.folds:
            # 1) grid on IS
            is_rows = self._grid_eval(fold, grid_list)
            all_rows.extend(is_rows)

            # 2) select best
            scored = []
            for r in is_rows:
                sc = self._score(r)
                if sc is not None:
                    scored.append((sc, r))
            if not scored:
                # fallback to raw best by metric
                scored = [((r.get(self.objective.metric, 0.0) or -1e9), r) for r in is_rows]
            scored.sort(key=lambda x: x[0], reverse=True)
            top_row = scored[0][1]
            best_params = json.loads(top_row["params"])
            best_params_by_fold.append({"fold_id": fold.fold_id, "params": best_params, "is_metrics": {k: top_row[k] for k in top_row if k not in ("phase","params","fold_id","start","end")}})

            # 3) purge gap then OOS eval
            oos_row = self._eval_oos(fold, best_params)
            all_rows.append(oos_row)
            oos_rows.append(oos_row)

        # 4) aggregate OOS
        oos_df = pd.DataFrame(oos_rows)
        if not oos_df.empty:
            summary = {
                "oos_avg_total_return": float(oos_df["total_return"].mean() if "total_return" in oos_df else 0.0),
                "oos_avg_sharpe": float(oos_df["sharpe"].mean() if "sharpe" in oos_df else 0.0),
                "oos_avg_max_drawdown": float(oos_df["max_drawdown"].mean() if "max_drawdown" in oos_df else 0.0),
                "oos_avg_win_rate": float(oos_df["win_rate"].mean() if "win_rate" in oos_df else 0.0),
                "oos_avg_profit_factor": float(oos_df["profit_factor"].mean() if "profit_factor" in oos_df else 0.0),
                "oos_trades": int(oos_df["trades"].sum() if "trades" in oos_df else 0),
                "folds": int(oos_df["fold_id"].nunique())
            }
        else:
            summary = {}

        return {
            "rows": all_rows,
            "oos_summary": summary,
            "best_params_by_fold": best_params_by_fold,
        }

    # ------------------ Exports ------------------

    @staticmethod
    def export_tables(result: Dict[str, Any], out_dir: str) -> Dict[str, str]:
        """
        Save rows & summaries to CSV/Parquet; return paths.
        """
        _ensure_dir(out_dir)
        rows = pd.DataFrame(result.get("rows", []))
        paths = {}
        if not rows.empty:
            p_csv = os.path.join(out_dir, "rows.csv")
            rows.to_csv(p_csv, index=False)
            paths["rows_csv"] = p_csv
            try:
                p_par = os.path.join(out_dir, "rows.parquet")
                rows.to_parquet(p_par, index=False)
                paths["rows_parquet"] = p_par
            except Exception:
                pass

            # Leaderboards
            is_df = rows[rows["phase"] == "IS"].copy() if "phase" in rows else pd.DataFrame()
            oos_df = rows[rows["phase"] == "OOS"].copy() if "phase" in rows else pd.DataFrame()
            if not is_df.empty:
                p_is = os.path.join(out_dir, "leaderboard_is.csv")
                is_df.sort_values(["fold_id", "sharpe"], ascending=[True, False]).to_csv(p_is, index=False)
                paths["leaderboard_is_csv"] = p_is
            if not oos_df.empty:
                p_oos = os.path.join(out_dir, "leaderboard_oos.csv")
                oos_df.sort_values(["fold_id", "sharpe"], ascending=[True, False]).to_csv(p_oos, index=False)
                paths["leaderboard_oos_csv"] = p_oos

        # Summaries
        p_sum = os.path.join(out_dir, "oos_summary.json")
        with open(p_sum, "w", encoding="utf-8") as f:
            json.dump(result.get("oos_summary", {}), f, indent=2, ensure_ascii=False)
        paths["oos_summary_json"] = p_sum

        p_best = os.path.join(out_dir, "best_params_by_fold.json")
        with open(p_best, "w", encoding="utf-8") as f:
            json.dump(result.get("best_params_by_fold", []), f, indent=2, ensure_ascii=False)
        paths["best_params_by_fold_json"] = p_best

        # Champion statistics: count of param choices among best
        champs = pd.DataFrame(result.get("best_params_by_fold", []))
        if not champs.empty and "params" in champs.columns:
            param_dicts = champs["params"].apply(lambda d: d)
            exp = pd.json_normalize(param_dicts)
            exp.insert(0, "fold_id", champs["fold_id"])
            p_champ = os.path.join(out_dir, "champion_params.csv")
            exp.to_csv(p_champ, index=False)
            paths["champion_params_csv"] = p_champ

        return paths
''')

# -*- coding: utf-8 -*-
"""
hpo_cli.py
----------
CLI runner for walk-forward HPO using wf_hpo.WalkForwardHPO.

Usage:
  python hpo_cli.py --config /path/to/config.json --out_dir /path/to/outdir --print-summary

Config format: see hpo_config_example.json or hpo_config_autofolds.json

from __future__ import annotations
import os, sys, json, argparse

def _load_config(path: str):
    if path.endswith(".json"):

    try:
        import yaml  # optional
        with open(path, "r", encoding="utf-8") as f:

        print("\nOOS Summary:")
