# 1) 策略模板：直接消费 `ml_signal`

**`strategies/model_signal_strategy.py`**

```python
from __future__ import annotations
from libs.strategy.base import StrategyAlgorithm, DecisionEnvelope

class ModelSignalStrategy(StrategyAlgorithm):
    """
    读取 features['ml_signal'] / ['ml_conf']，映射为目标仓位/下单。
    适用：XGBoost/LSTM 推理已由 InferenceManager 写入 features。

    extras:
      symbol: "TSLA"
      lot: 100                      # 下单最小手数
      max_qty: 1000                 # 单票最大数量（策略层软约束，执行侧仍有 Overlay 硬约束）
      conf_floor: 0.0               # 低置信度直接不下单
      scale: 1.0                    # 仓位缩放
      order_type: "MKT" | "LMT"
      limit_offset_bps: 3           # 限价相对 ref 的偏移（bps）
    """
    async def initialize(self):
        self.cfg = self.params.get("extras", {}) or {}
        self.symbol = self.cfg.get("symbol")
        self.lot = int(self.cfg.get("lot", 100))
        self.max_qty = int(self.cfg.get("max_qty", 1000))
        self.conf_floor = float(self.cfg.get("conf_floor", 0.0))
        self.scale = float(self.cfg.get("scale", 1.0))
        self.order_type = str(self.cfg.get("order_type", "MKT")).upper()
        self.limit_offset_bps = float(self.cfg.get("limit_offset_bps", 3.0))

    def _target_from_signal(self, sig: float, conf: float, px: float) -> int:
        if conf < self.conf_floor:
            return 0
        # 将 -1..1 的信号映射到目标手数
        base = int(round(sig * self.max_qty * self.scale))
        # 向 lot 对齐
        if base == 0:
            return 0
        return int(max(self.lot, abs(base)) * (1 if base > 0 else -1))

    async def on_features(self, symbol: str, features: dict[str, any]):
        if symbol != self.symbol:
            return
        sig = float(features.get("ml_signal", 0.0))
        conf = float(features.get("ml_conf", 0.0))
        px = float(features.get("close", 0.0))
        tgt = self._target_from_signal(sig, conf, px)
        if tgt == 0:
            return
        side = "BUY" if tgt > 0 else "SELL"
        qty = abs(tgt)

        order_kwargs = {}
        if self.order_type == "LMT":
            # 限价按当前参考价偏移
            ref = float(features.get("open", px) or px)
            off = ref * (self.limit_offset_bps / 1e4)
            limit_px = ref - off if side == "BUY" else ref + off
            order_kwargs["order_type"] = "LMT"
            order_kwargs["limit_price"] = float(limit_px)

        decision = DecisionEnvelope(
            decision_id=self.generate_decision_id(),
            strategy_id=self.strategy_id,
            symbol=self.symbol,
            side=side,
            target_qty=qty,
            **order_kwargs
        )
        await self.engine.submit_intent(self.strategy_id, decision)

    async def on_fill(self, symbol: str, qty: int, price: float, side: str):
        # 可选：记录成交/更新内部状态
        return
```

# 2) 实时推理：InferenceManager（带时延/错误诊断）

**更新 `inference/manager.py`（替换原文件）**

```python
from __future__ import annotations
from typing import Any, Dict, Optional, Tuple
import os, json, time
import pandas as pd
import numpy as np
from models.base import DecisionScheme
from models.xgb_model import XGBModel
from models.lstm_model import LSTMModel

class InferenceManager:
    """
    热加载 ./models/active 下的模型，对每条 features 生成 ml_signal/ml_pred/ml_conf。
    返回 (features_row, diag)；diag 包含 infer_ms / error 标志。
    """
    def __init__(self, cfg: Dict[str, Any]):
        self.cfg = cfg or {}
        self.path = self.cfg.get("active_path", "./models/active")
        self.scheme = DecisionScheme(**(self.cfg.get("decision") or {}))
        self.model = None
        self.meta = {}
        self._last_load_ts = 0
        self._load_if_changed()

    def _load_if_changed(self):
        meta_p = os.path.join(self.path, "meta.json")
        if not os.path.exists(meta_p):
            return
        mtime = os.path.getmtime(meta_p)
        if mtime <= self._last_load_ts:
            return
        meta = json.load(open(meta_p, "r", encoding="utf-8"))
        mtype = meta.get("type")
        if mtype == "XGBModel":
            mdl = XGBModel(task=meta.get("task", "reg"), hparams=meta.get("hparams"))
        elif mtype == "LSTMModel":
            mdl = LSTMModel(task=meta.get("task", "reg"), hparams=meta.get("hparams"))
        else:
            return
        mdl.load(self.path)
        self.model, self.meta = mdl, meta
        self._last_load_ts = mtime

    def infer_row(self, features_row: Dict[str, Any]) -> Tuple[Dict[str, Any], Dict[str, Any]]:
        self._load_if_changed()
        out = dict(features_row)
        diag = {"infer_ms": 0.0, "error": 0}
        if self.model is None:
            if self.cfg.get("fallback_zero", True):
                out["ml_signal"], out["ml_pred"], out["ml_conf"] = 0.0, 0.0, 0.0
            return out, diag

        cols = self.model.feature_cols()
        if not cols:
            return out, diag

        df = pd.DataFrame([out])
        t0 = time.perf_counter()
        try:
            pred = self.model.predict(df)[-1]
            infer_ms = (time.perf_counter() - t0) * 1e3
            diag["infer_ms"] = float(infer_ms)
            out["ml_pred"] = float(pred)
            if self.model.task == "reg":
                out["ml_conf"] = float(min(1.0, abs(pred)))
            else:
                out["ml_conf"] = float(min(1.0, abs(pred - 0.5) * 2))
            sig = self.model.to_signal(np.array([pred]), self.scheme)[0]
            out["ml_signal"] = float(sig)
        except Exception:
            diag["error"] = 1
            if self.cfg.get("fallback_zero", True):
                out["ml_signal"], out["ml_pred"], out["ml_conf"] = 0.0, 0.0, 0.0
        return out, diag
```

# 3) 监控扩展：推理时延/错误率 + ClickHouse 落库

**更新 `monitoring/slo.py`（在原基础上新增推理指标）**

```python
from __future__ import annotations
from collections import deque
from dataclasses import dataclass, field
from typing import Deque, Tuple, Dict, Any, Optional
import time

@dataclass
class SLOThresholds:
    max_data_lag_ms: int = 5_000
    max_order_error_rate: float = 0.02
    max_shadow_real_diff_bps: float = 5.0
    window_sec: int = 300
    # NEW:
    max_infer_latency_ms: int = 20         # 单点推理均值 <= 20ms
    max_infer_error_rate: float = 0.01     # 推理失败率 <= 1%

@dataclass
class SLOState:
    latencies: Deque[Tuple[int,int]] = field(default_factory=deque)
    order_attempts: Deque[int] = field(default_factory=deque)
    order_failures: Deque[int] = field(default_factory=deque)
    slippages_bps: Deque[Tuple[int,float]] = field(default_factory=deque)
    shadow_real_bps: Deque[Tuple[int,float]] = field(default_factory=deque)
    # NEW:
    infer_lat_ms: Deque[Tuple[int,float]] = field(default_factory=deque)
    infer_attempts: Deque[int] = field(default_factory=deque)
    infer_errors: Deque[int] = field(default_factory=deque)

class SLOMonitor:
    def __init__(self, thresholds: SLOThresholds, alert_cb=None, ch_client=None, run_id: str = "run"):
        self.t = thresholds; self.s = SLOState()
        self.alert = alert_cb or (lambda title, text: None)
        self.ch = ch_client; self.run_id = run_id

    def _now_ms(self): return int(time.time()*1000)
    def _trim(self, q: Deque, now_ms: int):
        cutoff = now_ms - self.t.window_sec*1000
        while q and (q[0][0] if isinstance(q[0], tuple) else q[0]) < cutoff:
            q.popleft()

    # ---- RECORDERS ----
    def record_data_point_latency(self, market_ts_ms: int, recv_ts_ms: Optional[int] = None):
        now = recv_ts_ms or self._now_ms(); lag = max(0, now - market_ts_ms)
        self.s.latencies.append((now, lag)); self._trim(self.s.latencies, now)

    def record_order_attempt(self):
        now = self._now_ms(); self.s.order_attempts.append(now); self._trim(self.s.order_attempts, now)

    def record_order_failure(self):
        now = self._now_ms(); self.s.order_failures.append(now); self._trim(self.s.order_failures, now)

    def record_slippage_bps(self, ref_px: float, fill_px: float, side: str):
        if ref_px <= 0 or fill_px <= 0: return
        bps = abs((fill_px - ref_px)/ref_px)*1e4
        now = self._now_ms(); self.s.slippages_bps.append((now, bps)); self._trim(self.s.slippages_bps, now)

    def record_shadow_real_diff_bps(self, shadow_fill: float, real_fill: float):
        if shadow_fill <= 0 or real_fill <= 0: return
        bps = abs((real_fill - shadow_fill)/shadow_fill)*1e4
        now = self._now_ms(); self.s.shadow_real_bps.append((now, bps)); self._trim(self.s.shadow_real_bps, now)

    # NEW:
    def record_infer_attempt(self): 
        now = self._now_ms(); self.s.infer_attempts.append(now); self._trim(self.s.infer_attempts, now)

    def record_infer_result(self, infer_ms: float, error: bool):
        now = self._now_ms()
        self.s.infer_lat_ms.append((now, float(infer_ms))); self._trim(self.s.infer_lat_ms, now)
        if error:
            self.s.infer_errors.append(now); self._trim(self.s.infer_errors, now)

    # ---- COMPUTE ----
    def compute_window_metrics(self) -> Dict[str, float]:
        now = self._now_ms()
        def avg(q: Deque[Tuple[int,float]]) -> float:
            if not q: return 0.0
            return float(sum(v for _, v in q)/len(q))
        lag_avg = avg(self.s.latencies)
        err_rate = (len(self.s.order_failures)/max(1,len(self.s.order_attempts)))
        slip_avg = avg(self.s.slippages_bps)
        diff_avg = avg(self.s.shadow_real_bps)
        infer_lat_avg = avg(self.s.infer_lat_ms)
        infer_err_rate = (len(self.s.infer_errors)/max(1,len(self.s.infer_attempts)))
        return {
            "data_lag_ms_avg": lag_avg,
            "order_error_rate": err_rate,
            "slippage_bps_avg": slip_avg,
            "shadow_real_diff_bps_avg": diff_avg,
            "infer_latency_ms_avg": infer_lat_avg,
            "infer_error_rate": infer_err_rate,
            "window_sec": self.t.window_sec
        }

    def evaluate_and_alert(self):
        m = self.compute_window_metrics()
        alerts = []
        if m["data_lag_ms_avg"] > self.t.max_data_lag_ms:
            alerts.append(f"DataLag {m['data_lag_ms_avg']:.0f}ms > {self.t.max_data_lag_ms}ms")
        if m["order_error_rate"] > self.t.max_order_error_rate:
            alerts.append(f"OrderError {m['order_error_rate']:.3f} > {self.t.max_order_error_rate}")
        if m["shadow_real_diff_bps_avg"] > self.t.max_shadow_real_diff_bps:
            alerts.append(f"ShadowRealDiff {m['shadow_real_diff_bps_avg']:.1f}bps > {self.t.max_shadow_real_diff_bps}bps")
        if m["infer_latency_ms_avg"] > self.t.max_infer_latency_ms:
            alerts.append(f"InferLatency {m['infer_latency_ms_avg']:.1f}ms > {self.t.max_infer_latency_ms}ms")
        if m["infer_error_rate"] > self.t.max_infer_error_rate:
            alerts.append(f"InferError {m['infer_error_rate']:.3f} > {self.t.max_infer_error_rate}")
        for msg in alerts:
            self.alert("SLO Breach", f"run={self.run_id} {msg}")

    def flush_to_clickhouse(self):
        if not self.ch: return
        m = self.compute_window_metrics()
        row = [[self.run_id, int(self._now_ms()),
                float(m["data_lag_ms_avg"]), float(m["order_error_rate"]),
                float(m["slippage_bps_avg"]), float(m["shadow_real_diff_bps_avg"]),
                float(m["infer_latency_ms_avg"]), float(m["infer_error_rate"]),
                int(m["window_sec"])]]
        self.ch.insert("live_metrics", row, column_names=[
            "run_id","ts_ms",
            "data_lag_ms_avg","order_error_rate",
            "slippage_bps_avg","shadow_real_diff_bps_avg",
            "infer_latency_ms_avg","infer_error_rate","window_sec"
        ])
```

# 4) 模型注册与一键升级（从 WF/HPO 到 Production）

**`models/promotion.py`**

```python
from __future__ import annotations
import os, json, shutil, time, hashlib
from typing import Dict, Any
from models.xgb_model import XGBModel
from models.lstm_model import LSTMModel

def _mkdir(p): os.makedirs(p, exist_ok=True)

def _hash(s: str) -> str: return hashlib.sha256(s.encode("utf-8")).hexdigest()[:10]

def save_and_register_model(mdl, save_dir: str, meta_extra: Dict[str, Any]):
    _mkdir = _mkdir if False else _mkdir  # keep linter calm
    _mkdir(save_dir)
    mdl.save(save_dir)
    meta_path = os.path.join(save_dir, "meta.json")
    meta = json.load(open(meta_path, "r", encoding="utf-8"))
    meta.update(meta_extra or {})
    json.dump(meta, open(meta_path, "w", encoding="utf-8"))
    return meta

def promote_if_pass(client, metrics: Dict[str, float], thresholds: Dict[str, float],
                    model_dir: str, active_symlink: str = "./models/active"):
    """
    thresholds 例：
      {"oos_sharpe_min":1.0, "oos_psr_min":0.8, "oos_pbo_max":0.15, "max_dd_max":0.35}
    通过则切换 symlink -> model_dir，并写 model_registry。
    """
    ok = True
    ok &= metrics.get("oos_avg_sharpe", 0.0) >= thresholds.get("oos_sharpe_min", -1e9)
    ok &= metrics.get("oos_psr", 0.0)        >= thresholds.get("oos_psr_min", -1e9)
    ok &= metrics.get("oos_pbo", 1.0)        <= thresholds.get("oos_pbo_max", 1e9)
    ok &= metrics.get("oos_avg_max_drawdown", 1.0) <= thresholds.get("max_dd_max", 1.0)

    # 写注册表
    if client:
        row = [[
          metrics.get("model_id","unknown"),
          int(time.time()),
          str(metrics.get("model_type","unknown")),
          str(metrics.get("task","reg")),
          str(model_dir),
          metrics.get("feature_cols",[]),
          int(metrics.get("lookback",0)),
          json.dumps(metrics.get("hparams",{})),
          int(metrics.get("train_end_ts",0)),
          {"oos_avg_sharpe":float(metrics.get("oos_avg_sharpe",0.0)),
           "oos_psr":float(metrics.get("oos_psr",0.0)),
           "oos_pbo":float(metrics.get("oos_pbo",0.0)),
           "oos_avg_max_drawdown":float(metrics.get("oos_avg_max_drawdown",0.0))},
          "active" if ok else "candidate"
        ]]
        client.insert("model_registry", row, column_names=[
          "model_id","created","type","task","path","features","lookback","hparams","train_end_ts","metrics","stage"
        ])

    if not ok:
        return False

    # 切换 active
    try:
        if os.path.islink(active_symlink) or os.path.exists(active_symlink):
            try:
                if os.path.islink(active_symlink):
                    os.unlink(active_symlink)
                else:
                    shutil.rmtree(active_symlink)
            except Exception:
                pass
        os.symlink(os.path.abspath(model_dir), active_symlink, target_is_directory=True)
    except OSError:
        # Windows 无 symlink 权限时，改为复制
        shutil.copytree(model_dir, active_symlink, dirs_exist_ok=True)
    return True
```

## 现在的状态

* **线上/纸交易**：实时拿到 `ml_signal` → **策略**生成决策 → **Overlay** 硬约束执行 → **SLO** 同时监控数据延迟/下单失败/影子-真实偏差/推理时延&错误。
* **研究→生产**：WF/HPO 选择冠军 → 自动**重训+落盘+登记+切换 active**，并写入 `model_registry`；Grafana/CH 可同时追踪模型版本和指标。
* **回退**：Kill-switch 仍然一键降级；active 模型回滚只要把 symlink 指回上一版或 `promotion` 参数拒绝晋级即可。

如果你愿意，我还可以把 **ClickHouse 实时特征 feed（direct tailer）**也贴上（从 `features_1m` 轮询/订阅），但按“最快 production”的策略，上面这套已经能直接跑：
**WF/HPO → Promote → Paper/Prod（推理+策略+执行+风控+监控）**。
