````python
# llm/llm_client.py
from __future__ import annotations
from typing import Dict, Any, List, Optional
import json, os, time
from ops.secrets import get_secret

class LLMClient:
    """
    统一 LLM 调用（默认 OpenAI 兼容），也支持“本地/离线回落”（返回内置 safe 计划）。
    环境变量：
      - LLM_API_KEY      : API key（ops/secrets 已白名单）
      - LLM_BASE_URL     : 可选，自定义基址
      - LLM_MODEL        : 默认 gpt-4o-mini / gpt-4.1 / 自定义
    """
    def __init__(self, model: Optional[str] = None, base_url: Optional[str] = None, timeout: int = 60):
        self.model = model or os.getenv("LLM_MODEL", "gpt-4o-mini")
        self.base_url = base_url or os.getenv("LLM_BASE_URL", "https://api.openai.com/v1")
        self.api_key = get_secret("LLM_API_KEY", required=False)  # 没有就走离线回落
        self.timeout = timeout

    def generate(self, system_prompt: str, user_prompt: str, schema_hint: Optional[str] = None) -> Dict[str, Any]:
        """
        返回 JSON(dict)。如果接口失败或没密钥，走 _offline_fallback()。
        """
        try:
            if not self.api_key:
                return self._offline_fallback(user_prompt)
            import requests
            url = f"{self.base_url}/chat/completions"
            headers = {"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"}
            messages = [{"role": "system", "content": system_prompt},
                        {"role": "user", "content": user_prompt}]
            data = {"model": self.model, "messages": messages, "temperature": 0.2}
            r = requests.post(url, headers=headers, json=data, timeout=self.timeout)
            r.raise_for_status()
            txt = r.json()["choices"][0]["message"]["content"]
            # 允许 LLM 输出包裹在 ```json ...```，做 strip
            txt = txt.strip().strip("`")
            if txt.startswith("json"):
                txt = txt[4:].strip()
            return json.loads(txt)
        except Exception:
            return self._offline_fallback(user_prompt)

    # --------- 离线回落：给一个安全、保守、可跑通的默认“配方+计划” ----------
    def _offline_fallback(self, user_prompt: str) -> Dict[str, Any]:
        return {
            "feature_recipe": {
                "use_table": "features_1m",
                "base_features": ["close","vwap","ema_9","ema_21","rsi_14","atr_14","volume"],
                "derived": [
                    {"alias":"ret_1","expr":"log(close/lag(close,1))"},
                    {"alias":"mom_9","expr":"ema_9 - ema_21"},
                    {"alias":"atr_norm","expr":"atr_14 / nullif(close,0)"}
                ],
                "filters": ["close > 0", "volume > 0"]
            },
            "training_plan": {
                "symbol": "TSLA",
                "horizon": "5m",
                "task": "reg",
                "target": {"alias":"y", "expr":"log(lead(close,5)/close)"},
                "model_candidates": [
                    {"model_type":"xgb","model_params":{"n_estimators":400,"max_depth":6,"learning_rate":0.05,"subsample":0.9,"colsample_bytree":0.8,"early_stopping_rounds":50}},
                    {"model_type":"lstm","model_params":{"lookback":32,"hidden":64,"num_layers":1,"dropout":0.0,"lr":0.001,"batch_size":256,"epochs":80,"patience":8}}
                ],
                "decision": {"mode":"quantile","q_long":0.7,"q_short":0.3},
                "optuna_hints": {"n_trials": 40}
````

# 2) 将“特征配方/训练计划”编译为 ClickHouse SQL（含白名单与防注入）

**`pi/feature_sql.py`**

```python
# pi/feature_sql.py
from __future__ import annotations
from typing import Dict, Any, List, Tuple
import re

_ALLOWED_FUNCS = {
    # 安全函数白名单（ClickHouse 常用 + 简单算子）
    "abs","log","exp","pow","least","greatest","lag","lead","coalesce","nullif","if","caseWithExpression",
    "toFloat64","toFloat32","toInt32","toDateTime","toUnixTimestamp","round","roundBankers",
}
_ALLOWED_OPS = set("+-*/()=<>!%,")  # 允许运算符
_IDENT = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*$")

def _is_safe_ident(s: str) -> bool:
    return bool(_IDENT.match(s))

def _sanitize_expr(expr: str, feature_catalog: List[str]) -> str:
    """
    极简 sanitizer：
    - 允许的标识符：特征列、safe 函数、数字字面量、运算符。
    - 禁止分号、注释、子查询、DDL/DML 关键字。
    """
    if ";" in expr or "--" in expr or "/*" in expr:
        raise ValueError("Expr contains illegal tokens.")
    # 拆 token：字母串 / 数字 / 标点
    tokens = re.findall(r"[A-Za-z_][A-Za-z0-9_]*|\d+\.\d+|\d+|.", expr)
    out = []
    for t in tokens:
        if re.fullmatch(r"\d+\.\d+|\d+", t):
            out.append(t); continue
        if t in _ALLOWED_OPS or t.isspace():
            out.append(t); continue
        if t.lower() in _ALLOWED_FUNCS:
            out.append(t); continue
        if _is_safe_ident(t) and t in feature_catalog:
            out.append(t); continue
        # 别名引用允许
        if _is_safe_ident(t):
            out.append(t); continue
        raise ValueError(f"Illegal token: {t}")
    return "".join(out)

def build_select_sql(
    table: str,
    symbol: str,
    start_ts: str, end_ts: str,
    recipe: Dict[str, Any],
    target: Dict[str, Any],
    feature_catalog: List[str],
    ts_col: str = "ts",
    symbol_col: str = "symbol",
    cutoff_col: str = "feature_cutoff_ts"
) -> str:
    base = recipe.get("base_features", [])
    derived = recipe.get("derived", [])
    filters = recipe.get("filters", [])
    # 列清单
    select_cols = [symbol_col, ts_col] + base
    # 目标列
    if target:
        t_alias = target["alias"]
        t_expr = _sanitize_expr(target["expr"], feature_catalog + base)
        select_cols.append(f"{t_expr} AS {t_alias}")
    # 派生列
    for d in derived:
        alias = d["alias"]; expr = _sanitize_expr(d["expr"], feature_catalog + base)
        select_cols.append(f"{expr} AS {alias}")

    where = [f"{symbol_col} = '{symbol}'", f"{ts_col} >= toDateTime('{start_ts}')", f"{ts_col} < toDateTime('{end_ts}')"]
    if cutoff_col:
        where.append(f"{ts_col} >= {cutoff_col}")  # point-in-time
    for f in filters:
        where.append(_sanitize_expr(f, feature_catalog + base))

    sql = f"SELECT {', '.join(select_cols)} FROM {table} WHERE " + " AND ".join(where) + " ORDER BY {ts} ASC".format(ts=ts_col)
    return sql
```

```python
# pi/pi_orchestrator.py
from __future__ import annotations
from typing import Dict, Any, List, Tuple, Optional
import json, time, os
import clickhouse_connect
import pandas as pd
from llm.llm_client import LLMClient
from pi.feature_sql import build_select_sql
from grid_guard import sanitize_and_constrain_grid

_PI_SYS = """你是量化平台的首席科学家(Principal Investigator)。你必须返回JSON，严格遵守字段名和结构。不能有解释性文字。"""
_PI_USER_TMPL = """目标：{goal}
可用表：{table}
符号：{symbol}
时间范围：{start_ts} ~ {end_ts}
特征目录(列名)：{feature_cols}

请输出 JSON，包含：
- feature_recipe: {{"use_table":"{table}","base_features":[...],"derived":[{{"alias":"x","expr":"..."}},...],"filters":[...]}}
- training_plan: {{
    "symbol":"{symbol}",
    "horizon":"{horizon}",
    "task":"reg|clf",
    "target":{{"alias":"y","expr":"..."}},
    "model_candidates":[{{"model_type":"xgb"|"lstm","model_params":{{...}}}},...],
    "decision":{{"mode":"quantile","q_long":0.7,"q_short":0.3}},
    "optuna_hints":{{"n_trials": 40}}
}}"""

def _ch_client(cfg: Dict[str, Any]):
    return clickhouse_connect.get_client(
        host=cfg.get("host","localhost"),
        port=cfg.get("port",8123),
        username=cfg.get("username","default"),
        password=cfg.get("password",""),
        database=cfg.get("database","default"),
    )

def fetch_feature_catalog(client, table: str) -> List[str]:
    q = f"SELECT name FROM system.columns WHERE table = '{table}'"
    df = client.query_df(q)
    return sorted(df["name"].tolist())

class PrincipalInvestigator:
    def __init__(self, ch_cfg: Dict[str, Any], llm_cfg: Dict[str, Any]):
        self.client = _ch_client(ch_cfg)
        self.llm = LLMClient(
            model=llm_cfg.get("model"),
            base_url=llm_cfg.get("base_url")
        )

    def propose(self, *, table: str, symbol: str, start_ts: str, end_ts: str,
                goal: str, horizon: str, guard_cfg: Dict[str, Any] | None = None) -> Dict[str, Any]:
        catalog = fetch_feature_catalog(self.client, table)
        # 提示 LLM
        user = _PI_USER_TMPL.format(
            goal=goal, table=table, symbol=symbol, start_ts=start_ts, end_ts=end_ts,
            feature_cols=",".join(catalog), horizon=horizon
        )
        j = self.llm.generate(system_prompt=_PI_SYS, user_prompt=user)
        # 基础校验
        rec = j.get("feature_recipe") or {}
        plan = j.get("training_plan") or {}
        # 生成 SQL
        sql = build_select_sql(
            table=rec.get("use_table", table),
            symbol=plan.get("symbol", symbol),
            start_ts=start_ts, end_ts=end_ts,
            recipe=rec, target=plan.get("target"),
            feature_catalog=catalog
        )
        # 拉数据
        df = self.client.query_df(sql)
        # 把 LLM 候选模型组装为 HPO grid（+护栏）
        grid = {
            "model_type": [],
            "feature_subset": [list(set((rec.get("base_features") or []) + [d["alias"] for d in rec.get("derived",[])] + ["close"]))],
            "task": [plan.get("task","reg")],
            "decision": [plan.get("decision", {"mode":"quantile","q_long":0.7,"q_short":0.3})],
            "model_params": []
        }
        for c in (plan.get("model_candidates") or []):
            grid["model_type"].append(c.get("model_type","xgb"))
            grid["model_params"].append(c.get("model_params", {}))
        # 护栏（模型参数空间）
        grid_guard_cfg = (guard_cfg or {}) | {
            "allow_params": ["model_type","feature_subset","task","decision","model_params"]
        }
        grid_sane, _rep = sanitize_and_constrain_grid(grid, grid_guard_cfg)  # 重用你的护栏
        return {
            "feature_recipe": rec,
            "training_plan": plan,
            "sql": sql,
            "dataset_df": df,        # 直接给 WF/HPO 用（也可改成落 parquet）
            "hpo_grid": grid_sane
        }
```

```python
# === Paste this whole method into your wf_hpo.py, inside the WalkForwardHPO class ===
def run_with_dataframe(self, df):
    """
    Run Walk-Forward + (optional) HPO directly on a Pandas DataFrame (no DB I/O).
    Expected df columns:
      - 'ts' (datetime-like or int/str convertible to pandas datetime)
      - 'symbol'
      - 'y'   (target; if missing, a fallback label will be created from 'close')
      - feature columns (numeric)
    Uses self.cfg:
      - cfg["symbols"]           : list[str], will filter to first symbol if provided
      - cfg["grid"]              : dict[str, list], sanitized grid (supports model_type/xgb/lstm branch)
      - cfg["objective"]["formula"]: str, e.g. "0.6*sharpe + 0.25*sortino - 1.6*max_drawdown"
      - cfg["objective"]["maximize"]: bool
      - cfg["folds"]             : int, number of chronological folds (default=5)
      - cfg["embargo"]           : str like "5m" (optional)
      - cfg["cost_bps"]          : float, execution cost per position change (default=0)
      - cfg["annualize"]         : bool (default True)
      - cfg["label_horizon_steps"]: int (fallback when 'y' missing; default=5 -> lead 5 bars)
    Returns dict:
      {
        "rows": [ per-combo metrics & details ],
        "oos_summary": { aggregated metrics of the selected best combo },
        "best_combo": {grid combo dict},
        "best_model": <trained XGBModel/LSTMModel or None>
      }
    """
    import math, itertools, json, time
    import numpy as np
    import pandas as pd

    # Lazy imports from your model hooks
    from wf_model_hooks import train_predict_to_signal, build_model

    # ------------------------
    # 0) Utilities
    # ------------------------
    def _to_dt(s):
        x = pd.to_datetime(s, utc=False, errors="coerce")
        if x.isna().any():
            # try ms epoch
            try:
                x = pd.to_datetime(s.astype(np.int64), unit="ms", utc=False, errors="coerce")
            except Exception:
                pass
        return x

    def _parse_minutes(spec: str) -> int:
        """Return seconds for simple 'Xm'/'Xs' strings."""
        if not isinstance(spec, str) or not spec:
            return 0
        spec = spec.strip().lower()
        if spec.endswith("m"):
            return int(float(spec[:-1]) * 60)
        if spec.endswith("s"):
            return int(float(spec[:-1]))
        if spec.endswith("h"):
            return int(float(spec[:-1]) * 3600)
        return 0

    def _bars_per_year(ts_ser: pd.Series) -> float:
        t = _to_dt(ts_ser).sort_values()
        if len(t) < 3:
            return 252.0 * 390.0  # fallback: 1m bars
        step = (t.diff().dt.total_seconds().median() or 60.0)
        return (365.0 * 24.0 * 3600.0) / float(step)

    def _equity_from_signal(ts_idx: pd.Series, y: np.ndarray, sig: np.ndarray, cost_bps: float = 0.0):
        """
        y: simple returns per bar (NOT log). sig in {-1,0,1}. Costs charged when position changes.
        """
        y = np.asarray(y, dtype=float)
        sig = np.asarray(sig, dtype=float)
        sig_lag = np.roll(sig, 1); sig_lag[0] = 0.0
        # trading returns
        r = sig_lag * y
        # turnover cost (per change)
        dpos = np.abs(sig - sig_lag)
        cost = dpos * (cost_bps / 1e4)
        r_net = r - cost
        equity = np.cumprod(1.0 + r_net)
        return r_net, equity

    def _max_drawdown(eq: np.ndarray) -> float:
        if len(eq) == 0:
            return 0.0
        peak = np.maximum.accumulate(eq)
        dd = 1.0 - (eq / np.maximum(peak, 1e-12))
        return float(np.max(dd))

    def _omega_ratio(ret: np.ndarray, thr: float = 0.0) -> float:
        up = np.sum(np.clip(ret - thr, 0.0, None))
        dn = np.sum(np.clip(thr - ret, 0.0, None))
        return float(up / dn) if dn > 0 else float("inf")

    def _cagr(eq: np.ndarray, ts_ser: pd.Series) -> float:
        if len(eq) < 2:
            return 0.0
        t0, t1 = _to_dt(ts_ser).iloc[0], _to_dt(ts_ser).iloc[-1]
        span_sec = max(1.0, (t1 - t0).total_seconds())
        years = span_sec / (365.0 * 24.0 * 3600.0)
        ending = float(eq[-1])
        if ending <= 0:
            return -1.0
        return float(ending ** (1.0 / max(1e-9, years)) - 1.0)

    def _prob_sharpe_ratio(ret: np.ndarray, sr_bench: float = 0.0, ann: float = 1.0) -> float:
        # Bailey & López de Prado approximation
        from math import sqrt
        n = len(ret)
        if n < 5:
            return 0.5
        r = ret - np.nanmean(ret)
        m1 = np.nanmean(ret)
        s = np.nanstd(ret) + 1e-12
        sr = (m1 / s) * sqrt(ann)
        # sample skewness & kurtosis
        z = r / (np.nanstd(ret) + 1e-12)
        skew = float(np.nanmean(z**3))
        kurt = float(np.nanmean(z**4))
        num = (sr - sr_bench) * math.sqrt(n - 1)
        den = math.sqrt(1.0 - skew * sr + 0.5 * (kurt - 1.0) * (sr ** 2))
        zsc = num / max(1e-9, den)
        # Normal CDF
        return 0.5 * (1.0 + math.erf(zsc / math.sqrt(2.0)))

    def _deflated_sharpe_ratio(best_sr: float, all_srs: list[float], ret_len: int, ann: float = 1.0) -> float:
        """
        Approximate DSR following López de Prado:
        SR* = mu_SR + sigma_SR * z_(1 - 1/N)
        DSR = Phi( (best_sr - SR*) * sqrt(T-1)/sqrt(1 - skew*SR + 0.5*(kurt-1)*SR^2) )
        Here we lack per-strategy skew/kurt → use Gaussian correction.
        """
        import numpy as np, math
        N = max(1, len(all_srs))
        mu = float(np.nanmean(all_srs))
        sig = float(np.nanstd(all_srs) + 1e-12)
        # z_{1-1/N}
        from math import sqrt
        from math import erf
        def _norm_ppf(p):
            # Beasley-Springer/Moro-ish quick approx not needed; use binary search on CDF
            lo, hi = -8.0, 8.0
            for _ in range(60):
                mid = 0.5*(lo+hi)
                cdf = 0.5*(1.0+math.erf(mid/math.sqrt(2.0)))
                if cdf < p: lo = mid
                else: hi = mid
            return 0.5*(lo+hi)
        zq = _norm_ppf(1.0 - 1.0/max(2.0, float(N)))
        sr_star = mu + sig * zq
        if ret_len < 5:
            return 0.5
        zscore = (best_sr - sr_star) * math.sqrt(ret_len - 1)
        return 0.5 * (1.0 + math.erf(zscore / math.sqrt(2.0)))

    def _safe_eval(formula: str, vars_dict: dict) -> float:
        allowed = {k: float(v) for k, v in vars_dict.items()}
        allowed_names = set(allowed.keys()) | {"min","max","abs"}
        code = compile(formula, "<obj>", "eval")
        for name in code.co_names:
            if name not in allowed_names:
                raise ValueError(f"Illegal name in objective: {name}")
        return float(eval(code, {"__builtins__": {"min": min, "max": max, "abs": abs}}, allowed))

    def _metrics_from_series(ts_ser, ret_series, equity, ann_factor):
        ret = np.asarray(ret_series, dtype=float)
        eq = np.asarray(equity, dtype=float)
        mu = float(np.nanmean(ret))
        sd = float(np.nanstd(ret) + 1e-12)
        sharpe = (mu / sd) * math.sqrt(ann_factor) if sd > 0 else 0.0
        neg = ret[ret < 0]
        sortino = (mu / (np.nanstd(neg) + 1e-12)) * math.sqrt(ann_factor) if neg.size > 0 else 0.0
        mdd = _max_drawdown(eq)
        cagr = _cagr(eq, ts_ser)
        calmar = (cagr / (mdd + 1e-12)) if mdd > 0 else float("inf")
        win_mask = ret > 0
        win_rate = float(np.mean(win_mask)) if ret.size > 0 else 0.0
        gross_pos = float(np.sum(ret[ret > 0]))
        gross_neg = float(np.sum(-ret[ret < 0]))
        profit_factor = (gross_pos / gross_neg) if gross_neg > 0 else float("inf")
        omega = _omega_ratio(ret, 0.0)
        trades = int(np.sum(np.abs(np.diff(np.clip(np.sign(ret), -1, 1))) > 0))
        psr = _prob_sharpe_ratio(ret, 0.0, ann_factor)
        return {
            "sharpe": sharpe,
            "sortino": sortino,
            "max_drawdown": mdd,
            "cagr": cagr,
            "calmar": calmar,
            "win_rate": win_rate,
            "profit_factor": profit_factor,
            "omega": omega,
            "total_return": float(eq[-1] - 1.0) if len(eq) else 0.0,
            "psr": psr,
            "trades": trades
        }

    # ------------------------
    # 1) Prepare DataFrame
    # ------------------------
    cfg = getattr(self, "cfg", {}) or {}
    df = df.copy()
    if "ts" not in df.columns:
        raise ValueError("DataFrame must include 'ts' column.")
    df["ts"] = _to_dt(df["ts"])
    df = df.sort_values(["ts"]).reset_index(drop=True)

    # symbol filter
    symbols = cfg.get("symbols") or list(sorted(set(df.get("symbol", "SYMB").astype(str))))
    if symbols:
        sym = symbols[0]
        if "symbol" in df.columns:
            df = df[df["symbol"] == sym].copy()

    # target fallback
    if "y" not in df.columns:
        if "close" not in df.columns:
            raise ValueError("Missing 'y' and 'close'. Provide target or close price.")
        steps = int(cfg.get("label_horizon_steps", 5))
        close = df["close"].astype(float).values
        # simple return over horizon
        y = np.zeros(len(close), dtype=float)
        y[:-steps] = (close[steps:] / np.maximum(close[:-steps], 1e-12)) - 1.0
        df["y"] = y

    # feature candidates (exclude meta/target columns)
    base_exclude = {"ts", "symbol", "y"}
    feat_all = [c for c in df.columns if c not in base_exclude]

    # ------------------------
    # 2) Build folds (chronological)
    # ------------------------
    n_folds = int(cfg.get("folds", 5))
    if n_folds < 2:
        n_folds = 2
    t = df["ts"].values
    idx = np.arange(len(df))
    fold_bounds = []
    # Split OOS windows evenly across timeline
    oos_splits = np.linspace(0.6, 1.0, n_folds + 1)  # last 40% timeline split into n folds
    t0 = int(len(df) * 0.0)
    mid = int(len(df) * 0.6)
    for i in range(n_folds):
        oos_start = int(len(df) * oos_splits[i])
        oos_end = int(len(df) * oos_splits[i + 1])
        is_start = t0
        is_end = oos_start  # embargo applied below
        fold_bounds.append((is_start, is_end, oos_start, oos_end))
    embargo_sec = _parse_minutes(cfg.get("embargo", ""))

    # ------------------------
    # 3) Iterate grid combos
    # ------------------------
    grid = cfg.get("grid", {}) or {}
    # If grid is empty → default no-model path using all features
    if not grid:
        grid = {"model_type": ["xgb"], "feature_subset": [feat_all], "task": ["reg"], "model_params": [{}], "decision": [{"mode":"quantile","q_long":0.7,"q_short":0.3}]}

    def _cartesian(d):
        keys = list(d.keys())
        vals = [d[k] if isinstance(d[k], list) else [d[k]] for k in keys]
        for prod in itertools.product(*vals):
            yield {k: prod[i] for i, k in enumerate(keys)}

    ann_factor = _bars_per_year(df["ts"]) if bool(cfg.get("annualize", True)) else 1.0
    cost_bps = float(cfg.get("cost_bps", 0.0))
    objective = (cfg.get("objective") or {}).get("formula", "sharpe - max_drawdown")
    maximize = bool((cfg.get("objective") or {}).get("maximize", True))

    rows = []
    per_combo_sr_is = []   # for DSR/PBO proxy
    per_combo_sr_oos = []

    best_obj = -1e99 if maximize else 1e99
    best_combo = None
    best_fold_metrics = None

    # For promotion later
    best_model_instance = None

    for combo in _cartesian(grid):
        model_type = combo.get("model_type")
        feature_subset = combo.get("feature_subset", feat_all)
        task = combo.get("task", "reg")
        decision = combo.get("decision", {"mode":"quantile","q_long":0.7,"q_short":0.3})
        hparams = combo.get("model_params", {})

        fold_metrics = []
        sr_is_list = []
        sr_oos_list = []

        # --- Each fold ---
        for (is_s, is_e, oos_s, oos_e) in fold_bounds:
            # embargo: remove last 'embargo' seconds from IS tail
            if embargo_sec > 0:
                # find time threshold
                oos_start_time = df["ts"].iloc[oos_s]
                cutoff = oos_start_time - pd.Timedelta(seconds=embargo_sec)
                is_e_idx = int(df.index[df["ts"] < cutoff].max()) if (df["ts"] < cutoff).any() else max(is_s, oos_s - 1)
            else:
                is_e_idx = is_e

            df_IS = df.iloc[is_s:is_e_idx].copy()
            df_OOS = df.iloc[oos_s:oos_e].copy()

            # Skip too short splits
            if len(df_IS) < 100 or len(df_OOS) < 50:
                continue

            # Make sure features exist
            fcols = [c for c in feature_subset if c in df.columns]
            if len(fcols) == 0:
                continue

            y_IS = df_IS["y"].astype(float).values
            y_OOS = df_OOS["y"].astype(float).values

            # MODEL branch (xgb/lstm) OR (fallback none)
            if model_type in ("xgb", "lstm"):
                sig_is, sig_oos, _meta = train_predict_to_signal(
                    df_IS[fcols + ["ts","symbol"] + (["y"] if "y" in df_IS.columns else [])],
                    df_OOS[fcols + ["ts","symbol"] + (["y"] if "y" in df_OOS.columns else [])],
                    y_IS, y_OOS,
                    features=fcols,
                    model_type=model_type,
                    task=task,
                    hparams=hparams,
                    decision=decision
                )
            else:
                # No-model: zero signal
                sig_is = np.zeros(len(df_IS), dtype=float)
                sig_oos = np.zeros(len(df_OOS), dtype=float)

            # Convert label (y) to simple returns (assume 'y' is log-return if |y|<0.3 often)
            def _to_simple_ret(v):
                v = np.asarray(v, dtype=float)
                if np.nanmedian(np.abs(v)) < 0.3:
                    return np.expm1(v)  # treat as log-ret
                return v
            r_IS_simple = _to_simple_ret(y_IS)
            r_OOS_simple = _to_simple_ret(y_OOS)

            # Build equity & metrics
            ret_is, eq_is = _equity_from_signal(df_IS["ts"], r_IS_simple, sig_is, cost_bps=cost_bps)
            ret_oos, eq_oos = _equity_from_signal(df_OOS["ts"], r_OOS_simple, sig_oos, cost_bps=cost_bps)

            m_is = _metrics_from_series(df_IS["ts"], ret_is, eq_is, ann_factor)
            m_oos = _metrics_from_series(df_OOS["ts"], ret_oos, eq_oos, ann_factor)
            fold_metrics.append({"is": m_is, "oos": m_oos})

            sr_is_list.append(m_is["sharpe"])
            sr_oos_list.append(m_oos["sharpe"])

        if not fold_metrics:
            continue

        # Aggregate OOS
        def _mean_of(key):
            vals = [fm["oos"][key] for fm in fold_metrics if key in fm["oos"]]
            return float(np.nanmean(vals)) if vals else 0.0

        oos_agg = {
            "oos_avg_sharpe": _mean_of("sharpe"),
            "oos_avg_sortino": _mean_of("sortino"),
            "oos_avg_max_drawdown": _mean_of("max_drawdown"),
            "oos_avg_total_return": _mean_of("total_return"),
            "oos_avg_win_rate": _mean_of("win_rate"),
            "oos_avg_profit_factor": _mean_of("profit_factor"),
            "oos_avg_omega": _mean_of("omega"),
            "oos_avg_calmar": _mean_of("calmar"),
            "oos_trades": int(np.nanmean([fm["oos"]["trades"] for fm in fold_metrics])),
        }

        # PSR (OOS)
        all_oos_rets = []
        for (is_s, is_e, oos_s, oos_e), fm in zip(fold_bounds, fold_metrics):
            # approximate ret length via df_OOS slice length
            # ret series already net; rebuild again for PSR length
            pass
        # Instead, rebuild from last fold for PSR estimate (length representative)
        oos_agg["oos_psr"] = float(np.nanmean([fm["oos"]["psr"] for fm in fold_metrics]))

        # DSR using per-combo IS SRs
        dsr = _deflated_sharpe_ratio(
            best_sr=float(np.nanmean(sr_oos_list)),
            all_srs=sr_is_list,
            ret_len=int(np.nanmean([len(df.iloc[oos_s:oos_e]) for (_, _, oos_s, oos_e) in fold_bounds])),
            ann=ann_factor
        )
        oos_agg["oos_dsr"] = float(dsr)

        # crude PBO proxy from rank correlation (lower correlation → higher PBO)
        per_combo_sr_is.append(float(np.nanmean(sr_is_list)))
        per_combo_sr_oos.append(float(np.nanmean(sr_oos_list)))

        # Objective
        obj_vars = {
            "sharpe": oos_agg["oos_avg_sharpe"],
            "sortino": oos_agg["oos_avg_sortino"],
            "max_drawdown": oos_agg["oos_avg_max_drawdown"],
            "total_return": oos_agg["oos_avg_total_return"],
            "psr": oos_agg["oos_psr"],
            "dsr": oos_agg["oos_dsr"],
            "omega": oos_agg["oos_avg_omega"],
            "calmar": oos_agg["oos_avg_calmar"],
            "profit_factor": oos_agg["oos_avg_profit_factor"],
            "win_rate": oos_agg["oos_avg_win_rate"],
            "trades": float(oos_agg["oos_trades"]),
        }
        try:
            score = _safe_eval(objective, obj_vars)
        except Exception:
            score = oos_agg["oos_avg_sharpe"] - oos_agg["oos_avg_max_drawdown"]

        rows.append({
            "combo": {
                "model_type": model_type, "task": task,
                "feature_subset": feature_subset, "model_params": hparams, "decision": decision
            },
            "fold_metrics": fold_metrics,
            "oos_summary": oos_agg,
            "objective": float(score)
        })

        better = (score > best_obj) if maximize else (score < best_obj)
        if better:
            best_obj = score
            best_combo = rows[-1]
            best_fold_metrics = fold_metrics

    if not rows:
        return {"rows": [], "oos_summary": {}, "best_combo": None, "best_model": None}

    # PBO proxy across all combos (Spearman corr IS vs OOS SR)
    try:
        from scipy.stats import spearmanr
        rho, _ = spearmanr(per_combo_sr_is, per_combo_sr_oos)
        rho = float(0.0 if np.isnan(rho) else rho)
        pbo = max(0.0, min(1.0, 0.5 * (1.0 - rho)))  # heuristic: anticorrelation → overfitting risk
    except Exception:
        pbo = 0.2  # fallback
    # Attach to each row's oos_summary for consistency
    for r in rows:
        r["oos_summary"]["oos_pbo"] = float(pbo)

    # ------------------------
    # 4) Retrain best model on full dataset (for promotion)
    # ------------------------
    best = best_combo
    best_model_instance = None
    if best and best["combo"].get("model_type") in ("xgb","lstm"):
        model_type = best["combo"]["model_type"]
        task = best["combo"]["task"]
        fcols = [c for c in best["combo"]["feature_subset"] if c in df.columns]
        hparams = best["combo"]["model_params"] or {}
        # Build & train on full df (use 90/10 split for early stop)
        mdl = build_model(model_type, task, hparams)
        if not fcols:
            fcols = [c for c in df.columns if c not in {"ts","symbol","y"}]
        mdl.set_feature_cols(fcols)

        # Prepare y (simple returns)
        y_full = df["y"].astype(float).values
        y_full = np.expm1(y_full) if np.nanmedian(np.abs(y_full)) < 0.3 else y_full

        cut = int(len(df) * 0.9)
        df_tr, y_tr = df.iloc[:cut], y_full[:cut]
        df_va, y_va = df.iloc[cut:], y_full[cut:]
        try:
            mdl.fit(df_tr, y_tr, df_valid=df_va, y_valid=y_va)
            best_model_instance = mdl
        except Exception:
            best_model_instance = None

    # ------------------------
    # 5) Final best summary
    # ------------------------
    best_oos = dict(best["oos_summary"])
    best_oos["oos_pbo"] = float(pbo)

    return {
        "rows": rows,
        "oos_summary": best_oos,
        "best_combo": best["combo"] if best else None,
        "best_model": best_model_instance
    }
```
