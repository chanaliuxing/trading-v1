# -*- coding: utf-8 -*-
"""
alpha_loop.py
-------------
End-to-end "Chain-of-Alpha" skeleton:
LLM/ML proposes parameter grids -> run Walk-Forward HPO -> summarize ->
LLM critiques and proposes refined grids -> iterate.

Dependencies:
- wf_hpo.py (already provided)
- backtest_engine.py, sim_adapter.py (already provided)
- Optional: openai (pip install openai) for LLM; scikit-learn for ML surrogate

Usage:
  python alpha_loop.py --config /path/to/alpha_config.json --gens 3 --out_dir ./alpha_out --k-top 8 --provider llm

Config format: see alpha_config_example.json
"""

from __future__ import annotations
import os, sys, json, argparse, random
from typing import Any, Dict, List, Tuple
from copy import deepcopy
from pathlib import Path

from wf_hpo import WalkForwardHPO, make_folds

# ---------- Utils ----------

def _ensure_dir(p: str) -> None:
    Path(p).mkdir(parents=True, exist_ok=True)

def _load_json(path: str) -> Dict[str, Any]:
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def _dump_json(obj: Any, path: str) -> None:
    with open(path, "w", encoding="utf-8") as f:
        json.dump(obj, f, indent=2, ensure_ascii=False)

def _topk_rows(rows: List[Dict[str, Any]], k: int, key: str) -> List[Dict[str, Any]]:
    rows2 = [r for r in rows if r.get(key) is not None]
    rows2.sort(key=lambda r: r.get(key, 0.0), reverse=True)
    return rows2[:k]

def _expand_numeric(val, scale=0.2):
    try:
        v = float(val)
        lo = v * (1.0 - scale)
        hi = v * (1.0 + scale)
        return [round(lo, 6), round(v, 6), round(hi, 6)]
    except Exception:
        return [val]

def _infer_grid_from_top(rows: List[Dict[str, Any]], param_keys: List[str], widen=0.2) -> Dict[str, List[Any]]:
    # Create a new grid centered around median of top params
    if not rows:
        return {}
    import statistics as st
    params_list = [json.loads(r["params"]) if isinstance(r.get("params"), str) else r["params"] for r in rows]
    out = {}
    for k in param_keys:
        vals = [p.get(k) for p in params_list if k in p]
        vals = [v for v in vals if v is not None]
        if not vals:
            continue
        # numeric vs categorical
        try:
            nums = [float(v) for v in vals]
            med = st.median(nums)
            out[k] = _expand_numeric(med, scale=widen)
        except Exception:
            # categorical: keep top unique values (cap 5)
            uniq = []
            for v in vals:
                if v not in uniq:
                    uniq.append(v)
            out[k] = uniq[:5]
    return out

# ---------- LLM glue (optional) ----------

def _llm_available() -> bool:
    try:
        import openai  # type: ignore
        return True
    except Exception:
        return False

def _llm_propose_grid(context: Dict[str, Any], model: str = "gpt-4o-mini") -> Tuple[Dict[str, List[Any]], str]:
    """
    Ask LLM to propose a new grid. Returns (grid_dict, rationale_text).
    If unavailable, returns ({}, "fallback").
    """
    if not _llm_available():
        return ({}, "LLM unavailable; fallback.")
    try:
        from llm_client import ask_for_grid  # our thin wrapper
        return ask_for_grid(context, model=model)
    except Exception as e:
        return ({}, f"LLM error: {e}")

# ---------- ML surrogate (optional) ----------

def _ml_available() -> bool:
    try:
        import sklearn  # type: ignore
        return True
    except Exception:
        return False

def _ml_suggest(rows: List[Dict[str, Any]], target_key: str, n: int = 16) -> Dict[str, List[Any]]:
    """
    Very light surrogate using RandomForestRegressor (if sklearn available).
    Suggest n points via random search around promising areas.
    Returns a grid dict (param -> values) built from suggested points.
    """
    # Parse historical IS rows
    import numpy as np
    import pandas as pd
    try:
        from sklearn.ensemble import RandomForestRegressor  # type: ignore
    except Exception:
        return {}

    if not rows:
        return {}
    df = pd.DataFrame(rows)
    # Expand params JSON
    try:
        pexp = df["params"].apply(lambda s: json.loads(s) if isinstance(s, str) else (s or {}))
    except Exception:
        return {}
    P = pd.json_normalize(pexp)
    y = df[target_key].astype(float)

    # Encode categoricals (very simple)
    X = P.copy()
    for c in X.columns:
        if not np.issubdtype(X[c].dtype, np.number):
            X[c] = X[c].astype("category").cat.codes

    # Train RF
    try:
        rf = RandomForestRegressor(n_estimators=200, random_state=42)
        rf.fit(X, y)
    except Exception:
        return {}

    # Sample around top rows
    top = _topk_rows(rows, k=min(20, len(rows)), key=target_key)
    top_params = [json.loads(r["params"]) if isinstance(r["params"], str) else r["params"] for r in top]
    candidates = []
    rng = np.random.default_rng(42)
    for _ in range(n * 3):
        cand = {}
        for k in X.columns:
            # derive original values distribution
            col = P[k]
            if np.issubdtype(col.dtype, np.number):
                mu, sd = float(col.mean()), float(col.std() or 1.0)
                cand[k] = float(rng.normal(mu, sd))
            else:
                choices = list(col.dropna().unique())
                cand[k] = rng.choice(choices) if choices else None
        candidates.append(cand)

    # Score candidates
    C = pd.DataFrame(candidates)
    for c in C.columns:
        if not np.issubdtype(C[c].dtype, np.number):
            C[c] = C[c].astype("category").cat.codes
    preds = rf.predict(C.fillna(0))
    C["pred"] = preds
    C = C.sort_values("pred", ascending=False).head(n)

    # Convert top candidates to grid (per-dim unique values)
    grid: Dict[str, List[Any]] = {}
    for _, row in C.iterrows():
        for k, v in row.drop(labels=["pred"]).items():
            grid.setdefault(k, [])
            if v not in grid[k]:
                grid[k].append(v)
    # Try to keep 3-5 choices per key
    for k in grid:
        grid[k] = grid[k][:5]
    return grid

# ---------- Main loop ----------

def run_loop(cfg: Dict[str, Any], out_dir: str, gens: int, k_top: int, provider: str) -> None:
    _ensure_dir(out_dir)
    base_hpo_cfg = cfg["hpo"]  # a valid wf_hpo config body
    objective_key = base_hpo_cfg.get("objective", {}).get("metric", "sharpe")
    summary_key = "sharpe" if base_hpo_cfg.get("objective", {}).get("formula") is None else "total_return"

    history: List[Dict[str, Any]] = []  # per generation summary

    for g in range(1, gens+1):
        gen_dir = os.path.join(out_dir, f"gen_{g:02d}")
        _ensure_dir(gen_dir)

        # 1) Build HPO config for this generation
        hpo_cfg = deepcopy(base_hpo_cfg)

        # If we have previous rows, create a refined grid via LLM/ML
        if history:
            prev = history[-1]
            is_rows = [r for r in prev["rows"] if r["phase"] == "IS"]
            top_is = _topk_rows(is_rows, k=min(k_top, len(is_rows)), key=objective_key)

            # Prepare context for LLM
            context = {
                "symbols": hpo_cfg["symbols"],
                "objective": hpo_cfg.get("objective", {}),
                "top_is": top_is,
                "base_grid": hpo_cfg["grid"],
                "notes": "Propose a JSON 'grid' mapping param->list of candidate values, grounded in the top IS rows. Keep lists <=5 values; prefer numeric ranges around winners; avoid exploding combinations."
            }

            new_grid = {}
            rationale = ""

            if provider in ("llm", "hybrid"):
                grid_llm, rationale = _llm_propose_grid(context, model=cfg.get("llm", {}).get("model", "gpt-4o-mini"))
                if grid_llm:
                    new_grid = grid_llm

            if not new_grid and provider in ("ml", "hybrid"):
                # ML surrogate
                new_grid = _ml_suggest(is_rows, target_key=objective_key, n=cfg.get("ml", {}).get("n_points", 16))

            if not new_grid:
                # Heuristic fallback: median-centered expansion for keys present in base grid
                param_keys = list(hpo_cfg["grid"].keys())
                new_grid = _infer_grid_from_top(top_is, param_keys, widen=cfg.get("heuristic", {}).get("widen", 0.2))
                rationale = rationale or "Heuristic median-centered expansion."

            # Merge: prefer new grid but keep any keys absent in suggestion
            merged = dict(hpo_cfg["grid"])
            merged.update(new_grid)
            hpo_cfg["grid"] = merged

            # Write rationale
            _dump_json({"rationale": rationale, "new_grid": new_grid}, os.path.join(gen_dir, "refine.json"))

        # 2) Run HPO (walk-forward)
        runner = WalkForwardHPO(hpo_cfg)
        result = runner.run()
        artifacts = runner.export_tables(result, gen_dir)
        try:
            plots = runner.export_plots(result, gen_dir)
            artifacts.update(plots)
        except Exception as e:
            artifacts["plot_error"] = str(e)

        # 3) Persist this generation
        payload = {
            "gen": g,
            "hpo_cfg": hpo_cfg,
            "artifacts": artifacts,
            "oos_summary": result.get("oos_summary", {}),
            "best_params_by_fold": result.get("best_params_by_fold", []),
            "rows": result.get("rows", [])
        }
        _dump_json(payload, os.path.join(gen_dir, "gen_summary.json"))
        history.append(payload)

    # Write a final index
    idx = {
        "generations": [{
            "gen": p["gen"],
            "oos_summary": p["oos_summary"],
            "artifacts": p["artifacts"]
        } for p in history]
    }
    _dump_json(idx, os.path.join(out_dir, "index.json"))

# ---------- CLI ----------

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--config", required=True, help="alpha loop config json")
    ap.add_argument("--out_dir", required=False, default="./alpha_out", help="output directory")
    ap.add_argument("--gens", type=int, default=3, help="number of generations")
    ap.add_argument("--k-top", type=int, default=8, help="top-K IS rows to guide refinement")
    ap.add_argument("--provider", choices=["llm","ml","hybrid","none"], default="hybrid", help="who proposes the next grid")
    args = ap.parse_args()

    cfg = _load_json(args.config)
    run_loop(cfg, args.out_dir, gens=args.gens, k_top=args.k_top, provider=args.provider)

if __name__ == "__main__":
    main()
''')

# -*- coding: utf-8 -*-
"""
llm_client.py
-------------
Minimal LLM wrapper. Uses OpenAI if available.
Environment:
  export OPENAI_API_KEY=...
"""

from __future__ import annotations
import os, json
from typing import Any, Dict, List, Tuple

PROMPT = """You are a quantitative research assistant. You will receive:
- objective: dict describing scoring and constraints
- base_grid: current grid (param -> list of values)
- top_is: array of top in-sample rows: each has keys ["params", "sharpe", "total_return", "max_drawdown", "profit_factor", "win_rate", "sortino", "omega", "calmar"]
Goal:
- Propose a refined parameter grid as pure JSON: {"grid": {<param>: [values...]}, "notes": "<short rationale>"}
Rules:
- Keep <= 5 values per param; prefer numeric refinements around winners; prune weak params
- Do NOT add commentary outside JSON. Return ONLY JSON.
"""

def ask_for_grid(context: Dict[str, Any], model: str = "gpt-4o-mini") -> Tuple[Dict[str, List[Any]], str]:
    try:
        import openai  # type: ignore
    except Exception as e:
        raise RuntimeError("openai package not available") from e

    client = openai.OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

    msg = [
        {"role": "system", "content": PROMPT},
        {"role": "user", "content": json.dumps(context, ensure_ascii=False)}
    ]
    resp = client.chat.completions.create(
        model=model,
        messages=msg,
        temperature=0.2,
        response_format={"type":"json_object"}
    )
    txt = resp.choices[0].message.content
    data = json.loads(txt)
    grid = data.get("grid", {})
    notes = data.get("notes", "")
    return grid, notes
''')

## 这套骨架如何闭环

1. **生成参数/公式**

   * LLM：基于上一代 Top-K 的 IS 明细，按你的 `objective`（或公式）提出**新网格**（<=5 值/维，围绕赢家收缩/扩展）。
   * ML：小型随机森林替代模型（可选），在历史 IS 数据上拟合，提出**高分候选点**再转为网格。
   * 无依赖回退：自动用**中位数中心扩展**的启发式。

2. **严格回测裁判**

   * 多标的**严格 MTM**、**交易账本**，胜率/PF 更准确；
   * Walk-Forward + Purge Gap；并行网格搜索；支持 Sortino/Omega/Calmar/自定义公式。

3. **分析与进化**

   * 导出榜单、曲线与热图；
   * 将结果摘要再喂给 LLM/ML → 产生下一代网格 → 重复。
