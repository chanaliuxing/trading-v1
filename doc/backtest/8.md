# portfolio_overlay.py
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Dict, Any, Optional, Tuple, List
import math
import time
from collections import deque

@dataclass
class OverlayConfig:
    vol_target_annual: float = 0.15       # 年化目标波动(15%)
    vol_window: int = 390                 # 1m bar 下约1日窗口（按需改）
    max_position_pct: float = 0.08        # 单票市值占NAV上限
    max_gross: float = 2.0                # 总多空名义敞口/NAV
    max_net: float = 0.5                  # 净敞口/NAV
    rebalance: str = "bar"                # "bar"|"daily"
    max_drawdown: float = 0.35            # 回撤熔断（35%）
    turnover_bps_budget: float = 30.0     # 单bar 目标换手(成本)预算（软约束）
    min_lot: int = 1                      # 最小成交单位
    vol_floor: float = 0.05               # 估算波动的下界，避免过度加杠杆
    vol_cap: float = 0.8                  # 上界，避免过度减仓
    enable: bool = True

@dataclass
class PortfolioState:
    # rolling portfolio returns for vol estimate
    rets: deque = field(default_factory=lambda: deque(maxlen=390*5))  # ~1周
    peak_nav: float = 1.0
    halted: bool = False
    last_rebalance_day: Optional[int] = None
    leverage: float = 1.0

class PortfolioOverlay:
    """
    组合层风控与杠杆控制：在执行前裁剪/缩放目标交易量。
    外部接口：
      - on_bar(symbol, px, nav, positions, ts)
      - limit_order(order, nav, positions) -> adjusted_qty
    """
    def __init__(self, cfg: OverlayConfig):
        self.cfg = cfg
        self.state = PortfolioState()
        self._px: Dict[str, float] = {}

    def _estimate_annual_vol(self) -> float:
        # 简易：用滚动std(rets) * sqrt(252)
        if len(self.state.rets) < 10:
            return self.cfg.vol_target_annual  # 冷启动
        import numpy as np
        s = float(np.std(self.state.rets))
        return max(1e-9, s * math.sqrt(252.0))

    def _recalc_leverage(self):
        cur_vol = self._estimate_annual_vol()
        # 目标杠杆 ≈ vol_target / max(cur_vol, vol_floor)
        lev = self.cfg.vol_target_annual / max(self.cfg.vol_floor, cur_vol)
        # 限幅
        lev = min(max(lev, 1.0/self.cfg.vol_cap), 1.0/self.cfg.vol_floor)
        self.state.leverage = float(lev)

    def on_bar(self, symbol: str, px: float, nav: float, positions: Dict[str, Dict[str, float]], ts: int):
        if not self.cfg.enable: 
            return
        self._px[symbol] = float(px)
        # 估算组合收益：ΔNAV/NAV_{t-1}，外部建议每bar调用一次并传当前NAV
        # 这里做近似：由持仓变动带来的市值变化除以 NAV（需要外部确保按bar顺序调用）
        # 若你在上层有更精确的组合收益，直接替换为该值即可。
        # 简化：不重复计算，由上层传入更稳妥；此处用 px 变动近似（可选）
        # ——为稳健，若外部未提供组合收益，这里跳过，仅用vol_floor冷启动
        # 回撤熔断
        self.state.peak_nav = max(self.state.peak_nav, nav)
        dd = 0.0 if self.state.peak_nav <= 0 else (self.state.peak_nav - nav)/self.state.peak_nav
        if dd >= self.cfg.max_drawdown:
            self.state.halted = True
        # rebalance节律
        if self.cfg.rebalance == "daily":
            day = int(ts // (1000*60*60*24))
            if self.state.last_rebalance_day != day:
                self.state.last_rebalance_day = day
                self._recalc_leverage()
        else:
            self._recalc_leverage()

    def _projected_exposures(self, order: Dict[str, Any], nav: float, positions: Dict[str, Dict[str, float]]) -> Tuple[float,float,float]:
        """
        返回 (gross_after, net_after, pos_pct_after_sym)
        gross = sum(|qty_i*px_i|)/NAV; net = sum(qty_i*px_i)/NAV
        """
        sym = order["symbol"]
        side = order["side"].upper()
        qty  = int(order["qty"])
        px   = float(order.get("price_hint") or order.get("bar", {}).get("close") or order.get("bar", {}).get("open") or self._px.get(sym, 0.0))
        if px <= 0 or nav <= 0:
            return 0.0, 0.0, 0.0

        cur = {s: p.get("qty",0) for s, p in positions.items()}
        cur_val = {s: (p.get("qty",0) * self._px.get(s, px)) for s,p in positions.items()}
        cur_gross = sum(abs(v) for v in cur_val.values()) / nav
        cur_net   = sum(v for v in cur_val.values()) / nav
        sym_new_qty = cur.get(sym, 0) + (qty if side=="BUY" else -qty)
        sym_after_val = sym_new_qty * px
        gross_after = ( (cur_gross*nav) - abs(cur.get(sym,0)*px) + abs(sym_after_val) ) / nav
        net_after   = ( (cur_net*nav)   - (cur.get(sym,0)*px) + sym_after_val ) / nav
        pos_pct_after = abs(sym_after_val)/nav
        return float(gross_after), float(net_after), float(pos_pct_after)

    def limit_order(self, order: Dict[str, Any], nav: float, positions: Dict[str, Dict[str, float]]) -> int:
        """
        返回调整后的 qty（可能缩小或为0）
        """
        if not self.cfg.enable or self.state.halted:
            return 0 if self.state.halted else int(order["qty"])

        qty = int(order["qty"])
        if qty == 0 or nav <= 0:
            return 0

        # 杠杆缩放（soft）
        lev = self.state.leverage
        qty = int(max(self.cfg.min_lot, round(qty * lev)))

        # 硬约束：单票上限、gross/net 上限
        # 二分缩量直至满足
        lo, hi = 0, qty
        best = 0
        tries = 0
        while lo <= hi and tries < 20:
            mid = (lo + hi)//2
            tmp = dict(order); tmp["qty"] = mid
            g_after, n_after, pos_pct = self._projected_exposures(tmp, nav, positions)
            ok_pos  = pos_pct <= self.cfg.max_position_pct + 1e-9
            ok_gross= g_after <= self.cfg.max_gross + 1e-9
            ok_net  = abs(n_after) <= self.cfg.max_net + 1e-9
            if ok_pos and ok_gross and ok_net:
                best = mid; lo = mid + 1
            else:
                hi = mid - 1
            tries += 1

        # 换手预算（软约束）：如需，可在此再缩量（此处留钩子）
        return int(max(0, best))

# --------- Alerts ---------
class AlertManager:
    def __init__(self, webhooks: list[str] | None = None, enabled: bool = True):
        self.webhooks = webhooks or []
        self.enabled = enabled

    def send(self, title: str, text: str):

async def csv_feed(path: str) -> AsyncGenerator[Dict[str, Any], None]:
    """
    读取 CSV (或 Parquet) 的增量脚本/准实时文件，按顺序吐出 features 行。
    要求列包含：ts (ms), symbol, open, high, low, close, volume, 以及策略需要的特征列。

    if not p.exists():
        raise FileNotFoundError(path)
    if p.suffix.lower() == ".parquet":
        df = pd.read_parquet(p)
    else:
        df = pd.read_csv(p)
    df = df.sort_values("ts")
    for _, row in df.iterrows():
        yield row.to_dict()

# --------- Strategy loader ---------
def load_strategy_from_file(py_path: str, class_name: str, engine, params: Dict[str, Any]) -> StrategyAlgorithm:
    spec = importlib.util.spec_from_file_location("user_strategy", py_path)
    module = importlib.util.module_from_spec(spec)
    assert spec and spec.loader
    spec.loader.exec_module(module)  # type: ignore
    cls = getattr(module, class_name)
    return cls(strategy_id=params.get("strategy_id","paper"), engine=engine, params=params)

# --------- Paper Engine ---------
class PaperEngine:
    def __init__(self, cfg: Dict[str, Any]):
        self.cfg = cfg
        self.run_id = cfg.get("run_id") or f"paper_{int(time.time())}"
        self.sim = SimAdapter(  # 已具有组合Overlay与费用/滑点
            commission_per_share=cfg.get("commission_per_share",0.005),
            commission_min=cfg.get("commission_min",1.0),

            dynamic_spread_bps=cfg.get("dynamic_spread_bps",0.0),

            fill_ref=cfg.get("fill_ref","open"),

        self.dd_halt = float(cfg.get("halt", {}).get("max_drawdown", 0.35))
        self.min_nav = float(cfg.get("halt", {}).get("min_nav", 1000.0))
        self.client = _ch_client_or_none(cfg)
        _ch_ensure_live_tables(self.client)
        # Strategy
        strat_cfg = cfg.get("strategy", {})
        self.strategy = load_strategy_from_file(
            strat_cfg["path"],
            strat_cfg["class"],
            engine=self,
            params={"extras": strat_cfg.get("params", {}), "strategy_id": self.run_id}
        )
        # ledger hook
        self._fills_since_last = []

    # 提供给策略的提交接口
    async def submit_intent(self, strategy_id: str, decision):
        """
        decision: DecisionEnvelope，或等价 dict，包含 symbol/side/target_qty 等。
        Paper 引擎将它翻译成 SimAdapter 的订单（市价/限价根据策略字段决定）。
        """
        sym = getattr(decision, "symbol", None) or decision.get("symbol")
        side = getattr(decision, "side", None) or decision.get("side")
        qty  = int(getattr(decision, "target_qty", 0) or decision.get("target_qty", 0))
        typ  = getattr(decision, "order_type", "MKT") if hasattr(decision, "order_type") else decision.get("order_type","MKT")
        lmt  = getattr(decision, "limit_price", None) if hasattr(decision, "limit_price") else decision.get("limit_price")

        order = {"symbol": sym, "side": side, "qty": qty, "type": typ, "limit_price": lmt, "ts": int(time.time()*1000)}
        await self.sim.submit_order(order)
        # 记录（可选入库）
        # fill 在 SimAdapter 的 ledger 中，统一在 on_market_data 后批量落库

    async def run(self, feed_kind: str, feed_path: str):
        # 初始化策略
        if hasattr(self.strategy, "initialize"):
            await self.strategy.initialize()

        # feed
        if feed_kind.lower() in ("csv","parquet"):
            agen = csv_feed(feed_path)
        else:
            raise ValueError(f"unsupported feed: {feed_kind}")

        stop_flag = False
        loop = asyncio.get_running_loop()
        for s in (signal.SIGINT, signal.SIGTERM):
            try:
                loop.add_signal_handler(s, lambda: self._set_stop())
            except Exception:
                pass

        async for feat in agen:
            if stop_flag or getattr(self, "_stop", False):
                break
            # 1) 推给 SimAdapter 做 MTM
            await self.sim.on_market_data(feat)
            # 2) 调策略
            await self.strategy.on_features(feat.get("symbol"), feat)
            # 3) 熔断&落库&告警
            nav = self.sim.current_nav({feat["symbol"]: float(feat.get("close", 0.0))})
            peak = getattr(self.sim.overlay.state, "peak_nav", max(1.0, nav))
            dd = 0.0 if peak <= 0 else (peak - nav)/peak
            halted = dd >= self.dd_halt or nav <= self.min_nav or self.sim.overlay.state.halted
            # 告警
            if halted:
                self.alerts.send("HALT", f"run={self.run_id} dd={dd:.3f} nav={nav:.2f}")
                self._insert_live_runs(nav, peak, dd, 1, notes="halt")
                break
            # 常规落库
            self._insert_live_runs(nav, peak, dd, 0, notes="")
            # flush fills
            self._flush_fills_to_ch()

        # 结束
        self._flush_fills_to_ch(final=True)

    def _insert_live_runs(self, nav: float, peak_nav: float, dd: float, halted: int, notes: str):
        if self.client is None: return
        row = [[self.run_id, "paper", notes or "", float(nav), float(peak_nav), float(dd), int(halted), notes or ""]]
        self.client.insert("live_runs", row, column_names=["run_id","mode","ts","nav","peak_nav","dd","halted","notes"])

    def _flush_fills_to_ch(self, final: bool=False):
        if self.client is None: return
        # 读取 SimAdapter.ledger 的新增记录
        batch = []
        while self.sim.ledger:
            ev = self.sim.ledger.pop(0)
            ts = int(ev.get("ts") or int(time.time()*1000))
            fill_px = float(ev.get("price", 0.0))
            fee = float(ev.get("fee", 0.0))
            side = str(ev.get("side",""))
            qty = int(ev.get("qty", 0))
            sym = str(ev.get("symbol",""))
            batch.append([self.run_id, ts, sym, side, qty, fill_px, fill_px, fee])
        if batch:
            self.client.insert("live_orders", batch, column_names=["run_id","ts_ms","symbol","side","qty","ref_px","fill_px","fee"])

    def _set_stop(self):
        setattr(self, "_stop", True)

from __future__ import annotations

@dataclass
class SLOThresholds:

    order_attempts: Deque[int] = field(default_factory=deque)        # ts
    order_failures: Deque[int] = field(default_factory=deque)        # ts
    slippages_bps: Deque[Tuple[int,float]] = field(default_factory=deque)   # (ts, bps)
    shadow_real_bps: Deque[Tuple[int,float]] = field(default_factory=deque) # (ts, bps)
    last_flush_ts: int = 0

class SLOMonitor:
    def __init__(self, thresholds: SLOThresholds, alert_cb=None, ch_client=None, run_id: str = "run"):
        self.t = thresholds
        self.s = SLOState()
        self.alert = alert_cb or (lambda title, text: None)
        self.ch = ch_client
        self.run_id = run_id

    def _now_ms(self) -> int:
        return int(time.time()*1000)

    def _trim(self, q: Deque, now_ms: int):
        cutoff = now_ms - self.t.window_sec*1000
        while q and (q[0][0] if isinstance(q[0], tuple) else q[0]) < cutoff:
            q.popleft()

    # ---- 记录指标 ----
    def record_data_point_latency(self, market_ts_ms: int, recv_ts_ms: Optional[int] = None):
        now = recv_ts_ms or self._now_ms()
        lag = max(0, now - market_ts_ms)
        self.s.latencies.append((now, lag))
        self._trim(self.s.latencies, now)

    def record_order_attempt(self):
        now = self._now_ms()
        self.s.order_attempts.append(now)
        self._trim(self.s.order_attempts, now)

    def record_order_failure(self):
        now = self._now_ms()
        self.s.order_failures.append(now)
        self._trim(self.s.order_failures, now)

    def record_slippage_bps(self, ref_px: float, fill_px: float, side: str):
        if ref_px <= 0 or fill_px <= 0: return
        # 买：fill>=ref 为负向滑点；卖：fill<=ref 为负向滑点，统一用绝对bps评价
        bps = abs((fill_px - ref_px)/ref_px)*1e4
        now = self._now_ms()
        self.s.slippages_bps.append((now, bps))
        self._trim(self.s.slippages_bps, now)

    def record_shadow_real_diff_bps(self, shadow_fill: float, real_fill: float):
        if shadow_fill <= 0 or real_fill <= 0: return
        bps = abs((real_fill - shadow_fill)/shadow_fill)*1e4
        now = self._now_ms()
        self.s.shadow_real_bps.append((now, bps))
        self._trim(self.s.shadow_real_bps, now)

    # ---- 计算 & 告警 ----
    def compute_window_metrics(self) -> Dict[str, float]:
        now = self._now_ms()
        def avg(q: Deque[Tuple[int,float]]) -> float:
            if not q: return 0.0
            return float(sum(v for _, v in q)/len(q))
        lag_avg = avg(self.s.latencies)
        err_rate = (len(self.s.order_failures)/max(1,len(self.s.order_attempts)))
        slip_avg = avg(self.s.slippages_bps)
        diff_avg = avg(self.s.shadow_real_bps)
        return {
            "data_lag_ms_avg": lag_avg,
            "order_error_rate": err_rate,
            "slippage_bps_avg": slip_avg,
            "shadow_real_diff_bps_avg": diff_avg,
            "window_sec": self.t.window_sec
        }

    def evaluate_and_alert(self):
        m = self.compute_window_metrics()
        alerts = []
        if m["data_lag_ms_avg"] > self.t.max_data_lag_ms:
            alerts.append(f"DataLag {m['data_lag_ms_avg']:.0f}ms > {self.t.max_data_lag_ms}ms")
        if m["order_error_rate"] > self.t.max_order_error_rate:
            alerts.append(f"OrderError {m['order_error_rate']:.3f} > {self.t.max_order_error_rate}")
        if m["shadow_real_diff_bps_avg"] > self.t.max_shadow_real_diff_bps:
            alerts.append(f"ShadowRealDiff {m['shadow_real_diff_bps_avg']:.1f}bps > {self.t.max_shadow_real_diff_bps}bps")
        for msg in alerts:
            self.alert("SLO Breach", f"run={self.run_id} {msg}")

    # ---- ClickHouse 落库（可选）----
    def flush_to_clickhouse(self):
        if not self.ch: return
        m = self.compute_window_metrics()
        row = [[self.run_id, int(self._now_ms()), 
                float(m["data_lag_ms_avg"]), float(m["order_error_rate"]),
                float(m["slippage_bps_avg"]), float(m["shadow_real_diff_bps_avg"]),
                int(m["window_sec"])]]
        self.ch.insert("live_metrics", row, column_names=[
            "run_id","ts_ms","data_lag_ms_avg","order_error_rate","slippage_bps_avg","shadow_real_diff_bps_avg","window_sec"
        ])
```

class SecretError(Exception): pass

def _load_local_vault(path: str) -> dict:
    if not path or not os.path.exists(path): return {}

    except Exception:
        return {}

def get_secret(name: str, vault_path: Optional[str] = None, required: bool = True) -> str:
    """
    读取优先级：环境变量 > 本地只读 vault.json
    仅允许白名单键名（避免误读其他敏感变量），并且不在日志中打印。
    """
    ALLOW = {
        "LLM_API_KEY","CLICKHOUSE_PASSWORD","BROKER_KEY","BROKER_SECRET","WEBHOOK_TOKEN"
    }
    if name not in ALLOW:
        raise SecretError(f"Secret {name} not allowed (whitelist).")
    v = os.getenv(name)
    if v: return v
    vault = _load_local_vault(vault_path or os.getenv("ALPHA_VAULT","/etc/alpha/vault.json"))
    v = vault.get(name)
    if not v and required:
        raise SecretError(f"Missing secret: {name}")
    return v or ""

from __future__ import annotations
from typing import Dict, Any, Optional
from services.backtest.sim_adapter import SimAdapter
from broker.adapter import BrokerAdapter

class DualRouter:
    """
    mode:
      - "shadow": 只影子成交（SimAdapter）
      - "canary": 部分比例真实，其余影子
      - "prod":   全真实
    """
    def __init__(self, sim: SimAdapter, real: BrokerAdapter, mode: str="shadow", canary_ratio: float=0.1, slo=None):
        self.sim, self.real = sim, real
        self.mode = mode
        self.canary_ratio = float(canary_ratio)
        self.slo = slo

    async def route(self, order: Dict[str,Any], ref_px: float):
        sym, side, qty = order["symbol"], order["side"], int(order["qty"])
        if self.mode == "prod":
            r = self.real.place_order(sym, side, qty, order.get("type","MKT"), order.get("limit_price"))
            if not r.get("ok"):
                if self.slo: self.slo.record_order_failure()
            return {"real": r, "shadow": None}
        elif self.mode == "canary":
            q_real = max(0, int(round(qty * self.canary_ratio)))
            q_shadow = qty - q_real
            r_real = self.real.place_order(sym, side, q_real, order.get("type","MKT"), order.get("limit_price")) if q_real>0 else {"ok":True}
            if q_shadow>0:
                o2 = dict(order); o2["qty"] = q_shadow
                await self.sim.submit_order(o2)
            return {"real": r_real, "shadow": {"ok": True}}
        else:  # shadow
            await self.sim.submit_order(order)
            return {"real": None, "shadow": {"ok": True}}
```

from typing import List, Dict, Any
import time

def realtime_compare_and_log(run_id: str, ch_client, shadow_fills: List[Dict[str,Any]], real_fills: List[Dict[str,Any]], slo=None):
    """
    简化匹配：按 (symbol, side, qty) 在近窗口内近似匹配，记录 shadow vs real 价格差。
    """
    # 索引真实成交
    real_ix: Dict[tuple, List[Dict[str,Any]]] = {}
    for f in real_fills:
        key = (f.get("symbol"), f.get("side"), int(f.get("qty",0)))
        real_ix.setdefault(key, []).append(f)
    inserts = []
    for s in shadow_fills:
        key = (s.get("symbol"), s.get("side"), int(s.get("qty",0)))
        candidates = real_ix.get(key, [])
        if not candidates: 
            continue
        r = candidates.pop(0)
        try:
            shadow_px = float(s.get("price",0.0)); real_px = float(r.get("price",0.0))
            diff_bps = abs((real_px - shadow_px)/shadow_px)*1e4 if shadow_px>0 else 0.0
            inserts.append([run_id, int(time.time()*1000), str(s.get("symbol")), str(s.get("side")), int(s.get("qty",0)), shadow_px, real_px, diff_bps])
            if slo: slo.record_shadow_real_diff_bps(shadow_px, real_px)
        except Exception:
            pass
    if inserts and ch_client:
        ch_client.insert("exec_compare", inserts, column_names=["run_id","ts_ms","symbol","side","qty","shadow_px","real_px","diff_bps"])

def eod_reconcile_positions(ch_client, run_id: str, internal_positions: Dict[str,Dict[str,float]], broker_positions: Dict[str,Dict[str,float]]):
    """
    T+1 对账：按 symbol 对数量与估值比对，记录差异（此处省略落库表，按需扩展）
    返回差异 dict。
    """
    diff = {}
    for sym, pos in internal_positions.items():
        q_int = int(pos.get("qty",0))
        q_real = int((broker_positions.get(sym) or {}).get("qty",0))
        if q_int != q_real:
            diff[sym] = {"internal": q_int, "broker": q_real}
    return diff
