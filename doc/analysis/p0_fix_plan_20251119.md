def _get_model_confidence_multiplier(
    action: Dict[str, Any],
    symbol: str,
    model_family: str = "",
    horizon: str = ""
) -> float:
    """
    根据模型真实战绩计算 confidence 倍数。

    Returns:
        float: confidence multiplier (0.0 ~ 2.0)
               - 0.0: 模型太烂，禁用
               - 0.5: 模型一般，降权
               - 1.0: 模型正常
               - 1.5-2.0: 模型表现优异，加权
    """
    from scripts.ai_trader.brain.policy.memory import get_model_stats_summary_from_ch

    # 获取模型统计
    stats = get_model_stats_summary_from_ch(
        symbol=symbol,
        model_family=model_family,
        horizon=horizon,
        window_days=0  # 使用默认 30 天
    )

    # Gating 逻辑
    sharpe = stats.get("latest_sharpe", 0.0)
    success_rate = stats.get("success_rate_30d", 0.5)
    fail_streak = stats.get("fail_streak", 0)
    attempts = stats.get("attempts_30d", 0)

    # 硬门槛：直接禁用
    if fail_streak >= 5:
        logger.warning(
            "model_disabled_fail_streak",
            symbol=symbol,
            model_family=model_family,
            fail_streak=fail_streak,
            reason="连续失败次数过多"
        )
        return 0.0  # 禁用

    if attempts >= 10 and sharpe < -0.5:
        logger.warning(
            "model_disabled_negative_sharpe",
            symbol=symbol,
            model_family=model_family,
            sharpe=sharpe,
            reason="Sharpe 严重负值"
        )
        return 0.0  # 禁用

    if attempts >= 10 and success_rate < 0.3:
        logger.warning(
            "model_disabled_low_success_rate",
            symbol=symbol,
            model_family=model_family,
            success_rate=success_rate,
            reason="成功率过低"
        )
        return 0.0  # 禁用

    # 软门槛：调整权重
    if attempts < 5:
        # 样本不足，使用中性权重
        return 1.0

    # 基于 Sharpe 的权重调整
    if sharpe >= 1.5:
        multiplier = 1.5  # 优秀模型，加权
    elif sharpe >= 1.0:
        multiplier = 1.2  # 良好模型，略微加权
    elif sharpe >= 0.5:
        multiplier = 1.0  # 正常模型
    elif sharpe >= 0.0:
        multiplier = 0.7  # 平庸模型，降权
    else:
        multiplier = 0.5  # 表现较差，大幅降权

    # 基于成功率的微调
    if success_rate > 0.6:
        multiplier *= 1.1  # 成功率高，额外加成
    elif success_rate < 0.4:
        multiplier *= 0.9  # 成功率低，额外惩罚

    logger.info(
        "model_confidence_calibrated",
        symbol=symbol,
        model_family=model_family,
        sharpe=round(sharpe, 3),
        success_rate=round(success_rate, 3),
        fail_streak=fail_streak,
        attempts=attempts,
        multiplier=round(multiplier, 3)
    )

    return multiplier
```

# 原来的代码（伪代码示例）
if action_type == "HPO" or action_type == "ALPHA":
    ms = memory_summary.get(str(action_type)) or {}

    # 基于文件扫描的 memory_summary 做决策
    if ms.get("recent_failures", 0) > 3:
        return {"approved": False, "reason": "too_many_failures"}

# 新代码
if action_type in ["HPO", "ALPHA", "OOF"]:
    # 1. 提取模型信息
    symbol = action.get("symbol", "")
    model_family = action.get("model_family", "xgb")
    horizon = action.get("horizon", "h5")

    # 2. 获取 confidence multiplier
    multiplier = _get_model_confidence_multiplier(
        action=action,
        symbol=symbol,
        model_family=model_family,
        horizon=horizon
    )

    # 3. 应用 gating
    if multiplier == 0.0:
        return {
            "approved": False,
            "reason": "model_performance_bad",
            "metadata": {
                "confidence_multiplier": 0.0,
                "source": "clickhouse_model_stats"
            }
        }

    # 4. 调整 confidence
    original_confidence = action.get("confidence", 0.5)
    adjusted_confidence = original_confidence * multiplier
    action["confidence"] = adjusted_confidence
    action["confidence_multiplier"] = multiplier

    logger.info(
        "action_confidence_adjusted",
        action_type=action_type,
        symbol=symbol,
        original_confidence=original_confidence,
        adjusted_confidence=adjusted_confidence,
        multiplier=multiplier
    )
```
